     21 const useSequentialAudio = (
     22   initialState: number,
     23   audioSources: Record<number, string>,
     24   onComplete: () => void,
     25 ) => {
     26   const [sceneState, setSceneState] = useState(initialState);
     27   const audioRef = useRef<HTMLAudioElement | null>(null);
     28 
     29   const currentAudioUrl = audioSources[sceneState];
     30 
     31   useEffect(() => {
     32     if (sceneState > Math.max(...Object.keys(audioSources).map(Number))) {
     33       console.log("All audio finished. Calling onComplete.");
     34       onComplete();
     35       return;
     36     }
     37 
     38     if (!currentAudioUrl) {
     39       return;
     40     }
     41 
     42     console.log(`Starting setup for state ${sceneState}: ${currentAudioUrl}`);
     43 
     44     const existingAudio = audioRef.current;
     45     if (existingAudio) {
     46       existingAudio.pause();
     47     }
     48 
     49     const audio = new Audio(currentAudioUrl);
     50     audioRef.current = audio;
     51 
     52     const handleAudioEnded = () => {
     53       console.log(`Audio ${sceneState} finished. Transitioning state.`);
     54       // 次の状態へ遷移 (例: 1 -> 2, 2 -> 3)
     55       setSceneState((prev) => prev + 1);
     56     };
     57 
     58     audio.addEventListener("ended", handleAudioEnded);
     59 
     60     audio.play().catch((e) => {
     61       console.warn(
     62         `Audio playback failed for state ${sceneState} (requires user interaction):`,
     63         e,
     64       );
     65     });
     66 
     67     return () => {
     68       if (audio === audioRef.current) {
     69         audio.pause();
     70         audio.removeEventListener("ended", handleAudioEnded);
     71       }
     72     };
     73   }, [sceneState, audioSources, onComplete]);
     74   return sceneState;
     75 };

     84 function SceneController({ sceneState }: { sceneState: number }) {
     85   const isFoggy = sceneState === 2;
     86 
     87   const directionalLightIntensity = isFoggy ? 0.1 : 2;
     88 
     89   const ambientLightIntensity = isFoggy ? 0.05 : 0.5;
     90 
     91   useFrame(({ scene }) => {
     92     if (isFoggy) {

     94     } else {
     95       scene.fog = null;
     96     }
     97   });
     98 
     99   return (
    100     <>
    101       <ambientLight intensity={ambientLightIntensity} />
    102 
    103       <directionalLight
    104         position={[5, 10, 5]}
    105         intensity={directionalLightIntensity} // ここで制御

    119 function AnimatedSky({ sceneState }: { sceneState: number }) {
    120   const [sunPosition, setSunPosition] = useState<[number, number, number]>([

    123 
    124   useFrame(({ clock }) => {

    126     const x = Math.sin(time) * 100;
    127     const z = Math.cos(time) * 100;

    129   });
    130 
    131   if (sceneState === 2) {
    132     return null;
    133   }
    134 
    135   return (

    149 function RandomClouds({

    155   const cloudColor = sceneState === 2 ? "#080808" : "#fff";

    157 
    158   const cloudConfigs = useMemo(() => {

    160       const angle = Math.random() * Math.PI * 2;
    161       const distance = Math.sqrt(Math.random()) * radius;
    162 
    163       const x = Math.cos(angle) * distance;
    164       const z = Math.sin(angle) * distance;
    165 
    166       return {
    167         position: [x, height + (Math.random() - 0.5) * 5, z] as [
    168           number,
    169           number,
    170           number,
    171         ],
    172 
    173         bounds: [
    174           Math.random() * 10 + 10,
    175           Math.random() * 2 + 1,
    176           Math.random() * 10 + 10,
    177         ] as [number, number, number],
    178         seed: Math.floor(Math.random() * 1000) + i,
    179         volume: Math.random() * 20 + 5,
    180         opacity: Math.random() * 0.4 + baseOpacity,
    181         color: cloudColor,
    182       };
    183     });
    184   }, [count, radius, height]);
    185 
    186   return (
    187     <>
    188       {cloudConfigs.map((config, index) => (
