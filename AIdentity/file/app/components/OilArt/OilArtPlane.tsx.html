     16 // Simplex Noise (Based on Stefan Gustavson's implementation)
     17 // -----------------------------------------------------------
     18 
     19 vec3 mod289(vec3 x) {
     20 	return x - floor(x * (1.0 / 289.0)) * 289.0;
     21 }
     22 
     23 vec2 mod289(vec2 x) {
     24 	return x - floor(x * (1.0 / 289.0)) * 289.0;
     25 }
     26 
     27 vec3 permute(vec3 x) {
     28 	return mod289(((x*34.0)+1.0)*x);
     29 }
     30 
     31 // 2D Simplex Noise
     32 float noise(vec2 v)
     33 {
     34 	const vec4 C = vec4(0.211324865405187,  // (3.0-sqrt(3.0))/6.0
     35 						0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
     36 	-0.577350269189626,  // -1.0 + 2.0 * (3.0-sqrt(3.0))/6.0
     37 	0.024390243902439); // 1.0/41.0
     38 	// First corner
     39 	vec2 i  = floor(v + dot(v, C.yy) );
     40 	vec2 x0 = v -   i + dot(i, C.xx) ;
     41 
     42 	// Other corners
     43 	vec2 i1;
     44 	//i1.x = step( x0.y, x0.x ); // x.y > x.x ? 1.0 : 0.0
     45 	//i1.y = 1.0 - i1.x;
     46 	i1 = (x0.x > x0.y) ? vec2(1.0, 0.0) : vec2(0.0, 1.0);
     47 	// x0 is the point after skewing and de-skewing to form the right isosceles triangle.
     48 
     49 	vec4 x12 = x0.xyxy + C.xxzz;
     50 	x12.xy -= i1;
     51 
     52 	// Permutations
     53 	i = mod289(i); // Avoids truncation effects in GLSL
     54 	vec3 p = permute( permute( i.y + vec3(0.0, i1.y, 1.0 ))
     55 					 + i.x + vec3(0.0, i1.x, 1.0 ) );
     56 
     57 					 vec3 m = max(0.5 - vec3(dot(x0,x0), dot(x12.xy,x12.xy), dot(x12.zw,x12.zw)), 0.0);
     58 					 m = m*m ;
     59 					 m = m*m ;
     60 
     61 					 // Gradients: 41 points on a unit square
     62 					 vec3 x = 2.0 * fract(p * C.www) - 1.0;
     63 					 vec3 y = abs(x) - 0.4;
     64 					 vec3 h = floor(x + 0.5);
     65 					 vec3 ox = h * C.z;
     66 					 vec3 a0 = x - h;
     67 
     68 					 // Final gradient computation for the three simplex corners
     69 					 m *= 1.79284291400159 - 0.85373472095314 * ( a0*a0 + y*y );
     70 
     71 				 // Gradients can be written concisely as dot( vec*x, vec*y )
     72 				 vec3 g;
     73 				 g.x  = a0.x  * x0.x  + y.x  * x0.y;
     74 				 g.yz = a0.yz * x12.xz + y.yz * x12.yw;
     75 				 return 130.0 * dot(m, g);
     76 }

     80 	vec2 uv = gl_FragCoord.xy / u_resolution.xy;
     81 
     82 	// --- ノイズベクトル生成 ---

     85 		noise(uv * 20.0 - u_time * 0.2)
     86 	) - 0.5;
     87 
     88 	vec2 total_displacement = (u_tiltDirection * 0.005) + (noise_direction * 0.001); 
     89

     91 	shifted_uv = clamp(shifted_uv, 0.001, 0.999);
     92 
     93 	vec4 prev_oil = texture2D(u_feedbackTexture, shifted_uv);
     94 	vec3 current_oil_color = prev_oil.rgb;
     95 	float current_alpha = prev_oil.a; 
     96 
     97 	// 💡 減衰（フェードアウト）係数を調整 (よりゆっくり淡くなるように1.0に近づける)
     98 	// ただし、完全に1.0だと色が残るため、非常に小さい値で調整
     99 	float decay_factor = 0.998; // 例: 非常にゆっくりと色が減衰
    100 	current_oil_color *= decay_factor; 
    101 	current_alpha *= decay_factor;
    102 
    103 	// --- 2. 滴下処理（大幅変更） ---
    104 	if (u_dropPosition.x > -0.9) { 
    105 		vec2 drop_uv = (u_dropPosition + 1.0) / 2.0;
    106 		vec2 relative_pos = uv - drop_uv; // 滴下点からの相対位置
    107 
    108 		// 1. 滴下点からの標準距離を計算
    109 		float standard_dist = length(relative_pos);
    110 
    111 		// 2. 角度(theta)を計算
    112 		// atan2(y, x) は [-PI, PI] の角度を返す
    113 		float theta = atan(relative_pos.y, relative_pos.x); 
    114 
    115 		// 3. 角度と時間に基づいてノイズを生成
    116 		// ノイズは [-1, 1] の範囲で、輪郭に複雑な歪みを与える
    117 		float noise_freq = 5.0; // ノイズの波の数（輪郭のギザギザの細かさ）
    118 		float noise_val = noise(vec2(theta * noise_freq, u_time * 0.5)) * 2.0 - 1.0; 
    119 
    120 		// 4. ノイズによる距離補正 (歪曲)
    121 		// 歪み係数: 0.1 は歪みの強さ (この値を大きくすると r'(θ) が急になります)
    122 		float distortion_strength = 0.001; 
    123 
    124 		// r(θ) のイメージ: 標準の半径 u_dropRadius にノイズを加える
    125 		// この計算により、ノイズ値が r(θ) の変化率 r'(θ) に大きく影響を与える
    126 		float distorted_radius = u_dropRadius + (noise_val * distortion_strength);
    127 
    128 		// 5. ブレンド計算
    129 		// 標準距離 standard_dist と歪んだ半径 distorted_radius を比較
    130 		if (standard_dist < distorted_radius) {
    131 			// スムースステップで中心から輪郭までのグラデーションを制御
    132 			float blend_factor = 1.0 - smoothstep(distorted_radius * 0.9, distorted_radius, standard_dist);
    133 
    134 			// 輪郭を暗くする処理（維持）
    135 			float edge_darkness = 1.0 - smoothstep(distorted_radius * 0.4, distorted_radius, standard_dist);
    136 			vec3 darker_drop_color = u_dropColor * (0.25 + edge_darkness * 0.75); 
    137 
    138 			// 混色ではなく、色の上書きブレンド
    139 			current_oil_color = mix(current_oil_color, darker_drop_color, blend_factor);
    140 			current_alpha = mix(current_alpha, 1.0, blend_factor);
    141 		}
    142 	}
    143 
    144 	// 💡 鮮やかさ（ビビッドさ）の強調処理
    145 	// 1. RGBの平均輝度（グレースケール値）を計算
    146 	float average_light = (current_oil_color.r + current_oil_color.g + current_oil_color.b) / 3.0;
    147 
    148 	// 2. 彩度を強調する係数 (Saturation Boost Factor)
    149 	// 1.0より大きい値にすると彩度が増します。2.0〜3.0が推奨値です。
    150 	float saturation_boost = 1.035; 
    151 
    152 	// 3. 平均輝度と現在の色を mix し、彩度を上げる
    153 	// (current_oil_color - average_light) は色の「差」成分。
    154 	// この差を大きくすることで彩度が増します。
    155 	vec3 vivid_color = mix(vec3(average_light), current_oil_color, saturation_boost);
    156 
    157 	// 結果をクランプして色を [0, 1] の範囲に収める
    158 	current_oil_color = clamp(vivid_color, 0.0, 1.0);
    159 
    160 	gl_FragColor = vec4(current_oil_color, current_alpha);
    161 }

    219 export const OilArtPlane: React.FC<OilArtPlaneProps> = ({ onTriggerAPI }) => {
    220 	const materialRef = useRef<OilArtMaterialRef>(null!);
    221 	const meshRef = useRef<THREE.Mesh>(null!);
    222 	const { gl, size, viewport, camera } = useThree();

    226 	const [rtB] = useState(() => new THREE.WebGLRenderTarget(size.width, size.height));
    227 	const currentRenderTexture = useRef(rtA);
    228 	const feedbackRenderTexture = useRef(rtB);
    229 
    230 	// 初期化とリサイズ処理

    232 		if (materialRef.current) {
    233 			materialRef.current.uniforms.u_resolution.value.set(size.width, size.height);
    234 			rtA.setSize(size.width, size.height);
    235 			rtB.setSize(size.width, size.height);
    236 
    237 			// 初回は基本色でテクスチャをクリア (任意)
    238 			gl.setRenderTarget(rtA);
    239 
    240 			// 💡 修正点: setClearColor(Color, alpha) を使用
    241 			// THREE.Colorオブジェクトを使って色を設定
    242 			const baseColor = new THREE.Color(0.0, 0.0, 0.002); 
    243 			const alpha = 0.7; 
    244 			gl.setClearColor(baseColor, alpha); 
    245 
    246 			// gl.clear() で設定した色とアルファ値を使ってクリア
    247 			gl.clear(); 
    248 			gl.setRenderTarget(null); // デフォルトに戻す
    249 		}
    250 	}, [gl, size, rtA, rtB]);
    251 
    252 	// 滴下処理: positionはビューポート座標 (-1.0〜1.0)
    253 	const dropOil = useCallback((position: { x: number; y: number }, color: THREE.Color) => {
    254 		if (materialRef.current) {
    255 			materialRef.current.uniforms.u_dropPosition.value.set(position.x, position.y);
    256 			materialRef.current.uniforms.u_dropColor.value.copy(color);
    257 			// ドロップをトリガー
    258 		}
    259 	}, []);
    260 
    261 	// 傾き処理: directionはベクトル (-1.0〜1.0)
    262 	const tilt = useCallback((direction: { x: number; y: number }) => {
    263 		if (materialRef.current) {
    264 			materialRef.current.uniforms.u_tiltDirection.value.set(direction.x, direction.y);
    265 		}
    266 	}, []);
    267 
    268 	// 外部APIの公開
    269 	useEffect(() => {
    270 		onTriggerAPI({ dropOil, tilt });
    271 	}, [onTriggerAPI, dropOil, tilt]);
    272 
    273 	useFrame(({ clock }) => {
    274 		if (!materialRef.current || !meshRef.current) return;
    275 
    276 		materialRef.current.uniforms.u_time.value = clock.getElapsedTime();
    277 
    278 		// 1. **オフスクリーンレンダリング**: 次のフレームの状態を計算し、`feedbackRenderTexture`に書き込む
    279 		gl.setRenderTarget(feedbackRenderTexture.current);
    280 		// 書き込む際には、前のフレームの状態を`u_feedbackTexture`としてシェーダーに渡す
    281 		materialRef.current.uniforms.u_feedbackTexture.value = currentRenderTexture.current.texture;
    282 		gl.render(meshRef.current, camera); // Ping-Pongバッファの計算
    283 
    284 		// 2. **画面への最終描画**: 計算したばかりのテクスチャを画面に表示
    285 		gl.setRenderTarget(null); // デフォルトのレンダーターゲット（画面）に戻す
    286 		materialRef.current.uniforms.u_feedbackTexture.value = feedbackRenderTexture.current.texture;
    287 		gl.render(meshRef.current, camera);
    288 
    289 		// 3. **バッファのスワップ**: 次のフレームのために currentとfeedbackを入れ替える
    290 		[currentRenderTexture.current, feedbackRenderTexture.current] = [feedbackRenderTexture.current, currentRenderTexture.current];
    291 
    292 		// ドロップをリセット: 1フレーム描画したら無効な位置に戻す
    293 		// これにより、毎フレーム新しい色が滴下されるのではなく、トリガー時のみ追加される
    294 		materialRef.current.uniforms.u_dropPosition.value.set(-1, -1);
    295 	});
    296 
    297 	return (
    298 		<mesh ref={meshRef} scale={[viewport.width, viewport.height, 1]}>
    299 		<planeGeometry args={[1, 1]} />
    300 		<oilArtMaterial ref={materialRef} />
    301 		</mesh>
    302 	);
    303 };
