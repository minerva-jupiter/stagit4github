     60       const offsetY = (stageHeight - effectiveSize) / 2;
     61       return {
     62         x: (p.x - offsetX) * scale,
     63         y: (p.y - offsetY) * scale,
     64       };
     65     },
     66     [stageWidth, stageHeight, viewBoxSize],
     67   );

     71       const effectiveSize = Math.min(stageWidth, stageHeight);
     72       const scale = effectiveSize / viewBoxSize;
     73       const offsetX = (stageWidth - effectiveSize) / 2;
     74       const offsetY = (stageHeight - effectiveSize) / 2;
     75       return {
     76         x: p.x * scale + offsetX,
     77         y: p.y * scale + offsetY,
     78       };
     79     },
     80     [stageWidth, stageHeight, viewBoxSize],
     81   );
     82   const getPointerPosition = useCallback(
     83     (
     84       e: React.MouseEvent | React.TouchEvent | MouseEvent | TouchEvent,

     92         clientY = e.touches[0].clientY;
     93       } else {
     94         clientX = e.clientX;
     95         clientY = e.clientY;
     96       }
     97

    124         let foundDValue = null;

    130           }

    138         );

    141       }

    143     if (isClient && backgroundPathD === "") {
    144       fetchAndParseSvg();
