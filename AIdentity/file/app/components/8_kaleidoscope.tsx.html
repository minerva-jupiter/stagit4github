     11 // 🌟 1. 頂点シェーダー (Vertex Shader) 🌟
     12 // ----------------------------------------------------
     13 const vertexShader = `
     14 // NOTE: Three.jsが自動的に以下の組み込み変数を渡すため、
     15 // 以下の行を削除します:
     16 // attribute vec3 position;
     17 // attribute vec2 uv;
     18 
     19 // Three.jsの組み込み変数を利用する宣言は残す
     20 // Three.jsはジオメトリから取得したuvを自動的にvarying vUvに渡すための
     21 // 組み込み変数として扱ってくれるため、カスタムシェーダーではvaryingのみを定義します。
     22 
     23 varying vec2 vUv; 
     24 // 組み込み変数 (attributes) はカスタムシェーダーでは宣言せず、
     25 // 以下のように直接使用します。
     26 // * Three.jsの組み込み attribute: position, uv, normal...
     27 // * Three.jsの組み込み uniform: modelMatrix, projectionMatrix...
     28 
     29 void main() {
     30     // 組み込みの uv を varying vUv に格納
     31     vUv = uv; 
     32     
     33     // 組み込みの行列を使って頂点位置を計算
     34     gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
     35 }
     36 `;

     40 // 🌟 2. フラグメントシェーダー (Fragment Shader) 🌟
     41 // ----------------------------------------------------
     42 const fragmentShader = `
     43 
     44 uniform float u_time;
     45 uniform vec2 u_resolution;
     46 varying vec2 vUv;
     47 const float PI = 3.14159265359;
     48 
     49 // ---------------------------
     50 // 歪み用のノイズ関数 (Simplex Noise 簡易版)
     51 // ---------------------------
     52 // ノイズは万華鏡の中心座標をぐにゃぐにゃと歪ませるために使う
     53 float hash(vec2 p) {
     54     return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
     55 }
     56 
     57 float noise(vec2 p) {
     58     vec2 i = floor(p);
     59     vec2 f = fract(p);
     60     
     61     // 🌟 修正: Quintic (6x^5) から Cubic (3x^2) へ変更 🌟
     62     vec2 u = f * f * (3.0 - 2.0 * f); 
     63     // u = smoothstep(0.0, 1.0, f); と同じ
     64 
     65     float a = hash(i);
     66     float b = hash(i + vec2(1.0, 0.0));
     67     float c = hash(i + vec2(0.0, 1.0));
     68     float d = hash(i + vec2(1.0, 1.0));
     69 
     70     return mix(mix(a, b, u.x), mix(c, d, u.x), u.y);
     71 }
     72 
     73 // ---------------------------
     74 // 万華鏡パターン生成関数
     75 // ---------------------------
     76 vec2 kaleidoscope(vec2 p, float segments) {
     77     // 1. 極座標に変換
     78     float radius = length(p);
     79     float theta = atan(p.y, p.x);
     80     
     81     // 2. 時間で回転
     82     theta += u_time * 3.0; 
     83 
     84     // 3. 分割数で角度を正規化し、ミラーリングを適用
     85     theta /= (2.0 * PI / segments);
     86     theta = fract(theta) - 0.5;
     87     theta = abs(theta) * 2.0;
     88     
     89     // 4. 新しい極座標を返す
     90     return vec2(theta, radius);
     91 }
     92 
     93 
     94 void main() {
     95     // 1. 座標の準備とアスペクト比の補正
     96     vec2 p = vUv * 2.0 - 1.0; 
     97     float aspect = u_resolution.x / u_resolution.y;
     98     p.x *= aspect; 
     99     
    100     // ---------------------------
    101     // 🌟 歪みの追加 🌟
    102     // ---------------------------
    103     // 時間と共にノイズが変化し、万華鏡全体を歪ませる
    104     vec2 noiseUV = p * 1.5 + u_time * 0.1;
    105     // ノイズをサンプリングし、座標 p に加算して歪みを発生させる
    106     float distortion = noise(noiseUV) * 0.25; 
    107     p += distortion; 
    108     
    109     
    110     // ---------------------------
    111     // 🌟 1. 中心からのパターン 🌟
    112     // ---------------------------
    113     vec2 k1 = kaleidoscope(p, 10.0); // 分割数 10
    114     
    115     // パターン1の色
    116     float h1 = k1.x + k1.y * 0.5 + u_time * 0.2;
    117     vec3 color1 = 0.5 + 0.5 * cos(6.283 * (vec3(h1, h1 + 0.33, h1 + 0.66) * 0.3 + 0.5));
    118     
    119     
    120     // ---------------------------
    121     // 🌟 2. 中心でない場所からの放射状パターン 🌟
    122     // ---------------------------
    123     // 新しい中心を設定 (画面左上に時間でゆっくりと移動)
    124     vec2 offsetCenter = vec2(-0.8 * aspect, 0.6);
    125     
    126     // p から offsetCenter へのベクトルを計算
    127     vec2 p2 = p - offsetCenter; 
    128 
    129     // 新しい中心から万華鏡パターンを生成 (分割数 6)
    130     vec2 k2 = kaleidoscope(p2, 6.0); 
    131 
    132     // パターン2の色
    133     // k2.y (半径) の値が小さいほど明るくなるように調整
    134     float brightness = smoothstep(0.5, 0.0, k2.y);
    135     vec3 color2 = vec3(1.0, 0.7, 0.3) * brightness; // 暖色系の放射
    136     
    137     
    138     // ---------------------------
    139     // 🌟 3. 色の合成 🌟
    140     // ---------------------------
    141     // パターン1とパターン2を単純に加算合成 (Add Blend)
    142     vec3 finalColor = color1 + color2 * 1.5; 
    143 
    144     // 外側を暗くする (ケラレ効果)
    145     finalColor *= (1.0 - length(p) * 0.3);
    146 
    147     gl_FragColor = vec4(finalColor, 1.0);
    148 }
    149 `;

    154 const KaleidoscopeMesh = () => {
    155   const meshRef = useRef<THREE.Mesh>(null!);
    156   
    157   const uniforms = useMemo(
    158     () => ({
    159       u_time: { value: 0.0 },
    160       u_resolution: { value: new THREE.Vector2(0, 0) },
    161       // 🌟 追加: 中心点 (初期値は少し左上にずらす) 🌟
    162       u_center: { value: new THREE.Vector2(-0.2, 0.3) }, 
    163     }),
    164     []
    165   );
    166 
    167   useFrame(({ clock, size }) => {
    168     if (meshRef.current) {
    169         const material = meshRef.current.material as THREE.ShaderMaterial;
    170         material.uniforms.u_time.value = clock.getElapsedTime();
    171         material.uniforms.u_resolution.value.set(size.width, size.height);
    172         
    173         // 🌟 おまけ: 中心点を時間でゆっくり動かす 🌟
    174         // Wevalの曲のような有機的な動きを追加
    175         const time = clock.getElapsedTime();
    176         material.uniforms.u_center.value.x = Math.sin(time * 0.3) * 0.5;
    177         material.uniforms.u_center.value.y = Math.cos(time * 0.25) * 0.4;
    178     }
    179   });
    180 
    181   return (
    182     <mesh ref={meshRef} position={[0, 0, 0]} scale={2}>
    183       <planeGeometry args={[1, 1]} /> 
    184       <shaderMaterial
    185         vertexShader={vertexShader}
    186         fragmentShader={fragmentShader}
    187         uniforms={uniforms}
    188         side={THREE.DoubleSide}
    189       />
    190     </mesh>
    191   );
    192 };
