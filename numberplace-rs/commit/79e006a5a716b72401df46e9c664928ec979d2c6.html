+use rand::RngExt;

+pub const GRID_SIZE: usize = 9;
+
+#[derive(Debug, Clone, Copy, PartialEq)]
+pub struct Cell {
+    pub value: Option<u8>,
+    pub is_fixed: bool,
+}
+
+impl Cell {
+    pub fn new(value: Option<u8>, is_fixed: bool) -> Self {
+        Self { value, is_fixed }
+    }
+}
+
+pub struct Board {
+    pub cells: [[Cell; GRID_SIZE]; GRID_SIZE],
+}
+
+impl Board {
+    pub fn new() -> Self {
+        let cells = [[Cell::new(None, false); GRID_SIZE]; GRID_SIZE];
+        Self { cells }
+    }
+
+    pub fn set_value(&mut self, row: usize, col: usize, value: Option<u8>) {
+        if !self.cells[row][col].is_fixed {
+            self.cells[row][col].value = value;
+        }
+    }
+
+    pub fn is_valid(&self, row: usize, col: usize, val: u8) -> bool {
+        // Check row
+        for c in 0..GRID_SIZE {
+            if c != col && self.cells[row][c].value == Some(val) {
+                return false;
+            }
+        }
+
+        // Check column
+        for r in 0..GRID_SIZE {
+            if r != row && self.cells[r][col].value == Some(val) {
+                return false;
+            }
+        }
+
+        // Check 3x3 box
+        let start_row = (row / 3) * 3;
+        let start_col = (col / 3) * 3;
+        for r in start_row..start_row + 3 {
+            for c in start_col..start_col + 3 {
+                if (r != row || c != col) && self.cells[r][c].value == Some(val) {
+                    return false;
+                }
+            }
+        }
+
+        true
+    }
+
+    pub fn is_conflict(&self, row: usize, col: usize) -> bool {
+        if let Some(val) = self.cells[row][col].value {
+            !self.is_valid(row, col, val)
+        } else {
+            false
+        }
+    }
+
+    pub fn is_full(&self) -> bool {
+        self.cells
+            .iter()
+            .all(|row| row.iter().all(|c| c.value.is_some()))
+    }
+
+    pub fn is_solved(&self) -> bool {
+        if !self.is_full() {
+            return false;
+        }
+        for r in 0..GRID_SIZE {
+            for c in 0..GRID_SIZE {
+                if self.is_conflict(r, c) {
+                    return false;
+                }
+            }
+        }
+        true
+    }
+
+    pub fn solve(&mut self) -> bool {
+        let mut empty_cell = None;
+        for r in 0..GRID_SIZE {
+            for c in 0..GRID_SIZE {
+                if self.cells[r][c].value.is_none() {
+                    empty_cell = Some((r, c));
+                    break;
+                }
+            }
+            if empty_cell.is_some() {
+                break;
+            }
+        }
+
+        let (r, c) = match empty_cell {
+            Some(pos) => pos,
+            None => return true, // Solved
+        };
+
+        for val in 1..=9 {
+            if self.is_valid(r, c, val) {
+                self.cells[r][c].value = Some(val);
+                if self.solve() {
+                    return true;
+                }
+                self.cells[r][c].value = None;
+            }
+        }
+
+        false
+    }
+
+    pub fn generate(&mut self) {
+        // Clear board
+        *self = Board::new();
+
+        // Fill diagonal 3x3 boxes
+        for i in (0..GRID_SIZE).step_by(3) {
+            self.fill_box(i, i);
+        }
+
+        // Solve the rest
+        self.solve();
+
+        // Set all as fixed for now (as the base)
+        for r in 0..GRID_SIZE {
+            for c in 0..GRID_SIZE {
+                self.cells[r][c].is_fixed = true;
+            }
+        }
+
+        // Randomly remove cells
+        let mut rng = rand::rng();
+        let cells_to_remove = 40;
+        let mut count = 0;
+        while count < cells_to_remove {
+            let r = rng.random_range(0..GRID_SIZE);
+            let c = rng.random_range(0..GRID_SIZE);
+            if self.cells[r][c].value.is_some() {
+                self.cells[r][c].value = None;
+                self.cells[r][c].is_fixed = false;
+                count += 1;
+            }
+        }
+    }
+
+    fn fill_box(&mut self, row: usize, col: usize) {
+        use rand::seq::SliceRandom;
+        let mut nums: Vec<u8> = (1..=9).collect();
+        let mut rng = rand::rng();
+        nums.shuffle(&mut rng);
+        for r in 0..3 {
+            for c in 0..3 {
+                self.cells[row + r][col + c].value = Some(nums[r * 3 + c]);
+            }
+        }
+    }

+fn run_app<B: Backend>(
+    terminal: &mut Terminal<B>,
+    app: &mut App,
+) -> Result<(), Box<dyn std::error::Error>>
+where
+    B::Error: std::error::Error + 'static,
+{
+    loop {
+        terminal.draw(|f| ui(f, app))?;
+
+        if let Event::Key(key) = event::read()? {
+            match key.code {
+                KeyCode::Char('q') => return Ok(()),
+                KeyCode::Char('n') if app.board.is_solved() => {
+                    app.board.generate();
+                    app.cursor = (0, 0);
+                }
+                KeyCode::Char('s') => {
+                    app.board.solve();
+                }
+                KeyCode::Char('h') | KeyCode::Left => app.move_cursor(0, -1),
+                KeyCode::Char('j') | KeyCode::Down => app.move_cursor(1, 0),
+                KeyCode::Char('k') | KeyCode::Up => app.move_cursor(-1, 0),
+                KeyCode::Char('l') | KeyCode::Right => app.move_cursor(0, 1),

+        }
