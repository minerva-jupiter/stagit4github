+import json
+
+with open("./winget.json","r") as fs:
+    wingetJson = json.load(fs)
+    fs.close
+
+with open("wingetRemoveList.json") as fl:
+    wingetRemoveListJson = json.load(fl)
+    fl.close
+
+removeList = []
+
+for i in range(len(wingetJson["Sources"][0]["Packages"])):
+    for j in range(len(wingetRemoveListJson["wingetRemoveList"])):
+        if wingetJson["Sources"][0]["Packages"][i]["PackageIdentifier"] == wingetRemoveListJson["wingetRemoveList"][j]:
+            removeList.append(wingetJson["Sources"][0]["Packages"][i])
+            print(wingetJson["Sources"][0]["Packages"][i]["PackageIdentifier"] + " was removed from winget.json")
+
+wingetJson["Sources"][0]["Packages"] = [i for i in wingetJson["Sources"][0]["Packages"] if i not in removeList]
+
+with open('./winget.json', 'w') as f:
+    json.dump(wingetJson, f, indent=2)
+    f.close+
