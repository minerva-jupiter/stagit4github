+import pandas as pd
+
+
+def convert_to_bitwarden_csv(input_file, output_file):
+    if input_file.endswith(".xlsx"):
+        df = pd.read_excel(input_file)
+    elif input_file.endswith(".md"):
+        df = pd.read_table(input_file, sep="|").iloc[1:]
+        df = df.dropna(axis=1, how="all").map(
+            lambda x: x.strip() if isinstance(x, str) else x
+        )
+        df.columns = [col.strip() for col in df.columns]
+
+    bw_df = pd.DataFrame()
+    bw_df["folder"] = None
+    bw_df["favorite"] = 0
+    bw_df["type"] = "login"
+    bw_df["name"] = df["サービス名"]
+    bw_df["notes"] = None
+    bw_df["fields"] = None
+    bw_df["login_uri"] = None
+    bw_df["login_username"] = df["アカウント名"]
+    bw_df["login_password"] = df["パスワード"]
+    bw_df["login_totp"] = None
+
+    bw_df.to_csv(output_file, index=False, encoding="utf-8")
+
+
+if __name__ == "__main__":
+    convert_to_bitwarden_csv(r"password.xlsx", "bitwarden_import.csv")

+import csv
+import os
+import re
+import sys
+
+
+def convert_md_file_to_bitwarden_csv(input_path):
+    if not os.path.exists(input_path):
+        print(f"Error: {input_path} not found.")
+        return
+
+    with open(input_path, "r", encoding="utf-8") as f:
+        lines = f.readlines()
+
+    data = []
+    for line in lines:
+        if re.match(r"^\s*\|?[:\-\s|]+\|?\s*$", line):
+            continue
+        columns = [col.strip() for col in line.split("|") if col.strip()]
+        if columns:
+            data.append(columns)
+
+    if len(data) < 2:
+        print("Error: No valid data found in table.")
+        return
+
+    headers = data[0]
+    rows = data[1:]
+    output_file = "bitwarden_import.csv"
+    bw_headers = [
+        "folder",
+        "favorite",
+        "type",
+        "name",
+        "notes",
+        "fields",
+        "login_uri",
+        "login_username",
+        "login_password",
+        "login_totp",
+    ]
+
+    with open(output_file, "w", newline="", encoding="utf-8") as f:
+        writer = csv.DictWriter(f, fieldnames=bw_headers)
+        writer.writeheader()
+
+        for row in rows:
+            item = dict(zip(headers, row))
+            writer.writerow(
+                {
+                    "folder": "Main",
+                    "favorite": 0,
+                    "type": "login",
+                    "name": item.get("サービス名", ""),
+                    "login_username": item.get("ユーザー名", ""),
+                    "login_password": item.get("パスワード", ""),
+                }
+            )
+
+    print(f"Success: {output_file} has been created.")
+
+
+if __name__ == "__main__":
+    # 実行時にファイル名を指定するか、デフォルトの passwords.md を読み込む
+    target_file = sys.argv[1] if len(sys.argv) > 1 else "password01.md"
+    convert_md_file_to_bitwarden_csv(target_file)
