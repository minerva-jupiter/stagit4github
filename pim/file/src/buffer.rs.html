     40         }

     42         let swap_path = path.with_extension(format!(
     43             "{}.swp",
     44             path.extension().and_then(|e| e.to_str()).unwrap_or("txt")
     45         ));

     47         let mut deltas = HashMap::new();

     52             for line in swap_reader.lines() {
     53                 let line = line?;
     54                 if line == "SAVE" {

     57                 }

     63                 }
     64             }

     67         }
