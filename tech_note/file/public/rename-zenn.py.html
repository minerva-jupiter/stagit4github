      7 def has_japanese(text):
      8     return bool(re.search(r'[^\x00-\x7F]', text))

     10 def to_snake_case(text):
     11     text = text.lower()
     12     text = re.sub(r'[^a-z0-9\s]', '', text)
     13     text = re.sub(r'\s+', '_', text.strip())
     14     return text

     33                 continue

     38                 english_title = title

     47             qiita_matter = {
     48                 'title': title,
     49                 'tags': zenn_matter.get('topics', []),
