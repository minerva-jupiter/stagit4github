    824         public void CreateUnitProcessor_LimitMode()
    825         {
    826             string resourceName = "xResourceName";
    827             string moduleName = "xModuleName";
    828             Version version = new Version("1.0");
    829 
    830             var processorEnvMock = new Mock<IProcessorEnvironment>();
    831             processorEnvMock.Setup(
    832                     m => m.GetDscResource(It.Is<ConfigurationUnitAndModule>(c => c.Unit.Type == resourceName)))
    833                 .Returns(new DscResourceInfoInternal(resourceName, moduleName, version))
    834                 .Verifiable();
    835 
    836             var limitSet = new ConfigurationSet();
    837             var limitUnit = new ConfigurationUnit
    838             {
    839                 Type = resourceName,
    840                 Intent = ConfigurationUnitIntent.Apply,
    841             };

    843             limitUnit.Metadata.Add("version", version.ToString());
    844             limitSet.Units = new List<ConfigurationUnit> { limitUnit };
    845 
    846             var configurationSetProcessor = new PowerShellConfigurationSetProcessor(
    847                 processorEnvMock.Object,
    848                 limitSet,
    849                 true);
    850 
    851             // Calling with unit different from limit set should throw.
    852             var unitDifferentContent = new ConfigurationUnit
    853             {
    854                 Type = "differentResourceName",
    855                 Intent = ConfigurationUnitIntent.Apply,
    856             };
    857 
    858             Assert.Throws<System.InvalidOperationException>(() => configurationSetProcessor.CreateUnitProcessor(unitDifferentContent));
    859             Assert.Throws<System.InvalidOperationException>(() => configurationSetProcessor.GetUnitProcessorDetails(unitDifferentContent, ConfigurationUnitDetailFlags.Load));

    868 
    869             // Calling CreateProcessor again should thow. Calling GetProcessorDetails multiple times is ok.
    870             Assert.Throws<System.InvalidOperationException>(() => configurationSetProcessor.CreateUnitProcessor(limitUnit));
    871             var processorDetails2 = configurationSetProcessor.GetUnitProcessorDetails(limitUnit, ConfigurationUnitDetailFlags.Load);
    872             Assert.NotNull(processorDetails2);
    873             Assert.Equal(moduleName, processorDetails2.ModuleName);
