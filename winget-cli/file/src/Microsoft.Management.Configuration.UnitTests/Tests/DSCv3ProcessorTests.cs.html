     42         /// <summary>
     43         /// Tests for the unit details caching.
     44         /// </summary>
     45         [Fact]
     46         public void Set_UnitPropertyDetailsCached()
     47         {
     48             var (factory, dsc) = CreateTestFactory();
     49             var set = this.ConfigurationSet();
     50             string type1 = "Type1";
     51             string type2 = "Type2";
     52             var unit1 = this.ConfigurationUnit().Assign(new { Type = type1 });
     53             var unit2 = this.ConfigurationUnit().Assign(new { Type = type2 });
     54 
     55             var setProcessor = factory.CreateSetProcessor(set);
     56 
     57             // Initially, no details
     58             var details = setProcessor.GetUnitProcessorDetails(unit1, ConfigurationUnitDetailFlags.Local);
     59             Assert.Null(details);
     60 
     61             // Null result not cached
     62             dsc.GetResourceByTypeResult = new TestResourceListItem() { Type = type1 };
     63             details = setProcessor.GetUnitProcessorDetails(unit1, ConfigurationUnitDetailFlags.Local);
     64             Assert.NotNull(details);
     65             Assert.Equal(type1, details.UnitType);
     66 
     67             // Not-null result cached
     68             dsc.GetResourceByTypeResult = null;
     69             dsc.GetResourceByTypeDelegate = s => throw new System.Exception("Shouldn't be called");
     70             details = setProcessor.GetUnitProcessorDetails(unit1, ConfigurationUnitDetailFlags.Local);
     71             Assert.NotNull(details);
     72             Assert.Equal(type1, details.UnitType);
     73 
     74             // Different type, no details
     75             dsc.GetResourceByTypeDelegate = null;
     76             details = setProcessor.GetUnitProcessorDetails(unit2, ConfigurationUnitDetailFlags.Local);
     77             Assert.Null(details);
     78 
     79             // Null result not cached
     80             dsc.GetResourceByTypeResult = new TestResourceListItem() { Type = type2 };
     81             details = setProcessor.GetUnitProcessorDetails(unit2, ConfigurationUnitDetailFlags.Local);
     82             Assert.NotNull(details);
     83             Assert.Equal(type2, details.UnitType);
     84 
     85             // First type is still first type
     86             dsc.GetResourceByTypeResult = null;
     87             dsc.GetResourceByTypeDelegate = s => throw new System.Exception("Shouldn't be called");
     88             details = setProcessor.GetUnitProcessorDetails(unit1, ConfigurationUnitDetailFlags.Local);
     89             Assert.NotNull(details);
     90             Assert.Equal(type1, details.UnitType);
     91         }

    246         /// <summary>
    247         /// Test for unit export with complex data.
    248         /// </summary>
    249         [Fact]
    250         public void GetAllUnits_Complex()
    251         {
    252             var (factory, dsc) = CreateTestFactory();
    253             var processor = this.CreateConfigurationProcessorWithDiagnostics(factory);
    254 
    255             string type1 = "Type1";
    256             string type2 = "Type2";
    257 
    258             string name1 = "1";
    259             string name2 = "2";
    260 
    261             var unit1 = this.ConfigurationUnit().Assign(new { Type = type1 });
    262 
    263             dsc.GetResourceByTypeDelegate = (type) =>
    264             {
    265                 Assert.Equal(type1, type);
    266                 return new TestResourceListItem() { Type = type1 };
    267             };
    268 
    269             ValueSet set1 = new ValueSet();
    270             set1.Add("key1", "val1");
    271 
    272             ValueSet metadata1 = new ValueSet();
    273             metadata1.Add("met1", "val11");
    274 
    275             ValueSet set2 = new ValueSet();
    276             set2.Add("key2", "val2");
    277 
    278             List<string> dependencies2 = new List<string>();
    279             dependencies2.Add(name1);
    280 
    281             dsc.ExportResourceResult = new List<IResourceExportItem>()
    282             {
    283                 new TestResourceExportItem() { Type = type1, Name = name1, Settings = set1, Metadata = metadata1 },
    284                 new TestResourceExportItem() { Type = type2, Name = name2, Settings = set2, Dependencies = dependencies2 },
    285             };
    286 
    287             var result = processor.GetAllUnits(unit1);
    288 
    289             Assert.NotNull(result);
    290             Assert.NotNull(result.ResultInformation);
    291             Assert.Null(result.ResultInformation.ResultCode);
    292 
    293             Assert.Equal(2, result.Units.Count);
    294 
    295             var result1 = result.Units.Single(unit => unit.Identifier == name1);
    296             var result2 = result.Units.Single(unit => unit.Identifier == name2);
    297 
    298             Assert.Equal(type1, result1.Type);
    299             Assert.Equal(type2, result2.Type);
    300 
    301             Assert.Contains(set1.First(), result1.Settings);
    302             Assert.Contains(set2.First(), result2.Settings);
    303 
    304             Assert.Contains(metadata1.First(), result1.Metadata);
    305             Assert.Empty(result2.Metadata);
    306 
    307             Assert.Empty(result1.Dependencies);
    308             Assert.Contains(dependencies2.First(), result2.Dependencies);
    309         }
