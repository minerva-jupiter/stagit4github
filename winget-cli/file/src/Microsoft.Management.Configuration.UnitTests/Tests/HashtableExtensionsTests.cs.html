     39         /// <summary>
     40         /// Tests ToValueSet with simple types.
     41         /// </summary>
     42         [Fact]
     43         public void ToValueSet_Test()
     44         {
     45             var ht = new Hashtable()
     46             {
     47                 { "key1", "value1" },
     48                 { "key2", 2 },
     49                 { "key3", true },
     50             };
     51 
     52             var valueSet = ht.ToValueSet();
     53 
     54             Assert.True(valueSet.ContainsKey("key1"));
     55             Assert.Equal("value1", (string)valueSet["key1"]);
     56 
     57             Assert.True(valueSet.ContainsKey("key2"));
     58             Assert.Equal(2, (int)valueSet["key2"]);
     59 
     60             Assert.True(valueSet.ContainsKey("key3"));
     61             Assert.True((bool)valueSet["key3"]);
     62         }
     63
