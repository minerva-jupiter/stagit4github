     99         {
    100             var configurationProcessorFactory = new PowerShellConfigurationSetProcessorFactory();
    101 
    102             var properties = configurationProcessorFactory.As<IPowerShellConfigurationProcessorFactoryProperties>();
    103             properties.Location = location;
    104 
    105             if (properties.Location == PowerShellConfigurationProcessorLocation.Custom)
    106             {
    107                 properties.CustomLocation = @"c:\this\is\a\module\path";
    108             }
    109 
    110             var configurationSet = new ConfigurationSet();
    111 
    112             var setProcessor = configurationProcessorFactory.CreateSetProcessor(configurationSet) as PowerShellConfigurationSetProcessor;
    113             Assert.NotNull(setProcessor);
    114 
    115             var modulePath = setProcessor.ProcessorEnvironment.GetVariable<string>(Variables.PSModulePath);
    116             Assert.Contains($"{PowerShellConfigurationSetProcessorFactory.GetWinGetModulePath()};", modulePath);

    140         /// <summary>
    141         /// Tests the configuration set processor in limitation mode.
    142         /// </summary>
    143         [Fact]
    144         public void CreateSetProcessor_LimitMode()
    145         {
    146             var configurationProcessorFactory = new PowerShellConfigurationSetProcessorFactory();
    147             var configurationSet = new ConfigurationSet();
    148             configurationProcessorFactory.LimitationSet = configurationSet;
    149 
    150             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.LimitationSet = configurationSet);
    151             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.ProcessorType = PowerShellConfigurationProcessorType.Default);
    152             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.AdditionalModulePaths = new List<string>());
    153             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.ImplicitModulePaths = new List<string>());
    154             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.Policy = PowerShellConfigurationProcessorPolicy.Unrestricted);
    155             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.Location = PowerShellConfigurationProcessorLocation.Custom);
    156             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.CustomLocation = @"c:\this\is\a\module\path");
    157 
    158             var setProcessor = configurationProcessorFactory.CreateSetProcessor(configurationSet) as PowerShellConfigurationSetProcessor;
    159             Assert.NotNull(setProcessor);
    160             Assert.True(setProcessor.IsLimitMode);
    161 
    162             // Create processor again in limit mode should fail
    163             Assert.Throws<System.InvalidOperationException>(() => configurationProcessorFactory.CreateSetProcessor(configurationSet));
    164         }

    166         /// <summary>
    167         /// Tests the configuration set processor factory with ImplicitModulePaths.
    168         /// </summary>
    169         [Fact]
    170         public void ImplicitModulePaths()
    171         {
    172             var configurationProcessorFactory = new PowerShellConfigurationSetProcessorFactory();
    173 
    174             // When ImplicitModulePaths module paths are not set
    175             configurationProcessorFactory.AdditionalModulePaths = new List<string> { @"c:\this\is\additional" };
    176             Assert.Equal(configurationProcessorFactory.AdditionalModulePaths, new List<string> { @"c:\this\is\additional" });
    177 
    178             // Implicit ModulePaths are set, it automatically populates AdditionalModulePaths
    179             configurationProcessorFactory.ImplicitModulePaths = new List<string> { @"c:\this\is\implicit" };
    180             Assert.Equal(configurationProcessorFactory.AdditionalModulePaths, new List<string> { @"c:\this\is\additional", @"c:\this\is\implicit" });
    181 
    182             // Set AdditionalModulePaths when ImplicitModulePaths module paths are set
    183             configurationProcessorFactory.AdditionalModulePaths = new List<string> { @"c:\this\is\additional\2" };
    184             Assert.Equal(configurationProcessorFactory.AdditionalModulePaths, new List<string> { @"c:\this\is\additional\2", @"c:\this\is\implicit" });
    185         }
