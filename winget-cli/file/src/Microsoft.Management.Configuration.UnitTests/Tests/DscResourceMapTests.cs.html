     56         /// <summary>
     57         /// Test GetResource for resources in the map.
     58         /// </summary>
     59         [Fact]
     60         public void DscResourcesMap_GetResource()
     61         {
     62             var dscResourceMap = new DscResourcesMap(this.CreateDscResourceInfo());
     63 
     64             // Get just by name.
     65             var zoroResult = dscResourceMap.GetResource(ResourceZoro, null, null);
     66             Assert.NotNull(zoroResult);
     67             Assert.Equal(ResourceZoro, zoroResult.Name);
     68             Assert.Equal(VersionZoro, zoroResult.Version);
     69 
     70             // Name not normalized.
     71             var zoroResultNormalize = dscResourceMap.GetResource("XRESOURCEZORO", null, null);
     72             Assert.NotNull(zoroResultNormalize);
     73             Assert.Equal(ResourceZoro, zoroResultNormalize.Name);
     74             Assert.Equal(VersionZoro, zoroResultNormalize.Version);
     75 
     76             var yamatoResult = dscResourceMap.GetResource(ModuleOniResourceYamato, null, null);
     77             Assert.NotNull(yamatoResult);
     78             Assert.Equal(ModuleOniResourceYamato, yamatoResult.Name);
     79             Assert.Equal(ModuleOni, yamatoResult.ModuleName);
     80             Assert.Equal(VersionYamato, yamatoResult.Version);
     81 
     82             // Just by name and module get latest.
     83             var luffyResult5 = dscResourceMap.GetResource(ModuleMugiwaraResourceLuffy, ModuleMugiwara, null);
     84             Assert.NotNull(luffyResult5);
     85             Assert.Equal(ModuleMugiwaraResourceLuffy, luffyResult5.Name);
     86             Assert.Equal(ModuleMugiwara, luffyResult5.ModuleName);
     87             Assert.Equal(VersionLuffyGear5, luffyResult5.Version);
     88 
     89             // Specific version.
     90             var luffyResult4 = dscResourceMap.GetResource(ModuleMugiwaraResourceLuffy, ModuleMugiwara, VersionLuffyGear4);
     91             Assert.NotNull(luffyResult4);
     92             Assert.Equal(ModuleMugiwaraResourceLuffy, luffyResult4.Name);
     93             Assert.Equal(ModuleMugiwara, luffyResult4.ModuleName);
     94             Assert.Equal(VersionLuffyGear4, luffyResult4.Version);
     95 
     96             // Module name not normalized
     97             var luffyResult4Normalized = dscResourceMap.GetResource(ModuleMugiwaraResourceLuffy, "XMUGIWARAMODULE", VersionLuffyGear4);
     98             Assert.NotNull(luffyResult4Normalized);
     99             Assert.Equal(ModuleMugiwaraResourceLuffy, luffyResult4Normalized.Name);
    100             Assert.Equal(ModuleMugiwara, luffyResult4Normalized.ModuleName);
    101             Assert.Equal(VersionLuffyGear4, luffyResult4Normalized.Version);
    102         }

    104         /// <summary>
    105         /// Tests GetResource when resources are not in the maps.
    106         /// </summary>
    107         [Fact]
    108         public void DscResourcesMap_GetResource_NoResource()
    109         {
    110             var dscResourceMap = new DscResourcesMap(this.CreateDscResourceInfo());
    111 
    112             // Zoro is lost.
    113             var zoroResult = dscResourceMap.GetResource(ResourceZoro, ModuleMugiwara, null);
    114             Assert.Null(zoroResult);
    115 
    116             // Yamato didn't join (spoilers)
    117             var yamatoResult = dscResourceMap.GetResource(ModuleOniResourceYamato, ModuleMugiwara, VersionYamato);
    118             Assert.Null(yamatoResult);
    119 
    120             // Gear 6 is not a thing (?)
    121             var luffyResult = dscResourceMap.GetResource(ModuleMugiwaraResourceLuffy, ModuleMugiwara, Version.Parse("6.0.0.0"));
    122             Assert.Null(luffyResult);
    123 
    124             // Wrong universe.
    125             var gokuResult = dscResourceMap.GetResource("xResourceGoku", null, null);
    126             Assert.Null(gokuResult);
    127         }
