     78         {
     79             ConfigurationSet configurationSet = this.ConfigurationSet();
     80             ConfigurationUnit configurationUnitWorks = this.ConfigurationUnit();
     81             ConfigurationUnit configurationUnitThrows = this.ConfigurationUnit();
     82             configurationSet.Units = new ConfigurationUnit[] { configurationUnitWorks, configurationUnitThrows };
     83 
     84             TestConfigurationProcessorFactory factory = new TestConfigurationProcessorFactory();
     85             TestConfigurationSetProcessor setProcessor = factory.CreateTestProcessor(configurationSet);
     86             var thrownException = new InvalidDataException();
     87             setProcessor.Exceptions.Add(configurationUnitThrows, thrownException);
     88 
     89             ConfigurationProcessor processor = this.CreateConfigurationProcessorWithDiagnostics(factory);
     90 
     91             GetConfigurationSetDetailsResult result = processor.GetSetDetails(configurationSet, ConfigurationUnitDetailFlags.Local);
     92             var unitResults = result.UnitResults;
     93             Assert.Equal(2, unitResults.Count);
     94 
     95             Assert.Equal(configurationUnitWorks, unitResults[0].Unit);
     96             Assert.Null(unitResults[0].ResultInformation.ResultCode);
     97             Assert.NotNull(configurationUnitWorks.Details);
     98 
     99             Assert.Equal(configurationUnitThrows, unitResults[1].Unit);
    100             Assert.NotNull(unitResults[1].ResultInformation.ResultCode);
    101             Assert.Equal(thrownException.HResult, unitResults[1].ResultInformation.ResultCode.HResult);
    102             Assert.Null(configurationUnitThrows.Details);

    180         public void GetSettings_ProcessorSettingsSuccess()
    181         {
    182             ConfigurationUnit configurationUnit = this.ConfigurationUnit();
    183 
    184             TestConfigurationProcessorFactory factory = new TestConfigurationProcessorFactory();
    185             factory.NullProcessor = new TestConfigurationSetProcessor(null);
    186             TestConfigurationUnitProcessor unitProcessor = factory.NullProcessor.CreateTestProcessor(configurationUnit);
    187             GetSettingsResultInstance getSettingsResult = new GetSettingsResultInstance(configurationUnit);
    188             getSettingsResult.Settings = new Windows.Foundation.Collections.ValueSet();
    189             getSettingsResult.Settings.Add("key", "value");
    190             unitProcessor.GetSettingsDelegate = () => getSettingsResult;
    191 
    192             ConfigurationProcessor processor = this.CreateConfigurationProcessorWithDiagnostics(factory);
    193 
    194             GetConfigurationUnitSettingsResult result = processor.GetUnitSettings(configurationUnit);
    195 
    196             Assert.NotNull(result);
    197             Assert.NotNull(result.ResultInformation);
    198             Assert.Null(result.ResultInformation.ResultCode);
    199             Assert.Empty(result.ResultInformation.Description);
    200             Assert.NotNull(result.Settings);
    201             Assert.NotEmpty(result.Settings);
    202             Assert.Contains("key", result.Settings);
    203             Assert.IsType<string>(result.Settings["key"]);
    204             Assert.Equal("value", (string)result.Settings["key"]);
