    531         private static void AssertExistingPackageResourceData(PackageResourceData output, string version, bool ignoreLatest = false)
    532         {
    533             Assert.IsNotNull(output);
    534             Assert.True(output.Exist);
    535             Assert.AreEqual(DefaultPackageIdentifier, output.Identifier);
    536             Assert.AreEqual(version, output.Version);
    537 
    538             if (!ignoreLatest)
    539             {
    540                 if (version == DefaultPackageHighVersion)
    541                 {
    542                     Assert.True(output.UseLatest);
    543                 }
    544                 else
    545                 {
    546                     Assert.False(output.UseLatest);
    547                 }
    548             }
