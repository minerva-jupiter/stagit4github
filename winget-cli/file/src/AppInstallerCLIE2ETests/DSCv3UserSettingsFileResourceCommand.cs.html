    298     /// <summary>
    299     /// Gets the current user settings from the settings file.
    300     /// </summary>
    301     /// <returns>The current user settings as a JsonObject.</returns>
    302     private static JsonObject GetCurrentUserSettings()
    303     {
    304         var settingsContent = File.ReadAllText(WinGetSettingsHelper.GetUserSettingsPath());
    305         return JsonNode.Parse(settingsContent).AsObject();

    309     /// Adds or modifies mock properties in the settings.
    310     /// </summary>
    311     /// <param name="settings">Target settings.</param>
    312     /// <param name="value">The mock value.</param>
    313     private static void AddOrModifyMockProperties(JsonObject settings, string value)
    314     {
    315         settings[SettingsMock] = value;
    316         settings[SettingsMockObject] ??= new JsonObject();
    317         settings[SettingsMockObject][SettingsMockNested] = value;

    320     /// <summary>
    321     /// Asserts that the settings contain the expected mock properties.
    322     /// </summary>
    323     /// <param name="settings">Target settings.</param>
    324     /// <param name="value">The expected mock value.</param>
    325     private static void AssertMockProperties(JsonObject settings, string value)
    326     {
    327         Assert.IsNotNull(settings);
    328         Assert.IsTrue(settings.ContainsKey(SettingsMock));
    329         Assert.AreEqual(settings[SettingsMock].ToString(), value);
    330         Assert.IsTrue(settings.ContainsKey(SettingsMockObject));
    331         Assert.IsTrue(settings[SettingsMockObject].AsObject().ContainsKey(SettingsMockNested));
    332         Assert.AreEqual(settings[SettingsMockObject][SettingsMockNested].ToString(), value);
    333     }
