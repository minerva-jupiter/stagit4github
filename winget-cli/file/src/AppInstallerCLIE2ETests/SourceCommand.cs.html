     76         /// <summary>
     77         /// Test source add with explicit flag. Packages should only appear if the source is explicitly declared.
     78         /// </summary>
     79         [Test]
     80         public void SourceAddWithExplicit()
     81         {
     82             // Remove the test source.
     83             TestCommon.RunAICLICommand("source remove", Constants.TestSourceName);
     84 
     85             var result = TestCommon.RunAICLICommand("source add", $"SourceTest {Constants.TestSourceUrl} --trust-level trusted --explicit");
     86             Assert.AreEqual(Constants.ErrorCode.S_OK, result.ExitCode);
     87             Assert.True(result.StdOut.Contains("Done"));
     88 
     89             var searchResult = TestCommon.RunAICLICommand("search", "TestExampleInstaller");
     90             Assert.AreEqual(Constants.ErrorCode.ERROR_NO_SOURCES_DEFINED, searchResult.ExitCode);
     91             Assert.True(searchResult.StdOut.Contains("No sources defined; add one with 'source add' or reset to defaults with 'source reset'"));
     92 
     93             var searchResult2 = TestCommon.RunAICLICommand("search", "TestExampleInstaller --source SourceTest");
     94             Assert.AreEqual(Constants.ErrorCode.S_OK, searchResult2.ExitCode);
     95             Assert.True(searchResult2.StdOut.Contains("TestExampleInstaller"));
     96             Assert.True(searchResult2.StdOut.Contains("AppInstallerTest.TestExampleInstaller"));
     97             TestCommon.RunAICLICommand("source remove", $"-n SourceTest");
     98         }
