     35         {

     37             string packageId, commandAlias, fileName, packageDirName, productCode;
     38             packageId = "AppInstallerTest.TestPortableExe";

     40             commandAlias = fileName = "AppInstallerTestExeInstaller.exe";

     42             var result = TestCommon.RunAICLICommand("install", "AppInstallerTest.TestPortableExe -v 1.0.0.0");

     48             Assert.True(result2.StdOut.Contains("Successfully installed"));
     49             TestCommon.VerifyPortablePackage(Path.Combine(installDir, packageDirName), commandAlias, fileName, productCode, true);

     52         /// <summary>
     53         /// Test upgrade portable package with arp mismatch.
     54         /// </summary>
     55         [Test]
     56         public void UpgradePortableARPMismatch()
     57         {
     58             string packageId = "AppInstallerTest.TestPortableExe";
     59             string productCode = packageId + "_" + Constants.TestSourceIdentifier;
     60 
     61             var installResult = TestCommon.RunAICLICommand("install", "AppInstallerTest.TestPortableExe -v 1.0.0.0");
     62             Assert.AreEqual(Constants.ErrorCode.S_OK, installResult.ExitCode);
     63             Assert.True(installResult.StdOut.Contains("Successfully installed"));
     64 
     65             // Modify packageId to cause mismatch.
     66             TestCommon.ModifyPortableARPEntryValue(productCode, Constants.WinGetPackageIdentifier, "testPackageId");
     67 
     68             var upgradeResult = TestCommon.RunAICLICommand("upgrade", $"{packageId} -v 2.0.0.0");
     69 
     70             // Reset and perform uninstall cleanup
     71             TestCommon.ModifyPortableARPEntryValue(productCode, Constants.WinGetPackageIdentifier, packageId);
     72             TestCommon.RunAICLICommand("uninstall", $"--product-code {productCode}");
     73 
     74             Assert.AreNotEqual(Constants.ErrorCode.S_OK, upgradeResult.ExitCode);
     75             Assert.True(upgradeResult.StdOut.Contains("Portable package from a different source already exists"));
     76         }
