     69         /// <summary>
     70         /// Test to find a package and verify the CatalogPackageMetadata COM output.
     71         /// </summary>
     72         [Test]
     73         public void FindPackagesVerifyDefaultLocaleFields()
     74         {
     75             var searchResult = this.FindAllPackages(this.testSource, PackageMatchField.Id, PackageFieldMatchOption.Equals, "AppInstallerTest.CatalogPackageMetadata");
     76 
     77             Assert.AreEqual(1, searchResult.Count);
     78 
     79             var catalogPackage = searchResult[0].CatalogPackage;
     80             var packageVersionId = catalogPackage.AvailableVersions[0];
     81             var packageVersionInfo = catalogPackage.GetPackageVersionInfo(packageVersionId);
     82             var catalogPackageMetadata = packageVersionInfo.GetCatalogPackageMetadata();
     83 
     84             Assert.AreEqual("testAuthor", catalogPackageMetadata.Author);
     85             Assert.AreEqual("AppInstallerTest", catalogPackageMetadata.Publisher);
     86             Assert.AreEqual("https://testPublisherUrl.com", catalogPackageMetadata.PublisherUrl);
     87             Assert.AreEqual("https://testPublisherSupportUrl.com", catalogPackageMetadata.PublisherSupportUrl);
     88             Assert.AreEqual("https://testPrivacyUrl.com", catalogPackageMetadata.PrivacyUrl);
     89 
     90             Assert.AreEqual("https://testPackageUrl.com", catalogPackageMetadata.PackageUrl);
     91             Assert.AreEqual("testLicense", catalogPackageMetadata.License);
     92             Assert.AreEqual("https://testLicenseUrl.com", catalogPackageMetadata.LicenseUrl);
     93             Assert.AreEqual("testCopyright", catalogPackageMetadata.Copyright);
     94             Assert.AreEqual("https://testCopyrightUrl.com", catalogPackageMetadata.CopyrightUrl);
     95             Assert.AreEqual("testDescription", catalogPackageMetadata.Description);
     96             Assert.AreEqual("testShortDescription", catalogPackageMetadata.ShortDescription);
     97             Assert.AreEqual("https://testPurchaseUrl.com", catalogPackageMetadata.PurchaseUrl);
     98 
     99             var tags = catalogPackageMetadata.Tags;
    100             Assert.AreEqual(2, tags.Count);
    101             Assert.AreEqual("tag1", tags[0]);
    102             Assert.AreEqual("tag2", tags[1]);
    103             Assert.AreEqual("testReleaseNotes", catalogPackageMetadata.ReleaseNotes);
    104             Assert.AreEqual("https://testReleaseNotes.net", catalogPackageMetadata.ReleaseNotesUrl);
    105             Assert.AreEqual("testInstallationNotes", catalogPackageMetadata.InstallationNotes);
    106 
    107             var packageAgreements = catalogPackageMetadata.Agreements;
    108             Assert.AreEqual(1, packageAgreements.Count);
    109 
    110             var agreement = packageAgreements[0];
    111             Assert.AreEqual("testAgreementLabel", agreement.Label);
    112             Assert.AreEqual("testAgreementText", agreement.Text);
    113             Assert.AreEqual("https://testAgreementUrl.net", agreement.Url);
    114 
    115             var documentations = catalogPackageMetadata.Documentations;
    116             Assert.AreEqual(1, documentations.Count);
    117 
    118             var documentation = documentations[0];
    119             Assert.AreEqual("testDocumentLabel", documentation.DocumentLabel);
    120             Assert.AreEqual("https://testDocumentUrl.com", documentation.DocumentUrl);
    121 
    122             var icons = catalogPackageMetadata.Icons;
    123             Assert.AreEqual(1, icons.Count);
    124 
    125             var icon = icons[0];
    126             Assert.AreEqual("https://testIcon", icon.Url);
    127             Assert.AreEqual(IconFileType.Ico, icon.FileType);
    128             Assert.AreEqual(IconTheme.Default, icon.Theme);
    129             Assert.AreEqual(IconResolution.Custom, icon.Resolution);
    130             Assert.AreEqual(Convert.FromHexString("69D84CA8899800A5575CE31798293CD4FEBAB1D734A07C2E51E56A28E0DF8123"), icon.Sha256);
    131         }
