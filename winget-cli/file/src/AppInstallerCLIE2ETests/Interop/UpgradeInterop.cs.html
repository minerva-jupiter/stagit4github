     65             string productCode = Constants.PortableExePackageDirName;

     69             // Find package

     71 
     72             // Configure install options

    299         // Cannot use foreach or Linq for out-of-process IVector
    300         // Bug: https://github.com/microsoft/CsWinRT/issues/1205
    301         private static T First<T>(IReadOnlyList<T> list, Func<T, bool> condition)
    302         {
    303             if (list == null || condition == null)
    304             {
    305                 throw new ArgumentNullException();
    306             }
    307 
    308             for (int i = 0; i < list.Count; ++i)
    309             {
    310                 var item = list[i];
    311                 if (condition(item))
    312                 {
    313                     return item;
    314                 }
    315             }
    316 
    317             throw new InvalidOperationException("No element satisfies the condition");
    318         }
