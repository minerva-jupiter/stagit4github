     57         /// <summary>
     58         /// Test Repair MSI Installer.
     59         /// </summary>
     60         /// <returns>A <see cref="Task"/> representing the asynchronous unit test. </returns>
     61         [Test]
     62         public async Task RepairMSIInstaller()
     63         {
     64             string msiInstallerPackageId = "AppInstallerTest.TestMsiRepair";
     65             var searchResult = await this.FindAndInstallPackage(msiInstallerPackageId, this.installDir, null);
     66 
     67             // Note: The 'msiexec repair' command requires the original installer file to be present at the location registered in the ARP (Add/Remove Programs).
     68             // In our test scenario, the MSI installer file is initially placed in a temporary location and then deleted, which can cause the repair operation to fail.
     69             // To work around this, we copy the installer file to the ARP source directory before running the repair command.
     70             // A more permanent solution would be to modify the MSI installer to cache the installer file in a known location and register that location as the installer source.
     71             // This would allow the 'msiexec repair' command to function as expected.
     72             string installerSourceDir = TestCommon.CopyInstallerFileToARPInstallSourceDirectory(TestCommon.GetTestDataFile("AppInstallerTestMsiInstallerV2.msi"), Constants.MsiInstallerProductCode, true);
     73 
     74             // Repair the package
     75             await this.RepairPackageAndValidateStatus(searchResult.CatalogPackage, this.TestFactory.CreateRepairOptions(), RepairResultStatus.Ok);
     76 
     77             // Cleanup
     78             Assert.True(TestCommon.VerifyTestMsiInstalledAndCleanup(this.installDir));
     79 
     80             if (installerSourceDir != null && Directory.Exists(installerSourceDir))
     81             {
     82                 Directory.Delete(installerSourceDir, true);
     83             }
     84         }

    289         private async Task<MatchResult> FindAndInstallPackage(string packageId, string installDir, string replacementInstallerArguments)
    290         {
    291             // Find a package
    292             var searchResult = this.FindOnePackage(this.compositeSource, PackageMatchField.Id, PackageFieldMatchOption.Equals, packageId);
    293 
    294             // Configure installation
    295             var installOptions = this.TestFactory.CreateInstallOptions();
    296             installOptions.PackageInstallMode = PackageInstallMode.Silent;
    297 
    298             if (!string.IsNullOrEmpty(installDir))
    299             {
    300                 installOptions.PreferredInstallLocation = installDir;
    301             }
    302 
    303             if (!string.IsNullOrEmpty(replacementInstallerArguments))
    304             {
    305                 installOptions.ReplacementInstallerArguments = replacementInstallerArguments;
    306             }
    307 
    308             // Install the package
    309             var installResult = await this.packageManager.InstallPackageAsync(searchResult.CatalogPackage, installOptions);
    310             Assert.AreEqual(InstallResultStatus.Ok, installResult.Status);
    311 
    312             // Find package again, but this time it should be detected as installed
    313             searchResult = this.FindOnePackage(this.compositeSource, PackageMatchField.Id, PackageFieldMatchOption.Equals, packageId);
    314             Assert.NotNull(searchResult.CatalogPackage.InstalledVersion);
    315 
    316             // Return the search result
    317             return searchResult;
    318         }

    320         private async Task RepairPackageAndValidateStatus(CatalogPackage package, RepairOptions repairOptions, RepairResultStatus expectedStatus, int expectedErrorCode = 0)
    321         {
    322             var repairResult = await this.packageManager.RepairPackageAsync(package, repairOptions);
    323             Assert.AreEqual(expectedStatus, repairResult.Status);
    324 
    325             if (expectedStatus != RepairResultStatus.Ok)
    326             {
    327                 Assert.AreEqual(expectedErrorCode, repairResult.ExtendedErrorCode.HResult);
    328             }
    329         }

    331         private string GetReplacementArguments(string installDir, string version, string displayName, bool useHKLM = false, bool noModify = false, bool noRepair = false)
    332         {
    333             var replacementArguments = new StringBuilder($"/InstallDir {installDir} /Version {version} /DisplayName {displayName}");
    334 
    335             // Machine scope install.
    336             if (useHKLM)
    337             {
    338                 replacementArguments.Append($" /UseHKLM");
    339             }
    340 
    341             // Instructs test installer to set NoModify ARP flag.
    342             if (noModify)
    343             {
    344                 replacementArguments.Append($" /NoModify");
    345             }
    346 
    347             // Instructs test installer to set NoRepair ARP flag.
    348             if (noRepair)
    349             {
    350                 replacementArguments.Append($" /NoRepair");
    351             }
    352 
    353             return replacementArguments.ToString();
    354         }
