     65         /// <summary>
     66         /// Simple test to confirm that a resource that is already locally available shows that way.
     67         /// </summary>
     68         /// <param name="location">The location the module should be before running.</param>
     69         [TestCase(TestCommon.TestModuleLocation.CurrentUser)]
     70         [TestCase(TestCommon.TestModuleLocation.AllUsers)]
     71         [TestCase(TestCommon.TestModuleLocation.WinGetModulePath)]
     72         [TestCase(TestCommon.TestModuleLocation.Custom)]
     73         public void ShowDetailsFromLocal(TestCommon.TestModuleLocation location)
     74         {
     75             TestCommon.EnsureModuleState(Constants.SimpleTestModuleName, present: true, repository: Constants.TestRepoName, location: location);
     76 
     77             string args = $"{TestCommon.GetTestDataFile("Configuration\\ShowDetails_TestRepo.yml")} --verbose";
     78             if (location == TestCommon.TestModuleLocation.Custom)
     79             {
     80                 args += " --module-path " + TestCommon.GetExpectedModulePath(location);
     81             }
     82 
     83             var result = TestCommon.RunAICLICommand("configure show", args);
     84             Assert.AreEqual(0, result.ExitCode);
     85             Assert.True(result.StdOut.Contains(Constants.LocalModuleDescriptor));
     86         }

    160         /// <summary>
    161         /// Simple test to confirm that a resource is discoverable with DSC v3.
    162         /// </summary>
    163         [Test]
    164         public void ShowDetails_DSCv3()
    165         {
    166             var result = TestCommon.RunAICLICommand("configure show", $"{TestCommon.GetTestDataFile("Configuration\\ShowDetails_DSCv3.yml")} --verbose");
    167             Assert.AreEqual(0, result.ExitCode);
    168 
    169             var outputLines = result.StdOut.Split('\n');
    170             int startLine = -1;
    171             for (int i = 0; i < outputLines.Length; ++i)
    172             {
    173                 if (outputLines[i].Trim() == "Microsoft.WinGet.Dev/TestFile [Test File]")
    174                 {
    175                     startLine = i;
    176                 }
    177             }
    178 
    179             Assert.AreNotEqual(-1, startLine);
    180             Assert.LessOrEqual(3, outputLines.Length - startLine);
    181 
    182             // outputLines[1] should contain the discovered resource string if working properly.
    183             Assert.AreEqual("Description 1.", outputLines[startLine + 2].Trim());
    184         }
