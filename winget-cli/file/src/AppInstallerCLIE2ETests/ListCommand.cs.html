     30         /// <summary>
     31         /// Test list after installing a package.
     32         /// </summary>
     33         [Test]
     34         public void ListAfterInstall()
     35         {
     36             System.Guid guid = System.Guid.NewGuid();
     37             string productCode = guid.ToString();
     38             var installDir = TestCommon.GetRandomTestDir();
     39 
     40             // DisplayName must be set to avoid conflicts with other packages that use the same exe installer.
     41             string displayName = "TestExeInstaller";
     42             string localAppDataPath = System.Environment.GetEnvironmentVariable(Constants.LocalAppData);
     43             string logFilePath = System.IO.Path.Combine(localAppDataPath, Constants.E2ETestLogsPathPackaged);
     44             logFilePath = System.IO.Path.Combine(logFilePath, "ListAfterInstall-" + System.IO.Path.GetRandomFileName() + ".log");
     45 
     46             var result = TestCommon.RunAICLICommand("list", productCode);
     47             Assert.AreEqual(Constants.ErrorCode.ERROR_NO_APPLICATIONS_FOUND, result.ExitCode);
     48 
     49             result = TestCommon.RunAICLICommand("install", $"AppInstallerTest.TestExeInstaller --override \"/InstallDir {installDir} /ProductID {productCode} /LogFile {logFilePath} /DisplayName {displayName}\"");
     50             Assert.AreEqual(Constants.ErrorCode.S_OK, result.ExitCode);
     51 
     52             result = TestCommon.RunAICLICommand("list", productCode);
     53             Assert.AreEqual(Constants.ErrorCode.S_OK, result.ExitCode);
     54             Assert.True(result.StdOut.Contains("AppInstallerTest.TestExeInstaller"));
     55             Assert.True(result.StdOut.Contains("1.0.0.0"));
     56             Assert.True(result.StdOut.Contains("2.0.0.0"));
     57         }

    230         private void ArpVersionMappingTest(string packageIdentifier, string displayNameOverride, string displayVersionOverride, string expectedListVersion, string notExpectedListVersion = "")
    231         {
    232             System.Guid guid = System.Guid.NewGuid();
    233             string productCode = guid.ToString();
    234             var installDir = TestCommon.GetRandomTestDir();
    235 
    236             var result = TestCommon.RunAICLICommand("list", productCode);
    237             Assert.AreEqual(Constants.ErrorCode.ERROR_NO_APPLICATIONS_FOUND, result.ExitCode);
    238 
    239             result = TestCommon.RunAICLICommand("install", $"{packageIdentifier} --override \"/InstallDir {installDir} /ProductID {productCode} /DisplayName {displayNameOverride} /Version {displayVersionOverride}\"");
    240             Assert.AreEqual(Constants.ErrorCode.S_OK, result.ExitCode);
    241 
    242             result = TestCommon.RunAICLICommand("list", productCode);
    243             Assert.AreEqual(Constants.ErrorCode.S_OK, result.ExitCode);
    244 
    245             Assert.True(result.StdOut.Contains(packageIdentifier));
    246             Assert.True(result.StdOut.Contains(expectedListVersion));
    247             if (!string.IsNullOrEmpty(notExpectedListVersion))
    248             {
    249                 Assert.False(result.StdOut.Contains(notExpectedListVersion));
    250             }
    251 
    252             // Try clean up
    253             if (File.Exists(Path.Combine(installDir, Constants.TestExeInstalledFileName)))
    254             {
    255                 TestCommon.RunCommand(Path.Combine(installDir, Constants.TestExeUninstallerFileName));
    256             }
    257         }
