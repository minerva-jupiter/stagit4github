    158         /// Calls `test` on the `source` resource with a argument that matches.
    159         /// </summary>
    160         /// <param name="useDefaultArgument">The argument to use when adding the existing source.</param>

    170             var setup = TestCommon.RunAICLICommand("source add", $"--name {DefaultSourceName} --arg {(useDefaultArgument ? DefaultSourceArgForCmdLine : NonDefaultSourceArgForCmdLine)} --type {DefaultSourceType} --trust-level {trustLevel} {(isExplicit ? "--explicit" : string.Empty)}");

    179                     break;
    180                 case TypePropertyName:
    181                     resourceData.Type = DefaultSourceType;
    182                     break;
    183                 case TrustLevelPropertyName:
    184                     resourceData.TrustLevel = trustLevel;
    185                     break;
    186                 case ExplicitPropertyName:

    454         private static string CreateSourceArgument(bool forCommandLine = false, int openHR = 0, int searchHR = 0)
    455         {
    456             const string CommandLineFormat = @"""{{""""OpenHR"""": {0}, """"SearchHR"""": {1} }}""";
    457             const string DirectFormat = @"{{""OpenHR"": {0}, ""SearchHR"": {1} }}";
