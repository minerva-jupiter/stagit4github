     41         // Only used in preview manifest
     42         std::vector<Manifest::string_t> SplitMultiValueField(const std::string& input)
     43         {
     44             if (input.empty())
     45             {
     46                 return {};
     47             }
     48 
     49             std::vector<Manifest::string_t> result;
     50             size_t currentPos = 0;
     51             while (currentPos < input.size())
     52             {
     53                 size_t splitPos = input.find(',', currentPos);
     54                 if (splitPos == std::string::npos)
     55                 {
     56                     splitPos = input.size();
     57                 }
     58 
     59                 std::string splitVal = input.substr(currentPos, splitPos - currentPos);
     60                 Utility::Trim(splitVal);
     61                 if (!splitVal.empty())
     62                 {
     63                     result.emplace_back(std::move(splitVal));
     64                 }
     65                 currentPos = splitPos + 1;
     66             }
     67 
     68             return result;
     69         }

     71         std::vector<Manifest::string_t> ProcessStringSequenceNode(const YAML::Node& node, bool trim = true)
     72         {
     73             THROW_HR_IF(E_INVALIDARG, !node.IsSequence());
     74 
     75             std::vector<Manifest::string_t> result;
     76 
     77             for (auto const& entry : node.Sequence())
     78             {
     79                 std::string value = entry.as<std::string>();
     80                 if (trim)
     81                 {
     82                     Utility::Trim(value);
     83                 }
     84 
     85                 result.emplace_back(std::move(value));
     86             }
     87 
     88             return result;

    811     ValidationErrors ManifestYamlPopulator::ValidateAndProcessFields(
    812         const YAML::Node& rootNode,
    813         const std::vector<FieldProcessInfo>& fieldInfos,
    814         const VariantManifestPtr& v)
    815     {
    816         ValidationErrors resultErrors;
    817 
    818         if (!rootNode.IsMap() || rootNode.size() == 0)
    819         {
    820             resultErrors.emplace_back(ManifestError::InvalidRootNode, "", "", m_isMergedManifest ? 0 : rootNode.Mark().line, m_isMergedManifest ? 0 : rootNode.Mark().column);
    821             return resultErrors;
    822         }
    823 
    824         // Keeps track of already processed fields. Used to check duplicate fields.
    825         std::set<std::string> processedFields;
    826 
    827         for (auto const& keyValuePair : rootNode.Mapping())
    828         {
    829             std::string key = keyValuePair.first.as<std::string>();
    830             const YAML::Node& valueNode = keyValuePair.second;
    831 
    832             // We'll do case-insensitive search first and validate correct case later.
    833             auto fieldIter = std::find_if(fieldInfos.begin(), fieldInfos.end(),
    834                 [&](auto const& s)
    835                 {
    836                     return Utility::CaseInsensitiveEquals(s.Name, key);
    837                 });
    838 
    839             if (fieldIter != fieldInfos.end())
    840             {
    841                 const FieldProcessInfo& fieldInfo = *fieldIter;
    842 
    843                 // Make sure the found key is in Pascal Case
    844                 if (key != fieldInfo.Name)
    845                 {
    846                     resultErrors.emplace_back(ManifestError::FieldIsNotPascalCase, key, "", m_isMergedManifest ? 0 : keyValuePair.first.Mark().line, m_isMergedManifest ? 0 : keyValuePair.first.Mark().column);
    847                 }
    848 
    849                 // Make sure it's not a duplicate key
    850                 if (!processedFields.insert(fieldInfo.Name).second)
    851                 {
    852                     resultErrors.emplace_back(ManifestError::FieldDuplicate, fieldInfo.Name, "", m_isMergedManifest ? 0 : keyValuePair.first.Mark().line, m_isMergedManifest ? 0 : keyValuePair.first.Mark().column);
    853                 }
    854 
    855                 if (fieldInfo.RequireVerifiedPublisher)
    856                 {
    857                     resultErrors.emplace_back(ManifestError::FieldRequireVerifiedPublisher, fieldInfo.Name, "",
    858                         m_isMergedManifest ? 0 : keyValuePair.first.Mark().line, m_isMergedManifest ? 0 : keyValuePair.first.Mark().column,
    859                         m_validateOption.ErrorOnVerifiedPublisherFields ? ValidationError::Level::Error : ValidationError::Level::Warning);
    860                 }
    861 
    862                 if (!valueNode.IsNull())
    863                 {
    864                     try
    865                     {
    866                         auto errors = fieldInfo.ProcessFunc(valueNode, v);
    867                         std::move(errors.begin(), errors.end(), std::inserter(resultErrors, resultErrors.end()));
    868                     }
    869                     catch (const std::exception&)
    870                     {
    871                         resultErrors.emplace_back(ManifestError::FieldFailedToProcess, fieldInfo.Name);
    872                     }
    873                 }
    874             }
    875             else
    876             {
    877                 // For full validation, also reports unrecognized fields as warning
    878                 if (m_validateOption.FullValidation)
    879                 {
    880                     resultErrors.emplace_back(ManifestError::FieldUnknown, key, "", m_isMergedManifest ? 0 : keyValuePair.first.Mark().line, m_isMergedManifest ? 0 : keyValuePair.first.Mark().column, ValidationError::Level::Warning);
    881                 }
    882             }
    883         }
    884 
    885         return resultErrors;
    886     }

   1094     ValidationErrors ManifestYamlPopulator::PopulateManifestInternal()
   1095     {
   1096         const YAML::Node& rootNode = m_rootNode;
   1097         ValidationErrors resultErrors;
   1098 
   1099         // Prepare field infos
   1100         RootFieldInfos = GetRootFieldProcessInfo();
   1101         InstallerFieldInfos = GetInstallerFieldProcessInfo();
   1102         SwitchesFieldInfos = GetSwitchesFieldProcessInfo();
   1103         ExpectedReturnCodesFieldInfos = GetExpectedReturnCodesFieldProcessInfo();
   1104         DependenciesFieldInfos = GetDependenciesFieldProcessInfo();
   1105         PackageDependenciesFieldInfos = GetPackageDependenciesFieldProcessInfo();
   1106         LocalizationFieldInfos = GetLocalizationFieldProcessInfo();
   1107         AgreementFieldInfos = GetAgreementFieldProcessInfo();
   1108         MarketsFieldInfos = GetMarketsFieldProcessInfo();
   1109         AppsAndFeaturesEntryFieldInfos = GetAppsAndFeaturesEntryFieldProcessInfo();
   1110         DocumentationFieldInfos = GetDocumentationFieldProcessInfo();
   1111         IconFieldInfos = GetIconFieldProcessInfo();
   1112         NestedInstallerFileFieldInfos = GetNestedInstallerFileFieldProcessInfo();
   1113         InstallationMetadataFieldInfos = GetInstallationMetadataFieldProcessInfo();
   1114         InstallationMetadataFilesFieldInfos = GetInstallationMetadataFilesFieldProcessInfo();
   1115         AuthenticationFieldInfos = GetAuthenticationFieldInfos();
   1116         MicrosoftEntraIdAuthenticationInfoFieldInfos = GetMicrosoftEntraIdAuthenticationInfoFieldInfos();
   1117 
   1118         resultErrors = ValidateAndProcessFields(rootNode, RootFieldInfos, VariantManifestPtr(&(m_manifest.get())));
   1119 
   1120         if (!m_p_installersNode)
   1121         {
   1122             return resultErrors;
   1123         }
   1124 
   1125         // Populate installers
   1126         for (auto const& entry : m_p_installersNode->Sequence())
   1127         {
   1128             ManifestInstaller installer = m_manifest.get().DefaultInstallerInfo;
   1129 
   1130             // Clear these defaults as PackageFamilyName, ProductCode, AppsAndFeaturesEntries need to be copied based on InstallerType
   1131             installer.PackageFamilyName.clear();
   1132             installer.ProductCode.clear();
   1133             installer.AppsAndFeaturesEntries.clear();
   1134             // Clear dependencies as installer overrides root dependencies
   1135             installer.Dependencies.Clear();
   1136             // Clear nested installers as it should only be copied for zip installerType.
   1137             installer.NestedInstallerType = InstallerTypeEnum::Unknown;
   1138             installer.NestedInstallerFiles.clear();
   1139 
   1140             auto errors = ValidateAndProcessFields(entry, InstallerFieldInfos, VariantManifestPtr(&installer));
   1141             std::move(errors.begin(), errors.end(), std::inserter(resultErrors, resultErrors.end()));
   1142 
   1143             // Copy in system reference strings from the root if not set in the installer and appropriate
   1144             if (installer.PackageFamilyName.empty() && DoesInstallerTypeUsePackageFamilyName(installer.EffectiveInstallerType()))
   1145             {
   1146                 installer.PackageFamilyName = m_manifest.get().DefaultInstallerInfo.PackageFamilyName;
   1147             }
   1148 
   1149             if (installer.ProductCode.empty() && DoesInstallerTypeUseProductCode(installer.EffectiveInstallerType()))
   1150             {
   1151                 installer.ProductCode = m_manifest.get().DefaultInstallerInfo.ProductCode;
   1152             }
   1153 
   1154             if (installer.AppsAndFeaturesEntries.empty() && DoesInstallerTypeWriteAppsAndFeaturesEntry(installer.EffectiveInstallerType()))
   1155             {
   1156                 installer.AppsAndFeaturesEntries = m_manifest.get().DefaultInstallerInfo.AppsAndFeaturesEntries;
   1157             }
   1158 
   1159             if (IsArchiveType(installer.BaseInstallerType))
   1160             {
   1161                 if (installer.NestedInstallerFiles.empty())
   1162                 {
   1163                     installer.NestedInstallerFiles = m_manifest.get().DefaultInstallerInfo.NestedInstallerFiles;
   1164                 }
   1165 
   1166                 if (installer.NestedInstallerType == InstallerTypeEnum::Unknown)
   1167                 {
   1168                     installer.NestedInstallerType = m_manifest.get().DefaultInstallerInfo.NestedInstallerType;
   1169                 }
   1170             }
   1171 
   1172             // If there are no dependencies on installer use default ones
   1173             if (!installer.Dependencies.HasAny())
   1174             {
   1175                 installer.Dependencies = m_manifest.get().DefaultInstallerInfo.Dependencies;
   1176             }
   1177 
   1178             // Populate installer default switches if not exists
   1179             auto defaultSwitches = GetDefaultKnownSwitches(installer.EffectiveInstallerType());
   1180             for (auto const& defaultSwitch : defaultSwitches)
   1181             {
   1182                 if (installer.Switches.find(defaultSwitch.first) == installer.Switches.end())
   1183                 {
   1184                     installer.Switches[defaultSwitch.first] = defaultSwitch.second;
   1185                 }
   1186             }
   1187 
   1188             // Populate installer default return codes if not present in ExpectedReturnCodes and InstallerSuccessCodes
   1189             auto defaultReturnCodes = GetDefaultKnownReturnCodes(installer.EffectiveInstallerType());
   1190             for (auto const& defaultReturnCode : defaultReturnCodes)
   1191             {
   1192                 if (installer.ExpectedReturnCodes.find(defaultReturnCode.first) == installer.ExpectedReturnCodes.end() &&
   1193                     std::find(installer.InstallerSuccessCodes.begin(), installer.InstallerSuccessCodes.end(), defaultReturnCode.first) == installer.InstallerSuccessCodes.end())
   1194                 {
   1195                     installer.ExpectedReturnCodes[defaultReturnCode.first].ReturnResponseEnum = defaultReturnCode.second;
   1196                 }
   1197             }
   1198 
   1199             m_manifest.get().Installers.emplace_back(std::move(installer));
   1200         }
   1201 
   1202         // Populate additional localizations
   1203         if (m_p_localizationsNode && m_p_localizationsNode->IsSequence())
   1204         {
   1205             for (auto const& entry : m_p_localizationsNode->Sequence())
   1206             {
   1207                 ManifestLocalization localization;
   1208                 auto errors = ValidateAndProcessFields(entry, LocalizationFieldInfos, VariantManifestPtr(&localization));
   1209                 std::move(errors.begin(), errors.end(), std::inserter(resultErrors, resultErrors.end()));
   1210                 m_manifest.get().Localizations.emplace_back(std::move(std::move(localization)));
   1211             }
   1212         }
   1213 
   1214         return resultErrors;
   1215     }

   1217     ValidationErrors ManifestYamlPopulator::InsertShadow(const YAML::Node& shadowNode)
   1218     {
   1219         Manifest shadowManifest;
   1220 
   1221         // Process shadow node.
   1222         auto resultErrors = ValidateAndProcessFields(shadowNode, GetShadowRootFieldProcessInfo(), VariantManifestPtr(&shadowManifest));
   1223 
   1224         // Merge.
   1225         if (m_manifestVersion.get() >= ManifestVer{ s_ManifestVersionV1_5 })
   1226         {
   1227             // Default localization
   1228             if (Utility::ICUCaseInsensitiveEquals(m_manifest.get().DefaultLocalization.Locale, shadowManifest.DefaultLocalization.Locale))
   1229             {
   1230                 // Icons
   1231                 if (!m_manifest.get().DefaultLocalization.Contains(Localization::Icons) &&
   1232                     shadowManifest.DefaultLocalization.Contains(Localization::Icons))
   1233                 {
   1234                     m_manifest.get().DefaultLocalization.Add<Localization::Icons>(std::move(shadowManifest.DefaultLocalization.Get<Localization::Icons>()));
   1235 
   1236                     YAML::Node key{ YAML::Node::Type::Scalar, "", YAML::Mark() };
   1237                     key.SetScalar("Icons");
   1238                     YAML::Node value = shadowNode.GetChildNode("Icons");
   1239                     m_rootNode.get().AddMappingNode(std::move(key), std::move(value));
   1240                 }
   1241             }
   1242 
   1243             // Localizations
   1244             if (!shadowManifest.Localizations.empty())
   1245             {
   1246                 // Merge manifest object
   1247                 for (auto const& shadowLocalization : shadowManifest.Localizations)
   1248                 {
   1249                     // Manifest
   1250                     if (auto iter = std::find_if(m_manifest.get().Localizations.begin(), m_manifest.get().Localizations.end(), [&](auto const& l) { return Utility::ICUCaseInsensitiveEquals(l.Locale, shadowLocalization.Locale); }); iter != m_manifest.get().Localizations.end())
   1251                     {
   1252                         if (!(*iter).Contains(Localization::Icons) &&
   1253                             shadowLocalization.Contains(Localization::Icons))
   1254                         {
   1255                             (*iter).Add<Localization::Icons>(std::move(shadowLocalization.Get<Localization::Icons>()));
   1256                         }
   1257                     }
   1258                     else
   1259                     {
   1260                         ManifestLocalization localization = shadowLocalization;
   1261                         m_manifest.get().Localizations.emplace_back(std::move(std::move(localization)));
   1262                     }
   1263                 }
   1264 
   1265                 // Merge yaml
   1266                 auto shadowLocalizationsNode = shadowNode.GetChildNode("Localization");
   1267                 if (m_p_localizationsNode)
   1268                 {
   1269                     m_rootNode.get().GetChildNode("Localization").MergeSequenceNode(shadowLocalizationsNode, "PackageLocale", true);
   1270                 }
   1271                 else
   1272                 {
   1273                     YAML::Node key{ YAML::Node::Type::Scalar, "", YAML::Mark() };
   1274                     key.SetScalar("Localization");
   1275                     m_rootNode.get().AddMappingNode(std::move(key), std::move(shadowLocalizationsNode));
   1276                 }
   1277             }
   1278         }
   1279 
   1280         return resultErrors;
   1281     }

   1283     ValidationErrors ManifestYamlPopulator::PopulateManifest(
   1284         YAML::Node& rootNode,
   1285         Manifest& manifest,
   1286         const ManifestVer& manifestVersion,
   1287         ManifestValidateOption validateOption,
   1288         const std::optional<YAML::Node>& shadowNode)
