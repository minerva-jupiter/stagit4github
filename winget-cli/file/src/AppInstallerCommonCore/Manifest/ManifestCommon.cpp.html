     40     ManifestVer::ManifestVer(std::string_view version)
     41     {
     42         bool validationSuccess = true;
     43 
     44         // Separate the extensions out
     45         size_t hyphenPos = version.find_first_of('-');
     46         if (hyphenPos != std::string_view::npos)
     47         {
     48             // The first part is the main version
     49             Assign(std::string{ version.substr(0, hyphenPos) }, ".");
     50 
     51             // The second part is the extensions
     52             hyphenPos += 1;
     53             while (hyphenPos < version.length())
     54             {
     55                 size_t newPos = version.find_first_of('-', hyphenPos);
     56 
     57                 size_t length = (newPos == std::string::npos ? version.length() : newPos) - hyphenPos;
     58                 m_extensions.emplace_back(std::string{ version.substr(hyphenPos, length) }, ".");
     59 
     60                 hyphenPos += length + 1;
     61             }
     62         }
     63         else
     64         {
     65             Assign(std::string{ version }, ".");
     66         }
     67 
     68         if (m_parts.size() > 3)
     69         {
     70             validationSuccess = false;
     71         }
     72         else
     73         {
     74             for (size_t i = 0; i < m_parts.size(); i++)
     75             {
     76                 if (!m_parts[i].Other.empty())
     77                 {
     78                     validationSuccess = false;
     79                     break;
     80                 }
     81             }
     82 
     83             for (const RawVersion& ext : m_extensions)
     84             {
     85                 if (ext.GetParts().empty() || ext.GetParts()[0].Integer != 0)
     86                 {
     87                     validationSuccess = false;
     88                     break;
     89                 }
     90             }
     91         }
     92 
     93         if (!validationSuccess)
     94         {
     95             std::vector<ValidationError> errors;
     96             errors.emplace_back(ManifestError::InvalidFieldValue, "ManifestVersion", std::string{ version });
     97             THROW_EXCEPTION(ManifestException(std::move(errors)));
     98         }
     99     }

    106     bool ManifestVer::HasExtension(std::string_view extension) const
    107     {
    108         for (const RawVersion& ext : m_extensions)
    109         {
    110             const auto& parts = ext.GetParts();
    111             if (!parts.empty() && parts[0].Integer == 0 && parts[0].Other == extension)
    112             {
    113                 return true;
    114             }
    115         }
    116 
    117         return false;
    118     }

    976     bool IsInstallerTypeCompatible(InstallerTypeEnum type1, InstallerTypeEnum type2)
    977     {
    978         // Unknown type cannot be compatible with any other
    979         if (type1 == InstallerTypeEnum::Unknown || type2 == InstallerTypeEnum::Unknown)
    980         {
    981             return false;
    982         }
    983 
    984         // Not unknown, so must be compatible
    985         if (type1 == type2)
    986         {
    987             return true;
    988         }
    989 
    990         CompatibilitySet set1 = GetCompatibilitySet(type1);
    991         CompatibilitySet set2 = GetCompatibilitySet(type2);
    992 
    993         // If either is none, they can't be compatible
    994         if (set1 == CompatibilitySet::None || set2 == CompatibilitySet::None)
    995         {
    996             return false;
    997         }
    998 
    999         return set1 == set2;
   1000     }

   1119     void DependencyList::Add(const Dependency& newDependency)
   1120     {
   1121         Dependency* existingDependency = this->HasDependency(newDependency);
   1122 
   1123         if (existingDependency != NULL)
   1124         {
   1125             if (newDependency.MinVersion > existingDependency->MinVersion)
   1126             {
   1127                 existingDependency->MinVersion = newDependency.MinVersion;
   1128             }
   1129         }
   1130         else
   1131         {
   1132             m_dependencies.push_back(newDependency);
   1133         }
   1134     }
