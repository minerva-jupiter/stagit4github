     38     std::optional<OSVersion> MsixPackageManifest::GetMinimumOSVersionForSupportedPlatforms() const
     39     {
     40         std::optional<OSVersion> minOSVersion;
     41         auto targetDeviceFamilies = GetTargetDeviceFamilies();
     42 
     43         for (const auto& targetDeviceFamily : targetDeviceFamilies)
     44         {
     45             if (IsSupportedPlatform(targetDeviceFamily.GetPlatform()))
     46             {
     47                 auto targetDeviceFamilyMinVersion = targetDeviceFamily.GetMinVersion();
     48                 if (!minOSVersion.has_value() || targetDeviceFamilyMinVersion < minOSVersion.value())
     49                 {
     50                     minOSVersion = targetDeviceFamilyMinVersion;
     51                 }
     52             }
     53         }
     54 
     55         return minOSVersion;
     56     }

     58     std::vector<MsixPackageManifestTargetDeviceFamily> MsixPackageManifest::GetTargetDeviceFamilies() const
     59     {
     60         std::vector<MsixPackageManifestTargetDeviceFamily> targetDeviceFamilies;
     61         ComPtr<IAppxManifestReader3> manifestReader3;
     62         THROW_IF_FAILED(m_manifestReader.As(&manifestReader3));
     63 
     64         ComPtr<IAppxManifestTargetDeviceFamiliesEnumerator> targetDeviceFamiliesIter;
     65         THROW_IF_FAILED(manifestReader3->GetTargetDeviceFamilies(&targetDeviceFamiliesIter));
     66 
     67         BOOL hasCurrent = FALSE;
     68         THROW_IF_FAILED(targetDeviceFamiliesIter->GetHasCurrent(&hasCurrent));
     69         while (hasCurrent)
     70         {
     71             ComPtr<IAppxManifestTargetDeviceFamily> targetDeviceFamily;
     72             THROW_IF_FAILED(targetDeviceFamiliesIter->GetCurrent(&targetDeviceFamily));
     73 
     74             targetDeviceFamilies.emplace_back(std::move(targetDeviceFamily));
     75 
     76             THROW_IF_FAILED(targetDeviceFamiliesIter->MoveNext(&hasCurrent));
     77         }
     78 
     79         return targetDeviceFamilies;
     80     }

     97     {
     98         auto name = GetName();
     99         if (Utility::CaseInsensitiveEquals(name, WindowsDesktopName))
    100         {
    101             return WindowsDesktop;

    103 
    104         if (Utility::CaseInsensitiveEquals(name, WindowsUniversalName))
