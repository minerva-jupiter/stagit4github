     23     // The authentication modes
     24     enum class AuthenticationMode
     25     {
     26         Unknown,
     27 
     28         // Always do interactive authentication on first request, following requests may use cached result.
     29         Interactive,
     30 
     31         // Try silent flow first. If failed, use interactive flow.
     32         SilentPreferred,
     33 
     34         // Only do silent flow. If failed, the authentication failed.
     35         Silent,
     36     };
