    163     struct ManifestException : public wil::ResultException
    164     {
    165         ManifestException(std::vector<ValidationError>&& errors = {}, HRESULT hr = APPINSTALLER_CLI_ERROR_MANIFEST_FAILED) :
    166             wil::ResultException(hr), m_errors(std::move(errors))
    167         {
    168             auto p = [&](ValidationError const& e) {
    169                 return e.ErrorLevel == ValidationError::Level::Error;
    170             };
    171 
    172             m_warningOnly = !m_errors.empty() && std::find_if(m_errors.begin(), m_errors.end(), p) == m_errors.end();
    173         }
    174 
    175         ManifestException(HRESULT hr) : ManifestException({}, hr) {}
    176 
    177         // Error message without wil diagnostic info
    178         const std::string& GetManifestErrorMessage() const noexcept
    179         {
    180             if (m_manifestErrorMessage.empty())
    181             {
    182                 if (m_errors.empty())
    183                 {
    184                     // Syntax error, yaml parser error is stored in FailureInfo
    185                     if (GetFailureInfo().pszMessage)
    186                     {
    187                         m_manifestErrorMessage = Utility::ConvertToUTF8(GetFailureInfo().pszMessage);
    188                     }
    189                 }
    190                 else
    191                 {
    192                     for (auto const& error : m_errors)
    193                     {
    194                         if (error.ErrorLevel == ValidationError::Level::Error)
    195                         {
    196                             m_manifestErrorMessage += ManifestError::ErrorMessagePrefix;
    197                         }
    198                         else if (error.ErrorLevel == ValidationError::Level::Warning)
    199                         {
    200                             m_manifestErrorMessage += ManifestError::WarningMessagePrefix;
    201                         }
    202                         m_manifestErrorMessage += error.GetErrorMessage();
    203 
    204                         if (!error.Context.empty())
    205                         {
    206                             m_manifestErrorMessage += " [" + error.Context + "]";
    207                         }
    208                         if (!error.Value.empty())
    209                         {
    210                             m_manifestErrorMessage += " Value: " + error.Value;
    211                         }
    212                         if (error.Line > 0 && error.Column > 0)
    213                         {
    214                             m_manifestErrorMessage += " Line: " + std::to_string(error.Line) + ", Column: " + std::to_string(error.Column);
    215                         }
    216                         if (!error.FileName.empty())
    217                         {
    218                             m_manifestErrorMessage += " File: " + error.FileName;
    219                         }
    220                         m_manifestErrorMessage += '\n';
    221                     }
    222                 }
    223             }
    224             return m_manifestErrorMessage;
    225         }
    226 
    227         bool IsWarningOnly() const noexcept
    228         {
    229             return m_warningOnly;
    230         }
    231 
    232         const char* what() const noexcept override
    233         {
    234             if (m_whatMessage.empty())
    235             {
    236                 m_whatMessage = ResultException::what();
    237 
    238                 if (!m_errors.empty())
    239                 {
    240                     m_whatMessage += GetManifestErrorMessage();
    241                 }
    242             }
    243             return m_whatMessage.c_str();
    244         }
    245 
    246         const std::vector<ValidationError>& Errors() const { return m_errors; }
    247 
    248     private:
    249         std::vector<ValidationError> m_errors;
    250         mutable std::string m_whatMessage;
    251         mutable std::string m_manifestErrorMessage;
    252         bool m_warningOnly;
    253     };
