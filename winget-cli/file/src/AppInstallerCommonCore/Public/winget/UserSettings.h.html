    133 #define SETTINGMAPPING_SPECIALIZATION_POLICY(_setting_, _json_, _value_, _default_, _path_, _valuePolicy_) \
    134         template <> \
    135         struct SettingMapping<_setting_> \
    136         { \
    137             using json_t = _json_; \
    138             using value_t = _value_; \
    139             inline static const value_t DefaultValue = _default_; \
    140             static constexpr std::string_view Path = _path_; \
    141             static std::optional<value_t> Validate(const json_t& value); \
    142             static constexpr ValuePolicy Policy = _valuePolicy_; \
    143             using policy_t = GroupPolicy::ValueType<Policy>; \
    144             static_assert(Policy == ValuePolicy::None || std::is_same<json_t, policy_t>::value); \
    145         }

    202         // Used to deduce the SettingVariant type; making a variant that includes std::monostate and all SettingMapping types.
    203         template <size_t... I>
    204         inline auto Deduce(std::index_sequence<I...>) { return std::variant<std::monostate, typename SettingMapping<static_cast<Setting>(I)>::value_t...>{}; }
    205 
    206         // Holds data of any type listed in a SettingMapping.
    207         using SettingVariant = decltype(Deduce(std::make_index_sequence<static_cast<size_t>(Setting::Max)>()));
    208 
    209         // Gets the index into the variant for the given Setting.
    210         constexpr inline size_t SettingIndex(Setting s) { return static_cast<size_t>(s) + 1; }

    245         // Gets setting value, if its not in the map it returns the default value.
    246         template <Setting S>
    247         typename details::SettingMapping<S>::value_t Get() const
    248         {
    249             auto itr = m_settings.find(S);
    250             if (itr == m_settings.end())
    251             {
    252                 return details::SettingMapping<S>::DefaultValue;
    253             }
    254 
    255             return std::get<details::SettingIndex(S)>(itr->second);
    256         }
