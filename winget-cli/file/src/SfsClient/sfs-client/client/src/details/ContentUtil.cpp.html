     11 bool contentutil::operator==(const ContentId& lhs, const ContentId& rhs)

     93 bool contentutil::operator==(const AppContent& lhs, const AppContent& rhs)
     94 {
     95     auto arePrerequisitesEqual = [&lhs, &rhs]() {
     96         return std::equal(
     97             lhs.GetPrerequisites().begin(),
     98             lhs.GetPrerequisites().end(),
     99             rhs.GetPrerequisites().begin(),
    100             rhs.GetPrerequisites().end(),
    101             [](const AppPrerequisiteContent& clhs, const AppPrerequisiteContent& crhs) { return clhs == crhs; });
    102     };
    103     auto areFilesEqual = [&lhs, &rhs]() {
    104         return std::is_permutation(lhs.GetFiles().begin(),
    105                                    lhs.GetFiles().end(),
    106                                    rhs.GetFiles().begin(),
    107                                    rhs.GetFiles().end(),
    108                                    [](const AppFile& flhs, const AppFile& frhs) { return flhs == frhs; });
    109     };
    110     return lhs.GetContentId() == rhs.GetContentId() && lhs.GetUpdateId() == rhs.GetUpdateId() &&
    111            arePrerequisitesEqual() && areFilesEqual();
    112 }
