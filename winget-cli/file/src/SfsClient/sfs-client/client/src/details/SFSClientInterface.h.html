     26 class SFSClientInterface

     28   public:

     33     //
     34     // Combined API calls for retrieval of metadata & download URLs
     35     //
     36 
     37     /**
     38      * @brief Retrieve combined metadata & download URLs from the latest version of specified products
     39      * @note At the moment only a single product request is supported
     40      * @param requestParams Parameters that define this request
     41      */
     42     virtual std::vector<Content> GetLatestDownloadInfo(const RequestParams& requestParams) const = 0;
     43 
     44     /**
     45      * @brief Retrieve combined metadata & download URLs from the latest version of specified apps
     46      * @note At the moment only a single product request is supported
     47      * @param requestParams Parameters that define this request
     48      */
     49     virtual std::vector<AppContent> GetLatestAppDownloadInfo(const RequestParams& requestParams) const = 0;
     50 
     51     //
     52     // Individual APIs 1:1 with service endpoints
     53     //
     54 
     55     /**
     56      * @brief Gets the metadata for the latest available version for the specified product request
     57      * @return Entity that describes the latest version of the product
     58      * @throws SFSException if the request fails
     59      */
     60     virtual std::unique_ptr<VersionEntity> GetLatestVersion(const ProductRequest& productRequest,
     61                                                             Connection& connection) const = 0;
     62 
     63     /**
     64      * @brief Gets the metadata for the latest available version for the specified product requests
     65      * @return Vector of entities that describe the latest version of the products
     66      * @throws SFSException if the request fails
     67      */
     68     virtual VersionEntities GetLatestVersionBatch(const std::vector<ProductRequest>& productRequests,
     69                                                   Connection& connection) const = 0;
     70 
     71     /**
     72      * @brief Gets the metadata for a specific version of the specified product
     73      * @return Entity that describes the latest version of the product
     74      * @throws SFSException if the request fails
     75      */
     76     virtual std::unique_ptr<VersionEntity> GetSpecificVersion(const std::string& product,
     77                                                               const std::string& version,
     78                                                               Connection& connection) const = 0;
     79 
     80     /**
     81      * @brief Gets the files metadata for a specific version of the specified product
     82      * @return Vector of File entities for the specific version of the product
     83      * @throws SFSException if the request fails
     84      */
     85     virtual FileEntities GetDownloadInfo(const std::string& product,
     86                                          const std::string& version,
     87                                          Connection& connection) const = 0;
     88 
     89     /**
     90      * @brief Returns a new Connection to be used by the SFSClient to make requests

     93     virtual std::unique_ptr<Connection> MakeConnection(const ConnectionConfig& config) const = 0;
