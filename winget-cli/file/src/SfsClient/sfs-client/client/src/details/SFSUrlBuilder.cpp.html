     11 SFSUrlBuilder::SFSUrlBuilder(const std::string& accountId,
     12                              std::string instanceId,
     13                              std::string nameSpace,

     34 std::string SFSUrlBuilder::GetLatestVersionUrl(const std::string& product)
     35 {
     36     SetVersionsUrlPath(product);
     37     AppendPath("latest");
     38     SetQuery("action", "select");
     39     return GetUrl();
     40 }

     42 std::string SFSUrlBuilder::GetLatestVersionBatchUrl()

     65 SFSUrlBuilder& SFSUrlBuilder::SetNamesUrlPath()
     66 {
     67     ResetPath().ResetQuery();
     68     AppendPath("api").AppendPath(c_apiVersion);
     69     AppendPath("contents").AppendPathEncoded(m_instanceId);
     70     AppendPath("namespaces").AppendPathEncoded(m_nameSpace);
     71     AppendPath("names");
     72     return *this;
     73 }

     75 SFSUrlBuilder& SFSUrlBuilder::SetVersionsUrlPath(const std::string& product)
     76 {
     77     SetNamesUrlPath();
     78     AppendPathEncoded(product);
     79     AppendPath("versions");
     80     return *this;
