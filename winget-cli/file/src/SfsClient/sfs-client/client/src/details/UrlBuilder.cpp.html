     13 #define THROW_IF_CURL_URL_ERROR(curlCall, error)                                                                       \
     14     do                                                                                                                 \
     15     {                                                                                                                  \
     16         auto __curlUrlCode = (curlCall);                                                                               \
     17         THROW_CODE_IF_NOT_LOG(error, __curlUrlCode == CURLUE_OK, m_handler, GetCurlUrlStrError(__curlUrlCode));        \
     18     } while ((void)0, 0)

     63     CurlCharPtr url;

     77 std::string UrlBuilder::GetQuery() const
     78 {
     79     CurlCharPtr query;
     80     char* queryPtr = query.get();
     81     const auto queryResult = curl_url_get(m_handle, CURLUPART_QUERY, &queryPtr, 0 /*flags*/);
     82     switch (queryResult)
     83     {
     84     case CURLUE_OK:
     85         return queryPtr;
     86     case CURLUE_NO_QUERY:
     87         return {};
     88     default:
     89         THROW_IF_CURL_URL_SETUP_ERROR(queryResult);
     90     }
     91     return {};
     92 }

    130 UrlBuilder& UrlBuilder::AppendPath(const std::string& path, bool encode)
    131 {
    132     THROW_CODE_IF_LOG(InvalidArg, path.empty(), m_handler, "Path must not empty");
    133     if (!m_path.empty() && m_path.back() != '/')
    134     {
    135         m_path += '/';
    136     }
    137 
    138     if (encode)
    139     {
    140         m_path += URLEncode(path);
    141     }
    142     else
    143     {
    144         m_path += path;
    145     }
    146 
    147     THROW_IF_CURL_URL_SETUP_ERROR(curl_url_set(m_handle, CURLUPART_PATH, m_path.c_str(), 0 /*flags*/));
    148     return *this;
    149 }

    187 std::string UrlBuilder::URLEncode(const std::string& str) const
    188 {
    189     CurlCharPtr encodedStr{curl_easy_escape(nullptr /*ignored*/, str.c_str(), static_cast<int>(str.length()))};
    190     THROW_CODE_IF_NOT_LOG(ConnectionUrlSetupFailed, encodedStr, m_handler, "Failed to URL-encode string");
    191     return encodedStr.get();
