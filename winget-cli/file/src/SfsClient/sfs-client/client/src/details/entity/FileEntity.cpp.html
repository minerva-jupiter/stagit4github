     24 HashType HashTypeFromString(const std::string& hashType, const ReportingHandler& handler)
     25 {
     26     if (AreEqualI(hashType, "Sha1"))
     27     {
     28         return HashType::Sha1;
     29     }
     30     else if (AreEqualI(hashType, "Sha256"))
     31     {
     32         return HashType::Sha256;
     33     }
     34     else
     35     {
     36         THROW_LOG(Result(Result::Unexpected, "Unknown hash type: " + hashType), handler);
     37         return HashType::Sha1; // Unreachable code, but the compiler doesn't know that.
     38     }
     39 }

     41 Architecture ArchitectureFromString(const std::string& arch, const ReportingHandler& handler)
     42 {
     43     if (AreEqualI(arch, "None"))
     44     {
     45         return Architecture::None;
     46     }
     47     else if (AreEqualI(arch, "x86"))
     48     {
     49         return Architecture::x86;
     50     }
     51     else if (AreEqualI(arch, "amd64"))
     52     {
     53         return Architecture::Amd64;
     54     }
     55     else if (AreEqualI(arch, "arm"))
     56     {
     57         return Architecture::Arm;
     58     }
     59     else if (AreEqualI(arch, "arm64"))
     60     {
     61         return Architecture::Arm64;
     62     }
     63     else
     64     {
     65         THROW_LOG(Result(Result::Unexpected, "Unknown architecture: " + arch), handler);
     66         return Architecture::None; // Unreachable code, but the compiler doesn't know that.
     67     }
     68 }

     70 void ValidateContentType(const FileEntity& entity, ContentType expectedType, const ReportingHandler& handler)
     71 {
     72     THROW_CODE_IF_LOG(Result::ServiceUnexpectedContentType,
     73                       entity.GetContentType() != expectedType,
     74                       handler,
     75                       "The service returned file \"" + entity.fileId + "\" with content type [" +
     76                           ToString(entity.GetContentType()) + "] while the expected type was [" +
     77                           ToString(expectedType) + "]");

     81 std::unique_ptr<FileEntity> FileEntity::FromJson(const nlohmann::json& file, const ReportingHandler& handler)
     82 {
     83     // Expected format for a generic file entity:
     84     // {
     85     //   "FileId": <fileid>,
     86     //   "Url": <url>,
     87     //   "SizeInBytes": <size>,
     88     //   "Hashes": {
     89     //     "Sha1": <sha1>,
     90     //     "Sha256": <sha2>
     91     //   },
     92     //   "DeliveryOptimization": {} // ignored, not used by the client.
     93     // }
     94     //
     95     // Expected extra elements for an app version entity:
     96     // {
     97     //   ...
     98     //   "ApplicabilityDetails": {
     99     //     "Architectures": [
    100     //       "<arch>"
    101     //     ],
    102     //     "PlatformApplicabilityForPackage": [
    103     //       "<app>"
    104     //     ]
    105     //   },
    106     //   "FileMoniker": "<moniker>",
    107     // }
    108 
    109     THROW_INVALID_RESPONSE_IF_NOT(file.is_object(), "File is not a JSON object", handler);
    110 
    111     std::unique_ptr<FileEntity> tmp;
    112     const bool isAppEntity = file.contains("FileMoniker");
    113     if (isAppEntity)
    114     {
    115         tmp = std::make_unique<AppFileEntity>();
    116     }
    117     else
    118     {
    119         tmp = std::make_unique<GenericFileEntity>();
    120     }
    121 
    122     THROW_INVALID_RESPONSE_IF_NOT(file.contains("FileId"), "Missing File.FileId in response", handler);
    123     THROW_INVALID_RESPONSE_IF_NOT(file["FileId"].is_string(), "File.FileId is not a string", handler);
    124     tmp->fileId = file["FileId"];
    125 
    126     THROW_INVALID_RESPONSE_IF_NOT(file.contains("Url"), "Missing File.Url in response", handler);
    127     THROW_INVALID_RESPONSE_IF_NOT(file["Url"].is_string(), "File.Url is not a string", handler);
    128     tmp->url = file["Url"];
    129 
    130     THROW_INVALID_RESPONSE_IF_NOT(file.contains("SizeInBytes"), "Missing File.SizeInBytes in response", handler);
    131     THROW_INVALID_RESPONSE_IF_NOT(file["SizeInBytes"].is_number_unsigned(),
    132                                   "File.SizeInBytes is not an unsigned number",
    133                                   handler);
    134     tmp->sizeInBytes = file["SizeInBytes"];
    135 
    136     THROW_INVALID_RESPONSE_IF_NOT(file.contains("Hashes"), "Missing File.Hashes in response", handler);
    137     THROW_INVALID_RESPONSE_IF_NOT(file["Hashes"].is_object(), "File.Hashes is not an object", handler);
    138 
    139     for (const auto& [hashType, hashValue] : file["Hashes"].items())
    140     {
    141         THROW_INVALID_RESPONSE_IF_NOT(hashValue.is_string(), "File.Hashes object value is not a string", handler);
    142         tmp->hashes[hashType] = hashValue;
    143     }
    144 
    145     if (isAppEntity)
    146     {
    147         auto appEntity = dynamic_cast<AppFileEntity*>(tmp.get());
    148 
    149         THROW_INVALID_RESPONSE_IF_NOT(file["FileMoniker"].is_string(), "File.FileMoniker is not a string", handler);
    150         appEntity->fileMoniker = file["FileMoniker"];
    151 
    152         THROW_INVALID_RESPONSE_IF_NOT(file.contains("ApplicabilityDetails"),
    153                                       "Missing File.ApplicabilityDetails in response",
    154                                       handler);
    155 
    156         const auto& details = file["ApplicabilityDetails"];
    157         THROW_INVALID_RESPONSE_IF_NOT(details.is_object(), "File.ApplicabilityDetails is not an object", handler);
    158 
    159         THROW_INVALID_RESPONSE_IF_NOT(details.contains("Architectures"),
    160                                       "Missing File.ApplicabilityDetails.Architectures in response",
    161                                       handler);
    162         THROW_INVALID_RESPONSE_IF_NOT(details["Architectures"].is_array(),
    163                                       "File.ApplicabilityDetails.Architectures is not an array",
    164                                       handler);
    165         for (const auto& arch : details["Architectures"])
    166         {
    167             THROW_INVALID_RESPONSE_IF_NOT(arch.is_string(),
    168                                           "File.ApplicabilityDetails.Architectures array value is not a string",
    169                                           handler);
    170         }
    171         appEntity->applicabilityDetails.architectures = details["Architectures"];
    172 
    173         THROW_INVALID_RESPONSE_IF_NOT(details.contains("PlatformApplicabilityForPackage"),
    174                                       "Missing File.ApplicabilityDetails.PlatformApplicabilityForPackage in response",
    175                                       handler);
    176         THROW_INVALID_RESPONSE_IF_NOT(details["PlatformApplicabilityForPackage"].is_array(),
    177                                       "File.ApplicabilityDetails.PlatformApplicabilityForPackage is not an array",
    178                                       handler);
    179         for (const auto& app : details["PlatformApplicabilityForPackage"])
    180         {
    181             THROW_INVALID_RESPONSE_IF_NOT(
    182                 app.is_string(),
    183                 "File.ApplicabilityDetails.PlatformApplicabilityForPackage array value is not a string",
    184                 handler);
    185         }
    186         appEntity->applicabilityDetails.platformApplicabilityForPackage = details["PlatformApplicabilityForPackage"];
    187     }
    188 
    189     return tmp;
    190 }

    192 FileEntities FileEntity::DownloadInfoResponseToFileEntities(const nlohmann::json& data, const ReportingHandler& handler)
    193 {
    194     // Expected format is an array of FileEntity
    195     THROW_CODE_IF_NOT_LOG(ServiceInvalidResponse, data.is_array(), handler, "Response is not a JSON array");
    196 
    197     FileEntities tmp;
    198     for (const auto& fileData : data)
    199     {
    200         THROW_CODE_IF_NOT_LOG(ServiceInvalidResponse,
    201                               fileData.is_object(),
    202                               handler,
    203                               "Array element is not a JSON object");
    204         tmp.push_back(std::move(FileEntity::FromJson(fileData, handler)));
    205     }
    206 
    207     return tmp;
    208 }

    215 std::unique_ptr<File> GenericFileEntity::ToFile(FileEntity&& entity, const ReportingHandler& handler)
    216 {
    217     ValidateContentType(entity, ContentType::Generic, handler);
    218 
    219     std::unordered_map<HashType, std::string> hashes;
    220     for (auto& [hashType, hashValue] : entity.hashes)
    221     {
    222         hashes[HashTypeFromString(hashType, handler)] = std::move(hashValue);
    223     }
    224 
    225     std::unique_ptr<File> tmp;
    226     THROW_IF_FAILED_LOG(
    227         File::Make(std::move(entity.fileId), std::move(entity.url), entity.sizeInBytes, std::move(hashes), tmp),
    228         handler);
    229     return tmp;
    230 }

    248 std::unique_ptr<AppFile> AppFileEntity::ToAppFile(FileEntity&& entity, const ReportingHandler& handler)
    249 {
    250     ValidateContentType(entity, ContentType::App, handler);
    251 
    252     auto appEntity = dynamic_cast<AppFileEntity&&>(entity);
    253 
    254     std::unordered_map<HashType, std::string> hashes;
    255     for (auto& [hashType, hashValue] : appEntity.hashes)
    256     {
    257         hashes[HashTypeFromString(hashType, handler)] = std::move(hashValue);
    258     }
    259 
    260     std::vector<Architecture> architectures;
    261     for (auto& arch : appEntity.applicabilityDetails.architectures)
    262     {
    263         architectures.push_back(ArchitectureFromString(arch, handler));
    264     }
    265 
    266     std::unique_ptr<AppFile> tmp;
    267     THROW_IF_FAILED_LOG(AppFile::Make(std::move(appEntity.fileId),
    268                                       std::move(appEntity.url),
    269                                       appEntity.sizeInBytes,
    270                                       std::move(hashes),
    271                                       std::move(architectures),
    272                                       std::move(appEntity.applicabilityDetails.platformApplicabilityForPackage),
    273                                       std::move(appEntity.fileMoniker),
    274                                       tmp),
    275                         handler);
    276     return tmp;
    277 }
