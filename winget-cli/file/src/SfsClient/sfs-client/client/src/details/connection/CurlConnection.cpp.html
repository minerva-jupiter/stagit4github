     38 // Curl callback for writing data to a std::string. Must return the number of bytes written.
     39 // This callback may be called multiple times for a single request, and will keep appending
     40 // to userData until the request is complete. The data received is not null-terminated.
     41 // For SFS, this data will likely be a JSON string.
     42 size_t WriteCallback(char* contents, size_t sizeInBytes, size_t numElements, void* userData)
     43 {
     44     auto readBufferPtr = static_cast<std::string*>(userData);
     45     if (readBufferPtr)
     46     {
     47         size_t totalSize = sizeInBytes * numElements;
     48 
     49         // Checking final response size to avoid unexpected amounts of data
     50         if ((readBufferPtr->length() + totalSize) > MAX_RESPONSE_CHARACTERS)
     51         {
     52             return CURL_WRITEFUNC_ERROR;
     53         }
     54 
     55         readBufferPtr->append(contents, totalSize);
     56         return totalSize;
     57     }
     58     return CURL_WRITEFUNC_ERROR;
     59 }

    104 Result CurlCodeToResult(CURLcode curlCode, char* errorBuffer)
    105 {
    106     Result::Code code;
    107     switch (curlCode)
    108     {
    109     case CURLE_OPERATION_TIMEDOUT:
    110         code = Result::HttpTimeout;
    111         break;
    112     default:
    113         code = Result::ConnectionUnexpectedError;
    114         break;
    115     }
    116 
    117     const bool isErrorStringRegistered = errorBuffer && errorBuffer[0] != '\0';
    118     std::string message = isErrorStringRegistered ? errorBuffer : "Curl error";
    119 
    120     return Result(code, std::move(message));
    121 }

    128 Result HttpCodeToResult(long httpCode)
    129 {
    130     if (IsSuccessfulSFSHttpCode(httpCode))
    131     {
    132         return Result::Success;
    133     }
    134 
    135     switch (httpCode)
    136     {
    137     case 400:
    138     {
    139         return Result(Result::HttpBadRequest, "400 Bad Request");
    140     }
    141     case 404:
    142     {
    143         return Result(Result::HttpNotFound, "404 Not Found");
    144     }
    145     case 405:
    146     {
    147         return Result(Result::HttpMethodNotAllowed, "405 Method Not Allowed");
    148     }
    149     case 429:
    150     {
    151         return Result(Result::HttpTooManyRequests, "429 Too Many Requests");
    152     }
    153     case 503:
    154     {
    155         return Result(Result::HttpServiceNotAvailable, "503 Service Unavailable");
    156     }
    157     default:
    158     {
    159         return Result(Result::HttpUnexpected, "Unexpected HTTP code " + std::to_string(httpCode));
    160     }
    161     }

    164 bool IsRetriableHttpError(long httpCode)
    165 {
    166     switch (httpCode)
    167     {
    168     case 429: // Too Many Requests - Rate Limiting
    169     case 500: // InternalServerError - Can be triggered within server timeouts, network issue
    170     case 502: // BadGateway - Likely an issue with routing
    171     case 503: // ServerBusy
    172     case 504: // GatewayTimeout
    173         return true;
    174     default:
    175         return false;
    176     }

    179 std::optional<std::string> GetResponseHeader(CURL* handle,
    180                                              HttpHeader httpHeader,
    181                                              const ReportingHandler& reportingHandler)
    182 {
    183     const std::string headerName = ToString(httpHeader);
    184 
    185     // This struct only represents data inside the CURL handle, and must not be manually freed
    186     curl_header* header;
    187     const int lastRequest = -1;
    188     CURLHcode curlhCode = curl_easy_header(handle, headerName.c_str(), 0 /*index*/, CURLH_HEADER, lastRequest, &header);
    189     switch (curlhCode)
    190     {
    191     case CURLHE_OK:
    192         return header->value;
    193     case CURLHE_BADINDEX:
    194     case CURLHE_MISSING:
    195     case CURLHE_NOHEADERS:
    196         return std::nullopt;
    197     default:
    198         THROW_LOG(
    199             Result(Result::ConnectionUnexpectedError,
    200                    "Failed to get response header " + headerName + " with CURLH code " + std::to_string(curlhCode)),
    201             reportingHandler);
    202     }
    203     return std::nullopt;
    204 }

    206 std::chrono::milliseconds ParseRetryAfterValue(const std::string& retryAfter, const ReportingHandler& reportingHandler)
    207 {
    208     LOG_VERBOSE(reportingHandler, "Parsing Retry-After value [%s]", retryAfter.c_str());
    209     std::chrono::seconds retryAfterSec{0};
    210     try
    211     {
    212         retryAfterSec = std::chrono::seconds(std::stoi(retryAfter));
    213     }
    214     catch (std::invalid_argument&)
    215     {
    216         // Value is not an integer, but may still be in HTTP Date format
    217         const time_t retryAfterSecSinceEpoch = curl_getdate(retryAfter.c_str(), nullptr /*unused*/);
    218         if (retryAfterSecSinceEpoch == -1)
    219         {
    220             THROW_LOG(Result(Result::ConnectionUnexpectedError,
    221                              "Retry-After header value could not be converted to an integer or an HTTP Date"),
    222                       reportingHandler);
    223         }
    224 
    225         // Get number of seconds since epoch for now to calculate the difference
    226         const auto epoch = std::chrono::system_clock::now().time_since_epoch();
    227         const auto nowSecSinceEpoch = std::chrono::duration_cast<std::chrono::seconds>(epoch);
    228 
    229         retryAfterSec = std::chrono::seconds(retryAfterSecSinceEpoch) - nowSecSinceEpoch;
    230     }
    231     catch (std::out_of_range&)
    232     {
    233         THROW_LOG(Result(Result::ConnectionUnexpectedError, "Retry-After header value is not in the expected range"),
    234                   reportingHandler);
    235     }
    236     if (retryAfterSec <= 0s)
    237     {
    238         THROW_LOG(Result(Result::ConnectionUnexpectedError, "Invalid Retry-After header value"), reportingHandler);
    239     }
    240     return retryAfterSec;
    241 }

    330 std::string CurlConnection::CurlPerform(const std::string& url, CurlHeaderList& headers)
    331 {
    332     THROW_IF_CURL_SETUP_ERROR(curl_easy_setopt(m_handle, CURLOPT_URL, url.c_str()));
    333 
    334     const std::string cv = m_cv.IncrementAndGet();
    335     headers.Add(HttpHeader::MSCV, cv);
    336     headers.Add(HttpHeader::UserAgent, GetUserAgentValue());
    337 
    338     THROW_IF_CURL_SETUP_ERROR(curl_easy_setopt(m_handle, CURLOPT_HTTPHEADER, headers.m_slist));
    339 
    340     // Setting up error buffer where error messages get written - this gets unset in the destructor
    341     CurlErrorBuffer errorBuffer(m_handle, m_handler);
    342 
    343     std::string readBuffer;
    344     THROW_IF_CURL_SETUP_ERROR(curl_easy_setopt(m_handle, CURLOPT_WRITEFUNCTION, WriteCallback));
    345     THROW_IF_CURL_SETUP_ERROR(curl_easy_setopt(m_handle, CURLOPT_WRITEDATA, &readBuffer));
    346 
    347     // Retry the connection a specified number of times
    348     const unsigned totalAttempts = 1 + m_maxRetries;
    349     for (unsigned i = 0; i < totalAttempts; i++)
    350     {
    351         const unsigned attempt = i + 1;
    352         LOG_INFO(m_handler, "Request attempt %u out of %u (cv: %s)", attempt, totalAttempts, cv.c_str());
    353         const bool lastAttempt = attempt == totalAttempts;
    354 
    355         // Clear the buffer before each attempt
    356         readBuffer.clear();
    357 
    358         // Perform the request
    359         const auto result = curl_easy_perform(m_handle);
    360         if (result != CURLE_OK)
    361         {
    362             THROW_LOG(CurlCodeToResult(result, errorBuffer.Get()), m_handler);
    363         }
    364 
    365         // Check request status to stop or retry
    366         long httpCode = 0;
    367         THROW_IF_CURL_UNEXPECTED_ERROR(curl_easy_getinfo(m_handle, CURLINFO_RESPONSE_CODE, &httpCode));
    368 
    369         if (IsSuccessfulSFSHttpCode(httpCode))
    370         {
    371             break;
    372         }
    373 
    374         const Result httpResult = HttpCodeToResult(httpCode);
    375         if (!CanRetryRequest(lastAttempt, httpCode))
    376         {
    377             THROW_LOG(httpResult, m_handler);
    378         }
    379 
    380         ProcessRetry(attempt, httpResult);
    381     }
    382 
    383     return readBuffer;
    384 }

    386 bool CurlConnection::CanRetryRequest(bool lastAttempt, long httpCode)
    387 {
    388     if (lastAttempt)
    389     {
    390         LOG_INFO(m_handler, "No retry as this is the last attempt");
    391         return false;
    392     }
    393 
    394     if (!IsRetriableHttpError(httpCode))
    395     {
    396         LOG_INFO(m_handler, "Error %ld is not retriable, stopping", httpCode);
    397         return false;
    398     }
    399 
    400     return true;
    401 }

    403 void CurlConnection::ProcessRetry(int attempt, const Result& httpResult)
    404 {
    405     // Wait before retrying. Prefer the Retry-After information if available
    406     std::chrono::milliseconds retryDelay{0};
    407     const std::optional<std::string> retryAfter = GetResponseHeader(m_handle, HttpHeader::RetryAfter, m_handler);
    408     if (retryAfter)
    409     {
    410         // TODO #93: Enforce Retry-After value across calls to avoid caller spamming server
    411         retryDelay = ParseRetryAfterValue(*retryAfter, m_handler);
    412     }
    413     else
    414     {
    415         // Apply exponential back-off with a factor of 2
    416         static std::chrono::milliseconds s_baseRetryDelay = 15s; // Value recommended as interval by the service
    417 
    418         std::chrono::milliseconds baseRetryDelay = s_baseRetryDelay;
    419 
    420         // Value can be overriden in tests
    421         if (auto override = test::GetTestOverrideAsInt(test::TestOverride::BaseRetryDelayMs))
    422         {
    423             baseRetryDelay = std::chrono::milliseconds{*override};
    424         }
    425 
    426         retryDelay = baseRetryDelay * (1 << (attempt - 1));
    427     }
    428 
    429     LOG_IF_FAILED(httpResult, m_handler);
    430     LOG_INFO(m_handler, "Sleeping for %lld ms", static_cast<long long>(retryDelay.count()));
    431     std::this_thread::sleep_for(retryDelay);
