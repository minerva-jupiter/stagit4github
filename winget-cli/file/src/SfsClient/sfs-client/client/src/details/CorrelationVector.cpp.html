     18 CorrelationVector::CorrelationVector(const std::string& cv, const ReportingHandler& handler)
     19 {
     20     THROW_CODE_IF_LOG(InvalidArg, cv.empty(), handler, "cv must not be empty");
     21 
     22     try
     23     {
     24         m_cv = std::make_unique<correlation_vector>(correlation_vector::extend(cv));
     25     }
     26     catch (std::invalid_argument& e)
     27     {
     28         THROW_LOG(Result(Result::InvalidArg, "baseCV is not a valid correlation vector: " + std::string(e.what())),
