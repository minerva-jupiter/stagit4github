     49     /**

     51      * @return Entity that describes the latest version of the product
     52      * @throws SFSException if the request fails

     57     /**

     62     VersionEntities GetLatestVersionBatch(const std::vector<ProductRequest>& productRequests,

     65     /**
     66      * @brief Gets the metadata for a specific version of the specified product
     67      * @return Entity that describes the latest version of the product
     68      * @throws SFSException if the request fails
     69      */
     70     std::unique_ptr<VersionEntity> GetSpecificVersion(const std::string& product,

     72                                                       Connection& connection) const override;

     74     /**

     77      * @throws SFSException if the request fails

     79     FileEntities GetDownloadInfo(const std::string& product,
