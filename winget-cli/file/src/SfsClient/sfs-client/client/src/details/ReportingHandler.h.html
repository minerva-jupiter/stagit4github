     11 #define MAX_LOG_MESSAGE_SIZE 1024

     13 #define LOG_INFO(handler, format, ...)                                                                                 \
     14     handler.LogWithSeverity(SFS::LogSeverity::Info, __FILE__, __LINE__, __FUNCTION__, format, ##__VA_ARGS__)
     15 #define LOG_WARNING(handler, format, ...)                                                                              \
     16     handler.LogWithSeverity(SFS::LogSeverity::Warning, __FILE__, __LINE__, __FUNCTION__, format, ##__VA_ARGS__)
     17 #define LOG_ERROR(handler, format, ...)                                                                                \
     18     handler.LogWithSeverity(SFS::LogSeverity::Error, __FILE__, __LINE__, __FUNCTION__, format, ##__VA_ARGS__)

     22 namespace SFS::details
     23 {
     24 /**
     25  * @brief This class enables thread-safe access to the externally set logging callback function.
     26  * @details Each SFSClient instance will have one ReportingHandler instance, and access to the logging callback function
     27  * is controlled by a mutex that makes sure that only one thread can access the logging callback function at a time.
     28  */
     29 class ReportingHandler
     30 {
     31   public:
     32     ReportingHandler() = default;
     33 
     34     ReportingHandler(const ReportingHandler&) = delete;
     35     ReportingHandler& operator=(const ReportingHandler&) = delete;
     36 
     37     /**
     38      * @brief Sets the logging callback function.
     39      * @details This function is thread-safe.
     40      * @param callback The logging callback function. To reset, pass a nullptr.
     41      */
     42     void SetLoggingCallback(LoggingCallbackFn&& callback);
     43 
     44     /**
     45      * @brief Logs a message with the given severity.
     46      * @details Prefer calling it with macros LOG_INFO, LOG_WARNING, LOG_ERROR, LOG_VERBOSE so file, line and function
     47      * are automatically populated and the message can be formatted.
     48      */
     49     template <typename... Args>
     50     void LogWithSeverity(LogSeverity severity,
     51                          const char* file,
     52                          unsigned line,
     53                          const char* function,
     54                          const char* format,
     55                          const Args&... args) const
     56     {
     57         constexpr std::size_t n = sizeof...(Args);
     58         if constexpr (n == 0)
     59         {
     60             CallLoggingCallback(severity, format, file, line, function);
     61         }
     62         else
     63         {
     64             char message[MAX_LOG_MESSAGE_SIZE];
     65             snprintf(message, MAX_LOG_MESSAGE_SIZE, format, args...);
     66             CallLoggingCallback(severity, message, file, line, function);
     67         }
     68     }
     69 
     70   private:
     71     void CallLoggingCallback(LogSeverity severity,
     72                              const char* message,
     73                              const char* file,
     74                              unsigned line,
     75                              const char* function) const;
     76 
     77     LoggingCallbackFn m_loggingCallbackFn;
     78     mutable std::mutex m_loggingCallbackFnMutex;
     79 };
     80 } // namespace SFS::details
