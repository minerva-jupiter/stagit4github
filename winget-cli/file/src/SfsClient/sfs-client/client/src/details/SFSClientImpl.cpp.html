     62 json ParseServerMethodStringToJson(const std::string& data, const std::string& method, const ReportingHandler& handler)
     63 {
     64     try
     65     {
     66         return json::parse(data);
     67     }
     68     catch (json::parse_error& ex)
     69     {
     70         THROW_LOG(
     71             Result(Result::ServiceInvalidResponse, "(" + method + ") JSON Parsing error: " + std::string(ex.what())),
     72             handler);
     73         return json(); // Unreachable code, but the compiler doesn't know that.
     74     }
     75 }

     77 VersionEntities ConvertLatestVersionBatchResponseToVersionEntities(const json& data, const ReportingHandler& handler)
     78 {
     79     // Expected format:
     80     // [
     81     //   {
     82     //     "ContentId": {
     83     //       "Namespace": <ns>,
     84     //       "Name": <name>,
     85     //       "Version": <version>
     86     //     }
     87     //   },
     88     //   ...
     89     // ]
     90     //
     91 
     92     ThrowInvalidResponseIfFalse(data.is_array(), "Response is not a JSON array", handler);
     93     ThrowInvalidResponseIfFalse(data.size() > 0, "Response does not have the expected size", handler);
     94 
     95     VersionEntities entities;
     96     for (const auto& obj : data)
     97     {
     98         entities.push_back(std::move(VersionEntity::FromJson(obj, handler)));
     99     }
    100 
    101     return entities;
    102 }

    122 void ValidateBatchVersionEntity(const VersionEntities& versionEntities,
    123                                 const std::string& nameSpace,
    124                                 const std::unordered_set<std::string>& requestedProducts,
    125                                 const ReportingHandler& handler)
    126 {
    127     for (const auto& entity : versionEntities)
    128     {
    129         THROW_CODE_IF_LOG(ServiceInvalidResponse,
    130                           requestedProducts.count(entity->contentId.name) == 0,
    131                           handler,
    132                           "Received product [" + entity->contentId.name +
    133                               "] which is not one of the requested products");
    134         THROW_CODE_IF_LOG(ServiceInvalidResponse,
    135                           AreNotEqualI(entity->contentId.nameSpace, nameSpace),
    136                           handler,
    137                           "Received product [" + entity->contentId.name + "] with a namespace [" +
    138                               entity->contentId.nameSpace + "] that does not match the requested namespace");
    139 
    140         LOG_INFO(handler,
    141                  "Received a response for product [%s] with version %s",
    142                  entity->contentId.name.c_str(),
    143                  entity->contentId.version.c_str());
    144     }
    145 }

    147 void ValidateRequestParams(const RequestParams& requestParams, const ReportingHandler& handler)
    148 {
    149     THROW_CODE_IF_LOG(InvalidArg, requestParams.productRequests.empty(), handler, "productRequests cannot be empty");
    150

    152     THROW_CODE_IF_LOG(NotImpl,
    153                       requestParams.productRequests.size() > 1,
    154                       handler,
    155                       "There cannot be more than 1 productRequest at the moment");
    156 
    157     for (const auto& [product, _] : requestParams.productRequests)
    158     {
    159         THROW_CODE_IF_LOG(InvalidArg, product.empty(), handler, "product must not be empty");
    160     }

    164 template <typename ConnectionManagerT>
    165 SFSClientImpl<ConnectionManagerT>::SFSClientImpl(ClientConfig&& config)
    166 {
    167     if (config.logCallbackFn)
    168     {
    169         m_reportingHandler.SetLoggingCallback(std::move(*config.logCallbackFn));
    170     }
    171 
    172     ValidateClientConfig(config, m_reportingHandler);
    173 
    174     m_accountId = std::move(config.accountId);
    175     m_instanceId =
    176         (config.instanceId && !config.instanceId->empty()) ? std::move(*config.instanceId) : c_defaultInstanceId;
    177     m_nameSpace = (config.nameSpace && !config.nameSpace->empty()) ? std::move(*config.nameSpace) : c_defaultNameSpace;
    178 
    179     static_assert(std::is_base_of<ConnectionManager, ConnectionManagerT>::value,
    180                   "ConnectionManagerT not derived from ConnectionManager");
    181     m_connectionManager = std::make_unique<ConnectionManagerT>(m_reportingHandler);
    182 
    183     LogIfTestOverridesAllowed(m_reportingHandler);
    184 }

    187 std::unique_ptr<VersionEntity> SFSClientImpl<ConnectionManagerT>::GetLatestVersion(const ProductRequest& productRequest,
    188                                                                                    Connection& connection) const
    189 try
    190 {
    191     const auto& [product, attributes] = productRequest;
    192     const std::string url{MakeUrlBuilder().GetLatestVersionUrl(product)};
    193 
    194     LOG_INFO(m_reportingHandler, "Requesting latest version of [%s] from URL [%s]", product.c_str(), url.c_str());
    195 
    196     const json body = {{"TargetingAttributes", attributes}};
    197     LOG_VERBOSE(m_reportingHandler, "Request body [%s]", body.dump().c_str());
    198 
    199     const std::string postResponse{connection.Post(url, body.dump())};
    200     const json versionResponse = ParseServerMethodStringToJson(postResponse, "GetLatestVersion", m_reportingHandler);
    201 
    202     auto versionEntity = VersionEntity::FromJson(versionResponse, m_reportingHandler);
    203     ValidateVersionEntity(*versionEntity, m_nameSpace, product, m_reportingHandler);
    204 
    205     LOG_INFO(m_reportingHandler, "Received a response with version %s", versionEntity->contentId.version.c_str());
    206 
    207     return versionEntity;
    208 }

    211 template <typename ConnectionManagerT>
    212 VersionEntities SFSClientImpl<ConnectionManagerT>::GetLatestVersionBatch(
    213     const std::vector<ProductRequest>& productRequests,
    214     Connection& connection) const
    215 try
    216 {
    217     const std::string url{MakeUrlBuilder().GetLatestVersionBatchUrl()};
    218 
    219     LOG_INFO(m_reportingHandler, "Requesting latest version of multiple products from URL [%s]", url.c_str());
    220 
    221     // Creating request body
    222     std::unordered_set<std::string> requestedProducts;
    223     json body = json::array();
    224     for (const auto& [product, attributes] : productRequests)
    225     {
    226         LOG_INFO(m_reportingHandler, "Product #%zu: [%s]", body.size() + size_t{1}, product.c_str());
    227         requestedProducts.insert(product);
    228 
    229         body.push_back({{"TargetingAttributes", attributes}, {"Product", product}});
    230     }
    231 
    232     LOG_VERBOSE(m_reportingHandler, "Request body [%s]", body.dump().c_str());
    233 
    234     const std::string postResponse{connection.Post(url, body.dump())};
    235 
    236     const json versionResponse =
    237         ParseServerMethodStringToJson(postResponse, "GetLatestVersionBatch", m_reportingHandler);
    238 
    239     auto entities = ConvertLatestVersionBatchResponseToVersionEntities(versionResponse, m_reportingHandler);
    240     ValidateBatchVersionEntity(entities, m_nameSpace, requestedProducts, m_reportingHandler);
    241 
    242     return entities;
    243 }

    302 template <typename ConnectionManagerT>
    303 std::vector<Content> SFSClientImpl<ConnectionManagerT>::GetLatestDownloadInfo(const RequestParams& requestParams) const
    304 try
    305 {
    306     ValidateRequestParams(requestParams, m_reportingHandler);
    307 
    308     const auto connection = MakeConnection(ConnectionConfig(requestParams));
    309 
    310     auto versionEntity = GetLatestVersion(requestParams.productRequests[0], *connection);
    311     auto contentId = VersionEntity::ToContentId(std::move(*versionEntity), m_reportingHandler);
    312 
    313     const auto& product = requestParams.productRequests[0].product;
    314     auto fileEntities = GetDownloadInfo(product, contentId->GetVersion(), *connection);
    315     auto files = GenericFileEntity::FileEntitiesToFileVector(std::move(fileEntities), m_reportingHandler);
    316 
    317     std::unique_ptr<Content> content;
    318     THROW_IF_FAILED_LOG(Content::Make(std::move(contentId), std::move(files), content), m_reportingHandler);
    319 
    320     std::vector<Content> contents;
    321     contents.push_back(std::move(*content));
    322 
    323     return contents;
    324 }

    327 template <typename ConnectionManagerT>
    328 std::vector<AppContent> SFSClientImpl<ConnectionManagerT>::GetLatestAppDownloadInfo(
    329     const RequestParams& requestParams) const
    330 try
    331 {
    332     ValidateRequestParams(requestParams, m_reportingHandler);
    333 
    334     // TODO #150: For now apps are only coming from the "storeapps" instanceId and the service has requested
    335     // we double check for it. In the future we should remove this check and allow the user to specify any instanceId
    336     THROW_CODE_IF_LOG(Unexpected,
    337                       AreNotEqualI(m_instanceId, "storeapps"),
    338                       m_reportingHandler,
    339                       "At this moment only the \"storeapps\" instanceId can send app requests");
    340 
    341     const auto connection = MakeConnection(ConnectionConfig(requestParams));
    342 
    343     auto versionEntity = GetLatestVersion(requestParams.productRequests[0], *connection);
    344 
    345     auto appVersionEntity = AppVersionEntity::GetAppVersionEntityPtr(versionEntity, m_reportingHandler);
    346     auto contentId = AppVersionEntity::ToContentId(std::move(*appVersionEntity), m_reportingHandler);
    347 
    348     LOG_INFO(m_reportingHandler, "Getting download info for main app content");
    349     const auto& product = requestParams.productRequests[0].product;
    350     auto fileEntities = GetDownloadInfo(product, contentId->GetVersion(), *connection);
    351     auto files = AppFileEntity::FileEntitiesToAppFileVector(std::move(fileEntities), m_reportingHandler);
    352 
    353     std::vector<AppPrerequisiteContent> prerequisites;
    354     for (auto& prereq : appVersionEntity->prerequisites)
    355     {
    356         LOG_INFO(m_reportingHandler, "Getting download info for prerequisite [%s]", prereq.contentId.name.c_str());
    357         auto prereqContentId = GenericVersionEntity::ToContentId(std::move(prereq), m_reportingHandler);
    358 
    359         auto prereqFileEntities =
    360             GetDownloadInfo(prereqContentId->GetName(), prereqContentId->GetVersion(), *connection);
    361         auto prereqFiles =
    362             AppFileEntity::FileEntitiesToAppFileVector(std::move(prereqFileEntities), m_reportingHandler);
    363 
    364         std::unique_ptr<AppPrerequisiteContent> prereqContent;
    365         THROW_IF_FAILED_LOG(
    366             AppPrerequisiteContent::Make(std::move(prereqContentId), std::move(prereqFiles), prereqContent),
    367             m_reportingHandler);
    368 
    369         prerequisites.push_back(std::move(*prereqContent));
    370     }
    371 
    372     std::unique_ptr<AppContent> content;
    373     THROW_IF_FAILED_LOG(AppContent::Make(std::move(contentId),
    374                                          std::move(appVersionEntity->updateId),
    375                                          std::move(prerequisites),
    376                                          std::move(files),
    377                                          content),
    378                         m_reportingHandler);
    379 
    380     std::vector<AppContent> contents;
    381     contents.push_back(std::move(*content));
    382 
    383     return contents;
    384 }

    403     {
    404         return SFSUrlBuilder(SFSCustomUrl(*envVar), m_instanceId, m_nameSpace, m_reportingHandler);
