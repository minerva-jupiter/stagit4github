    544 {
    545     ReportingHandler handler;
    546     handler.SetLoggingCallback(LogCallbackToTest);
    547 
    548     SECTION("GenericFileEntity")
    549     {
    550         auto CheckEntity = [&](const FileEntity& entity) {
    551             REQUIRE(entity.GetContentType() == ContentType::Generic);
    552             REQUIRE(entity.fileId == c_fileId);
    553             REQUIRE(entity.url == c_url);
    554             REQUIRE(entity.sizeInBytes == c_size);
    555             REQUIRE(entity.hashes.size() == 2);
    556             REQUIRE(entity.hashes.at("Sha1") == c_sha1);
    557             REQUIRE(entity.hashes.at("Sha256") == c_sha256);
    558         };
    559 
    560         SECTION("Correct")
    561         {
    562             const json fileJson = {{"FileId", c_fileId},
    563                                    {"Url", c_url},
    564                                    {"SizeInBytes", c_size},
    565                                    {"Hashes", {{"Sha1", c_sha1}, {"Sha256", c_sha256}}}};
    566 
    567             SECTION("1 file")
    568             {
    569                 const json downloadInfoResponse = json::array({fileJson});
    570                 auto entities = FileEntity::DownloadInfoResponseToFileEntities(downloadInfoResponse, handler);
    571                 REQUIRE(entities.size() == 1);
    572                 REQUIRE(entities[0] != nullptr);
    573                 CheckEntity(*entities[0]);
    574             }
    575 
    576             SECTION("2 files")
    577             {
    578                 const json downloadInfoResponse = json::array({fileJson, fileJson});
    579                 auto entities = FileEntity::DownloadInfoResponseToFileEntities(downloadInfoResponse, handler);
    580                 REQUIRE(entities.size() == 2);
    581                 REQUIRE(entities[0] != nullptr);
    582                 CheckEntity(*entities[0]);
    583                 REQUIRE(entities[1] != nullptr);
    584                 CheckEntity(*entities[1]);
    585             }
    586         }
    587     }
    588 
    589     SECTION("AppFileEntity")
    590     {
    591         const json applicabilityDetails = {{"Architectures", json::array({c_arch})},
    592                                            {"PlatformApplicabilityForPackage", json::array({c_applicability})}};
    593 
    594         auto CheckEntity = [&](const AppFileEntity& entity) {
    595             REQUIRE(entity.GetContentType() == ContentType::App);
    596             REQUIRE(entity.fileId == c_fileId);
    597             REQUIRE(entity.url == c_url);
    598             REQUIRE(entity.sizeInBytes == c_size);
    599             REQUIRE(entity.hashes.size() == 2);
    600             REQUIRE(entity.hashes.at("Sha1") == c_sha1);
    601             REQUIRE(entity.hashes.at("Sha256") == c_sha256);
    602             REQUIRE(entity.fileMoniker == c_fileMoniker);
    603             REQUIRE(entity.applicabilityDetails.architectures.size() == 1);
    604             REQUIRE(entity.applicabilityDetails.architectures[0] == c_arch);
    605             REQUIRE(entity.applicabilityDetails.platformApplicabilityForPackage.size() == 1);
    606             REQUIRE(entity.applicabilityDetails.platformApplicabilityForPackage[0] == c_applicability);
    607         };
    608 
    609         SECTION("Correct")
    610         {
    611             const json fileJson = {{"FileId", c_fileId},
    612                                    {"Url", c_url},
    613                                    {"SizeInBytes", c_size},
    614                                    {"Hashes", {{"Sha1", c_sha1}, {"Sha256", c_sha256}}},
    615                                    {"FileMoniker", c_fileMoniker},
    616                                    {"ApplicabilityDetails", applicabilityDetails}};
    617 
    618             SECTION("1 file")
    619             {
    620                 const json downloadInfoResponse = json::array({fileJson});
    621                 auto entities = FileEntity::DownloadInfoResponseToFileEntities(downloadInfoResponse, handler);
    622                 REQUIRE(entities.size() == 1);
    623                 REQUIRE(entities[0] != nullptr);
    624                 CheckEntity(dynamic_cast<AppFileEntity&>(*entities[0]));
    625             }
    626 
    627             SECTION("2 files")
    628             {
    629                 const json downloadInfoResponse = json::array({fileJson, fileJson});
    630                 auto entities = FileEntity::DownloadInfoResponseToFileEntities(downloadInfoResponse, handler);
    631                 REQUIRE(entities.size() == 2);
    632                 REQUIRE(entities[0] != nullptr);
    633                 CheckEntity(dynamic_cast<AppFileEntity&>(*entities[0]));
    634                 REQUIRE(entities[1] != nullptr);
    635                 CheckEntity(dynamic_cast<AppFileEntity&>(*entities[1]));
    636             }
    637         }
    638     }
