     17 TEST("Testing expected values in curl_version_info_data")
     18 {
     19     curl_version_info_data* ver = curl_version_info(CURLVERSION_NOW);
     20     REQUIRE(ver != nullptr);
     21     CHECK(ver->features & CURL_VERSION_SSL);
     22 
     23     // Checking thread safety is on
     24     CHECK(ver->features & CURL_VERSION_THREADSAFE);
     25 
     26     // For thread safety we need the DNS resolutions to be asynchronous (which happens because of c-ares)
     27     CHECK(ver->features & CURL_VERSION_ASYNCHDNS);
     28 }
