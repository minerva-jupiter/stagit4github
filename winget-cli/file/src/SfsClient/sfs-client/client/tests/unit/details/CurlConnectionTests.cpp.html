     24 class MockCurlConnection : public CurlConnection

     26   public:
     27     MockCurlConnection(const ReportingHandler& handler, Result::Code& responseCode, std::string& response)
     28         : CurlConnection({}, handler)
     29         , m_responseCode(responseCode)
     30         , m_response(response)
     31     {
     32     }

     34   protected:
     35     std::string CurlPerform(const std::string&, CurlHeaderList&) override

     37         if (m_responseCode == Result::Success)

     42     }

     46     std::string& m_response;

     50 TEST("Testing CurlConnection()")
     51 {
     52     ReportingHandler handler;
     53     handler.SetLoggingCallback(LogCallbackToTest);
     54     Result::Code responseCode = Result::HttpNotFound;
     55     std::string response = "expected";
     56     std::unique_ptr<Connection> connection = std::make_unique<MockCurlConnection>(handler, responseCode, response);
     57 
     58     SECTION("Testing CurlConnection::Get()")
     59     {
     60         std::string out;
     61         REQUIRE_THROWS_CODE(out = connection->Get("url"), HttpNotFound);
     62         REQUIRE(out.empty());
     63 
     64         responseCode = Result::Success;
     65         REQUIRE_NOTHROW(out = connection->Get("url"));
     66         REQUIRE(out == response);
     67     }
     68 
     69     SECTION("Testing CurlConnection::Post()")
     70     {
     71         std::string out;
     72         REQUIRE_THROWS_CODE(out = connection->Post("url"), HttpNotFound);
     73         REQUIRE(out.empty());
     74 
     75         const json body = {{{"dummy", {}}}};
     76         REQUIRE_THROWS_CODE(out = connection->Post("url", body.dump()), HttpNotFound);
     77         REQUIRE(out.empty());
     78 
     79         responseCode = Result::Success;
     80         REQUIRE_NOTHROW(out = connection->Get("url"));
     81         REQUIRE(out == response);
     82 
     83         response.clear();
     84         REQUIRE_NOTHROW(out = connection->Post("url", body.dump()));
     85         REQUIRE(out == response);
     86     }
     87 }
