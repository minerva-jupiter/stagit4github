     19 std::string TimestampToString(std::chrono::time_point<std::chrono::system_clock> time)
     20 {
     21     using namespace std::chrono;
     22 
     23     // get number of milliseconds for the current second
     24     // (remainder after division into seconds)
     25     auto ms = duration_cast<milliseconds>(time.time_since_epoch()) % 1000;
     26 
     27     auto timer = system_clock::to_time_t(time);
     28 
     29     std::stringstream timeStream;
     30     struct tm gmTime;
     31 #ifdef _WIN32
     32     gmtime_s(&gmTime, &timer); // gmtime_s is the safe version of gmtime, not available on Linux
     33 #else
     34     gmTime = (*std::gmtime(&timer));
     35 #endif
     36     timeStream << std::put_time(&gmTime, "%F %X"); // yyyy-mm-dd HH:MM:SS
     37     timeStream << '.' << std::setfill('0') << std::setw(3) << ms.count();
     38 
     39     return timeStream.str();
     40 }
