     52 void ProxyServerImpl::ConfigureRequestHandlers()
     53 {
     54     auto HandleRequest = [&](const httplib::Request& req, httplib::Response& res) {
     55         // As a proxy, we'll parse the URL and Path/Query so we can reuse them in httplib::Client
     56         ReportingHandler handler;
     57         UrlBuilder urlBuilder(req.target.c_str(), handler);
     58         const std::string path = urlBuilder.GetPath();
     59         const std::string query = urlBuilder.GetQuery();
     60         urlBuilder.ResetPath().ResetQuery();
     61 
     62         // URL may come back from UrlBuilder with a final /, which doesn't work with httplib::Client, so we remove it
     63         auto url = urlBuilder.GetUrl();
     64         if (url.at(url.size() - 1) == '/')
     65         {
     66             url.pop_back();
     67         }
     68         const std::string pathAndQuery = path + (query.empty() ? "" : ("?" + query));
     69         httplib::Client cli(url);
     70         httplib::Result result;
     71         if (req.method == "GET")
     72         {
     73             result = cli.Get(pathAndQuery, req.headers);
     74         }
     75         else if (req.method == "POST")
     76         {
     77             const auto length = std::stoi(req.get_header_value("Content-Length"));
     78             result =
     79                 cli.Post(pathAndQuery, req.headers, req.body.c_str(), length, req.get_header_value("Content-Type"));
     80         }
     81 
     82         if (!result)
     83         {
     84             BUFFER_LOG("Client error: " + to_string(result.error()));
     85             throw StatusCodeException(httplib::StatusCode::InternalServerError_500);
     86         }
     87         res = result.value();
     88     };
     89 
     90     m_server.Get(".*", HandleRequest);
     91     m_server.Post(".*", HandleRequest);
     92 }
