     75 std::string TimestampToHttpDateString(std::chrono::time_point<std::chrono::system_clock> time)
     76 {
     77     auto timer = system_clock::to_time_t(time);
     78 
     79     std::stringstream timeStream;
     80     struct tm gmTime;
     81 #ifdef _WIN32
     82     gmtime_s(&gmTime, &timer); // gmtime_s is the safe version of gmtime, not available on Linux
     83 #else
     84     gmTime = (*std::gmtime(&timer));
     85 #endif
     86     timeStream << std::put_time(&gmTime, "%a, %d %b %Y %X GMT"); // day, dd Mmm yyyy HH:MM:SS
     87 
     88     return timeStream.str();
     89 }

    412 TEST("Testing retry behavior")
    413 {
    414     if (!AreTestOverridesAllowed())
    415     {
    416         INFO("Skipping. Test overrides not enabled");
    417         return;
    418     }
    419 
    420     ReportingHandler handler;
    421     handler.SetLoggingCallback(LogCallbackToTest);
    422 
    423     MockWebServer server;
    424     CurlConnectionManager connectionManager(handler);
    425     SFSUrlBuilder urlBuilder(SFSCustomUrl(server.GetBaseUrl()), c_instanceId, c_namespace, handler);

    428     const std::string url = urlBuilder.GetSpecificVersionUrl(c_productName, c_version);
    429 
    430     SECTION("Test exponential backoff")
    431     {
    432         auto connection = connectionManager.MakeConnection({});
    433 
    434         INFO("Sets the retry delay to 50ms to speed up the test");
    435         ScopedTestOverride override(TestOverride::BaseRetryDelayMs, 50);
    436 
    437         const int retriableError = 503; // ServerBusy
    438 
    439         auto RunTimedGet = [&](bool success = true) -> long long {
    440             std::string out;
    441             auto begin = steady_clock::now();
    442             if (success)
    443             {
    444                 REQUIRE_NOTHROW(out = connection->Get(url));
    445                 REQUIRE_FALSE(out.empty());
    446             }
    447             else
    448             {
    449                 REQUIRE_THROWS_CODE(out = connection->Get(url), HttpServiceNotAvailable);
    450                 REQUIRE(out.empty());
    451             }
    452             auto end = steady_clock::now();
    453             return duration_cast<milliseconds>(end - begin).count();
    454         };
    455 
    456         long long allowedTimeDeviation = 200LL;
    457         std::queue<HttpCode> forcedHttpErrors({retriableError});
    458         SECTION("Should take at least 50ms with a single retriable error")
    459         {
    460             server.SetForcedHttpErrors(forcedHttpErrors);
    461             const auto time = RunTimedGet();
    462             REQUIRE(time >= 50LL);
    463             REQUIRE(time < 50LL + allowedTimeDeviation);
    464         }
    465 
    466         forcedHttpErrors.push(retriableError);
    467         SECTION("Should take at least 150ms (50ms + 2*50ms) with two retriable errors")
    468         {
    469             server.SetForcedHttpErrors(forcedHttpErrors);
    470             const auto time = RunTimedGet();
    471             REQUIRE(time >= 150LL);
    472             REQUIRE(time < 150LL + allowedTimeDeviation);
    473         }
    474 
    475         forcedHttpErrors.push(retriableError);
    476         SECTION("Should take at least 300ms (50ms + 2*50ms + 3*50ms) with three retriable errors")
    477         {
    478             server.SetForcedHttpErrors(forcedHttpErrors);
    479             const auto time = RunTimedGet();
    480             REQUIRE(time >= 300LL);
    481             REQUIRE(time < 300LL + allowedTimeDeviation);
    482         }
    483 
    484         forcedHttpErrors.push(retriableError);
    485         SECTION("Should take at least 300ms (50ms + 2*50ms + 3*50ms) with four retriable errors, but fail")
    486         {
    487             server.SetForcedHttpErrors(forcedHttpErrors);
    488             const auto time = RunTimedGet(false /*success*/);
    489             REQUIRE(time >= 300LL);
    490             REQUIRE(time < 300LL + allowedTimeDeviation);
    491         }
