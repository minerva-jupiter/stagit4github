     40 # Use this function to set warning level and warnings as errors for a given
     41 # target
     42 function(set_compile_options_for_target target)
     43     if(MSVC)
     44         # Enable some MSVC warnings that are off by default:
     45         #
     46         # * 4062: all enums are handled in switch statements without a default
     47         #   label
     48         # * 4191: unsafe conversion from 'type of expression' to 'type required'
     49         # * 4242: possible loss of data (identifier)
     50         # * 4254: possible loss of data (operator)
     51         # * 4287: 'operator': unsigned/negative constant mismatch
     52         # * 4296: 'operator': expression is always true (or false)
     53         # * 4388: signed/unsigned mismatch
     54         # * 4800: implicit conversion to bool; possible information loss
     55         # * 4946: reinterpret_cast used between related classes
     56 
     57         # cmake-format: off
     58         target_compile_options(${target} PRIVATE /W4 /WX /we4062 /we4191 /we4242 /we4254 /we4287 /we4296 /we4388 /we4800 /we4946)
     59         # cmake-format: on
     60     else()
     61         target_compile_options(
     62             ${target}
     63             PRIVATE -Wall
     64                     -Wextra
     65                     -Wpedantic
     66                     -Werror)
     67     endif()
     68 endfunction()
