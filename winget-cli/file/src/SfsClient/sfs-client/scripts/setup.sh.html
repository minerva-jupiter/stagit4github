     32 install_with_apt() {
     33     program_to_install=$1

     35     if ! sudo apt-get -qq install "$program_to_install"; then

     54 install_pip_dependencies() {
     55     header "\nInstalling dependencies using pip"
     56 
     57     # Upgrade pip and install requirements. Filter out output for dependencies that are already installed
     58     python3 -m pip install --upgrade pip | grep -v -e "already satisfied" -e "Defaulting to user installation"
     59     ret="${PIPESTATUS[0]}"
     60     if [ "$ret" -ne 0 ]; then
     61         return "$ret"
     62     fi
     63 
     64     pip_reqs="$script_dir/pip.requirements.txt"
     65 
     66     pip install -r "$pip_reqs" | grep -v -e "already satisfied" -e "Defaulting to user installation"
     67     ret="${PIPESTATUS[0]}"
     68     if [ "$ret" -ne 0 ]; then
     69         return "$ret"
     70     fi
     71 
     72     return 0
     73 }

     84 install_vcpkg() {
     85     header "\nSetting up vcpkg"
     86 
     87     vcpkg_dir="$git_root/vcpkg"
     88     if [ -d "$vcpkg_dir" ]; then
     89         echo "Checking if vcpkg repo has new commits"
     90         git -C "$vcpkg_dir" pull --show-forced-updates | grep -v 'Already up to date'
     91     else
     92         echo "Cloning vcpkg repo"
     93         git clone https://github.com/microsoft/vcpkg "$vcpkg_dir"
     94         # Needed for the bootstrap and for other vcpkg packages
     95         if ! sudo apt-get -qq install curl zip unzip tar pkg-config; then
     96             return
     97         fi
     98         "$vcpkg_dir/bootstrap-vcpkg.sh"
     99     fi
    100     return 0
    101 }

    116 set_git_hooks() {
    117     header "\nSetting Git hooks"
    118 
    119     hook_dest_dir="$git_root/.git/hooks"
    120     declare -A git_hooks=( ["pre-commit-wrapper.sh"]="pre-commit")
    121     for src in "${!git_hooks[@]}"; do
    122         hook_src="$git_root/$src"
    123         hook_dest="$hook_dest_dir/${git_hooks[$src]}"
    124 
    125         # If the destination doesn't exist or is different than the one in the source, we'll copy it over.
    126         if [ ! -f "$hook_dest" ] || ! cmp -s "$hook_src" "$hook_dest"; then
    127             cp -f "$hook_src" "$hook_dest"
    128             echo "Setup git ${git_hooks[$src]} hook with $hook_src"
    129         fi
    130     done
    131 }

    133 set_aliases() {
    134     header "\nSetting aliases"
    135 
    136     python_scripts_dir=$(python3 -c 'import os,sysconfig;print(sysconfig.get_path("scripts",f"{os.name}_user"))')
    137 
    138     declare -A aliases=( ["build"]="$git_root/scripts/build.sh"
    139         ["clang-format"]="$python_scripts_dir/clang-format"
    140         ["cmake-format"]="$python_scripts_dir/cmake-format"
    141         ["test"]="$git_root/scripts/test.sh"
    142         ["vcpkg"]="$git_root/vcpkg/vcpkg")
    143 
    144     for alias in "${!aliases[@]}"; do
    145         target="${aliases[$alias]}"
    146 
    147         # If the alias doesn't exist or is set to something different than the one we want to target, we'll add it.
    148 
    149         # If alias doesn't exist, the alias command returns a non-zero result
    150         alias_does_not_exist=false
    151         alias_exists_not_the_same=false
    152         if ! alias "$alias" > /dev/null 2>&1; then
    153             alias_does_not_exist=true
    154         else
    155             output=$(alias "$alias")
    156             expected_output="alias $alias='$target'"
    157             if [ "$output" != "$expected_output" ]; then
    158                 alias_exists_not_the_same=true
    159             fi
    160         fi
    161 
    162         if $alias_does_not_exist || $alias_exists_not_the_same; then
    163             eval "alias $alias=$target"
    164             echo "Setup alias $alias"
    165         fi
    166     done
    167 }
