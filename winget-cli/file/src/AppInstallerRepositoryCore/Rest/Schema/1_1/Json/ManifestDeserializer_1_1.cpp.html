     44     std::vector<Manifest::AppsAndFeaturesEntry> ManifestDeserializer::DeserializeAppsAndFeaturesEntries(const web::json::array& entries) const
     45     {
     46         std::vector<Manifest::AppsAndFeaturesEntry> result;
     47 
     48         for (auto& arpEntryNode : entries)
     49         {
     50             AppsAndFeaturesEntry arpEntry;
     51             arpEntry.DisplayName = JSON::GetRawStringValueFromJsonNode(arpEntryNode, JSON::GetUtilityString(DisplayName)).value_or("");
     52             arpEntry.Publisher = JSON::GetRawStringValueFromJsonNode(arpEntryNode, JSON::GetUtilityString(Publisher)).value_or("");
     53             arpEntry.DisplayVersion = JSON::GetRawStringValueFromJsonNode(arpEntryNode, JSON::GetUtilityString(DisplayVersion)).value_or("");
     54             arpEntry.ProductCode = JSON::GetRawStringValueFromJsonNode(arpEntryNode, JSON::GetUtilityString(ProductCode)).value_or("");
     55             arpEntry.UpgradeCode = JSON::GetRawStringValueFromJsonNode(arpEntryNode, JSON::GetUtilityString(UpgradeCode)).value_or("");
     56             arpEntry.InstallerType = ConvertToInstallerType(JSON::GetRawStringValueFromJsonNode(arpEntryNode, JSON::GetUtilityString(InstallerType)).value_or(""));
     57 
     58             // Only add when at least one field is valid
     59             if (!arpEntry.DisplayName.empty() || !arpEntry.Publisher.empty() || !arpEntry.DisplayVersion.empty() ||
     60                 !arpEntry.ProductCode.empty() || !arpEntry.UpgradeCode.empty() || arpEntry.InstallerType != InstallerTypeEnum::Unknown)
     61             {
     62                 result.emplace_back(std::move(arpEntry));
     63             }
     64         }
     65 
     66         return result;
     67     }

     69     Manifest::InstallerTypeEnum ManifestDeserializer::ConvertToInstallerType(std::string_view in) const
     70     {
     71         std::string inStrLower = Utility::ToLower(in);

     75             return InstallerTypeEnum::MSStore;

     77

     92             result = ExpectedReturnCodeEnum::InstallInProgress;
     93         }
     94         else if (inStrLower == "fileinuse")
     95         {
     96             result = ExpectedReturnCodeEnum::FileInUse;
     97         }
     98         else if (inStrLower == "missingdependency")
     99         {
    100             result = ExpectedReturnCodeEnum::MissingDependency;
    101         }
    102         else if (inStrLower == "diskfull")
    103         {
    104             result = ExpectedReturnCodeEnum::DiskFull;
    105         }
    106         else if (inStrLower == "insufficientmemory")
    107         {
    108             result = ExpectedReturnCodeEnum::InsufficientMemory;
    109         }
    110         else if (inStrLower == "nonetwork")
    111         {
    112             result = ExpectedReturnCodeEnum::NoNetwork;
    113         }
    114         else if (inStrLower == "contactsupport")
    115         {
    116             result = ExpectedReturnCodeEnum::ContactSupport;
    117         }
    118         else if (inStrLower == "rebootrequiredtofinish")
    119         {
    120             result = ExpectedReturnCodeEnum::RebootRequiredToFinish;
    121         }
    122         else if (inStrLower == "rebootrequiredforinstall")
    123         {
    124             result = ExpectedReturnCodeEnum::RebootRequiredForInstall;

    157     std::optional<Manifest::ManifestInstaller> ManifestDeserializer::DeserializeInstaller(const web::json::value& installerJsonObject) const
    158     {
    159         auto result = V1_0::Json::ManifestDeserializer::DeserializeInstaller(installerJsonObject);
    160 
    161         if (result)
    162         {
    163             auto& installer = result.value();
    164 
    165             installer.ProductId = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(MSStoreProductIdentifier)).value_or("");
    166             installer.ReleaseDate = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(ReleaseDate)).value_or("");
    167             installer.InstallerAbortsTerminal = JSON::GetRawBoolValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallerAbortsTerminal)).value_or(false);
    168             installer.InstallLocationRequired = JSON::GetRawBoolValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallLocationRequired)).value_or(false);
    169             installer.RequireExplicitUpgrade = JSON::GetRawBoolValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(RequireExplicitUpgrade)).value_or(false);
    170             installer.ElevationRequirement = Manifest::ConvertToElevationRequirementEnum(
    171                 JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(ElevationRequirement)).value_or(""));
    172 
    173             // list of unsupported OS architectures
    174             std::optional<std::reference_wrapper<const web::json::array>> unsupportedOSArchitectures = JSON::GetRawJsonArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(UnsupportedOSArchitectures));
    175             if (unsupportedOSArchitectures)
    176             {
    177                 for (auto& archValue : unsupportedOSArchitectures.value().get())
    178                 {
    179                     std::optional<std::string> arch = JSON::GetRawStringValueFromJsonValue(archValue);
    180                     if (JSON::IsValidNonEmptyStringValue(arch))
    181                     {
    182                         auto archEnum = Utility::ConvertToArchitectureEnum(arch.value());
    183 
    184                         if (archEnum == Utility::Architecture::Neutral)
    185                         {
    186                             AICLI_LOG(Repo, Error, << "Unsupported OS architectures cannot contain neutral value.");
    187                             return {};
    188                         }
    189 
    190                         if (archEnum != Utility::Architecture::Unknown)
    191                         {
    192                             installer.UnsupportedOSArchitectures.emplace_back(archEnum);
    193                         }
    194                     }
    195                 }
    196             }
    197 
    198             // Apps and Features Entries
    199             std::optional<std::reference_wrapper<const web::json::array>> arpEntriesNode = JSON::GetRawJsonArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(AppsAndFeaturesEntries));
    200             if (arpEntriesNode)
    201             {
    202                 installer.AppsAndFeaturesEntries = DeserializeAppsAndFeaturesEntries(arpEntriesNode.value());
    203             }
    204 
    205             // Markets
    206             std::optional<std::reference_wrapper<const web::json::value>> marketsNode = JSON::GetJsonValueFromNode(installerJsonObject, JSON::GetUtilityString(Markets));
    207             if (marketsNode && !marketsNode.value().get().is_null())
    208             {
    209                 installer.Markets.ExcludedMarkets = V1_0::Json::ManifestDeserializer::ConvertToManifestStringArray(
    210                     JSON::GetRawStringArrayFromJsonNode(marketsNode.value().get(), JSON::GetUtilityString(ExcludedMarkets)));
    211                 installer.Markets.AllowedMarkets = V1_0::Json::ManifestDeserializer::ConvertToManifestStringArray(
    212                     JSON::GetRawStringArrayFromJsonNode(marketsNode.value().get(), JSON::GetUtilityString(AllowedMarkets)));
    213             }
    214 
    215             // Expected return codes
    216             std::optional<std::reference_wrapper<const web::json::array>> expectedReturnCodesNode = JSON::GetRawJsonArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(ExpectedReturnCodes));
    217             if (expectedReturnCodesNode)
    218             {
    219                 for (auto& returnCodeNode : expectedReturnCodesNode.value().get())
    220                 {
    221                     DWORD installerReturnCode = static_cast<DWORD>(JSON::GetRawIntValueFromJsonNode(returnCodeNode, JSON::GetUtilityString(InstallerReturnCode)).value_or(0));
    222                     auto returnCodeInfo = DeserializeExpectedReturnCodeInfo(returnCodeNode);
    223 
    224                     // Only add when it is valid
    225                     if (installerReturnCode != 0 && returnCodeInfo.ReturnResponseEnum != ExpectedReturnCodeEnum::Unknown)
    226                     {
    227                         if (!installer.ExpectedReturnCodes.insert({ installerReturnCode, std::move(returnCodeInfo) }).second)
    228                         {
    229                             AICLI_LOG(Repo, Error, << "Expected return codes cannot have repeated value.");
    230                             return {};
    231                         }
    232                     }
    233                 }
    234             }
    235 
    236             // Populate installer default return codes if not present in ExpectedReturnCodes and InstallerSuccessCodes
    237             auto defaultReturnCodes = GetDefaultKnownReturnCodes(installer.EffectiveInstallerType());
    238             for (auto const& defaultReturnCode : defaultReturnCodes)
    239             {
    240                 if (installer.ExpectedReturnCodes.find(defaultReturnCode.first) == installer.ExpectedReturnCodes.end() &&
    241                     std::find(installer.InstallerSuccessCodes.begin(), installer.InstallerSuccessCodes.end(), defaultReturnCode.first) == installer.InstallerSuccessCodes.end())
    242                 {
    243                     installer.ExpectedReturnCodes[defaultReturnCode.first].ReturnResponseEnum = defaultReturnCode.second;
    244                 }
    245             }
    246         }
    247 
    248         return result;
    249     }

    251     std::optional<Manifest::ManifestLocalization> ManifestDeserializer::DeserializeLocale(const web::json::value& localeJsonObject) const
    252     {
    253         auto result = V1_0::Json::ManifestDeserializer::DeserializeLocale(localeJsonObject);
    254 
    255         if (result)
    256         {
    257             auto& locale = result.value();
    258 
    259             TryParseStringLocaleField<Manifest::Localization::ReleaseNotes>(locale, localeJsonObject, ReleaseNotes);
    260             TryParseStringLocaleField<Manifest::Localization::ReleaseNotesUrl>(locale, localeJsonObject, ReleaseNotesUrl);
    261 
    262             // Agreements
    263             auto agreementsNode = JSON::GetRawJsonArrayFromJsonNode(localeJsonObject, JSON::GetUtilityString(Agreements));
    264             if (agreementsNode)
    265             {
    266                 std::vector<Manifest::Agreement> agreements;
    267                 for (auto const& agreementNode : agreementsNode.value().get())
    268                 {
    269                     Manifest::Agreement agreementEntry;
    270 
    271                     agreementEntry.Label = JSON::GetRawStringValueFromJsonNode(agreementNode, JSON::GetUtilityString(AgreementLabel)).value_or("");
    272                     agreementEntry.AgreementText = JSON::GetRawStringValueFromJsonNode(agreementNode, JSON::GetUtilityString(Agreement)).value_or("");
    273                     agreementEntry.AgreementUrl = JSON::GetRawStringValueFromJsonNode(agreementNode, JSON::GetUtilityString(AgreementUrl)).value_or("");
    274 
    275                     if (!agreementEntry.Label.empty() || !agreementEntry.AgreementText.empty() || !agreementEntry.AgreementUrl.empty())
    276                     {
    277                         agreements.emplace_back(std::move(agreementEntry));
    278                     }
    279                 }
    280 
    281                 if (!agreements.empty())
    282                 {
    283                     locale.Add<Manifest::Localization::Agreements>(std::move(agreements));
    284                 }
    285             }
    286         }
    287 
    288         return result;
    289     }
