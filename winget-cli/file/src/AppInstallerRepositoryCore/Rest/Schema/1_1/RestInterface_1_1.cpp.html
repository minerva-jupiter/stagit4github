     50     std::map<std::string_view, std::string> Interface::GetValidatedQueryParams(const std::map<std::string_view, std::string>& params) const
     51     {
     52         std::map<std::string_view, std::string> result = params;
     53 
     54         for (auto const& param : m_information.RequiredQueryParameters)
     55         {
     56             if (params.end() == std::find_if(params.begin(), params.end(), [&](const auto& pair) { return Utility::CaseInsensitiveEquals(pair.first, param); }))
     57             {
     58                 if (Utility::CaseInsensitiveEquals(param, MarketQueryParam))
     59                 {
     60                     result.emplace(MarketQueryParam, Runtime::GetOSRegion());
     61                     continue;
     62                 }
     63 
     64                 AICLI_LOG(Repo, Error, << "Search request is not supported by the rest source. Required query Parameter: " << param);
     65                 throw UnsupportedRequestException({}, {}, {}, m_information.RequiredQueryParameters);
     66             }
     67         }
     68 
     69         for (auto const& param : m_information.UnsupportedQueryParameters)
     70         {
     71             if (params.end() != std::find_if(params.begin(), params.end(), [&](const auto& pair) { return Utility::CaseInsensitiveEquals(pair.first, param); }))
     72             {
     73                 AICLI_LOG(Repo, Error, << "Search request is not supported by the rest source. Unsupported query Parameter: " << param);
     74                 throw UnsupportedRequestException({}, {}, m_information.UnsupportedQueryParameters, {});
     75             }
     76         }
     77 
     78         return result;
     79     }

     81     web::json::value Interface::GetValidatedSearchBody(const SearchRequest& searchRequest) const
     82     {
     83         SearchRequest resultSearchRequest = searchRequest;
     84 
     85         for (auto const& field : m_information.RequiredPackageMatchFields)
     86         {
     87             PackageMatchField matchField = ConvertStringToPackageMatchField(field);
     88 
     89             if (searchRequest.Filters.end() == std::find_if(searchRequest.Filters.begin(), searchRequest.Filters.end(), [&](const PackageMatchFilter& filter) { return filter.Field == matchField; }))
     90             {
     91                 if (matchField == PackageMatchField::Market)
     92                 {
     93                     resultSearchRequest.Filters.emplace_back(PackageMatchFilter(PackageMatchField::Market, MatchType::CaseInsensitive, Runtime::GetOSRegion()));
     94                     continue;
     95                 }
     96 
     97                 AICLI_LOG(Repo, Error, << "Search request is not supported by the rest source. Required package match field: " << field);
     98                 throw UnsupportedRequestException({}, m_information.RequiredPackageMatchFields, {}, {});
     99             }
    100         }
    101 
    102         for (auto const& field : m_information.UnsupportedPackageMatchFields)
    103         {
    104             PackageMatchField matchField = ConvertStringToPackageMatchField(field);
    105 
    106             if (matchField == PackageMatchField::Unknown)
    107             {
    108                 continue;
    109             }
    110 
    111             if (searchRequest.Inclusions.end() != std::find_if(searchRequest.Inclusions.begin(), searchRequest.Inclusions.end(), [&](const PackageMatchFilter& inclusion) { return inclusion.Field == matchField; }))
    112             {
    113                 AICLI_LOG(Repo, Info, << "Search request Inclusions contains package match field not supported by the rest source. Ignoring the field. Unsupported package match field: " << field);
    114 
    115                 resultSearchRequest.Inclusions.erase(
    116                     std::remove_if(
    117                         resultSearchRequest.Inclusions.begin(), resultSearchRequest.Inclusions.end(),
    118                         [&](const PackageMatchFilter& inclusion) { return inclusion.Field == matchField; }),
    119                     resultSearchRequest.Inclusions.end());
    120             }
    121 
    122             if (searchRequest.Filters.end() != std::find_if(searchRequest.Filters.begin(), searchRequest.Filters.end(), [&](const PackageMatchFilter& filter) { return filter.Field == matchField; }))
    123             {
    124                 AICLI_LOG(Repo, Error, << "Search request is not supported by the rest source. Unsupported package match field: " << field);
    125                 throw UnsupportedRequestException(m_information.UnsupportedPackageMatchFields, {}, {}, {});
    126             }
    127         }
    128 
    129         return V1_0::Interface::GetValidatedSearchBody(resultSearchRequest);
    130     }

    139             auto unsupportedPackageMatchFields = JSON::GetRawStringArrayFromJsonNode(searchResponseObject, JSON::GetUtilityString(UnsupportedPackageMatchFields));

    144                 throw UnsupportedRequestException(std::move(unsupportedPackageMatchFields), std::move(requiredPackageMatchFields), {}, {});
    145             }
    146         }

    148         return result;
    149     }

    170     PackageMatchField Interface::ConvertStringToPackageMatchField(std::string_view field) const
    171     {
    172         std::string toLower = Utility::ToLower(field);
    173 
    174         if (toLower == "command")
    175         {
    176             return PackageMatchField::Command;
    177         }
    178         else if (toLower == "packageidentifier")
    179         {
    180             return PackageMatchField::Id;
    181         }
    182         else if (toLower == "moniker")
    183         {
    184             return PackageMatchField::Moniker;
    185         }
    186         else if (toLower == "packagename")
    187         {
    188             return PackageMatchField::Name;
    189         }
    190         else if (toLower == "tag")
    191         {
    192             return PackageMatchField::Tag;
    193         }
    194         else if (toLower == "packagefamilyname")
    195         {
    196             return PackageMatchField::PackageFamilyName;
    197         }
    198         else if (toLower == "productcode")
    199         {
    200             return PackageMatchField::ProductCode;
    201         }
    202         else if (toLower == "normalizedpackagenameandpublisher")
    203         {
    204             return PackageMatchField::NormalizedNameAndPublisher;
    205         }
    206         else if (toLower == "market")
    207         {
    208             return PackageMatchField::Market;
    209         }
    210 
    211         return PackageMatchField::Unknown;
    212     }
