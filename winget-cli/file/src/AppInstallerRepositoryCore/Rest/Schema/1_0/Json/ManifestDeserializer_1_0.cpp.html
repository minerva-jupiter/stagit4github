    101     std::vector<Manifest::Manifest> ManifestDeserializer::Deserialize(const web::json::value& responseJsonObject) const
    102     {
    103         if (responseJsonObject.is_null())
    104         {
    105             AICLI_LOG(Repo, Error, << "Missing json object.");
    106             THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    107         }
    108 
    109         try
    110         {
    111             std::optional<std::reference_wrapper<const web::json::value>> manifestObject =
    112                 JSON::GetJsonValueFromNode(responseJsonObject, JSON::GetUtilityString(Data));
    113 
    114             if (!manifestObject || manifestObject.value().get().is_null())
    115             {
    116                 AICLI_LOG(Repo, Verbose, << "No manifest results returned.");
    117                 return {};
    118             }
    119 
    120             return DeserializeData(manifestObject.value());
    121         }
    122         catch (const wil::ResultException&)
    123         {
    124             throw;
    125         }
    126         catch (const std::exception& e)
    127         {
    128             AICLI_LOG(Repo, Error, << "Error encountered while deserializing manifest. Reason: " << e.what());
    129         }
    130         catch (...)
    131         {
    132             AICLI_LOG(Repo, Error, << "Error encountered while deserializing manifest...");
    133         }
    134 
    135         // If we make it here, there was an exception above that we didn't throw.
    136         // This will convert it into our standard error.
    137         THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    138     }

    140     std::vector<Manifest::Manifest> ManifestDeserializer::DeserializeData(const web::json::value& dataJsonObject) const
    141     {
    142         THROW_HR_IF(E_INVALIDARG, dataJsonObject.is_null());
    143 
    144         std::vector<Manifest::Manifest> manifests;
    145 
    146         std::optional<std::string> id = JSON::GetRawStringValueFromJsonNode(dataJsonObject, JSON::GetUtilityString(PackageIdentifier));
    147         if (!JSON::IsValidNonEmptyStringValue(id))
    148         {
    149             AICLI_LOG(Repo, Error, << "Missing package identifier.");
    150             THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    151         }
    152 
    153         std::optional<std::reference_wrapper<const web::json::array>> versions = JSON::GetRawJsonArrayFromJsonNode(dataJsonObject, JSON::GetUtilityString(Versions));
    154         if (!versions || versions.value().get().size() == 0)
    155         {
    156             AICLI_LOG(Repo, Error, << "Missing versions in package: " << id.value());
    157             THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    158         }
    159 
    160         for (auto& versionItem : versions.value().get())
    161         {
    162             Manifest::Manifest manifest;
    163 
    164             manifest.ManifestVersion = GetManifestVersion();
    165 
    166             manifest.Id = id.value();
    167 
    168             std::optional<std::string> packageVersion = JSON::GetRawStringValueFromJsonNode(versionItem, JSON::GetUtilityString(PackageVersion));
    169             if (!JSON::IsValidNonEmptyStringValue(packageVersion))
    170             {
    171                 AICLI_LOG(Repo, Error, << "Missing package version in package: " << manifest.Id);
    172                 THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    173             }
    174             manifest.Version = std::move(packageVersion.value());
    175 
    176             manifest.Channel = JSON::GetRawStringValueFromJsonNode(versionItem, JSON::GetUtilityString(Channel)).value_or("");
    177 
    178             // Default locale
    179             std::optional<std::reference_wrapper<const web::json::value>> defaultLocale =
    180                 JSON::GetJsonValueFromNode(versionItem, JSON::GetUtilityString(DefaultLocale));
    181             if (!defaultLocale)
    182             {
    183                 AICLI_LOG(Repo, Error, << "Missing default locale in package: " << manifest.Id);
    184                 THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    185             }
    186             else
    187             {
    188                 std::optional<Manifest::ManifestLocalization> defaultLocaleObject = DeserializeLocale(defaultLocale.value().get());
    189                 if (!defaultLocaleObject)
    190                 {
    191                     AICLI_LOG(Repo, Error, << "Missing default locale in package: " << manifest.Id);
    192                     THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    193                 }
    194 
    195                 if (!defaultLocaleObject.value().Contains(Manifest::Localization::PackageName) ||
    196                     !defaultLocaleObject.value().Contains(Manifest::Localization::Publisher) ||
    197                     !defaultLocaleObject.value().Contains(Manifest::Localization::ShortDescription))
    198                 {
    199                     AICLI_LOG(Repo, Error, << "Missing PackageName, Publisher or ShortDescription in default locale: " << manifest.Id);
    200                     THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    201                 }
    202 
    203                 manifest.DefaultLocalization = std::move(defaultLocaleObject.value());
    204 
    205                 // Moniker is in Default locale
    206                 manifest.Moniker = JSON::GetRawStringValueFromJsonNode(defaultLocale.value().get(), JSON::GetUtilityString(Moniker)).value_or("");
    207             }
    208 
    209             // Installers
    210             std::optional<std::reference_wrapper<const web::json::array>> installers = JSON::GetRawJsonArrayFromJsonNode(versionItem, JSON::GetUtilityString(Installers));
    211             if (!installers || installers.value().get().size() == 0)
    212             {
    213                 AICLI_LOG(Repo, Error, << "Missing installers in package: " << manifest.Id);
    214                 THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    215             }
    216 
    217             for (auto& installer : installers.value().get())
    218             {
    219                 std::optional<Manifest::ManifestInstaller> installerObject = DeserializeInstaller(installer);
    220                 if (installerObject)
    221                 {
    222                     // Merge default switches after parsing.
    223                     auto defaultSwitches = Manifest::GetDefaultKnownSwitches(installerObject->EffectiveInstallerType());
    224                     for (auto const& defaultSwitch : defaultSwitches)
    225                     {
    226                         if (installerObject->Switches.find(defaultSwitch.first) == installerObject->Switches.end())
    227                         {
    228                             installerObject->Switches[defaultSwitch.first] = defaultSwitch.second;
    229                         }
    230                     }
    231 
    232                     manifest.Installers.emplace_back(std::move(installerObject.value()));
    233                 }
    234             }
    235 
    236             if (manifest.Installers.size() == 0)
    237             {
    238                 AICLI_LOG(Repo, Error, << "Missing valid installers in package: " << manifest.Id);
    239                 THROW_HR(APPINSTALLER_CLI_ERROR_RESTSOURCE_INVALID_DATA);
    240             }
    241 
    242             // Other locales
    243             std::optional<std::reference_wrapper<const web::json::array>> locales = JSON::GetRawJsonArrayFromJsonNode(versionItem, JSON::GetUtilityString(Locales));
    244             if (locales)
    245             {
    246                 for (auto& locale : locales.value().get())
    247                 {
    248                     std::optional<Manifest::ManifestLocalization> localeObject = DeserializeLocale(locale);
    249                     if (localeObject)
    250                     {
    251                         manifest.Localizations.emplace_back(std::move(localeObject.value()));
    252                     }
    253                 }
    254             }
    255 
    256             manifests.emplace_back(std::move(manifest));
    257         }
    258 
    259         return manifests;
    260     }

    272     std::optional<Manifest::ManifestLocalization> ManifestDeserializer::DeserializeLocale(const web::json::value& localeJsonObject) const
    273     {
    274         if (localeJsonObject.is_null())
    275         {
    276             return {};
    277         }
    278 
    279         Manifest::ManifestLocalization locale;
    280         std::optional<std::string> packageLocale = JSON::GetRawStringValueFromJsonNode(localeJsonObject, JSON::GetUtilityString(PackageLocale));
    281         if (!JSON::IsValidNonEmptyStringValue(packageLocale))
    282         {
    283             AICLI_LOG(Repo, Error, << "Missing package locale.");
    284             return {};
    285         }
    286         locale.Locale = std::move(packageLocale.value());
    287 
    288         TryParseStringLocaleField<Manifest::Localization::PackageName>(locale, localeJsonObject, PackageName);
    289         TryParseStringLocaleField<Manifest::Localization::Publisher>(locale, localeJsonObject, Publisher);
    290         TryParseStringLocaleField<Manifest::Localization::ShortDescription>(locale, localeJsonObject, ShortDescription);
    291         TryParseStringLocaleField<Manifest::Localization::PublisherUrl>(locale, localeJsonObject, PublisherUrl);
    292         TryParseStringLocaleField<Manifest::Localization::PublisherSupportUrl>(locale, localeJsonObject, PublisherSupportUrl);
    293         TryParseStringLocaleField<Manifest::Localization::PrivacyUrl>(locale, localeJsonObject, PrivacyUrl);
    294         TryParseStringLocaleField<Manifest::Localization::Author>(locale, localeJsonObject, Author);
    295         TryParseStringLocaleField<Manifest::Localization::PackageUrl>(locale, localeJsonObject, PackageUrl);
    296         TryParseStringLocaleField<Manifest::Localization::License>(locale, localeJsonObject, License);
    297         TryParseStringLocaleField<Manifest::Localization::LicenseUrl>(locale, localeJsonObject, LicenseUrl);
    298         TryParseStringLocaleField<Manifest::Localization::Copyright>(locale, localeJsonObject, Copyright);
    299         TryParseStringLocaleField<Manifest::Localization::CopyrightUrl>(locale, localeJsonObject, CopyrightUrl);
    300         TryParseStringLocaleField<Manifest::Localization::Description>(locale, localeJsonObject, Description);
    301 
    302         auto tags = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(localeJsonObject, JSON::GetUtilityString(Tags)));
    303         if (!tags.empty())
    304         {
    305             locale.Add<AppInstaller::Manifest::Localization::Tags>(tags);
    306         }
    307 
    308         return locale;
    309     }

    311     std::optional<Manifest::ManifestInstaller> ManifestDeserializer::DeserializeInstaller(const web::json::value& installerJsonObject) const
    312     {
    313         if (installerJsonObject.is_null())
    314         {
    315             return {};
    316         }
    317 
    318         Manifest::ManifestInstaller installer;
    319 
    320         installer.Url = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallerUrl)).value_or("");
    321 
    322         std::optional<std::string> sha256 = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallerSha256));
    323         if (JSON::IsValidNonEmptyStringValue(sha256))
    324         {
    325             installer.Sha256 = Utility::SHA256::ConvertToBytes(sha256.value());
    326         }
    327 
    328         std::optional<std::string> arch = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(Architecture));
    329         if (!JSON::IsValidNonEmptyStringValue(arch))
    330         {
    331             AICLI_LOG(Repo, Error, << "Missing installer architecture.");
    332             return {};
    333         }
    334         installer.Arch = Utility::ConvertToArchitectureEnum(arch.value());
    335 
    336         std::optional<std::string> installerType = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallerType));
    337         if (!JSON::IsValidNonEmptyStringValue(installerType))
    338         {
    339             AICLI_LOG(Repo, Error, << "Missing installer type.");
    340             return {};
    341         }
    342         installer.BaseInstallerType = ConvertToInstallerType(installerType.value());
    343         installer.Locale = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallerLocale)).value_or("");
    344 
    345         // platform
    346         std::optional<std::reference_wrapper<const web::json::array>> platforms = JSON::GetRawJsonArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(Platform));
    347         if (platforms)
    348         {
    349             for (auto& platform : platforms.value().get())
    350             {
    351                 std::optional<std::string> platformValue = JSON::GetRawStringValueFromJsonValue(platform);
    352                 if (platformValue)
    353                 {
    354                     installer.Platform.emplace_back(Manifest::ConvertToPlatformEnum(platformValue.value()));
    355                 }
    356             }
    357         }
    358 
    359         installer.MinOSVersion = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(MinimumOSVersion)).value_or("");
    360         std::optional<std::string> scope = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(Scope));
    361         if (scope)
    362         {
    363             installer.Scope = Manifest::ConvertToScopeEnum(scope.value());
    364         }
    365 
    366         std::optional<std::string> signatureSha256 = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(SignatureSha256));
    367         if (signatureSha256)
    368         {
    369             installer.SignatureSha256 = Utility::SHA256::ConvertToBytes(signatureSha256.value());
    370         }
    371 
    372         // Install modes
    373         std::optional<std::reference_wrapper<const web::json::array>> installModes = JSON::GetRawJsonArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallModes));
    374         if (installModes)
    375         {
    376             for (auto& mode : installModes.value().get())
    377             {
    378                 std::optional<std::string> modeObject = JSON::GetRawStringValueFromJsonValue(mode);
    379                 if (modeObject)
    380                 {
    381                     installer.InstallModes.emplace_back(Manifest::ConvertToInstallModeEnum(modeObject.value()));
    382                 }
    383             }
    384         }
    385 
    386         // Installer Switches
    387         std::optional<std::reference_wrapper<const web::json::value>> switches =
    388             JSON::GetJsonValueFromNode(installerJsonObject, JSON::GetUtilityString(InstallerSwitches));
    389         if (switches)
    390         {
    391             installer.Switches = DeserializeInstallerSwitches(switches.value().get());
    392         }
    393 
    394         // Installer SuccessCodes
    395         std::optional<std::reference_wrapper<const web::json::array>> installSuccessCodes = JSON::GetRawJsonArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(InstallerSuccessCodes));
    396         if (installSuccessCodes)
    397         {
    398             for (auto& code : installSuccessCodes.value().get())
    399             {
    400                 std::optional<int> codeValue = JSON::GetRawIntValueFromJsonValue(code);
    401                 if (codeValue)
    402                 {
    403                     installer.InstallerSuccessCodes.emplace_back(std::move(codeValue.value()));
    404                 }
    405             }
    406         }
    407 
    408         std::optional<std::string> updateBehavior = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(UpgradeBehavior));
    409         if (updateBehavior)
    410         {
    411             installer.UpdateBehavior = ConvertToUpdateBehavior(updateBehavior.value());
    412         }
    413 
    414         installer.Commands = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(Commands)));
    415         installer.Protocols = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(Protocols)));
    416         installer.FileExtensions = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(FileExtensions)));
    417 
    418         // Dependencies
    419         std::optional<std::reference_wrapper<const web::json::value>> dependenciesObject =
    420             JSON::GetJsonValueFromNode(installerJsonObject, JSON::GetUtilityString(Dependencies));
    421         if (dependenciesObject)
    422         {
    423             std::optional<Manifest::DependencyList> dependencyList = DeserializeDependency(dependenciesObject.value().get());
    424             if (dependencyList)
    425             {
    426                 installer.Dependencies = std::move(dependencyList.value());
    427             }
    428         }
    429 
    430         installer.PackageFamilyName = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(PackageFamilyName)).value_or("");
    431         installer.ProductCode = JSON::GetRawStringValueFromJsonNode(installerJsonObject, JSON::GetUtilityString(ProductCode)).value_or("");
    432         installer.Capabilities = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(Capabilities)));
    433         installer.RestrictedCapabilities = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(installerJsonObject, JSON::GetUtilityString(RestrictedCapabilities)));
    434 
    435         return installer;
    436     }

    453     std::optional<Manifest::DependencyList> ManifestDeserializer::DeserializeDependency(const web::json::value& dependenciesObject) const
    454     {
    455         if (dependenciesObject.is_null())
    456         {
    457             return {};
    458         }
    459 
    460         Manifest::DependencyList dependencyList;
    461 
    462         auto wfIds = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(dependenciesObject, JSON::GetUtilityString(WindowsFeatures)));
    463         for (auto&& id : wfIds)
    464         {
    465             dependencyList.Add(Dependency(DependencyType::WindowsFeature, std::move(id)));
    466         };
    467 
    468         const auto& wlIds = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(dependenciesObject, JSON::GetUtilityString(WindowsLibraries)));
    469         for (auto id : wlIds)
    470         {
    471             dependencyList.Add(Dependency(DependencyType::WindowsLibrary, id));
    472         };
    473 
    474         const auto& extIds = ConvertToManifestStringArray(JSON::GetRawStringArrayFromJsonNode(dependenciesObject, JSON::GetUtilityString(ExternalDependencies)));
    475         for (auto id : extIds)
    476         {
    477             dependencyList.Add(Dependency(DependencyType::External, id));
    478         };
    479 
    480         // Package Dependencies
    481         std::optional<std::reference_wrapper<const web::json::array>> packageDependencies = JSON::GetRawJsonArrayFromJsonNode(dependenciesObject, JSON::GetUtilityString(PackageDependencies));
    482         if (packageDependencies)
    483         {
    484             for (auto& packageDependency : packageDependencies.value().get())
    485             {
    486                 std::optional<std::string> id = JSON::GetRawStringValueFromJsonNode(packageDependency, JSON::GetUtilityString(PackageIdentifier));
    487                 if (id)
    488                 {
    489                     Dependency pkg{ DependencyType::Package, std::move(id.value()) ,  JSON::GetRawStringValueFromJsonNode(packageDependency, JSON::GetUtilityString(MinimumVersion)).value_or("") };
    490                     dependencyList.Add(std::move(pkg));
    491                 }
    492             }
    493         }
    494 
    495         return dependencyList;
    496     }

    498     Manifest::InstallerTypeEnum ManifestDeserializer::ConvertToInstallerType(std::string_view in) const
    499     {
    500         std::string inStrLower = Utility::ToLower(in);
    501 
    502         if (inStrLower == "inno")
    503         {
    504             return InstallerTypeEnum::Inno;
    505         }
    506         else if (inStrLower == "wix")
    507         {
    508             return InstallerTypeEnum::Wix;
    509         }
    510         else if (inStrLower == "msi")
    511         {
    512             return InstallerTypeEnum::Msi;
    513         }
    514         else if (inStrLower == "nullsoft")
    515         {
    516             return InstallerTypeEnum::Nullsoft;
    517         }
