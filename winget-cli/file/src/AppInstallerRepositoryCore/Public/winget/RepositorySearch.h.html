     75     // A match on a specific field to be performed during a search.
     76     struct PackageMatchFilter : public RequestMatch
     77     {
     78         PackageMatchField Field;
     79 
     80         PackageMatchFilter(PackageMatchField f, MatchType t) : RequestMatch(t), Field(f) { EnsureRequiredValues(); }
     81         PackageMatchFilter(PackageMatchField f, MatchType t, Utility::NormalizedString& v) : RequestMatch(t, v), Field(f) { EnsureRequiredValues(); }
     82         PackageMatchFilter(PackageMatchField f, MatchType t, const Utility::NormalizedString& v) : RequestMatch(t, v), Field(f) { EnsureRequiredValues(); }
     83         PackageMatchFilter(PackageMatchField f, MatchType t, Utility::NormalizedString&& v) : RequestMatch(t, std::move(v)), Field(f) { EnsureRequiredValues(); }
     84         PackageMatchFilter(PackageMatchField f, MatchType t, std::string_view v1, std::string_view v2) : RequestMatch(t, v1, v2), Field(f) { EnsureRequiredValues(); }
     85 
     86     protected:
     87         void EnsureRequiredValues()
     88         {
     89             // Ensure that the second value always exists when it should
     90             if (Field == PackageMatchField::NormalizedNameAndPublisher && !Additional)
     91             {
     92                 Additional = Utility::NormalizedString{};
     93             }
     94         }
     95     };

    108     // Container for data used to filter the available manifests in a source.
    109     // It can be thought of as:
    110     //  (Query || Inclusions...) && Filters...
    111     // If Query and Inclusions are both empty, the starting data set will be the entire database.
    112     //  Everything && Filters...
    113     struct SearchRequest
    114     {
    115         // The generic query matches against a source defined set of fields.
    116         std::optional<RequestMatch> Query;
    117 
    118         // Specific fields used to include more data.
    119         // If Query is defined, this can add more rows afterward.
    120         // If Query is not defined, this is the only set of data included.
    121         std::vector<PackageMatchFilter> Inclusions;
    122 
    123         // Specific fields used to filter the data further.
    124         std::vector<PackageMatchFilter> Filters;
    125 
    126         // The search purpose of the search request.
    127         SearchPurpose Purpose = SearchPurpose::Default;
    128 
    129         // The maximum number of results to return.
    130         // The default of 0 will place no limit.
    131         size_t MaximumResults{};
    132 
    133         // Returns a value indicating whether this request is for all available data.
    134         bool IsForEverything() const;
    135 
    136         // Returns a string summarizing the search request.
    137         std::string ToString() const;
    138     };

    371     // Does the equivalent of a dynamic_cast, but without it to allow RTTI to be disabled.
    372     // Example usage:
    373     //  bool IsSame(const IPackage* other) const override
    374     //  {
    375     //      const MyPackage* otherAsMyType = PackageCast<const MyPackage*>(other);
    376     //      ...
    377     //  }
    378     template <typename PackageType>
    379     PackageType PackageCast(const IPackage* package)
    380     {
    381         static_assert(std::is_pointer_v<PackageType>, "The target type of the PackageCast must be a pointer; use the same type as if this were a dynamic_cast.");
    382         if (!package)
    383         {
    384             return nullptr;
    385         }
    386         using ActualPackageType = std::remove_pointer_t<std::remove_cv_t<PackageType>>;
    387         return reinterpret_cast<PackageType>(package->CastTo(ActualPackageType::PackageType));
    388     }
