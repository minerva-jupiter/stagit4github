     12     // Get the relative path and hash for the package version data manifest.
     13     std::pair<std::filesystem::path, std::string> CreatePackageVersionDataRelativePath(const std::shared_ptr<SQLiteIndexSource>& source, SQLiteIndex::IdType packageRowId)
     14     {
     15         const SQLiteIndex& index = source->GetIndex();
     16 
     17         std::string identifier = index.GetPropertyByPrimaryId(packageRowId, PackageVersionProperty::Id).value();
     18         std::string hash = index.GetPropertyByPrimaryId(packageRowId, PackageVersionProperty::ManifestSHA256Hash).value();
     19         std::filesystem::path relativePath = Manifest::PackageVersionDataManifest::GetRelativeDirectoryPath(identifier, hash) / Manifest::PackageVersionDataManifest::VersionManifestCompressedFileName();
     20 
     21         return std::make_pair(std::move(relativePath), std::move(hash));
     22     }

     24     // Gets package version data for the given package in the index.
     25     Manifest::PackageVersionDataManifest GetPackageVersionData(const std::shared_ptr<SQLiteIndexSource>& source, SQLiteIndex::IdType packageRowId, const Caching::FileCache& fileCache)
     26     {
     27         auto pathAndHash = CreatePackageVersionDataRelativePath(source, packageRowId);
     28         auto fileStream = fileCache.GetFile(pathAndHash.first, SHA256::ConvertToBytes(pathAndHash.second));
     29         auto fileBytes = ReadEntireStreamAsByteArray(*fileStream);
     30 
     31         Manifest::PackageVersionDataManifest result;
     32         result.Deserialize(Manifest::PackageVersionDataManifest::CreateDecompressor().Decompress(fileBytes));
     33 
     34         return result;
     35     }

    181         void EnsurePackageVersionData() const
    182         {
    183             {
    184                 auto sharedLock = m_versionAndManifestLock.lock_shared();
    185                 if (m_packageVersionData)
    186                 {
    187                     return;
    188                 }
    189             }
    190 
    191             auto exclusiveLock = m_versionAndManifestLock.lock_exclusive();
    192             if (m_packageVersionData)
    193             {
    194                 return;
    195             }
    196 
    197             Manifest::PackageVersionDataManifest packageVersionDataManifest = GetPackageVersionData(GetReferenceSource(), m_packageRowId, *m_packageVersionDataCache);
    198 
    199             for (const auto& versionData : packageVersionDataManifest.Versions())
    200             {
    201                 // We should only ever be looking for the latest version here.
    202                 if (!m_packageVersionData || m_packageVersionData->Version < versionData.Version)
    203                 {
    204                     m_packageVersionData = versionData;
    205                 }
    206             }

    487     std::shared_ptr<IPackageVersion> SQLitePackage::GetVersion(const PackageVersionKey& versionKey) const
    488     {
    489         std::shared_ptr<SQLiteIndexSource> source = GetReferenceSource();
    490 
    491         // Ensure that this key targets this (or any) source
    492         if (!versionKey.SourceId.empty() && versionKey.SourceId != source->GetIdentifier())
    493         {
    494             return {};
    495         }
    496 
    497         std::optional<Manifest::PackageVersionDataManifest::VersionData> versionData;
    498 
    499         // Check for a latest version request.
    500         if (versionKey.IsDefaultLatest())
    501         {
    502             auto sharedLock = m_versionKeysLock.lock_shared();
    503             return std::make_shared<PackageVersion>(source, m_packageRowId, m_latestVersionData, m_manifestCache, m_packageVersionDataCache);
    504         }
    505 
    506         EnsurePackageVersionData(source);
    507 
    508         {
    509             MapKey requested{ versionKey.Version, versionKey.Channel };
    510             auto sharedLock = m_versionKeysLock.lock_shared();
    511 
    512             auto itr = m_versionKeysMap.find(requested);
    513             if (itr != m_versionKeysMap.end())
    514             {
    515                 versionData = itr->second;
    516             }
    517         }
    518 
    519         if (versionData)
    520         {
    521             return std::make_shared<PackageVersion>(source, m_packageRowId, std::move(versionData), m_manifestCache, m_packageVersionDataCache);
    522         }
    523 
    524         return {};
    525     }

    580     // Ensures that we have the package version data present.
    581     void SQLitePackage::EnsurePackageVersionData(const std::shared_ptr<SQLiteIndexSource>& source) const
    582     {
    583         {
    584             auto sharedLock = m_versionKeysLock.lock_shared();
    585 
    586             if (!m_versionKeys.empty())
    587             {
    588                 return;
    589             }
    590         }
    591 
    592         auto exclusiveLock = m_versionKeysLock.lock_exclusive();
    593 
    594         if (!m_versionKeys.empty())
    595         {
    596             return;
    597         }
    598 
    599         Manifest::PackageVersionDataManifest packageVersionDataManifest = GetPackageVersionData(source, m_packageRowId, *m_packageVersionDataCache);
    600 
    601         for (const auto& versionData : packageVersionDataManifest.Versions())
    602         {
    603             std::string version = versionData.Version.ToString();
    604             std::string channel;
    605             m_versionKeys.emplace_back(source->GetIdentifier(), version, channel);
    606             m_versionKeysMap.emplace(MapKey{ std::move(version), std::move(channel) }, versionData);
    607 
    608             if (!m_latestVersionData || m_latestVersionData->Version < versionData.Version)
    609             {
    610                 m_latestVersionData = versionData;
    611             }
    612         }
    613     }
