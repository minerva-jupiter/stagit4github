     16         void AddNormalizedName(
     17             const Utility::NameNormalizer& normalizer,
     18             const Manifest::string_t& name,
     19             std::vector<Utility::NormalizedString>& out,
     20             Utility::NormalizationField fieldsToInclude = Utility::NormalizationField::None)
     21         {
     22             Utility::NormalizedString value = normalizer.NormalizeName(Utility::FoldCase(name)).GetNormalizedName(fieldsToInclude);
     23             if (std::find(out.begin(), out.end(), value) == out.end())

     54         std::vector<Utility::NormalizedString> GetNormalizedNames(const Utility::NameNormalizer& normalizer, const Manifest::Manifest& manifest)
     55         {
     56             std::vector<Utility::NormalizedString> result;
     57 
     58             AddLocalizationNormalizedName(normalizer, manifest.DefaultLocalization, result);
     59             for (const auto& loc : manifest.Localizations)
     60             {
     61                 AddLocalizationNormalizedName(normalizer, loc, result);
     62             }
     63 
     64             // In addition to the names used for our display, add the display names from the ARP entries
     65             for (const auto& installer : manifest.Installers)
     66             {
     67                 for (const auto& appsAndFeaturesEntry : installer.AppsAndFeaturesEntries)
     68                 {
     69                     if (!appsAndFeaturesEntry.DisplayName.empty())
     70                     {
     71                         AddNormalizedName(normalizer, appsAndFeaturesEntry.DisplayName, result);
     72                         // For arp display name, also add a copy with architecture info for more accurate correlation.
     73                         AddNormalizedName(normalizer, appsAndFeaturesEntry.DisplayName, result, Utility::NormalizationField::Architecture);
     74                     }
     75                 }
     76             }
     77 
     78             return result;
     79         }

     81         std::vector<Utility::NormalizedString> GetNormalizedPublishers(const Utility::NameNormalizer& normalizer, const Manifest::Manifest& manifest)
     82         {
     83             std::vector<Utility::NormalizedString> result;
     84 
     85             AddLocalizationNormalizedPublisher(normalizer, manifest.DefaultLocalization, result);
     86             for (const auto& loc : manifest.Localizations)
     87             {
     88                 AddLocalizationNormalizedPublisher(normalizer, loc, result);
     89             }
     90 
     91             // In addition to the publishers used for our display, add the publishers from the ARP entries
     92             for (const auto& installer : manifest.Installers)
     93             {
     94                 for (const auto& appsAndFeaturesEntry : installer.AppsAndFeaturesEntries)
     95                 {
     96                     if (!appsAndFeaturesEntry.Publisher.empty())
     97                     {
     98                         AddNormalizedPublisher(normalizer, appsAndFeaturesEntry.Publisher, result);
     99                     }
    100                 }
    101             }
    102 
    103             return result;
    104         }

    106         // Update NormalizedNameAndPublisher with normalization and folding
    107         // Returns true if the normalized name contains normalization field of fieldsToInclude
    108         bool UpdateNormalizedNameAndPublisher(
    109             PackageMatchFilter& filter,
    110             const Utility::NameNormalizer& normalizer,
    111             Utility::NormalizationField fieldsToInclude)
    112         {
    113             Utility::NormalizedName normalized = normalizer.Normalize(Utility::FoldCase(filter.Value), Utility::FoldCase(filter.Additional.value()));
    114             filter.Value = normalized.GetNormalizedName(fieldsToInclude);
    115             filter.Additional = normalized.Publisher();
    116             return WI_AreAllFlagsSet(normalized.GetNormalizedFields(), fieldsToInclude);
    117         }
    118 
    119         // Update NormalizedNameAndPublisher with normalization and folding
    120         // Returns true if any of normalized name contains normalization field of fieldsToInclude
    121         bool UpdatePackageMatchFilters(
    122             std::vector<PackageMatchFilter>& filters,
    123             const Utility::NameNormalizer& normalizer,
    124             Utility::NormalizationField normalizedNameFieldsToFilter = Utility::NormalizationField::None)
    125         {
    126             bool normalizedNameFieldsFound = false;
    127             for (auto itr = filters.begin(); itr != filters.end();)
    128             {
    129                 if (itr->Field == PackageMatchField::NormalizedNameAndPublisher && itr->Type == MatchType::Exact)
    130                 {
    131                     if (!UpdateNormalizedNameAndPublisher(*itr, normalizer, normalizedNameFieldsToFilter))
    132                     {
    133                         // If not matched, this package match filter will be removed.
    134                         // For example, if caller is trying to search with arch info only, values without arch will be removed from search.
    135                         itr = filters.erase(itr);
    136                         continue;
    137                     }
    138 
    139                     normalizedNameFieldsFound = true;
    140                 }
    141 
    142                 ++itr;
    143             }
    144 
    145             return normalizedNameFieldsFound;
    146         }
    147     }

    158     void Interface::CreateTables(SQLite::Connection& connection, CreateOptions options)
    159     {
    160         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "createtables_v1_2");
    161 
    162         V1_1::Interface::CreateTables(connection, options);
    163 
    164         // While the name and publisher should be linked per-locale, we are not implementing that here.
    165         // This will mean that one can match cross locale name and publisher, but the chance that this
    166         // leads to a confusion between packages is very small. More likely would be intentional attempts
    167         // to confuse the correlation, which could be fairly easily carried out even with linked values.
    168         NormalizedPackageNameTable::Create(connection, GetOneToManyTableSchema());
    169         NormalizedPackagePublisherTable::Create(connection, GetOneToManyTableSchema());
    170 
    171         savepoint.Commit();
    172     }

    174     SQLite::rowid_t Interface::AddManifest(SQLite::Connection& connection, const Manifest::Manifest& manifest, const std::optional<std::filesystem::path>& relativePath)
    175     {
    176         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "addmanifest_v1_2");
    177 
    178         SQLite::rowid_t manifestId = V1_1::Interface::AddManifest(connection, manifest, relativePath);
    179 
    180         // Add the new 1.2 data
    181         // These normalized strings are all stored with their cases folded so that they can be
    182         // looked up ordinally; enabling the index to provide efficient searches.
    183         NormalizedPackageNameTable::EnsureExistsAndInsert(connection, anon::GetNormalizedNames(m_normalizer, manifest), manifestId);
    184         NormalizedPackagePublisherTable::EnsureExistsAndInsert(connection, anon::GetNormalizedPublishers(m_normalizer, manifest), manifestId);
    185 
    186         savepoint.Commit();
    187 
    188         return manifestId;

    273     ISQLiteIndex::SearchResult Interface::SearchInternal(const SQLite::Connection& connection, SearchRequest& request) const
    274     {
    275         if (request.Purpose == SearchPurpose::CorrelationToInstalled)
    276         {
    277             // Correlate from available package to installed package
    278             // For available package to installed package mapping, only one try is needed.
    279             // For example, if ARP DisplayName contains arch, then the installed package's ARP DisplayName should also include arch.
    280             auto candidateInclusionsWithArch = request.Inclusions;
    281             if (anon::UpdatePackageMatchFilters(candidateInclusionsWithArch, m_normalizer, Utility::NormalizationField::Architecture))
    282             {
    283                 // If DisplayNames contain arch, only use Inclusions with arch for search
    284                 request.Inclusions = candidateInclusionsWithArch;
    285             }
    286             else
    287             {
    288                 // Otherwise, just update the Inclusions with normalization
    289                 anon::UpdatePackageMatchFilters(request.Inclusions, m_normalizer);
    290             }
    291 
    292             return V1_1::Interface::SearchInternal(connection, request);
    293         }
    294         else if (request.Purpose == SearchPurpose::CorrelationToAvailable)
    295         {
    296             // For installed package to available package correlation,
    297             // try the search with NormalizedName with Arch first, if not found, try with all values.
    298             // This can be extended in the future for more granular search requests.
    299             std::vector<SearchRequest> candidateSearches;
    300             auto candidateSearchWithArch = request;
    301             if (anon::UpdatePackageMatchFilters(candidateSearchWithArch.Inclusions, m_normalizer, Utility::NormalizationField::Architecture))
    302             {
    303                 candidateSearches.emplace_back(std::move(candidateSearchWithArch));
    304             }
    305             anon::UpdatePackageMatchFilters(request.Inclusions, m_normalizer);
    306             candidateSearches.emplace_back(request);
    307 
    308             SearchResult result;
    309             for (auto& candidateSearch : candidateSearches)
    310             {
    311                 result = V1_1::Interface::SearchInternal(connection, candidateSearch);
    312                 if (!result.Matches.empty())
    313                 {
    314                     break;
    315                 }
    316             }
    317 
    318             return result;
    319         }
    320         else
    321         {
    322             anon::UpdatePackageMatchFilters(request.Inclusions, m_normalizer);
    323             anon::UpdatePackageMatchFilters(request.Filters, m_normalizer);
    324 
    325             return V1_1::Interface::SearchInternal(connection, request);
    326         }
    327     }
