     26         // Gets an existing manifest by its rowid., if it exists.
     27         std::optional<SQLite::rowid_t> GetExistingManifestId(const SQLite::Connection& connection, const Manifest::Manifest& manifest)
     28         {
     29             std::optional<SQLite::rowid_t> idId = IdTable::SelectIdByValue(connection, manifest.Id, true);
     30             if (!idId)
     31             {
     32                 AICLI_LOG(Repo, Verbose, << "Did not find an Id { " << manifest.Id << " }");
     33                 return {};
     34             }
     35 
     36             std::optional<SQLite::rowid_t> versionId = VersionTable::SelectIdByValue(connection, manifest.Version, true);
     37             if (!versionId)
     38             {
     39                 AICLI_LOG(Repo, Verbose, << "Did not find a Version { " << manifest.Version << " }");
     40                 return {};
     41             }
     42 
     43             std::optional<SQLite::rowid_t> channelId = ChannelTable::SelectIdByValue(connection, manifest.Channel, true);
     44             if (!channelId)
     45             {
     46                 AICLI_LOG(Repo, Verbose, << "Did not find a Channel { " << manifest.Channel << " }");
     47                 return {};
     48             }
     49 
     50             auto result = ManifestTable::SelectByValueIds<IdTable, VersionTable, ChannelTable>(connection, { idId.value(), versionId.value(), channelId.value() });
     51 
     52             if (!result)
     53             {
     54                 AICLI_LOG(Repo, Verbose, << "Did not find a manifest row for { " << manifest.Id << ", " << manifest.Version << ", " << manifest.Channel << " }");
     55             }
     56 
     57             return result;
     58         }

     60         // Gets a manifest id by the given key values.
     61         std::optional<SQLite::rowid_t> StaticGetManifestIdByKey(const SQLite::Connection& connection, SQLite::rowid_t id, std::string_view version = "", std::string_view channel = "")
     62         {
     63             std::optional<SQLite::rowid_t> channelIdOpt = ChannelTable::SelectIdByValue(connection, channel, true);
     64             if (!channelIdOpt && !channel.empty())
     65             {
     66                 AICLI_LOG(Repo, Info, << "Did not find a Channel { " << channel << " }");
     67                 return {};
     68             }
     69 
     70             std::optional<SQLite::rowid_t> versionIdOpt;
     71             std::vector<std::pair<SQLite::rowid_t, std::string>> versionStrings;
     72 
     73             if (channelIdOpt)
     74             {
     75                 versionStrings = ManifestTable::GetAllValuesByIds<VersionTable, IdTable, ChannelTable>(connection, { id, channelIdOpt.value() });
     76             }
     77             else
     78             {
     79                 versionStrings = ManifestTable::GetAllValuesByIds<VersionTable, IdTable>(connection, { id });
     80             }
     81 
     82             if (versionStrings.empty())
     83             {
     84                 AICLI_LOG(Repo, Info, << "Did not find any Versions { " << id << ", " << channel << " }");
     85                 return {};
     86             }
     87 
     88             // Convert the strings to Versions and sort them
     89             struct VersionAndRow
     90             {
     91                 SQLite::rowid_t Row = 0;
     92                 Utility::Version Version;
     93 
     94                 bool operator<(const VersionAndRow& other) const { return Version < other.Version; }
     95             };
     96 
     97             std::vector<VersionAndRow> versions;
     98             for (auto& v : versionStrings)
     99             {
    100                 versions.emplace_back(VersionAndRow{ v.first, std::move(v.second) });
    101             }
    102 
    103             std::sort(versions.begin(), versions.end());
    104 
    105             if (version.empty())
    106             {
    107                 // Get the last version in the list (the highest version)
    108                 versionIdOpt = versions.back().Row;
    109             }
    110             else
    111             {
    112                 VersionAndRow requested;
    113                 requested.Version = Utility::Version{ std::string(version) };
    114 
    115                 auto itr = std::lower_bound(versions.begin(), versions.end(), requested);
    116                 if (itr != versions.end() && itr->Version == requested.Version)
    117                 {
    118                     versionIdOpt = itr->Row;
    119                 }
    120             }
    121 
    122             if (!versionIdOpt)
    123             {
    124                 AICLI_LOG(Repo, Info, << "Did not find a Version for { " << version << " }");
    125                 return {};
    126             }
    127 
    128             if (channelIdOpt)
    129             {
    130                 return ManifestTable::SelectByValueIds<IdTable, VersionTable, ChannelTable>(connection, { id, versionIdOpt.value(), channelIdOpt.value() });
    131             }
    132             else
    133             {
    134                 return ManifestTable::SelectByValueIds<IdTable, VersionTable>(connection, { id, versionIdOpt.value() });
    135             }
    136         }

    146         {
    147             auto [oldValueId] = ManifestTable::GetIdsById<Table>(connection, manifestId);
    148

    192     SQLite::rowid_t Interface::AddManifest(SQLite::Connection& connection, const Manifest::Manifest& manifest, const std::optional<std::filesystem::path>& relativePath)
    193     {
    194         auto manifestResult = GetExistingManifestId(connection, manifest);
    195 
    196         // If this manifest is already present, we can't add it.
    197         THROW_HR_IF(HRESULT_FROM_WIN32(ERROR_ALREADY_EXISTS), manifestResult.has_value());
    198 
    199         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "addmanifest_v1_0");
    200 
    201         auto [pathAdded, pathLeafId] = PathPartTable::EnsurePathExists(connection, relativePath, true);
    202 
    203         // If we get false from the function, this manifest path already exists in the index.
    204         THROW_HR_IF(HRESULT_FROM_WIN32(ERROR_ALREADY_EXISTS), relativePath && !pathAdded);
    205 
    206         // Ensure that all of the 1:1 data exists.
    207         SQLite::rowid_t idId = IdTable::EnsureExists(connection, manifest.Id, true);
    208         SQLite::rowid_t nameId = NameTable::EnsureExists(connection, manifest.DefaultLocalization.Get<Manifest::Localization::PackageName>());
    209         SQLite::rowid_t monikerId = MonikerTable::EnsureExists(connection, manifest.Moniker);
    210         SQLite::rowid_t versionId = VersionTable::EnsureExists(connection, manifest.Version);
    211         SQLite::rowid_t channelId = ChannelTable::EnsureExists(connection, manifest.Channel);
    212 
    213         // Insert the manifest entry.
    214         SQLite::rowid_t manifestId = ManifestTable::Insert(connection, {
    215             { IdTable::ValueName(), idId},
    216             { NameTable::ValueName(), nameId },
    217             { MonikerTable::ValueName(), monikerId },
    218             { VersionTable::ValueName(), versionId },
    219             { ChannelTable::ValueName(), channelId },
    220             { PathPartTable::ValueName(), pathLeafId }
    221             });
    222 
    223         // Add all of the 1:N data.
    224         TagsTable::EnsureExistsAndInsert(connection, manifest.GetAggregatedTags(), manifestId);
    225         CommandsTable::EnsureExistsAndInsert(connection, manifest.GetAggregatedCommands(), manifestId);
    226 
    227         savepoint.Commit();
    228 
    229         return manifestId;
    230     }

    232     std::pair<bool, SQLite::rowid_t> Interface::UpdateManifest(SQLite::Connection& connection, const Manifest::Manifest& manifest, const std::optional<std::filesystem::path>& relativePath)
    233     {
    234         auto manifestResult = GetExistingManifestId(connection, manifest);
    235 
    236         // If the manifest doesn't actually exist, fail the update.
    237         THROW_HR_IF(E_NOT_SET, !manifestResult);
    238 
    239         SQLite::rowid_t manifestId = manifestResult.value();
    240 
    241         auto [idInIndex, nameInIndex, monikerInIndex, versionInIndex, channelInIndex] =
    242             ManifestTable::GetValuesById<IdTable, NameTable, MonikerTable, VersionTable, ChannelTable>(connection, manifestId);
    243 
    244         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "updatemanifest_v1_0");
    245         bool indexModified = false;
    246 
    247         // Id, Version, and Channel may have changed casing. If so, they too need to be updated.
    248         if (idInIndex != manifest.Id)
    249         {
    250             UpdateManifestValueById<IdTable>(connection, manifest.Id, manifestId, true);
    251             indexModified = true;
    252         }
    253 
    254         if (versionInIndex != manifest.Version)
    255         {
    256             UpdateManifestValueById<VersionTable>(connection, manifest.Version, manifestId);
    257             indexModified = true;
    258         }
    259 
    260         if (channelInIndex != manifest.Channel)
    261         {
    262             UpdateManifestValueById<ChannelTable>(connection, manifest.Channel, manifestId);
    263             indexModified = true;
    264         }
    265 
    266         auto packageName = manifest.DefaultLocalization.Get<Manifest::Localization::PackageName>();
    267         if (nameInIndex != packageName)
    268         {
    269             UpdateManifestValueById<NameTable>(connection, packageName, manifestId);
    270             indexModified = true;
    271         }
    272 
    273         if (monikerInIndex != manifest.Moniker)
    274         {
    275             UpdateManifestValueById<MonikerTable>(connection, manifest.Moniker, manifestId);
    276             indexModified = true;
    277         }
    278 
    279         // Update path table if necessary
    280         auto [existingPathLeafId] = ManifestTable::GetIdsById<PathPartTable>(connection, manifestId);
    281         auto [pathAdded, newPathLeafId] = PathPartTable::EnsurePathExists(connection, relativePath, true);
    282 
    283         if (relativePath && pathAdded)
    284         {
    285             // Path was added, so we need to update the manifest table and delete the old path
    286             ManifestTable::UpdateValueIdById<PathPartTable>(connection, manifestId, newPathLeafId);
    287             PathPartTable::RemovePathById(connection, existingPathLeafId);
    288             indexModified = true;
    289         }
    290         else
    291         {
    292             // The path already existed, so it must either match the existing manifest path or it is an error
    293             THROW_HR_IF(HRESULT_FROM_WIN32(ERROR_ALREADY_EXISTS), existingPathLeafId != newPathLeafId);
    294         }
    295 
    296         // Update all 1:N tables as necessary
    297         indexModified = TagsTable::UpdateIfNeededByManifestId(connection, manifest.GetAggregatedTags(), manifestId) || indexModified;
    298         indexModified = CommandsTable::UpdateIfNeededByManifestId(connection, manifest.GetAggregatedCommands(), manifestId) || indexModified;
    299 
    300         savepoint.Commit();
    301 
    302         return { indexModified, manifestId };
    303     }

    317     void Interface::RemoveManifestById(SQLite::Connection& connection, SQLite::rowid_t manifestId)
    318     {
    319         // Get the ids of the values from the manifest table
    320         auto [idId, nameId, monikerId, versionId, channelId, pathLeafId] = 
    321             ManifestTable::GetIdsById<IdTable, NameTable, MonikerTable, VersionTable, ChannelTable, PathPartTable>(connection, manifestId);
    322 
    323         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "RemoveManifestById_v1_0");
    324 
    325         // Remove the manifest row
    326         ManifestTable::DeleteById(connection, manifestId);
    327 
    328         // Remove all of the 1:1 data that is no longer referenced.
    329         if (NotNeeded(connection, IdTable::TableName(), IdTable::ValueName(), idId))
    330         {
    331             IdTable::DeleteById(connection, idId);
    332         }
    333         
    334         if (NotNeeded(connection, NameTable::TableName(), NameTable::ValueName(), nameId))
    335         {
    336             NameTable::DeleteById(connection, nameId);
    337         }
    338 
    339         if (NotNeeded(connection, MonikerTable::TableName(), MonikerTable::ValueName(), monikerId))
    340         {
    341             MonikerTable::DeleteById(connection, monikerId);
    342         }
    343         
    344         if (NotNeeded(connection, VersionTable::TableName(), VersionTable::ValueName(), versionId))
    345         {
    346             VersionTable::DeleteById(connection, versionId);
    347         }
    348         
    349         if (NotNeeded(connection, ChannelTable::TableName(), ChannelTable::ValueName(), channelId))
    350         {
    351             ChannelTable::DeleteById(connection, channelId);
    352         }
    353 
    354         // Remove the path
    355         PathPartTable::RemovePathById(connection, pathLeafId);
    356 
    357         // Remove all of the 1:N data that is no longer referenced.
    358         TagsTable::DeleteIfNotNeededByManifestId(connection, manifestId);
    359         CommandsTable::DeleteIfNotNeededByManifestId(connection, manifestId);
    360 
    361         savepoint.Commit();
    362     }

    432     ISQLiteIndex::SearchResult Interface::Search(const SQLite::Connection& connection, const SearchRequest& request) const
    433     {
    434         if (request.IsForEverything())
    435         {
    436             std::vector<SQLite::rowid_t> ids = IdTable::GetAllRowIds(connection, request.MaximumResults);
    437 
    438             SearchResult result;
    439             for (SQLite::rowid_t id : ids)
    440             {
    441                 result.Matches.emplace_back(std::make_pair(id, PackageMatchFilter(PackageMatchField::Id, MatchType::Wildcard)));
    442             }
    443 
    444             result.Truncated = (request.MaximumResults && IdTable::GetCount(connection) > request.MaximumResults);
    445 
    446             return result;
    447         }
    448 
    449         // First phase, create the search results table and populate it with the initial results.
    450         // If the Query is provided, we search across many fields and put results in together.
    451         // If Inclusions has fields, we add these to the data.
    452         // If neither is defined, we take the first filter and use it as the initial results search.
    453         std::unique_ptr<SearchResultsTable> resultsTable = CreateSearchResultsTable(connection);
    454         bool inclusionsAttempted = false;
    455 
    456         if (request.Query)
    457         {
    458             // Perform searches across multiple tables to populate the initial results.
    459             PerformQuerySearch(*resultsTable.get(), request.Query.value());
    460 
    461             inclusionsAttempted = true;
    462         }
    463 
    464         if (!request.Inclusions.empty())
    465         {
    466             for (auto include : request.Inclusions)
    467             {
    468                 for (MatchType match : GetDefaultMatchTypeOrder(include.Type))
    469                 {
    470                     include.Type = match;
    471                     resultsTable->SearchOnField(include);
    472                 }
    473             }
    474 
    475             inclusionsAttempted = true;
    476         }
    477 
    478         size_t filterIndex = 0;
    479         if (!inclusionsAttempted)
    480         {
    481             THROW_HR_IF(E_UNEXPECTED, request.Filters.empty());
    482 
    483             // Perform search for just the field matching the first filter
    484             PackageMatchFilter filter = request.Filters[0];
    485 
    486             for (MatchType match : GetDefaultMatchTypeOrder(filter.Type))
    487             {
    488                 filter.Type = match;
    489                 resultsTable->SearchOnField(filter);
    490             }
    491 
    492             // Skip the filter as we already know everything matches
    493             filterIndex = 1;
    494         }
    495 
    496         // Remove any duplicate manifest entries
    497         resultsTable->RemoveDuplicateManifestRows();
    498 
    499         // Second phase, for remaining filters, flag matching search results, then remove unflagged values.
    500         for (size_t i = filterIndex; i < request.Filters.size(); ++i)
    501         {
    502             PackageMatchFilter filter = request.Filters[i];
    503 
    504             resultsTable->PrepareToFilter();
    505 
    506             for (MatchType match : GetDefaultMatchTypeOrder(filter.Type))
    507             {
    508                 filter.Type = match;
    509                 resultsTable->FilterOnField(filter);
    510             }
    511 
    512             resultsTable->CompleteFilter();
    513         }
    514 
    515         return resultsTable->GetSearchResults(request.MaximumResults);
    516     }

    583         auto versionsAndChannels = ManifestTable::GetAllValuesById<IdTable, VersionTable, ChannelTable>(connection, id);
    584 
    585         std::vector<ISQLiteIndex::VersionKey> result;
    586         result.reserve(versionsAndChannels.size());
    587         for (auto&& vac : versionsAndChannels)
    588         {
    589             result.emplace_back(ISQLiteIndex::VersionKey{ Utility::VersionAndChannel{ Utility::Version{ std::move(std::get<1>(vac)) }, Utility::Channel{ std::move(std::get<2>(vac)) } }, std::get<0>(vac) });
    590         }
    591 
    592         std::sort(result.begin(), result.end());
    593 
    594         return result;
    595     }

    619     void Interface::PerformQuerySearch(SearchResultsTable& resultsTable, const RequestMatch& query) const
    620     {
    621         // Arbitrary values to create a reusable filter with the given value.
    622         PackageMatchFilter filter(PackageMatchField::Id, MatchType::Exact, query.Value);
    623 
    624         for (MatchType match : GetDefaultMatchTypeOrder(query.Type))
    625         {
    626             filter.Type = match;
    627 
    628             for (auto field : { PackageMatchField::Id, PackageMatchField::Name, PackageMatchField::Moniker, PackageMatchField::Command, PackageMatchField::Tag })
    629             {
    630                 filter.Field = field;
    631                 resultsTable.SearchOnField(filter);
    632             }
    633         }
    634     }

    636     std::optional<std::string> Interface::GetPropertyByManifestIdInternal(const SQLite::Connection& connection, SQLite::rowid_t manifestId, PackageVersionProperty property) const
    637     {
    638         switch (property)
    639         {
    640         case AppInstaller::Repository::PackageVersionProperty::Id:
    641             return ManifestTable::GetValueById<IdTable>(connection, manifestId);
    642         case AppInstaller::Repository::PackageVersionProperty::Name:
    643             return ManifestTable::GetValueById<NameTable>(connection, manifestId);
    644         case AppInstaller::Repository::PackageVersionProperty::Version:
    645             return ManifestTable::GetValueById<VersionTable>(connection, manifestId);
    646         case AppInstaller::Repository::PackageVersionProperty::Channel:
    647             return ManifestTable::GetValueById<ChannelTable>(connection, manifestId);
    648         case AppInstaller::Repository::PackageVersionProperty::RelativePath:
    649             return PathPartTable::GetPathById(connection, std::get<0>(ManifestTable::GetIdsById<PathPartTable>(connection, manifestId)));
    650         case AppInstaller::Repository::PackageVersionProperty::Moniker:
    651             return ManifestTable::GetValueById<MonikerTable>(connection, manifestId);
    652         default:
    653             return {};
    654         }
    655     }
