    176         {
    177             return details::ManifestTableGetAllValuesByIds(connection,
    178                 { SQLite::Builder::QualifiedColumn{ ValueTable::TableName(), SQLite::RowIDName }, SQLite::Builder::QualifiedColumn{ ValueTable::TableName(), ValueTable::ValueName() } },

    183         // Gets all values for rows that match the given id.
    184         template <typename IdTable, typename... ValueTables>
    185         static std::vector<std::tuple<SQLite::rowid_t, typename ValueTables::value_t...>> GetAllValuesById(const SQLite::Connection& connection, SQLite::rowid_t id)
    186         {
    187             auto stmt = details::ManifestTableGetAllValuesByIds_Statement(connection,
    188                 { SQLite::Builder::QualifiedColumn{ TableName(), SQLite::RowIDName }, SQLite::Builder::QualifiedColumn{ ValueTables::TableName(), ValueTables::ValueName() }... },
    189                 { SQLite::Builder::QualifiedColumn{ ValueTables::TableName(), ValueTables::ValueName() }... },
    190                 { IdTable::ValueName() }, { id });
    191             std::vector<std::tuple<SQLite::rowid_t, typename ValueTables::value_t...>> result;
    192             while (stmt.Step())
    193             {
    194                 result.emplace_back(stmt.GetRow<SQLite::rowid_t, typename ValueTables::value_t...>());
    195             }
    196             return result;
    197         }
