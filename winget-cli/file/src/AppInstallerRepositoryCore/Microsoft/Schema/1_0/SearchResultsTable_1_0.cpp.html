     72     void SearchResultsTable::SearchOnField(const PackageMatchFilter& filter)
     73     {
     74         using namespace SQLite::Builder;
     75 
     76         int sortOrdinal = m_sortOrdinalValue++;
     77 
     78         // Create an insert statement to select values into the table as requested.
     79         // The goal is a statement like this:
     80         //      INSERT INTO <tempTable>
     81         //      SELECT valueTable.m, <field>, <match>, valueTable.v, <sort>, <filter> FROM
     82         //      (SELECT manifest.rowid as m, manifest.id as v from manifest join ids on manifest.id = ids.rowid where ids.id = <value>) AS valueTable
     83         // Where the subselect is built by the owning table.
     84         StatementBuilder builder;
     85         builder.InsertInto(GetQualifiedName()).Select().
     86             Column(QualifiedColumn(s_SearchResultsTable_SubSelect_TableAlias, s_SearchResultsTable_SubSelect_ManifestAlias)).
     87             Value(filter.Field).
     88             Value(filter.Type).
     89             Column(QualifiedColumn(s_SearchResultsTable_SubSelect_TableAlias, s_SearchResultsTable_SubSelect_ValueAlias)).
     90             Value(sortOrdinal).
     91             Value(false).
     92         From().BeginParenthetical();
     93 
     94         // Add the field specific portion
     95         std::vector<int> bindIndex = BuildSearchStatement(builder, filter.Field, filter.Type);
     96 
     97         if (bindIndex.empty())
     98         {
     99             AICLI_LOG(Repo, Verbose, << "PackageMatchField not supported in this version: " << ToString(filter.Field));
    100             return;
    101         }
    102 
    103         builder.EndParenthetical().As(s_SearchResultsTable_SubSelect_TableAlias);
    104 
    105         SQLite::Statement statement = builder.Prepare(m_connection);
    106         BindStatementForMatchType(statement, filter, bindIndex);
    107         statement.Execute();
    108         AICLI_LOG(SQL, Verbose, << "Search found " << m_connection.GetChanges() << " rows");
    109     }

    111     void SearchResultsTable::RemoveDuplicateManifestRows()
    112     {
    113         using namespace SQLite::Builder;
    114 
    115         // Create a delete statement to leave only one row with a given manifest.
    116         // This will arbitrarily choose one of the rows if multiple have the same lowest sort order.
    117         // The goal is a statement like this:
    118         //      DELETE from <temp> where rowid not in (
    119         //          SELECT rowid from (
    120         //              SELECT rowid, min(sort) from <temp> group by manifest
    121         //          )
    122         //      )
    123         StatementBuilder builder;
    124         builder.DeleteFrom(GetQualifiedName()).Where(SQLite::RowIDName).Not().In().BeginParenthetical().
    125             Select(SQLite::RowIDName).From().BeginParenthetical().
    126                 Select().Column(SQLite::RowIDName).Column(Aggregate::Min, s_SearchResultsTable_SortValue).From(GetQualifiedName()).GroupBy(s_SearchResultsTable_Manifest).
    127             EndParenthetical().
    128         EndParenthetical();
    129 
    130         builder.Execute(m_connection);
    131         AICLI_LOG(SQL, Verbose, << "Removed " << m_connection.GetChanges() << " duplicate rows");
    132     }

    143     void SearchResultsTable::FilterOnField(const PackageMatchFilter& filter)
    144     {
    145         using namespace SQLite::Builder;
    146 
    147         // Create an update statement to mark rows that are found by the search.
    148         // This will arbitrarily choose one of the rows if multiple have the same lowest sort order.
    149         // The goal is a statement like this:
    150         //      UPDATE <temp> set filter = 1 where manifest in (
    151         //          SELECT m from (
    152         //              SELECT manifest.rowid as m, manifest.id as v from manifest join ids on manifest.id = ids.rowid where ids.id = <value>
    153         //          )
    154         //      )
    155         StatementBuilder builder;
    156         builder.Update(GetQualifiedName()).Set().Column(s_SearchResultsTable_Filter).Equals(true).Where(s_SearchResultsTable_Manifest).In().BeginParenthetical().
    157             Select(s_SearchResultsTable_SubSelect_ManifestAlias).From().BeginParenthetical();
    158 
    159         // Add the field specific portion
    160         std::vector<int> bindIndex = BuildSearchStatement(builder, filter.Field, filter.Type);
    161 
    162         if (bindIndex.empty())
    163         {
    164             AICLI_LOG(Repo, Verbose, << "PackageMatchField not supported in this version: " << ToString(filter.Field));
    165             return;
    166         }
    167 
    168         builder.EndParenthetical().EndParenthetical();
    169 
    170         SQLite::Statement statement = builder.Prepare(m_connection);
    171         BindStatementForMatchType(statement, filter, bindIndex);
    172         statement.Execute();
    173         AICLI_LOG(SQL, Verbose, << "Filter kept " << m_connection.GetChanges() << " rows");

    186     ISQLiteIndex::SearchResult SearchResultsTable::GetSearchResults(size_t limit)
    187     {
    188         constexpr std::string_view tempTableAlias = "t"sv;
    189 
    190         using namespace SQLite::Builder;
    191         using QCol = QualifiedColumn;
    192 
    193         // Select all unique ids from the results table, and their highest ordered match.
    194         // The goal is a statement like this:
    195         //  SELECT m.id, field, match, value, min(sort) from <temp> join manifest on rowid = manifest group by m.id order by t.sort
    196         // Through the "group by m.id", we will only ever have one row per id, and the "min(sort)" returns us one of the rows that matched
    197         // through the earliest search.  We also order by the sort value to have the earliest search matches first in the list
    198         StatementBuilder builder;
    199         builder.Select().
    200             Column(QCol(ManifestTable::TableName(), IdTable::ValueName())).
    201             Column(QCol(tempTableAlias, s_SearchResultsTable_MatchField)).
    202             Column(QCol(tempTableAlias, s_SearchResultsTable_MatchType)).
    203             Column(QCol(tempTableAlias, s_SearchResultsTable_MatchValue)).
    204             Column(Aggregate::Min, QCol(tempTableAlias, s_SearchResultsTable_SortValue)).
    205         From(GetQualifiedName()).As(tempTableAlias).
    206             Join(ManifestTable::TableName()).On(QCol(tempTableAlias, s_SearchResultsTable_Manifest), QCol(ManifestTable::TableName(), SQLite::RowIDName)).
    207             GroupBy(QCol(ManifestTable::TableName(), IdTable::ValueName())).OrderBy(QCol(tempTableAlias, s_SearchResultsTable_SortValue));
    208 
    209         SQLite::Statement select = builder.Prepare(m_connection);
    210 
    211         ISQLiteIndex::SearchResult result;
    212         while (select.Step())
    213         {
    214             if (limit && result.Matches.size() >= limit)
    215             {
    216                 break;
    217             }
    218 
    219             result.Matches.emplace_back(select.GetColumn<SQLite::rowid_t>(0), 
    220                 PackageMatchFilter(select.GetColumn<PackageMatchField>(1), select.GetColumn<MatchType>(2), select.GetColumn<std::string>(3)));
    221         }
    222 
    223         result.Truncated = (select.GetState() != SQLite::Statement::State::Completed);
    224 
    225         return result;
    226     }

    262     void SearchResultsTable::BindStatementForMatchType(SQLite::Statement& statement, MatchType match, int bindIndex, std::string_view value)
    263     {
    264         std::string valueToUse;
    265 
    266         if (MatchUsesLike(match))
    267         {
    268             valueToUse = SQLite::EscapeStringForLike(value);
    269         }
    270         else
    271         {
    272             valueToUse = value;
    273         }
    274 
    275         switch (match)
    276         {
    277         case AppInstaller::Repository::MatchType::StartsWith:
    278             valueToUse += '%';
    279             break;
    280         case AppInstaller::Repository::MatchType::Substring:
    281             valueToUse = "%"s + valueToUse + '%';
    282             break;
    283         default:
    284             // No changes required for others.
    285             break;
    286         }
    287 
    288         statement.Bind(bindIndex, valueToUse);
    289     }
