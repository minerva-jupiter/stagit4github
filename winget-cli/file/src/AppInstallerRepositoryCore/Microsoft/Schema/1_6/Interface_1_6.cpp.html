    115     void Interface::PerformQuerySearch(V1_0::SearchResultsTable& resultsTable, const RequestMatch& query) const
    116     {
    117         // First, do an exact match search for the folded system reference strings
    118         // We do this first because it is exact, and likely won't match anything else if it matches this.
    119         PackageMatchFilter filter(PackageMatchField::UpgradeCode, MatchType::Exact, Utility::FoldCase(query.Value));
    120         resultsTable.SearchOnField(filter);
    121 
    122         // Then do the 1.5 search
    123         V1_5::Interface::PerformQuerySearch(resultsTable, query);
    124     }

    126     ISQLiteIndex::SearchResult Interface::SearchInternal(const SQLite::Connection& connection, SearchRequest& request) const
    127     {
    128         // Update any system reference strings to be folded
    129         auto foldIfNeeded = [](PackageMatchFilter& filter)
    130         {
    131             if (filter.Field == PackageMatchField::UpgradeCode && filter.Type == MatchType::Exact)
    132             {
    133                 filter.Value = Utility::FoldCase(filter.Value);
    134             }
    135         };
    136 
    137         for (auto& inclusion : request.Inclusions)
    138         {
    139             foldIfNeeded(inclusion);
    140         }
    141 
    142         for (auto& filter : request.Filters)
    143         {
    144             foldIfNeeded(filter);
    145         }
    146 
    147         return V1_5::Interface::SearchInternal(connection, request);
    148     }
