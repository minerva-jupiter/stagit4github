    199     void Interface::PerformQuerySearch(V1_0::SearchResultsTable& resultsTable, const RequestMatch& query) const
    200     {
    201         // First, do an exact match search for the folded system reference strings
    202         // We do this first because it is exact, and likely won't match anything else if it matches this.
    203         PackageMatchFilter filter(PackageMatchField::PackageFamilyName, MatchType::Exact, Utility::FoldCase(query.Value));
    204         resultsTable.SearchOnField(filter);
    205 
    206         filter.Field = PackageMatchField::ProductCode;
    207         resultsTable.SearchOnField(filter);
    208 
    209         // Then do the 1.0 search
    210         V1_0::Interface::PerformQuerySearch(resultsTable, query);
    211     }

    218     ISQLiteIndex::SearchResult Interface::SearchInternal(const SQLite::Connection& connection, SearchRequest& request) const
    219     {
    220         // Update any system reference strings to be folded
    221         auto foldIfNeeded = [](PackageMatchFilter& filter)
    222         {
    223             if ((filter.Field == PackageMatchField::PackageFamilyName || filter.Field == PackageMatchField::ProductCode) &&
    224                 filter.Type == MatchType::Exact)
    225             {
    226                 filter.Value = Utility::FoldCase(filter.Value);
    227             }
    228         };
    229 
    230         for (auto& inclusion : request.Inclusions)
    231         {
    232             foldIfNeeded(inclusion);
    233         }
    234 
    235         for (auto& filter : request.Filters)
    236         {
    237             foldIfNeeded(filter);
    238         }
    239 
    240         return V1_0::Interface::Search(connection, request);
    241     }
