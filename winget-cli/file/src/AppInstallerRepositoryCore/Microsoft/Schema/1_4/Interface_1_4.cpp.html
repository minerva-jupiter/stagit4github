     61     void Interface::RemoveManifestById(SQLite::Connection& connection, SQLite::rowid_t manifestId)
     62     {
     63         // Get all versions that need cleaning from the version table.
     64         auto minVersions = DependenciesTable::GetDependenciesMinVersionsRowIdByManifestId(connection, manifestId);
     65 
     66         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "removemanifest_v1_4");
     67 
     68         // Removes dependences for the manifest id.
     69         DependenciesTable::RemoveDependencies(connection, manifestId);
     70 
     71         // Removes the manifest.
     72         V1_3::Interface::RemoveManifestById(connection, manifestId);
     73 
     74         // Remove the versions that are not needed.
     75         for (auto minVersion : minVersions)
     76         {
     77             if (NotNeeded(connection, Schema::V1_0::VersionTable::TableName(), Schema::V1_0::VersionTable::ValueName(), minVersion))
     78             {
     79                 Schema::V1_0::VersionTable::DeleteById(connection, minVersion);
     80             }
     81         }
     82 
     83         savepoint.Commit();
     84     }

    148     bool Interface::ValidateDependenciesWithMinVersions(const SQLite::Connection& connection, bool log) const
    149     {
    150         try
    151         {
    152             bool result = true;
    153             // A map to store already checked dependency package latest versions.
    154             std::map<SQLite::rowid_t, Utility::Version> checkedVersions;
    155 
    156             auto dependencies = DependenciesTable::GetAllDependenciesWithMinVersions(connection);
    157             for (auto const& dependency : dependencies)
    158             {
    159                 // If the dependency package has not been checked yet, add to the map.
    160                 if (checkedVersions.find(dependency.first) == checkedVersions.end())
    161                 {
    162                     auto versionKeys = GetVersionKeysById(connection, dependency.first);
    163                     THROW_HR_IF(E_UNEXPECTED, versionKeys.empty());
    164                     checkedVersions.emplace(dependency.first, versionKeys[0].VersionAndChannel.GetVersion());
    165                 }
    166 
    167                 // If the latest version is less than min version required, fail the validation.
    168                 if (checkedVersions[dependency.first] < Utility::Version{ dependency.second })
    169                 {
    170                     AICLI_LOG(Repo, Error, << "Dependency with min version not satisfied. Dependency package row id: " << dependency.first << " min version: " << dependency.second);
    171                     result = false;
    172 
    173                     if (!log)
    174                     {
    175                         break;
    176                     }
    177                 }
    178             }
    179 
    180             return result;
    181         }
    182         catch (...)
    183         {
    184             AICLI_LOG(Repo, Error, << "ValidateDependenciesWithMinVersions() encountered internal error. Returning false.");
    185             return false;
    186         }
    187     }
