     12         std::optional<Pinning::Pin> GetPinFromRow(
     13             std::string_view packageId,
     14             std::string_view sourceId,
     15             Pinning::PinType type,
     16             std::string_view version)
     17 
     18         {
     19             switch (type)
     20             {
     21             case Pinning::PinType::Blocking:
     22                 return Pinning::Pin::CreateBlockingPin({ packageId, sourceId });
     23             case Pinning::PinType::Pinning:
     24                 return Pinning::Pin::CreatePinningPin({ packageId, sourceId });
     25             case Pinning::PinType::Gating:
     26                 return Pinning::Pin::CreateGatingPin({ packageId, sourceId }, Utility::GatedVersion{ version });
     27             default:
     28                 return {};
     29             }
     30         }
     31     }

    156     {

    159             s_PinTable_PackageId_Column,

    161             s_PinTable_Type_Column,

    163             .From(s_PinTable_Table_Name);

    166 
    167         std::vector<Pinning::Pin> pins;
    168         while (select.Step())

    170             auto [packageId, sourceId, pinType, gatedVersion] =
    171                 select.GetRow<std::string, std::string, Pinning::PinType, std::string>();
    172             auto pin = GetPinFromRow(packageId, sourceId, pinType, gatedVersion);
    173             if (pin)
    174             {
    175                 pins.push_back(std::move(pin.value()));
    176             }
    177         }
    178 
    179         return pins;
    180     }
