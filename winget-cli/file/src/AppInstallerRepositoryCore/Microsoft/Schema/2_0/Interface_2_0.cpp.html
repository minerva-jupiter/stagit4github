     27         // Folds the values of the fields that are stored folded.
     28         void FoldPackageMatchFilters(std::vector<PackageMatchFilter>& filters)
     29         {
     30             for (auto& filter : filters)
     31             {
     32                 if ((filter.Field == PackageMatchField::PackageFamilyName || filter.Field == PackageMatchField::ProductCode || filter.Field == PackageMatchField::UpgradeCode) &&
     33                     filter.Type == MatchType::Exact)
     34                 {
     35                     filter.Value = Utility::FoldCase(filter.Value);
     36                 }
     37             }
     38         }

     40         // Update NormalizedNameAndPublisher with normalization and folding
     41         // Returns true if the normalized name contains normalization field of fieldsToInclude
     42         bool UpdateNormalizedNameAndPublisher(
     43             PackageMatchFilter& filter,
     44             const Utility::NameNormalizer& normalizer,
     45             Utility::NormalizationField fieldsToInclude)
     46         {
     47             Utility::NormalizedName normalized = normalizer.Normalize(Utility::FoldCase(filter.Value), Utility::FoldCase(filter.Additional.value()));
     48             filter.Value = normalized.GetNormalizedName(fieldsToInclude);
     49             filter.Additional = normalized.Publisher();
     50             return WI_AreAllFlagsSet(normalized.GetNormalizedFields(), fieldsToInclude);
     51         }

     53         bool UpdatePackageMatchFilters(
     54             std::vector<PackageMatchFilter>& filters,
     55             const Utility::NameNormalizer& normalizer,
     56             Utility::NormalizationField normalizedNameFieldsToFilter = Utility::NormalizationField::None)
     57         {
     58             bool normalizedNameFieldsFound = false;
     59             for (auto itr = filters.begin(); itr != filters.end();)
     60             {
     61                 if (itr->Field == PackageMatchField::NormalizedNameAndPublisher && itr->Type == MatchType::Exact)
     62                 {
     63                     if (!UpdateNormalizedNameAndPublisher(*itr, normalizer, normalizedNameFieldsToFilter))
     64                     {
     65                         // If not matched, this package match filter will be removed.
     66                         // For example, if caller is trying to search with arch info only, values without arch will be removed from search.
     67                         itr = filters.erase(itr);
     68                         continue;
     69                     }
     70 
     71                     normalizedNameFieldsFound = true;
     72                 }
     73 
     74                 ++itr;
     75             }
     76 
     77             return normalizedNameFieldsFound;
     78         }
     79     }

    216     std::optional<std::string> Interface::GetPropertyByPrimaryId(const SQLite::Connection& connection, SQLite::rowid_t primaryId, PackageVersionProperty property) const
    217     {
    218         EnsureInternalInterface(connection);
    219 
    220         if (m_internalInterface)
    221         {
    222             return m_internalInterface->GetPropertyByPrimaryId(connection, primaryId, property);
    223         }
    224 
    225         switch (property)
    226         {
    227         case PackageVersionProperty::Id:
    228             return PackagesTable::GetValueById<PackagesTable::IdColumn>(connection, primaryId);
    229         case PackageVersionProperty::Name:
    230             return PackagesTable::GetValueById<PackagesTable::NameColumn>(connection, primaryId);
    231         case PackageVersionProperty::Version:
    232             return PackagesTable::GetValueById<PackagesTable::LatestVersionColumn>(connection, primaryId);
    233         case PackageVersionProperty::Channel:
    234             return "";
    235         case PackageVersionProperty::ManifestSHA256Hash:
    236         {
    237             std::optional<SQLite::blob_t> hash = PackagesTable::GetValueById<PackagesTable::HashColumn>(connection, primaryId);
    238             return (!hash || hash->empty()) ? std::optional<std::string>{} : Utility::SHA256::ConvertToString(hash.value());
    239         }
    240         case PackageVersionProperty::ArpMinVersion:
    241             return PackagesTable::GetValueById<PackagesTable::ARPMinVersionColumn>(connection, primaryId);
    242         case PackageVersionProperty::ArpMaxVersion:
    243             return PackagesTable::GetValueById<PackagesTable::ARPMaxVersionColumn>(connection, primaryId);
    244         case PackageVersionProperty::Moniker:
    245             return PackagesTable::GetValueById<PackagesTable::MonikerColumn>(connection, primaryId);
    246         default:
    247             return {};
    248         }
    249     }

    386     bool Interface::MigrateFrom(SQLite::Connection& connection, const ISQLiteIndex* current)
    387     {
    388         THROW_HR_IF_NULL(E_POINTER, current);
    389 
    390         auto currentVersion = current->GetVersion();
    391         if (currentVersion.MajorVersion != 1 || currentVersion.MinorVersion != 7)
    392         {
    393             return false;
    394         }
    395 
    396         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "migrate_from_v2_0");
    397 
    398         // We only need to insert all of the existing packages into the update tracking table.
    399         PackageUpdateTrackingTable::EnsureExists(connection);
    400         SearchResult allPackages = current->Search(connection, {});
    401 
    402         for (const auto& packageMatch : allPackages.Matches)
    403         {
    404             std::vector<ISQLiteIndex::VersionKey> versionKeys = current->GetVersionKeysById(connection, packageMatch.first);
    405             ISQLiteIndex::VersionKey& latestVersionKey = versionKeys[0];
    406             PackageUpdateTrackingTable::Update(connection, current, current->GetPropertyByPrimaryId(connection, latestVersionKey.ManifestId, PackageVersionProperty::Id).value(), false);
    407         }
    408 
    409         savepoint.Commit();
    410         return true;
    411     }

    442     void Interface::PerformQuerySearch(SearchResultsTable& resultsTable, const RequestMatch& query) const
    443     {
    444         // First, do an exact match search for the folded system reference strings
    445         // We do this first because it is exact, and likely won't match anything else if it matches this.
    446         PackageMatchFilter filter(PackageMatchField::Unknown, MatchType::Exact, Utility::FoldCase(query.Value));
    447 
    448         for (PackageMatchField field : { PackageMatchField::PackageFamilyName, PackageMatchField::ProductCode, PackageMatchField::UpgradeCode })
    449         {
    450             filter.Field = field;
    451             resultsTable.SearchOnField(filter);
    452         }
    453 
    454         // Now search on the unfolded value
    455         filter.Value = query.Value;
    456 
    457         for (MatchType match : GetDefaultMatchTypeOrder(query.Type))
    458         {
    459             filter.Type = match;
    460 
    461             for (auto field : { PackageMatchField::Id, PackageMatchField::Name, PackageMatchField::Moniker, PackageMatchField::Command, PackageMatchField::Tag })
    462             {
    463                 filter.Field = field;
    464                 resultsTable.SearchOnField(filter);
    465             }
    466         }
    467     }

    474     ISQLiteIndex::SearchResult Interface::SearchInternal(const SQLite::Connection& connection, SearchRequest& request) const
    475     {
    476         anon::FoldPackageMatchFilters(request.Inclusions);
    477         anon::FoldPackageMatchFilters(request.Filters);
    478 
    479         if (request.Purpose == SearchPurpose::CorrelationToInstalled)
    480         {
    481             // Correlate from available package to installed package
    482             // For available package to installed package mapping, only one try is needed.
    483             // For example, if ARP DisplayName contains arch, then the installed package's ARP DisplayName should also include arch.
    484             auto candidateInclusionsWithArch = request.Inclusions;
    485             if (anon::UpdatePackageMatchFilters(candidateInclusionsWithArch, m_normalizer, Utility::NormalizationField::Architecture))
    486             {
    487                 // If DisplayNames contain arch, only use Inclusions with arch for search
    488                 request.Inclusions = candidateInclusionsWithArch;
    489             }
    490             else
    491             {
    492                 // Otherwise, just update the Inclusions with normalization
    493                 anon::UpdatePackageMatchFilters(request.Inclusions, m_normalizer);
    494             }
    495 
    496             return BasicSearchInternal(connection, request);
    497         }
    498         else if (request.Purpose == SearchPurpose::CorrelationToAvailable)
    499         {
    500             // For installed package to available package correlation,
    501             // try the search with NormalizedName with Arch first, if not found, try with all values.
    502             // This can be extended in the future for more granular search requests.
    503             std::vector<SearchRequest> candidateSearches;
    504             auto candidateSearchWithArch = request;
    505             if (anon::UpdatePackageMatchFilters(candidateSearchWithArch.Inclusions, m_normalizer, Utility::NormalizationField::Architecture))
    506             {
    507                 candidateSearches.emplace_back(std::move(candidateSearchWithArch));
    508             }
    509             anon::UpdatePackageMatchFilters(request.Inclusions, m_normalizer);
    510             candidateSearches.emplace_back(request);
    511 
    512             SearchResult result;
    513             for (auto& candidateSearch : candidateSearches)
    514             {
    515                 result = BasicSearchInternal(connection, candidateSearch);
    516                 if (!result.Matches.empty())
    517                 {
    518                     break;
    519                 }
    520             }
    521 
    522             return result;
    523         }
    524         else
    525         {
    526             anon::UpdatePackageMatchFilters(request.Inclusions, m_normalizer);
    527             anon::UpdatePackageMatchFilters(request.Filters, m_normalizer);
    528 
    529             return BasicSearchInternal(connection, request);
    530         }
    531     }

    533     ISQLiteIndex::SearchResult Interface::BasicSearchInternal(const SQLite::Connection& connection, const SearchRequest& request) const
    534     {
    535         if (request.IsForEverything())
    536         {
    537             std::vector<SQLite::rowid_t> ids = PackagesTable::GetAllRowIds(connection, PackagesTable::IdColumn::Name, request.MaximumResults);
    538 
    539             SearchResult result;
    540             for (SQLite::rowid_t id : ids)
    541             {
    542                 result.Matches.emplace_back(std::make_pair(id, PackageMatchFilter(PackageMatchField::Id, MatchType::Wildcard)));
    543             }
    544 
    545             result.Truncated = (request.MaximumResults && PackagesTable::GetCount(connection) > request.MaximumResults);
    546 
    547             return result;
    548         }
    549 
    550         // First phase, create the search results table and populate it with the initial results.
    551         // If the Query is provided, we search across many fields and put results in together.
    552         // If Inclusions has fields, we add these to the data.
    553         // If neither is defined, we take the first filter and use it as the initial results search.
    554         std::unique_ptr<SearchResultsTable> resultsTable = CreateSearchResultsTable(connection);
    555         bool inclusionsAttempted = false;
    556 
    557         if (request.Query)
    558         {
    559             // Perform searches across multiple tables to populate the initial results.
    560             PerformQuerySearch(*resultsTable.get(), request.Query.value());
    561 
    562             inclusionsAttempted = true;
    563         }
    564 
    565         if (!request.Inclusions.empty())
    566         {
    567             for (auto include : request.Inclusions)
    568             {
    569                 for (MatchType match : GetDefaultMatchTypeOrder(include.Type))
    570                 {
    571                     include.Type = match;
    572                     resultsTable->SearchOnField(include);
    573                 }
    574             }
    575 
    576             inclusionsAttempted = true;
    577         }
    578 
    579         size_t filterIndex = 0;
    580         if (!inclusionsAttempted)
    581         {
    582             THROW_HR_IF(E_UNEXPECTED, request.Filters.empty());
    583 
    584             // Perform search for just the field matching the first filter
    585             PackageMatchFilter filter = request.Filters[0];
    586 
    587             for (MatchType match : GetDefaultMatchTypeOrder(filter.Type))
    588             {
    589                 filter.Type = match;
    590                 resultsTable->SearchOnField(filter);
    591             }
    592 
    593             // Skip the filter as we already know everything matches
    594             filterIndex = 1;
    595         }
    596 
    597         // Remove any duplicate manifest entries
    598         resultsTable->RemoveDuplicatePackageRows();
    599 
    600         // Second phase, for remaining filters, flag matching search results, then remove unflagged values.
    601         for (size_t i = filterIndex; i < request.Filters.size(); ++i)
    602         {
    603             PackageMatchFilter filter = request.Filters[i];
    604 
    605             resultsTable->PrepareToFilter();
    606 
    607             for (MatchType match : GetDefaultMatchTypeOrder(filter.Type))
    608             {
    609                 filter.Type = match;
    610                 resultsTable->FilterOnField(filter);
    611             }
    612 
    613             resultsTable->CompleteFilter();
    614         }
    615 
    616         return resultsTable->GetSearchResults(request.MaximumResults);
    617     }

    619     void Interface::PrepareForPackaging(const SQLiteIndexContext& context, bool vacuum)
    620     {
    621         SQLite::Connection& connection = context.Connection;
    622 
    623         // Get the base time from metadata
    624         int64_t updateBaseTime = 0;
    625         std::optional<std::string> updateBaseTimeString = SQLite::MetadataTable::TryGetNamedValue<std::string>(connection, s_MetadataValueName_PackageUpdateTrackingBaseTime);
    626         if (updateBaseTimeString && !updateBaseTimeString->empty())
    627         {
    628             updateBaseTime = std::stoll(updateBaseTimeString.value());
    629         }
    630 
    631         // Get the output directory or use the file path
    632         std::filesystem::path baseOutputDirectory;
    633 
    634         if (context.Data.Contains(Property::IntermediateFileOutputPath))
    635         {
    636             baseOutputDirectory = context.Data.Get<Property::IntermediateFileOutputPath>();
    637         }
    638         else if (context.Data.Contains(Property::DatabaseFilePath))
    639         {
    640             baseOutputDirectory = context.Data.Get<Property::DatabaseFilePath>();
    641             baseOutputDirectory = baseOutputDirectory.parent_path();
    642         }
    643 
    644         THROW_WIN32_IF(ERROR_INVALID_STATE, baseOutputDirectory.empty() || baseOutputDirectory.is_relative());
    645 
    646         // Output all of the changed package version manifests since the base time to the target location
    647         for (const auto& packageData : PackageUpdateTrackingTable::GetUpdatesSince(connection, updateBaseTime))
    648         {
    649             std::filesystem::path packageDirectory = baseOutputDirectory /
    650                 Manifest::PackageVersionDataManifest::GetRelativeDirectoryPath(packageData.PackageIdentifier, Utility::SHA256::ConvertToString(packageData.Hash));
    651 
    652             std::filesystem::create_directories(packageDirectory);
    653 
    654             std::filesystem::path manifestPath = packageDirectory / Manifest::PackageVersionDataManifest::VersionManifestCompressedFileName();
    655 
    656             AICLI_LOG(Repo, Info, << "Writing PackageVersionDataManifest for [" << packageData.PackageIdentifier << "] to [" << manifestPath << "]");
    657 
    658             std::ofstream stream(manifestPath, std::ios_base::out | std::ios_base::binary | std::ios_base::trunc);
    659             THROW_LAST_ERROR_IF(stream.fail());
    660             stream.write(reinterpret_cast<const char*>(packageData.Manifest.data()), packageData.Manifest.size());
    661             THROW_LAST_ERROR_IF(stream.fail());
    662             stream.flush();
    663         }
    664 
    665         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "prepareforpackaging_v2_0");
    666 
    667         // Create the 2.0 data tables
    668         PackagesTable::Create<
    669             PackagesTable::IdColumn,
    670             PackagesTable::NameColumn,
    671             PackagesTable::MonikerColumn,
    672             PackagesTable::LatestVersionColumn,
    673             PackagesTable::ARPMinVersionColumn,
    674             PackagesTable::ARPMaxVersionColumn,
    675             PackagesTable::HashColumn
    676         >(connection);
    677 
    678         TagsTable::Create(connection, GetOneToManyTableSchema());
    679         CommandsTable::Create(connection, GetOneToManyTableSchema());
    680 
    681         PackageFamilyNameTable::Create(connection);
    682         ProductCodeTable::Create(connection);
    683         NormalizedPackageNameTable::Create(connection);
    684         NormalizedPackagePublisherTable::Create(connection);
    685         UpgradeCodeTable::Create(connection);
    686 
    687         // Copy data from 1.7 tables to 2.0 tables
    688         SearchResult allPackages = m_internalInterface->Search(connection, {});
    689 
    690         for (const auto& packageMatch : allPackages.Matches)
    691         {
    692             std::vector<ISQLiteIndex::VersionKey> versionKeys = m_internalInterface->GetVersionKeysById(connection, packageMatch.first);
    693             ISQLiteIndex::VersionKey& latestVersionKey = versionKeys[0];
    694 
    695             std::string packageIdentifier = m_internalInterface->GetPropertyByPrimaryId(connection, latestVersionKey.ManifestId, PackageVersionProperty::Id).value();
    696 
    697             std::vector<PackagesTable::NameValuePair> packageData{
    698                 { PackagesTable::IdColumn::Name, packageIdentifier },
    699                 { PackagesTable::NameColumn::Name, m_internalInterface->GetPropertyByPrimaryId(connection, latestVersionKey.ManifestId, PackageVersionProperty::Name).value() },
    700                 { PackagesTable::LatestVersionColumn::Name, latestVersionKey.VersionAndChannel.GetVersion().ToString() },
    701             };
    702 
    703             auto addIfPresent = [&](std::string_view name, std::optional<std::string>&& value)
    704                 {
    705                     if (value && !value->empty())
    706                     {
    707                         packageData.emplace_back(PackagesTable::NameValuePair{ name, std::move(value).value() });
    708                     }
    709                 };
    710 
    711             addIfPresent(PackagesTable::MonikerColumn::Name, m_internalInterface->GetPropertyByPrimaryId(connection, latestVersionKey.ManifestId, PackageVersionProperty::Moniker).value());
    712             addIfPresent(PackagesTable::ARPMinVersionColumn::Name, m_internalInterface->GetPropertyByPrimaryId(connection, latestVersionKey.ManifestId, PackageVersionProperty::ArpMinVersion).value());
    713             addIfPresent(PackagesTable::ARPMaxVersionColumn::Name, m_internalInterface->GetPropertyByPrimaryId(connection, latestVersionKey.ManifestId, PackageVersionProperty::ArpMaxVersion).value());
    714 
    715             SQLite::rowid_t packageId = PackagesTable::Insert(connection, packageData);
    716 
    717             PackagesTable::UpdateValueIdById<PackagesTable::HashColumn>(connection, packageId, PackageUpdateTrackingTable::GetDataHash(connection, packageIdentifier));
    718 
    719             for (const auto& versionKey : versionKeys)
    720             {
    721                 TagsTable::EnsureExistsAndInsert(connection, m_internalInterface->GetMultiPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionMultiProperty::Tag), packageId);
    722                 CommandsTable::EnsureExistsAndInsert(connection, m_internalInterface->GetMultiPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionMultiProperty::Command), packageId);
    723 
    724                 PackageFamilyNameTable::EnsureExists(connection, m_internalInterface->GetMultiPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionMultiProperty::PackageFamilyName), packageId);
    725                 ProductCodeTable::EnsureExists(connection, m_internalInterface->GetMultiPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionMultiProperty::ProductCode), packageId);
    726                 NormalizedPackageNameTable::EnsureExists(connection, m_internalInterface->GetMultiPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionMultiProperty::Name), packageId);
    727                 NormalizedPackagePublisherTable::EnsureExists(connection, m_internalInterface->GetMultiPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionMultiProperty::Publisher), packageId);
    728                 UpgradeCodeTable::EnsureExists(connection, m_internalInterface->GetMultiPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionMultiProperty::UpgradeCode), packageId);
    729             }
    730         }
    731 
    732         PackagesTable::PrepareForPackaging<
    733             PackagesTable::IdColumn,
    734             PackagesTable::NameColumn,
    735             PackagesTable::MonikerColumn,
    736             PackagesTable::LatestVersionColumn,
    737             PackagesTable::ARPMinVersionColumn,
    738             PackagesTable::ARPMaxVersionColumn,
    739             PackagesTable::HashColumn
    740         >(connection);
    741 
    742         TagsTable::PrepareForPackaging(connection);
    743         CommandsTable::PrepareForPackaging(connection);
    744 
    745         PackageUpdateTrackingTable::Drop(connection);
    746 
    747         // The tables based on SystemReferenceStringTable don't need a prepare currently
    748 
    749         // Drop 1.7 tables
    750         m_internalInterface->DropTables(connection);
    751 
    752         savepoint.Commit();
    753 
    754         m_internalInterface.reset();
    755 
    756         if (vacuum)
    757         {
    758             Vacuum(connection);
    759         }
    760     }

    762     void Interface::Vacuum(const SQLite::Connection& connection)
    763     {
    764         SQLite::Builder::StatementBuilder builder;
    765         builder.Vacuum();
    766         builder.Execute(connection);
    767     }

    769     void Interface::EnsureInternalInterface(const SQLite::Connection& connection, bool requireInternalInterface) const
    770     {
    771         if (!m_internalInterfaceChecked)
    772         {
    773             if (!PackagesTable::Exists(connection))
    774             {
    775                 m_internalInterface = CreateInternalInterface();
    776             }
    777 
    778             m_internalInterfaceChecked = true;
    779         }
    780 
    781         THROW_HR_IF(E_NOT_VALID_STATE, requireInternalInterface && !m_internalInterface);
    782     }
