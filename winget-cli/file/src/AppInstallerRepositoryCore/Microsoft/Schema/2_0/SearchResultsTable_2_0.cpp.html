     85         // Where the subselect is built by the owning table.
     86         StatementBuilder builder;
     87         builder.InsertInto(GetQualifiedName()).Select().
     88             Column(QualifiedColumn(s_SearchResultsTable_SubSelect_TableAlias, s_SearchResultsTable_SubSelect_PackageAlias)).

     92             Value(sortOrdinal).
     93             Value(false).

     95

    113     void SearchResultsTable::RemoveDuplicatePackageRows()
    114     {
    115         using namespace SQLite::Builder;
    116 
    117         // Create a delete statement to leave only one row with a given package.
    118         // This will arbitrarily choose one of the rows if multiple have the same lowest sort order.
    119         // The goal is a statement like this:
    120         //      DELETE from <temp> where rowid not in (
    121         //          SELECT rowid from (
    122         //              SELECT rowid, min(sort) from <temp> group by package
    123         //          )
    124         //      )
    125         StatementBuilder builder;
    126         builder.DeleteFrom(GetQualifiedName()).Where(SQLite::RowIDName).Not().In().BeginParenthetical().
    127             Select(SQLite::RowIDName).From().BeginParenthetical().
    128                 Select().Column(SQLite::RowIDName).Column(Aggregate::Min, s_SearchResultsTable_SortValue).From(GetQualifiedName()).GroupBy(s_SearchResultsTable_Package).
    129             EndParenthetical().
    130         EndParenthetical();
    131 
    132         builder.Execute(m_connection);
    133         AICLI_LOG(SQL, Verbose, << "Removed " << m_connection.GetChanges() << " duplicate rows");
    134     }

    145     void SearchResultsTable::FilterOnField(const PackageMatchFilter& filter)
    146     {
    147         using namespace SQLite::Builder;
    148 
    149         // Create an update statement to mark rows that are found by the search.
    150         // This will arbitrarily choose one of the rows if multiple have the same lowest sort order.
    151         // The goal is a statement like this:
    152         //      UPDATE <temp> set filter = 1 where package in (
    153         //          SELECT p from (
    154         //              SELECT packages.rowid as p, packages.id as v from packages where packages.id = <value>
    155         //          )
    156         //      )
    157         StatementBuilder builder;
    158         builder.Update(GetQualifiedName()).Set().Column(s_SearchResultsTable_Filter).Equals(true).Where(s_SearchResultsTable_Package).In().BeginParenthetical().
    159             Select(s_SearchResultsTable_SubSelect_PackageAlias).From().BeginParenthetical();
    160 
    161         // Add the field specific portion
    162         std::vector<int> bindIndex = BuildSearchStatement(builder, filter.Field, filter.Type);
    163 
    164         if (bindIndex.empty())
    165         {
    166             AICLI_LOG(Repo, Verbose, << "PackageMatchField not supported in this version: " << ToString(filter.Field));
    167             return;
    168         }
    169 
    170         builder.EndParenthetical().EndParenthetical();
    171 
    172         SQLite::Statement statement = builder.Prepare(m_connection);
    173         BindStatementForMatchType(statement, filter, bindIndex);
    174         statement.Execute();

    188     ISQLiteIndex::SearchResult SearchResultsTable::GetSearchResults(size_t limit)
    189     {
    190         using namespace SQLite::Builder;
    191         using QCol = QualifiedColumn;
    192 
    193         // Select all of the results from the table; it is expected that RemoveDuplicatePackageRows has been called.
    194         StatementBuilder builder;
    195         builder.Select({
    196             s_SearchResultsTable_Package,
    197             s_SearchResultsTable_MatchField,
    198             s_SearchResultsTable_MatchType,
    199             s_SearchResultsTable_MatchValue,
    200         }).
    201         From(GetQualifiedName()).OrderBy(s_SearchResultsTable_SortValue);
    202 
    203         SQLite::Statement select = builder.Prepare(m_connection);
    204 
    205         ISQLiteIndex::SearchResult result;
    206         while (select.Step())
    207         {
    208             if (limit && result.Matches.size() >= limit)
    209             {
    210                 break;
    211             }
    212 
    213             result.Matches.emplace_back(select.GetColumn<SQLite::rowid_t>(0), 
    214                 PackageMatchFilter(select.GetColumn<PackageMatchField>(1), select.GetColumn<MatchType>(2), select.GetColumn<std::string>(3)));
    215         }
    216 
    217         result.Truncated = (select.GetState() != SQLite::Statement::State::Completed);
    218 
    219         return result;
    220     }

    227     std::vector<int> SearchResultsTable::BuildSearchStatement(
    228         SQLite::Builder::StatementBuilder& builder,
    229         PackageMatchField field,
    230         std::string_view manifestAlias,
    231         std::string_view valueAlias,
    232         bool useLike) const
    233     {
    234         std::vector<int> result;

    236         switch (field)

    238         case PackageMatchField::Id:

    240             break;
    241         case PackageMatchField::Name:
    242             result.push_back(PackagesTable::BuildSearchStatement(builder, PackagesTable::NameColumn::Name, manifestAlias, valueAlias, useLike));
    243             break;
    244         case PackageMatchField::Moniker:
    245             result.push_back(PackagesTable::BuildSearchStatement(builder, PackagesTable::MonikerColumn::Name, manifestAlias, valueAlias, useLike));
    246             break;
    247         case PackageMatchField::Tag:
    248             result.push_back(TagsTable::BuildSearchStatement(builder, manifestAlias, valueAlias, useLike));
    249             break;
    250         case PackageMatchField::Command:
    251             result.push_back(CommandsTable::BuildSearchStatement(builder, manifestAlias, valueAlias, useLike));
    252             break;
    253         case PackageMatchField::PackageFamilyName:
    254             result.push_back(PackageFamilyNameTable::BuildSearchStatement(builder, manifestAlias, valueAlias, useLike));
    255             break;
    256         case PackageMatchField::ProductCode:
    257             result.push_back(ProductCodeTable::BuildSearchStatement(builder, manifestAlias, valueAlias, useLike));
    258             break;
    259         case PackageMatchField::UpgradeCode:
    260             result.push_back(UpgradeCodeTable::BuildSearchStatement(builder, manifestAlias, valueAlias, useLike));
    261             break;

    263             result = NormalizedPackageNameTable::BuildPairedSearchStatement<NormalizedPackagePublisherTable>(builder, manifestAlias, valueAlias, useLike);
    264             break;
    265         }
    266 
    267         return result;
    268     }

    275     void SearchResultsTable::BindStatementForMatchType(SQLite::Statement& statement, MatchType match, int bindIndex, std::string_view value)
    276     {
    277         std::string valueToUse;
    278 
    279         if (MatchUsesLike(match))
    280         {
    281             valueToUse = SQLite::EscapeStringForLike(value);
    282         }
    283         else
    284         {
    285             valueToUse = value;
    286         }
    287 
    288         switch (match)
    289         {
    290         case AppInstaller::Repository::MatchType::StartsWith:
    291             valueToUse += '%';
    292             break;
    293         case AppInstaller::Repository::MatchType::Substring:
    294             valueToUse = "%"s + valueToUse + '%';
    295             break;
    296         default:
    297             // No changes required for others.
    298             break;
    299         }
    300 
    301         statement.Bind(bindIndex, valueToUse);
    302     }

    305     {
