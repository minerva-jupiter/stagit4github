     33     SQLite::rowid_t Interface::AddManifest(SQLite::Connection& connection, const Manifest::Manifest& manifest, const std::optional<std::filesystem::path>& relativePath)
     34     {
     35         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "addmanifest_v1_5");
     36 
     37         SQLite::rowid_t manifestId = V1_4::Interface::AddManifest(connection, manifest, relativePath);
     38 
     39         auto arpVersionRange = manifest.GetArpVersionRange();
     40         Manifest::string_t arpMinVersion, arpMaxVersion;
     41 
     42         if (!arpVersionRange.IsEmpty())
     43         {
     44             // Check to see if adding this version range will create a conflict
     45             SQLite::rowid_t packageIdentifier = V1_0::ManifestTable::GetIdById<V1_0::IdTable>(connection, manifestId).value();
     46             std::vector<Utility::VersionRange> ranges = GetArpVersionRanges(connection, packageIdentifier);
     47             ranges.push_back(arpVersionRange);
     48 
     49             if (Utility::HasOverlapInVersionRanges(ranges))
     50             {
     51                 AICLI_LOG(Repo, Error, << "Overlapped Arp version ranges found for package. All ranges currently in index followed by new range:\n" << [&]() {
     52                         std::stringstream stream;
     53                         for (const auto& range : ranges)
     54                         {
     55                             stream << '[' << range.GetMinVersion().ToString() << "] - [" << range.GetMaxVersion().ToString() << "]\n";
     56                         }
     57                         return std::move(stream).str();
     58                     }());
     59                 THROW_HR(APPINSTALLER_CLI_ERROR_ARP_VERSION_VALIDATION_FAILED);
     60             }
     61 
     62             arpMinVersion = arpVersionRange.GetMinVersion().ToString();
     63             arpMaxVersion = arpVersionRange.GetMaxVersion().ToString();
     64         }
     65 
     66         SQLite::rowid_t arpMinVersionId = V1_0::VersionTable::EnsureExists(connection, arpMinVersion);
     67         SQLite::rowid_t arpMaxVersionId = V1_0::VersionTable::EnsureExists(connection, arpMaxVersion);
     68         V1_0::ManifestTable::UpdateValueIdById<ArpMinVersionVirtualTable>(connection, manifestId, arpMinVersionId);
     69         V1_0::ManifestTable::UpdateValueIdById<ArpMaxVersionVirtualTable>(connection, manifestId, arpMaxVersionId);
     70 
     71         savepoint.Commit();
     72 
     73         return manifestId;
     74     }

     76     std::pair<bool, SQLite::rowid_t> Interface::UpdateManifest(SQLite::Connection& connection, const Manifest::Manifest& manifest, const std::optional<std::filesystem::path>& relativePath)
     77     {
     78         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "updatemanifest_v1_5");
     79 
     80         auto [indexModified, manifestId] = V1_4::Interface::UpdateManifest(connection, manifest, relativePath);
     81 
     82         auto [oldMinVersionId, oldMaxVersionId] =
     83             V1_0::ManifestTable::GetIdsById<ArpMinVersionVirtualTable, ArpMaxVersionVirtualTable>(connection, manifestId);
     84 
     85         auto arpVersionRange = manifest.GetArpVersionRange();
     86         Manifest::string_t arpMinVersion = arpVersionRange.IsEmpty() ? "" : arpVersionRange.GetMinVersion().ToString();
     87         Manifest::string_t arpMaxVersion = arpVersionRange.IsEmpty() ? "" : arpVersionRange.GetMaxVersion().ToString();
     88 
     89         SQLite::rowid_t arpMinVersionId = V1_0::VersionTable::EnsureExists(connection, arpMinVersion);
     90         SQLite::rowid_t arpMaxVersionId = V1_0::VersionTable::EnsureExists(connection, arpMaxVersion);
     91 
     92         // For cleaning up the old entries after update if applicable
     93         bool cleanOldMinVersionId = false;
     94         bool cleanOldMaxVersionId = false;
     95 
     96         if (arpMinVersionId != oldMinVersionId)
     97         {
     98             V1_0::ManifestTable::UpdateValueIdById<ArpMinVersionVirtualTable>(connection, manifestId, arpMinVersionId);
     99             cleanOldMinVersionId = true;
    100             indexModified = true;
    101         }
    102 
    103         if (arpMaxVersionId != oldMaxVersionId)
    104         {
    105             V1_0::ManifestTable::UpdateValueIdById<ArpMaxVersionVirtualTable>(connection, manifestId, arpMaxVersionId);
    106             cleanOldMaxVersionId = true;
    107             indexModified = true;
    108         }
    109 
    110         if (!arpVersionRange.IsEmpty())
    111         {
    112             // Check to see if the new set of version ranges created a conflict.
    113             // We could have done this before attempting the update but it would be more complex and SQLite gives us easy rollback.
    114             SQLite::rowid_t packageIdentifier = V1_0::ManifestTable::GetIdById<V1_0::IdTable>(connection, manifestId).value();
    115             std::vector<Utility::VersionRange> ranges = GetArpVersionRanges(connection, packageIdentifier);
    116 
    117             if (Utility::HasOverlapInVersionRanges(ranges))
    118             {
    119                 AICLI_LOG(Repo, Error, << "Overlapped Arp version ranges found for package. Ranges that would be present with attempted upgrade:\n" << [&]() {
    120                         std::stringstream stream;
    121                         for (const auto& range : ranges)
    122                         {
    123                             stream << '[' << range.GetMinVersion().ToString() << "] - [" << range.GetMaxVersion().ToString() << "]\n";
    124                         }
    125                         return std::move(stream).str();
    126                     }());
    127                 THROW_HR(APPINSTALLER_CLI_ERROR_ARP_VERSION_VALIDATION_FAILED);
    128             }
    129         }
    130 
    131         if (cleanOldMinVersionId && NotNeeded(connection, V1_0::VersionTable::TableName(), V1_0::VersionTable::ValueName(), oldMinVersionId))
    132         {
    133             V1_0::VersionTable::DeleteById(connection, oldMinVersionId);
    134         }
    135 
    136         if (cleanOldMaxVersionId && oldMaxVersionId != oldMinVersionId && NotNeeded(connection, V1_0::VersionTable::TableName(), V1_0::VersionTable::ValueName(), oldMaxVersionId))
    137         {
    138             V1_0::VersionTable::DeleteById(connection, oldMaxVersionId);
    139         }
    140 
    141         savepoint.Commit();
    142 
    143         return { indexModified, manifestId };
    144      }

    146     void Interface::RemoveManifestById(SQLite::Connection& connection, SQLite::rowid_t manifestId)
    147     {
    148         // Get the old arp version ids of the values from the manifest table
    149         auto [arpMinVersionId, arpMaxVersionId] =
    150             V1_0::ManifestTable::GetIdsById<ArpMinVersionVirtualTable, ArpMaxVersionVirtualTable>(connection, manifestId);
    151 
    152         SQLite::Savepoint savepoint = SQLite::Savepoint::Create(connection, "RemoveManifestById_v1_5");
    153 
    154         // Removes the manifest.
    155         V1_4::Interface::RemoveManifestById(connection, manifestId);
    156 
    157         // Remove the versions that are not needed.
    158         if (NotNeeded(connection, V1_0::VersionTable::TableName(), V1_0::VersionTable::ValueName(), arpMinVersionId))
    159         {
    160             V1_0::VersionTable::DeleteById(connection, arpMinVersionId);
    161         }
    162 
    163         if (arpMaxVersionId != arpMinVersionId && NotNeeded(connection, V1_0::VersionTable::TableName(), V1_0::VersionTable::ValueName(), arpMaxVersionId))
    164         {
    165             V1_0::VersionTable::DeleteById(connection, arpMaxVersionId);
    166         }
    167 
    168         savepoint.Commit();
    169     }

    171     bool Interface::NotNeeded(const SQLite::Connection& connection, std::string_view tableName, std::string_view valueName, SQLite::rowid_t id) const
    172     {
    173         bool result = V1_4::Interface::NotNeeded(connection, tableName, valueName, id);
    174 
    175         if (result && tableName == V1_0::VersionTable::TableName())
    176         {
    177             if (valueName != V1_0::VersionTable::ValueName())
    178             {
    179                 result = !V1_0::ManifestTable::IsValueReferenced(connection, V1_0::VersionTable::ValueName(), id) && result;
    180             }
    181             if (valueName != ArpMinVersionVirtualTable::ManifestColumnName())
    182             {
    183                 result = !V1_0::ManifestTable::IsValueReferenced(connection, ArpMinVersionVirtualTable::ManifestColumnName(), id) && result;
    184             }
    185             if (valueName != ArpMaxVersionVirtualTable::ManifestColumnName())
    186             {
    187                 result = !V1_0::ManifestTable::IsValueReferenced(connection, ArpMaxVersionVirtualTable::ManifestColumnName(), id) && result;
    188             }
    189         }

    230     std::vector<Utility::VersionRange> Interface::GetArpVersionRanges(const SQLite::Connection& connection, SQLite::rowid_t packageIdentifier) const
    231     {
    232         std::vector<Utility::VersionRange> ranges;
    233         auto versionKeys = GetVersionKeysById(connection, packageIdentifier);
    234         for (auto const& versionKey : versionKeys)
    235         {
    236             auto arpMinVersion = GetPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionProperty::ArpMinVersion).value_or("");
    237             auto arpMaxVersion = GetPropertyByPrimaryId(connection, versionKey.ManifestId, PackageVersionProperty::ArpMaxVersion).value_or("");
    238 
    239             // Either both empty or both not empty
    240             THROW_HR_IF(E_UNEXPECTED, arpMinVersion.empty() != arpMaxVersion.empty());
    241 
    242             if (!arpMinVersion.empty() && !arpMaxVersion.empty())
    243             {
    244                 ranges.emplace_back(Utility::VersionRange{ Utility::Version{ std::move(arpMinVersion) }, Utility::Version{ std::move(arpMaxVersion) } });
    245             }
    246         }
    247         return ranges;
    248     }

    250     bool Interface::ValidateArpVersionConsistency(const SQLite::Connection& connection, bool log) const
    251     {
    252         try
    253         {
    254             bool result = true;
    255 
    256             // Search everything
    257             SearchRequest request;
    258             auto searchResult = Search(connection, request);
    259             for (auto const& match : searchResult.Matches)
    260             {
    261                 // Get arp version ranges for each package to check
    262                 std::vector<Utility::VersionRange> ranges = GetArpVersionRanges(connection, match.first);
    263 
    264                 // Check overlap
    265                 if (Utility::HasOverlapInVersionRanges(ranges))
    266                 {
    267                     AICLI_LOG(Repo, Error, << "Overlapped Arp version ranges found for package. PackageRowId: " << match.first);
    268                     result = false;
    269 
    270                     if (!log)
    271                     {
    272                         break;
    273                     }
    274                 }
    275             }
    276 
    277             return result;
    278         }
    279         catch (...)
    280         {
    281             AICLI_LOG(Repo, Error, << "ValidateArpVersionConsistency() encountered internal error. Returning false.");
    282             return false;
    283         }
