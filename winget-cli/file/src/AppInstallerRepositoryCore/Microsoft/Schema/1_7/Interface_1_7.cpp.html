     18         bool ShouldFoldPropertyLookup(const SQLite::Connection& connection)
     19         {
     20             // Get the metadata indicator that we folded these multi properties.
     21             // If it contains the value for folding these properties in the 1.7 manner, also fold the incoming manifest
     22             // to the same value that it would have been folded to so that all manifest entries will have all of these properties.
     23             std::optional<std::string> mapDataFolded = SQLite::MetadataTable::TryGetNamedValue<std::string>(connection, s_MetadataValueName_MapDataFolded);
     24 
     25             if (mapDataFolded)
     26             {
     27                 std::vector<std::string> foldedSplit = Utility::Split(mapDataFolded.value(), s_MetadataValue_MapDataFolded_Separator);
     28 
     29                 for (const std::string& splitValue : foldedSplit)
     30                 {
     31                     if (splitValue == Interface::MapDataFolded_VersionSpecifier)
     32                     {
     33                         return true;
     34                     }
     35                 }
     36             }
     37 
     38             return false;
     39         }
     40     }
