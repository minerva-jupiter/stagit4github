     83         void UpdateStatus(
     84             Connection& connection,
     85             const GUID& setInstanceIdentifier,
     86             const std::optional<int32_t>& state,
     87             const std::optional<bool>& inQueue,
     88             const std::optional<GUID>& unitInstanceIdentifier = std::nullopt,
     89             const std::optional<int32_t>& resultCode = std::nullopt,
     90             const std::optional<std::string>& resultDescription = std::nullopt,
     91             const std::optional<std::string>& resultDetails = std::nullopt,
     92             const std::optional<ConfigurationUnitResultSource>& resultSource = std::nullopt)
     93         {
     94             static constexpr std::string_view s_alias = "sub_expression";
     95 
     96             int64_t changeIdentifier = GetNextChangeIdentifier(connection);
     97             int64_t changeTime = GetCurrentUnixEpoch();
     98 
     99             // Statement like:
    100             // Update status_items set state = 1 from (Select rowid from set_info where instance_identifier = "foo") as sub where status_items.set_rowid = sub.rowid
    101             StatementBuilder updateBuilder;
    102             updateBuilder.Update(s_StatusItemTable_Table).Set();
    103 
    104             if (state)
    105             {
    106                 updateBuilder.Column(s_StatusItemTable_Column_State).Equals(state.value());
    107             }
    108 
    109             if (inQueue)
    110             {
    111                 updateBuilder.Column(s_StatusItemTable_Column_InQueue).Equals(inQueue.value());
    112             }
    113 
    114             if (resultCode)
    115             {
    116                 updateBuilder.Column(s_StatusItemTable_Column_ResultCode).Equals(resultCode.value());
    117             }
    118 
    119             if (resultDescription)
    120             {
    121                 updateBuilder.Column(s_StatusItemTable_Column_ResultDescription).Equals(resultDescription.value());
    122             }
    123 
    124             if (resultDetails)
    125             {
    126                 updateBuilder.Column(s_StatusItemTable_Column_ResultDetails).Equals(resultDetails.value());
    127             }
    128 
    129             if (resultSource)
    130             {
    131                 updateBuilder.Column(s_StatusItemTable_Column_ResultSource).Equals(resultSource.value());
    132             }
    133 
    134             updateBuilder.
    135                 Column(s_StatusItemTable_Column_ChangeIdentifier).Equals(changeIdentifier).
    136                 Column(s_StatusItemTable_Column_ChangeTimeLatest).Equals(changeTime).
    137             From().BeginParenthetical().
    138                 Select(RowIDName).From(V0_1::SetInfoTable::TableName()).Where(V0_1::SetInfoTable::InstanceIdentifierColumn()).Equals(setInstanceIdentifier).
    139             EndParenthetical().As(s_alias).Where(QualifiedColumn{ s_StatusItemTable_Table, s_StatusItemTable_Column_SetRowId }).Equals(QualifiedColumn{ s_alias, RowIDName }).
    140                 And(QualifiedColumn{ s_StatusItemTable_Table, s_StatusItemTable_Column_UnitInstanceIdentifier }).Equals(unitInstanceIdentifier);
    141 
    142             updateBuilder.Execute(connection);
    143 
    144             if (connection.GetChanges() == 0)
    145             {
    146                 // No change; we need to insert the status row
    147                 StatementBuilder insertBuilder;
    148                 insertBuilder.InsertInto(s_StatusItemTable_Table).Columns({
    149                     s_StatusItemTable_Column_ChangeIdentifier,
    150                     s_StatusItemTable_Column_ChangeTimeInitial,
    151                     s_StatusItemTable_Column_ChangeTimeLatest,
    152                     s_StatusItemTable_Column_SetRowId,

    157                     s_StatusItemTable_Column_ResultDescription,
    158                     s_StatusItemTable_Column_ResultDetails,
    159                     s_StatusItemTable_Column_ResultSource,
    160                 }).Select().
    161                     Value(changeIdentifier).
    162                     Value(changeTime).
    163                     Value(changeTime).
    164                     Column(QualifiedColumn{ V0_1::SetInfoTable::TableName(), RowIDName }).
    165                     Value(inQueue.value_or(false)).
    166                     Value(unitInstanceIdentifier).
    167                     Value(state.value_or(0)).
    168                     Value(resultCode).
    169                     Value(resultDescription).
    170                     Value(resultDetails).
    171                     Value(resultSource).
    172                 From(V0_1::SetInfoTable::TableName()).Where(QualifiedColumn{ V0_1::SetInfoTable::TableName(), V0_1::SetInfoTable::InstanceIdentifierColumn() }).Equals(setInstanceIdentifier);

    178         Statement PrepareSelectStatusValues(Connection& connection, const std::optional<GUID>& setInstanceIdentifier, const std::optional<GUID>& unitInstanceIdentifier, std::initializer_list<std::string_view> columns)
    179         {
    180             THROW_HR_IF(E_INVALIDARG, (setInstanceIdentifier && unitInstanceIdentifier) || (!setInstanceIdentifier && !unitInstanceIdentifier));
    181 
    182             StatementBuilder builder;
    183             builder.Select(columns).From(s_StatusItemTable_Table);
    184

    188                     Where(QualifiedColumn{ V0_1::SetInfoTable::TableName(), V0_1::SetInfoTable::InstanceIdentifierColumn() }).Equals(setInstanceIdentifier).
    189                     And(s_StatusItemTable_Column_UnitInstanceIdentifier).IsNull();
    190             }
    191             else
    192             {
    193                 builder.Where(s_StatusItemTable_Column_UnitInstanceIdentifier).Equals(unitInstanceIdentifier.value());
    194             }
    195 
    196             return builder.Prepare(connection);
    197         }
