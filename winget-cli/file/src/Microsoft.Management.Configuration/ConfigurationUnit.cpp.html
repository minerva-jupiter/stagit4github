     15         ValueSet Clone(const ValueSet& source)
     16         {
     17             ValueSet result;
     18 
     19             for (const auto& entry : source)
     20             {
     21                 ValueSet child = entry.Value().try_as<ValueSet>();
     22 
     23                 if (child)
     24                 {
     25                     result.Insert(entry.Key(), Clone(child));
     26                 }
     27                 else
     28                 {
     29                     result.Insert(entry.Key(), entry.Value());
     30                 }
     31             }
     32 
     33             return result;
     34         }

     36         Windows::Foundation::Collections::IVector<hstring> Clone(const Windows::Foundation::Collections::IVector<hstring>& value)
     37         {
     38             std::vector<hstring> temp{ value.Size() };
     39             value.GetMany(0, temp);
     40             return winrt::multi_threaded_vector<hstring>(std::move(temp));
     41         }
     42     }
