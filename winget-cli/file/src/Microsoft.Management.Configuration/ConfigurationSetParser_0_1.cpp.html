     16     void ConfigurationSetParser_0_1::Parse()
     17     {
     18         std::vector<Configuration::ConfigurationUnit> units;
     19         const Node& properties = m_document[GetConfigurationFieldName(ConfigurationField::Properties)];
     20         ParseConfigurationUnitsFromField(properties, ConfigurationField::Assertions, ConfigurationUnitIntent::Assert, units);
     21         ParseConfigurationUnitsFromField(properties, ConfigurationField::Parameters, ConfigurationUnitIntent::Inform, units);
     22         ParseConfigurationUnitsFromField(properties, ConfigurationField::Resources, ConfigurationUnitIntent::Apply, units);
     23 
     24         m_configurationSet = make_self<implementation::ConfigurationSet>();
     25         m_configurationSet->Units(std::move(units));
     26         m_configurationSet->SchemaVersion(GetSchemaVersion());
     27     }

     44     void ConfigurationSetParser_0_1::ParseConfigurationUnitsFromField(const Node& document, ConfigurationField field, ConfigurationUnitIntent intent, std::vector<Configuration::ConfigurationUnit>& result)
     45     {
     46         ParseSequence(document, field, false, Node::Type::Mapping, [&](const Node& item)
     47             {
     48                 auto configurationUnit = make_self<ConfigurationUnit>();
     49                 ParseConfigurationUnit(configurationUnit.get(), item, intent);
     50                 result.emplace_back(*configurationUnit);
     51             });
     52     }

     54     void ConfigurationSetParser_0_1::ParseConfigurationUnit(ConfigurationUnit* unit, const Node& unitNode, ConfigurationUnitIntent intent)
     55     {
     56         CHECK_ERROR(GetStringValueForUnit(unitNode, ConfigurationField::Resource, true, unit, &ConfigurationUnit::Type));
     57         CHECK_ERROR(GetStringValueForUnit(unitNode, ConfigurationField::Id, false, unit, &ConfigurationUnit::Identifier));
     58         unit->Intent(intent);
     59         CHECK_ERROR(GetStringArrayForUnit(unitNode, ConfigurationField::DependsOn, false, unit, &ConfigurationUnit::Dependencies));
     60         CHECK_ERROR(ParseValueSet(unitNode, ConfigurationField::Directives, false, unit->Metadata()));
     61         CHECK_ERROR(ParseValueSet(unitNode, ConfigurationField::Settings, false, unit->Settings()));
     62     }
