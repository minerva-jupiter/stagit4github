    117 void RequireLessThan(std::string_view a, std::string_view b)
    118 {
    119     Version vA{ std::string(a) };
    120     Version vB{ std::string(b) };
    121 
    122     REQUIRE(vA < vB);
    123     REQUIRE_FALSE(vB < vA);
    124     REQUIRE(vA <= vB);
    125     REQUIRE_FALSE(vB <= vA);
    126     REQUIRE(vB > vA);
    127     REQUIRE_FALSE(vA > vB);
    128     REQUIRE(vB >= vA);
    129     REQUIRE_FALSE(vA >= vB);
    130     REQUIRE_FALSE(vA == vB);
    131     REQUIRE(vA != vB);
    132 }

    193 TEST_CASE("VersionAndChannelSort", "[versions]")
    194 {
    195     std::vector<VersionAndChannel> sortedList =
    196     {
    197         { Version("15.0.0"), Channel("") },
    198         { Version("14.0.0"), Channel("") },
    199         { Version("13.2.1-bugfix"), Channel("") },
    200         { Version("13.2.0"), Channel("") },
    201         { Version("13.2.0-rc"), Channel("") },
    202         { Version("13.0.0"), Channel("") },
    203         { Version("16.0.0"), Channel("alpha") },
    204         { Version("15.8.0"), Channel("alpha") },
    205         { Version("15.1.0"), Channel("beta") },
    206     };
    207 
    208     std::vector<size_t> reorderList = { 4, 2, 1, 7, 6, 3, 8, 5, 0 };
    209     REQUIRE(sortedList.size() == reorderList.size());
    210 
    211     std::vector<VersionAndChannel> jumbledList;
    212     for (auto i : reorderList)
    213     {
    214         jumbledList.emplace_back(sortedList[i]);
    215     }
    216 
    217     std::sort(jumbledList.begin(), jumbledList.end());
    218 
    219     for (size_t i = 0; i < jumbledList.size(); ++i)
    220     {
    221         const VersionAndChannel& sortedVAC = sortedList[i];
    222         const VersionAndChannel& jumbledVAC = jumbledList[i];
    223 
    224         INFO(i);
    225         REQUIRE(sortedVAC.GetVersion().ToString() == jumbledVAC.GetVersion().ToString());
    226         REQUIRE(sortedVAC.GetChannel().ToString() == jumbledVAC.GetChannel().ToString());
    227     }
    228 }

    375     REQUIRE_NOTHROW(VersionRange{ Version{ "1.0" }, Version{ "1.0" } });

    384 
    385     // Empty
    386     REQUIRE(VersionRange{}.IsEmpty());
    387     REQUIRE_THROWS(VersionRange{}.GetMinVersion());
    388     REQUIRE_THROWS(VersionRange{}.GetMaxVersion());
    389 
    390     // Less than compare
    391     REQUIRE_THROWS(VersionRange{ Version{ "0.5" }, Version{ "1.0" } } < VersionRange{ Version{ "1.0" }, Version{ "2.0" } });
    392     REQUIRE_THROWS(VersionRange{} < VersionRange{ Version{ "1.0" }, Version{ "2.0" } });
    393     REQUIRE(VersionRange{ Version{ "0.5" }, Version{ "1.0" } } < VersionRange{ Version{ "1.5" }, Version{ "2.0" } });
    394     REQUIRE_FALSE(VersionRange{ Version{ "1.5" }, Version{ "2.0" } } < VersionRange{ Version{ "0.5" }, Version{ "1.0" } });
