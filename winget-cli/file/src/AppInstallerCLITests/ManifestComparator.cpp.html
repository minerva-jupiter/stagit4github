     28 const ManifestInstaller& AddInstaller(
     29     Manifest& manifest,
     30     Architecture architecture,
     31     InstallerTypeEnum installerType,
     32     ScopeEnum scope = ScopeEnum::Unknown,
     33     std::string minOSVersion = {},
     34     std::string locale = {},
     35     std::vector<Architecture> unsupportedOSArchitectures = {},
     36     MarketsInfo markets = {})
     37 {
     38     ManifestInstaller toAdd;
     39     toAdd.Arch = architecture;
     40     toAdd.BaseInstallerType = installerType;
     41     toAdd.Scope = scope;
     42     toAdd.MinOSVersion = minOSVersion;
     43     toAdd.Locale = locale;
     44     toAdd.UnsupportedOSArchitectures = unsupportedOSArchitectures;
     45     toAdd.Markets = markets;
     46 
     47     manifest.Installers.emplace_back(std::move(toAdd));
     48 
     49     return manifest.Installers.back();
     50 }

     52 template<typename T>
     53 void RequireVectorsEqual(const std::vector<T>& actual, const std::vector<T>& expected)
     54 {
     55     REQUIRE(actual.size() == expected.size());
     56     
     57     for (std::size_t i = 0; i < actual.size(); ++i)
     58     {
     59         REQUIRE(actual[i] == expected[i]);
     60     }
     61 }
     62 
     63 void RequireInstaller(const std::optional<ManifestInstaller>& actual, const ManifestInstaller& expected)
     64 {
     65     REQUIRE(actual);
     66     REQUIRE(actual->Arch == expected.Arch);
     67     REQUIRE(actual->EffectiveInstallerType() == expected.EffectiveInstallerType());
     68     REQUIRE(actual->Scope == expected.Scope);
     69     REQUIRE(actual->MinOSVersion == expected.MinOSVersion);
     70     REQUIRE(actual->Locale == expected.Locale);
     71 
     72     RequireVectorsEqual(actual->Markets.AllowedMarkets, expected.Markets.AllowedMarkets);
     73     RequireVectorsEqual(actual->Markets.ExcludedMarkets, expected.Markets.ExcludedMarkets);
     74 }
     75 
     76 void RequireInapplicabilities(const std::vector<InapplicabilityFlags>& inapplicabilities, const std::vector<InapplicabilityFlags>& expected)
     77 {
     78     RequireVectorsEqual(inapplicabilities, expected);
     79 }

     81 void RequireInapplicabilityType(const std::vector<InapplicabilityFlags>& inapplicabilities, InapplicabilityFlags expected)
     82 {
     83     REQUIRE(!inapplicabilities.empty());
     84     for (std::size_t i = 0; i < inapplicabilities.size(); ++i)
     85     {
     86         REQUIRE(inapplicabilities[i] == expected);
     87     }
     88 }
