     60 template <typename Enumerable, typename Op>
     61 void EnsureStringsAreLowercaseAndNoCollisions(const std::string& info, const Enumerable& e, Op& op, std::unordered_set<std::string>& values, bool requireLower = true)
     62 {
     63     INFO(info);
     64 
     65     for (const auto& val : e)
     66     {
     67         std::string valString = op(val);
     68         if (valString.empty())
     69         {
     70             continue;
     71         }
     72         INFO(valString);
     73 
     74         if (requireLower)
     75         {
     76             REQUIRE(StringIsLowercase(valString));
     77         }
     78 
     79         REQUIRE(values.find(valString) == values.end());
     80 
     81         values.emplace(std::move(valString));
     82     }
     83 }

     85 template <typename Enumerable, typename Op>
     86 void EnsureStringsAreLowercaseAndNoCollisions(const std::string& info, const Enumerable& e, Op& op, bool requireLower = true)
     87 {
     88     std::unordered_set<std::string> values;
     89     EnsureStringsAreLowercaseAndNoCollisions(info, e, op, values, requireLower);
     90 }

     92 template <typename Enumerable, typename Op>
     93 void EnsureVectorStringViewsAreLowercaseAndNoCollisions(const std::string& info, const Enumerable& e, Op& op, std::unordered_set<std::string>& values, bool requireLower = true)
     94 {
     95     INFO(info);
     96 
     97     for (const auto& val : e)
     98     { 
     99         std::vector<std::string_view> aliasVector = op(val);
    100         std::vector<std::string> valVector(aliasVector.begin(), aliasVector.end());
    101         if (valVector.empty())
    102         {
    103             continue;
    104         }
    105         // When op returns a vector, we need to ensure every value in the vector does not cause a collision
    106         for (auto& valString : valVector)
    107         {
    108             INFO(valString);
    109 
    110             if (requireLower)
    111             {
    112                 REQUIRE(StringIsLowercase(valString));
    113             }
    114 
    115             REQUIRE(values.find(valString) == values.end());
    116             values.emplace(std::move(valString));
    117         }
    118     }
    119 }

    121 void EnsureCommandConsistency(const Command& command)
    122 {
    123     // Command names and aliases exist in the same space, so both need to be checked as a set
    124     // However, collisions do not occur between levels, so the full name must be used to check for collision
    125     std::unordered_set<std::string> allCommandAliasNames; 
    126     EnsureStringsAreLowercaseAndNoCollisions(command.FullName() + " commands", command.GetCommands(), GetCommandName, allCommandAliasNames);
    127     EnsureVectorStringViewsAreLowercaseAndNoCollisions(command.FullName() + " aliases", command.GetCommands(), GetCommandAliases, allCommandAliasNames);
    128 
    129     auto args = command.GetArguments();
    130     Argument::GetCommon(args);
    131 
    132     // Argument names and alternate names exist in the same space, so both need to be checked as a set
    133     std::unordered_set<std::string> allArgumentNames;
    134     EnsureStringsAreLowercaseAndNoCollisions(command.FullName() + " argument names", args, GetArgumentName, allArgumentNames);
    135     EnsureStringsAreLowercaseAndNoCollisions(command.FullName() + " argument alternate name", args, GetArgumentAlternateName, allArgumentNames);
    136     // Argument aliases use a different space than the names and can be checked separately
    137     EnsureStringsAreLowercaseAndNoCollisions(command.FullName() + " argument alias", args, GetArgumentAlias, false);
    138 
    139     // No : allowed in commands
    140     for (const auto& comm : command.GetCommands())
    141     {
    142         INFO(command.FullName());
    143         INFO(comm->Name());
    144 
    145         REQUIRE(comm->Name().find_first_of(Command::ParentSplitChar) == std::string_view::npos);
    146     }
    147 
    148     // No = allowed in arguments
    149     // All positional args should be listed first
    150     bool foundNonPositional = false;
    151     bool queryArgPresent = false;
    152     bool multiQueryArgPresent = false;
    153     for (const auto& arg : command.GetArguments())
    154     {
    155         INFO(command.FullName());
    156         INFO(arg.Name());
    157 
    158         REQUIRE(arg.Name().find_first_of(APPINSTALLER_CLI_ARGUMENT_SPLIT_CHAR) == std::string_view::npos);
    159 
    160         if (arg.Type() == ArgumentType::Positional)
    161         {
    162             REQUIRE(!foundNonPositional);
    163         }
    164         else
    165         {
    166             foundNonPositional = true;
    167         }
    168 
    169         if (arg.ExecArgType() == Execution::Args::Type::Query)
    170         {
    171             queryArgPresent = true;
    172         }
    173 
    174         if (arg.ExecArgType() == Execution::Args::Type::MultiQuery)
    175         {
    176             multiQueryArgPresent = true;
    177         }
    178     }
    179 
    180     REQUIRE((!queryArgPresent || !multiQueryArgPresent));
    181 
    182     // Recurse for all subcommands
    183     for (const auto& sub : command.GetCommands())
    184     {
    185         EnsureCommandConsistency(*sub.get());
    186     }
    187 }

    220 // Matcher that lets us verify CommandExceptions.
    221 struct CommandExceptionMatcher : public Catch::Matchers::MatcherBase<CommandException>
    222 {
    223     CommandExceptionMatcher(CLI::Resource::LocString message) : m_expectedMessage(std::move(message)) {}
    224 
    225     bool match(const CommandException& ce) const override
    226     {
    227         return ce.Message() == m_expectedMessage;
    228     }
    229 
    230     std::string describe() const override
    231     {
    232         std::ostringstream result;
    233         result << "has message == " << m_expectedMessage;
    234         return result.str();
    235     }
    236 
    237 private:
    238     CLI::Resource::LocString m_expectedMessage;
    239 };

    252 void RequireValueParsedToArg(const std::string& value, const Argument& arg, const Args& args)
    253 {
    254     REQUIRE(args.Contains(arg.ExecArgType()));
    255     REQUIRE(value == args.GetArg(arg.ExecArgType()));
    256 }

    258 void RequireValuesParsedToArg(const std::vector<std::string>& values, Args::Type execArgType, const Args& args)
    259 {
    260     REQUIRE(args.Contains(execArgType));
    261     REQUIRE(args.GetCount(execArgType) == values.size());
    262 
    263     auto argValues = args.GetArgs(execArgType);
    264     for (size_t i = 0; i < values.size(); ++i)
    265     {
    266         REQUIRE(argValues->at(i) == values[i]);
    267     }
    268 }

    270 void RequireValuesParsedToArg(const std::vector<std::string>& values, const Argument& arg, const Args& args)
    271 {
    272     RequireValuesParsedToArg(values, arg.ExecArgType(), args);
