     19     static Source SimpleTestSetup(const std::string& filePath, SourceDetails& details, Manifest& manifest, std::string& relativePath)
     20     {
     21         SQLiteIndex index = SQLiteIndex::CreateNew(filePath, AppInstaller::SQLite::Version::Latest(), SQLiteIndex::CreateOptions::SupportPathless | SQLiteIndex::CreateOptions::DisableDependenciesSupport);
     22 
     23         TestDataFile testManifest("Manifest-Good.yaml");
     24         manifest = YamlParser::CreateFromPath(testManifest);
     25 
     26         relativePath = testManifest.GetPath().filename().u8string();
     27 
     28         index.AddManifest(manifest, relativePath);
     29 
     30         details.Identifier = "*SimpleTestSetup";
     31         details.Name = "TestName";
     32         details.Type = "TestType";
     33         details.Arg = testManifest.GetPath().parent_path().u8string();
     34         details.Data = "";
     35 
     36         auto result = std::make_shared<SQLiteIndexSource>(details, std::move(index));
     37 
     38         PackageTrackingCatalog::RemoveForSource(result->GetIdentifier());
     39 
     40         return { result };
     41     }

     67 TEST_CASE("TrackingCatalog_Install", "[tracking_catalog]")
     68 {
     69     TempFile tempFile{ "repolibtest_tempdb"s, ".db"s };
     70     INFO("Using temporary file named: " << tempFile.GetPath());
     71 
     72     SourceDetails details;
     73     Manifest manifest;
     74     std::string relativePath;
     75     auto source = SimpleTestSetup(tempFile, details, manifest, relativePath);
     76 
     77     PackageTrackingCatalog catalog = CreatePackageTrackingCatalogForSource(source);
     78 
     79     SearchRequest request;
     80     request.Filters.emplace_back(PackageMatchField::Id, MatchType::Exact, manifest.Id);
     81 
     82     SearchResult resultBefore = catalog.Search(request);
     83     REQUIRE(resultBefore.Matches.size() == 0);
     84 
     85     catalog.RecordInstall(manifest, manifest.Installers[0], false);
     86 
     87     SearchResult resultAfter = catalog.Search(request);
     88     REQUIRE(resultAfter.Matches.size() == 1);
     89     REQUIRE(resultAfter.Matches[0].Package->GetAvailable().size() == 1);
     90 
     91     auto trackingVersion = resultAfter.Matches[0].Package->GetAvailable()[0]->GetLatestVersion();
     92     REQUIRE(trackingVersion);
     93 
     94     auto metadata = trackingVersion->GetMetadata();
     95     REQUIRE(metadata.find(PackageVersionMetadata::TrackingWriteTime) != metadata.end());
     96 }
