     65 TEST_CASE("AdminSetting_AllSettingsAreImplemented", "[adminSettings]")
     66 {
     67     for (auto adminSetting : GetAllBoolAdminSettings())
     68     {
     69         // If we forget to add it to the conversion, it returns Unknown/None
     70         REQUIRE(AdminSettingToString(adminSetting) != AdminSettingToString(BoolAdminSetting::Unknown));
     71         REQUIRE(StringToBoolAdminSetting(AdminSettingToString(adminSetting)) != BoolAdminSetting::Unknown);
     72         REQUIRE(GetAdminSettingPolicy(adminSetting) != TogglePolicy::Policy::None);
     73 
     74         GroupPolicyTestOverride policies;
     75         policies.SetState(GetAdminSettingPolicy(adminSetting), PolicyState::NotConfigured);
     76 
     77         // We should be able to configure the state.
     78         // If we forget to add it, it won't persist.
     79         REQUIRE(EnableAdminSetting(adminSetting));
     80         REQUIRE(IsAdminSettingEnabled(adminSetting));
     81         REQUIRE(DisableAdminSetting(adminSetting));
     82         REQUIRE_FALSE(IsAdminSettingEnabled(adminSetting));
     83     }
