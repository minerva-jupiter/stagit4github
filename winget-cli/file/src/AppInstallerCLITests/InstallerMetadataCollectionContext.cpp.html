    167 
    168             auto metadataValue = JSON::GetJsonValueFromNode(input, L"metadata");
    169             if (metadataValue && !metadataValue->get().is_null())
    170             {
    171                 Metadata = std::make_optional<ProductMetadata>();
    172                 Metadata->FromJson(metadataValue->get());
    173             }
    174 
    175             auto diagnosticsValue = JSON::GetJsonValueFromNode(input, L"diagnostics");
    176             if (diagnosticsValue)
    177             {
    178                 auto errorHRNumber = JSON::GetRawIntValueFromJsonNode(diagnosticsValue.value(), L"errorHR");
    179                 if (errorHRNumber)
    180                 {
    181                     ErrorHR = std::move(errorHRNumber);
    182                 }

    184                 auto errorTextString = JSON::GetRawStringValueFromJsonNode(diagnosticsValue.value(), L"errorText");
