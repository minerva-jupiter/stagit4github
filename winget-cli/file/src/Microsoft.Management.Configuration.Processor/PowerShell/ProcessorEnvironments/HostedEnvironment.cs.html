     72         /// <inheritdoc/>
     73         public void ValidateRunspace()
     74         {
     75             // Only support PowerShell Core.
     76             if (this.GetVariable<string>(Variables.PSEdition) != Core)
     77             {
     78                 throw new NotSupportedException("Only PowerShell Core is supported.");
     79             }
     80 
     81             // If opening a runspace has failures, like one of the modules in ImportPSModule is not found, it won't throw but
     82             // write to the error output. This is not a fatal error, since we install PSDesiredStateConfiguration
     83             // module if not found, so unless there's a real reason keep it in verbose.
     84             var errors = this.GetVariable<ArrayList>(Variables.Error);
     85             if (errors.Count > 0)
     86             {
     87                 this.OnDiagnostics(
     88                     DiagnosticLevel.Verbose,
     89                     $"Error creating runspace '{string.Join("\n", errors.Cast<string>().ToArray())}'");
     90             }
     91 
     92             var powerShellGet = PowerShellHelpers.CreateModuleSpecification(
     93                     Modules.PowerShellGet,
     94                     minVersion: Modules.PowerShellGetMinVersion);
     95             if (!this.ValidateModule(powerShellGet))
     96             {
     97                 var previousVersion = this.GetAvailableModule(
     98                     PowerShellHelpers.CreateModuleSpecification(
     99                         Modules.PowerShellGet));
    100                 string message = $"Required '{powerShellGet}'";
    101                 if (previousVersion is not null)
    102                 {
    103                     message += $" Found '{previousVersion.Name} {previousVersion.Version}'";
    104                 }
    105 
    106                 throw new NotSupportedException(message);
    107             }
    108 
    109             // Make sure PSDesiredConfiguration is present.
    110             this.InstallModule(this.DscModule.ModuleSpecification);
    111         }

    240         /// <inheritdoc/>
    241         public PSObject? GetInstalledModule(ModuleSpecification moduleSpecification)
    242         {
    243             // Instead of Get-InstalledModule, we look for PSGetModuleInfo.xml and serialize it
    244             // if found. This allow us to get the information from Install-Module and Save-Module.
    245             var module = this.GetAvailableModule(moduleSpecification);
    246             if (module is null)
    247             {
    248                 return null;
    249             }
    250 
    251             var getModuleInfoFile = Path.Combine(module.ModuleBase, "PSGetModuleInfo.xml");
    252             if (!File.Exists(getModuleInfoFile))
    253             {
    254                 // Keep Get-InstalledModule behaviour.
    255                 return null;
    256             }
    257 
    258             using PowerShell pwsh = PowerShell.Create(this.Runspace);
    259             var installedModule = pwsh.AddCommand(Commands.ImportCliXml)
    260                                       .AddParameter(Parameters.Path, getModuleInfoFile)
    261                                       .Invoke()
    262                                       .FirstOrDefault();
    263 
    264             this.OnDiagnostics(DiagnosticLevel.Verbose, pwsh);
    265             return installedModule;
    266         }

    268         /// <inheritdoc/>
    269         public PSObject? FindModule(ConfigurationUnitInternal unitInternal)
    270         {
    271             // Don't use ModuleSpecification here. Each parameter is independent and
    272             // we need version even if a module was not specified.
    273             string? moduleName = unitInternal.GetDirective<string>(DirectiveConstants.Module);
    274 
    275             if (string.IsNullOrEmpty(moduleName))
    276             {
    277                 return null;
    278             }
    279 
    280             using PowerShell pwsh = PowerShell.Create(this.Runspace);
    281 
    282             var result = this.powerShellGet.FindModule(
    283                 pwsh,
    284                 moduleName,
    285                 unitInternal.GetSemanticVersion(),
    286                 unitInternal.GetSemanticMinVersion(),
    287                 unitInternal.GetSemanticMaxVersion(),
    288                 unitInternal.GetDirective<string>(DirectiveConstants.Repository),
    289                 unitInternal.GetDirective(DirectiveConstants.AllowPrerelease));
    290 
    291             this.OnDiagnostics(DiagnosticLevel.Verbose, pwsh);
    292             return result;
    293         }

    330         /// <inheritdoc/>
    331         public void InstallModule(PSObject inputObject)
    332         {
    333             if (this.location == PowerShellConfigurationProcessorLocation.Custom)
    334             {
    335                 if (string.IsNullOrEmpty(this.customLocation))
    336                 {
    337                     throw new ArgumentNullException(nameof(this.customLocation));
    338                 }
    339 
    340                 this.SaveModule(inputObject, this.customLocation);
    341             }
    342             else
    343             {
    344                 using PowerShell pwsh = PowerShell.Create(this.Runspace);
    345                 this.powerShellGet.InstallModule(pwsh, inputObject, this.location == PowerShellConfigurationProcessorLocation.AllUsers);
    346                 this.OnDiagnostics(DiagnosticLevel.Verbose, pwsh);
    347             }
    348         }

    350         /// <inheritdoc/>
    351         public void InstallModule(ModuleSpecification moduleSpecification)
    352         {
    353             // Maybe is already there.
    354             if (!this.ValidateModule(moduleSpecification))
    355             {
    356                 this.OnDiagnostics(DiagnosticLevel.Verbose, $"Installing module: {moduleSpecification.Name} ...");
    357 
    358                 // Ok, we have to get it.
    359                 if (this.location == PowerShellConfigurationProcessorLocation.Custom)
    360                 {
    361                     if (string.IsNullOrEmpty(this.customLocation))
    362                     {
    363                         throw new ArgumentNullException(nameof(this.customLocation));
    364                     }
    365 
    366                     this.OnDiagnostics(DiagnosticLevel.Verbose, $"... calling save module ...");
    367                     this.SaveModule(moduleSpecification, this.customLocation);
    368                 }
    369                 else
    370                 {
    371                     this.OnDiagnostics(DiagnosticLevel.Verbose, $"... calling install module ...");
    372                     using PowerShell pwsh = PowerShell.Create(this.Runspace);
    373                     this.powerShellGet.InstallModule(pwsh, moduleSpecification, this.location == PowerShellConfigurationProcessorLocation.AllUsers);
    374                     this.OnDiagnostics(DiagnosticLevel.Verbose, pwsh);
    375                 }
    376 
    377                 this.OnDiagnostics(DiagnosticLevel.Verbose, $" ... module installed.");
    378             }
    379         }

    381         /// <inheritdoc/>
    382         public List<Certificate> GetCertsOfValidSignedFiles(string[] paths)
    383         {
    384             using PowerShell pwsh = PowerShell.Create(this.Runspace);
    385 
    386             var signatures = pwsh.AddCommand(Commands.GetChildItem)
    387                                  .AddParameter(Parameters.Path, paths)
    388                                  .AddCommand(Commands.GetAuthenticodeSignature)
    389                                  .Invoke<Signature>();
    390 
    391             var thumbprint = new HashSet<string>();
    392             var certificates = new List<Certificate>();
    393             foreach (var signature in signatures)
    394             {
    395                 if (signature.Status == SignatureStatus.Valid)
    396                 {
    397                     if (thumbprint.Add(signature.SignerCertificate.Thumbprint))
    398                     {
    399                         IBuffer buffer = signature.SignerCertificate.GetRawCertData().AsBuffer();
    400                         certificates.Add(new Certificate(buffer));
    401                     }
    402                 }
    403             }
    404 
    405             this.OnDiagnostics(DiagnosticLevel.Verbose, pwsh);
    406             return certificates;
    407         }

    443         /// <inheritdoc/>
    444         public void PrependPSModulePaths(IReadOnlyList<string> paths)
    445         {
    446             var newPaths = paths.ToList();
    447             var oldModulePath = this.GetModulePaths();
    448             foreach (var newPath in paths)
    449             {
    450                 if (oldModulePath.Contains(newPath))
    451                 {
    452                     newPaths.Remove(newPath);
    453                 }
    454             }
    455 
    456             if (newPaths.Any())
    457             {
    458                 this.SetPSModulePath($"{string.Join(";", newPaths)};{string.Join(";", oldModulePath)}");
    459             }
    460         }

    516         private bool ValidateModule(ModuleSpecification moduleSpecification)
    517         {
    518             this.OnDiagnostics(DiagnosticLevel.Verbose, $"Validating module: {moduleSpecification.Name} ...");
    519 
    520             var loadedModule = this.GetImportedModule(moduleSpecification);
    521             if (loadedModule is not null)
    522             {
    523                 this.OnDiagnostics(DiagnosticLevel.Verbose, $" ... module is already imported.");
    524                 return true;
    525             }
    526 
    527             var availableModule = this.GetAvailableModule(moduleSpecification);
    528             if (availableModule is not null)
    529             {
    530                 this.OnDiagnostics(DiagnosticLevel.Verbose, $" ... module is available, importing ...");
    531                 this.ImportModule(moduleSpecification);
    532                 this.OnDiagnostics(DiagnosticLevel.Verbose, $" ... module imported.");
    533                 return true;
    534             }
    535 
    536             this.OnDiagnostics(DiagnosticLevel.Verbose, $" ... module not found.");
    537             return false;
    538         }
