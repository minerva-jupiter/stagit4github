     17     /// <summary>

     28         /// <summary>
     29         /// Initializes a new instance of the <see cref="ConfigurationUnitInternal"/> class.
     30         /// </summary>
     31         /// <param name="unit">Configuration unit.</param>
     32         /// <param name="configurationFilePath">The configuration file path.</param>
     33         public ConfigurationUnitInternal(
     34             ConfigurationUnit unit,
     35             string? configurationFilePath)
     36         {
     37             this.Unit = unit;
     38             this.InitializeDirectives();
     39             this.InitializeNames();
     40 
     41             if (!string.IsNullOrEmpty(configurationFilePath))
     42             {
     43                 if (!File.Exists(configurationFilePath))
     44                 {
     45                     throw new FileNotFoundException(configurationFilePath);
     46                 }
     47 
     48                 this.configurationFileRootPath = Path.GetDirectoryName(configurationFilePath);
     49             }
     50         }

     72         /// <summary>
     73         /// Gets the directive value from the unit taking into account the directives overlay.
     74         /// </summary>
     75         /// <param name="directiveName">Directive name.</param>
     76         /// <returns>Value of directive, null if not found.</returns>
     77         /// <typeparam name="TType">Directive type value.</typeparam>
     78         public TType? GetDirective<TType>(string directiveName)
     79             where TType : class
     80         {
     81             var normalizedDirectiveName = StringHelpers.Normalize(directiveName);
     82             if (this.normalizedDirectives.TryGetValue(normalizedDirectiveName, out object? value))
     83             {
     84                 return value as TType;
     85             }
     86 
     87             return null;
     88         }

    176         private string? ExpandConfigRoot(string? value, string settingName)
    177         {
    178             if (!string.IsNullOrEmpty(value))
    179             {
    180                 // TODO: since we only support one variable, this only finds and replace
    181                 // ${WingetConfigRoot} if found in the string when the work of expanding
    182                 // string is done it should take into account other operators like the subexpression operator $()
    183                 if (value.Contains(ConfigRootVar, StringComparison.OrdinalIgnoreCase))
    184                 {
    185                     if (string.IsNullOrEmpty(this.configurationFileRootPath))
    186                     {
    187                         throw new UnitSettingConfigRootException(this.QualifiedName, settingName);
    188                     }
    189 
    190                     return value.Replace(ConfigRootVar, this.configurationFileRootPath, StringComparison.OrdinalIgnoreCase);
    191                 }
    192             }
    193 
    194             return value;
    195         }

    197         private void InitializeDirectives()
    198         {
    199             foreach (var directive in this.Unit.Metadata)
    200             {
    201                 var normalizedKey = StringHelpers.Normalize(directive.Key);
    202                 this.normalizedDirectives.Add(normalizedKey, directive.Value);
    203             }
    204         }

    206         private string ConstructQualifiedName(string? moduleName)
    207         {
    208             return $"{(moduleName == null ? string.Empty : $"{moduleName}/")}{this.ResourceName}";
    209         }
    210 
    211         [MemberNotNull(nameof(ResourceName), nameof(QualifiedName))]
    212         private void InitializeNames()
    213         {
    214             // Determine ResourceName, QualifiedName, and the module directive
    215             string unitType = this.Unit.Type;
    216             string? moduleDirective = this.GetDirective<string>(DirectiveConstants.Module);
    217 
    218             if (this.UnitTypeIsResourceName)
    219             {
    220                 this.ResourceName = unitType;
    221                 this.QualifiedName = this.ConstructQualifiedName(moduleDirective);
    222                 return;
    223             }
    224 
    225             int unitTypeDividerPosition = unitType.IndexOf('/');
    226 
    227             if (unitTypeDividerPosition == unitType.Length - 1)
    228             {
    229                 throw new ArgumentException($"Invalid unit Type: {unitType}");
    230             }
    231 
    232             string? moduleName;
    233 
    234             if (unitTypeDividerPosition == -1)
    235             {
    236                 moduleName = moduleDirective;
    237                 this.ResourceName = unitType;
    238                 this.QualifiedName = this.ConstructQualifiedName(moduleDirective);
    239             }
    240             else
    241             {
    242                 moduleName = unitType.Substring(0, unitTypeDividerPosition);
    243                 this.ResourceName = unitType.Substring(unitTypeDividerPosition + 1);
    244                 this.QualifiedName = unitType;
    245             }
    246 
    247             if (moduleName != null)
    248             {
    249                 if (moduleDirective != null)
    250                 {
    251                     if (moduleName != moduleDirective)
    252                     {
    253                         throw new ArgumentException($"Mismatched module specifiers: {moduleName} != {moduleDirective}");
    254                     }
    255                 }
    256                 else
    257                 {
    258                     this.normalizedDirectives.Add(DirectiveConstants.Module, moduleName);
    259                 }
    260             }
    261         }
