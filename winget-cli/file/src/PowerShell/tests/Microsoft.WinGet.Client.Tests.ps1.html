     54     # This is a workaround to an issue where the server takes longer than expected to terminate when
     55     # running from PowerShell. This can cause other E2E tests to fail when attempting to reset the test source.
     56     function RemoveTestSource {
     57         try {
     58             # Source Remove requires admin privileges, this will only execute successfully in an elevated PowerShell.
     59             $testSource = Get-WinGetSource | Where-Object -Property 'Name' -eq 'TestSource'
     60             if ($null -ne $testSource)
     61             {
     62                 # Source Remove requires admin privileges
     63                 Remove-WinGetSource -Name 'TestSource'
     64             }
     65         }
     66         catch {
     67             # Non-admin
     68             Start-Process -FilePath $wingetExeName -ArgumentList "source remove TestSource"
     69         }
     70     }

    102     function WaitForWindowsPackageManagerServer([bool]$force = $false)
    103     {
    104         $processes = Get-Process | Where-Object { $_.Name -eq "WindowsPackageManagerServer" }
    105         foreach ($p in $processes)
    106         {
    107             if ($force)
    108             {
    109                 Stop-Process $p
    110             }
    111 
    112             $timeout = 300
    113             $secondsToWait = 5
    114             $time = 0
    115             while ($p.HasExited -eq $false)
    116             {
    117                 $time += $secondsToWait
    118                 if ($time -ge $timeout )
    119                 {
    120                     throw "Timeout waiting for $($p.Id) to exit"
    121                 }
    122                 Start-Sleep -Seconds 5
    123             }
    124         }
    125     }

    132     function Validate-WinGetResultCommonFields([psobject]$result, [psobject]$expected) {
    133         $result | Should -Not -BeNullOrEmpty -ErrorAction Stop
    134         $result.Id | Should -Be $expected.Id
    135         $result.Name | Should -Be $expected.Name
    136         $result.Source | Should -Be $expected.Source
    137         $result.Status | Should -Be $expected.Status
    138     }

    140     function Validate-WinGetPackageOperationResult([psobject]$result, [psobject]$expected, [string]$operationType)
    141     {
    142         Validate-WinGetResultCommonFields $result $expected
    143         $result.RebootRequired | Should -Be $expected.RebootRequired
    144 
    145         switch ($operationType) {
    146             'install' {
    147                 $result.InstallerErrorCode | Should -Be $expected.InstallerErrorCode
    148             }
    149             'update' {
    150                 $result.InstallerErrorCode | Should -Be $expected.InstallerErrorCode
    151             }
    152             'repair' {
    153                 $result.RepairErrorCode | Should -Be $expected.RepairErrorCode
    154             }
    155             'uninstall' {
    156                 $result.UninstallerErrorCode | Should -Be $expected.UninstallerErrorCode
    157             }
    158             default {
    159                 throw "Unknown operation type: $operationType"
    160             }
    161         }
    162     }

    913         $source | Should -Not -BeNullOrEmpty
    914         $source.Name | Should -Be 'TestSource'
    915 
    916         $process = Get-Process -Name "WindowsPackageManagerServer"
    917         $process | Should -Not -BeNullOrEmpty
    918 
    919         # At least one is running.
    920         $process | Where-Object { $_.HasExited -eq $false } | Should -Not -BeNullOrEmpty
    921 
    922         WaitForWindowsPackageManagerServer $true
    923 
    924         # From the ones we got, at least one exited
    925         $process | Where-Object { $_.HasExited -eq $true } | Should -Not -BeNullOrEmpty
    926 
    927         $source2 = Get-WinGetSource -Name 'TestSource'
    928         $source2 | Should -Not -BeNullOrEmpty
    929         $source2.Name | Should -Be 'TestSource'
    930 
    931         $process2 = Get-Process -Name "WindowsPackageManagerServer"
    932         $process2 | Should -Not -BeNullOrEmpty
    933         $process2.Id | Should -Not -Be $process.Id
    934     }
    935 
    936     # The Microsoft.WinGet.Client has static proxy objects of WindowsPackageManagerServer
    937     # This tests does all the Microsoft.WinGet.Client calls in a different pwsh instance.
    938     It 'Graceful termination' {
    939         $typeTable = [System.Management.Automation.Runspaces.TypeTable]::LoadDefaultTypeFiles()
    940         $oopRunspace = [System.Management.Automation.Runspaces.RunspaceFactory]::CreateOutOfProcessRunspace($typeTable)
    941         $oopRunspace.Open()
    942         $oopPwsh = [PowerShell]::Create()
    943         $oopPwsh.Runspace = $oopRunspace
    944         $oopPwshPid = $oopPwsh.AddScript("`$PID").Invoke()
    945         $oopPwshProcess = Get-Process -Id $oopPwshPid
    946         $oopPwshProcess.HasExited | Should -Be $false
    947 
    948         $source = $oopPwsh.AddScript("Get-WinGetSource -Name TestSource").Invoke()
    949         $source | Should -Not -BeNullOrEmpty
    950         $source.Name | Should -Be 'TestSource'
    951 
    952         $wingetProcess = Get-Process -Name "WindowsPackageManagerServer"
    953         $wingetProcess | Should -Not -BeNullOrEmpty
    954 
    955         # At least one is running.
    956         $wingetProcess | Where-Object { $_.HasExited -eq $false } | Should -Not -BeNullOrEmpty
    957 
    958         $oopRunspace.Close()
    959 
    960         Start-Sleep -Seconds 30
    961         $oopPwshProcess.HasExited | Should -Be $true
    962 
    963         # From the ones we got, at least one exited
    964         WaitForWindowsPackageManagerServer
    965         $wingetProcess | Where-Object { $_.HasExited -eq $true } | Should -Not -BeNullOrEmpty
    966     }
