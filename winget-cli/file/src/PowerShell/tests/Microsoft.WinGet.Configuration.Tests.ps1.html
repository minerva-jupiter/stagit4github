    101     function GetExpectedModulePath([TestModuleLocation]$testModuleLocation)
    102     {
    103         switch ($testModuleLocation)
    104         {
    105             ([TestModuleLocation]::CurrentUser)
    106             {
    107                 $path = [Environment]::GetFolderPath([Environment+SpecialFolder]::MyDocuments)
    108                 return Join-Path $path "PowerShell\Modules"
    109             }
    110             ([TestModuleLocation]::AllUsers)
    111             {
    112                 $path = [Environment]::GetFolderPath([Environment+SpecialFolder]::ProgramFiles)
    113                 return Join-Path $path "PowerShell\Modules"
    114             }
    115             ([TestModuleLocation]::DefaultLocation)
    116             {
    117                 $path = [Environment]::GetFolderPath([Environment+SpecialFolder]::LocalApplicationData)
    118                 return Join-Path $path "Microsoft\WinGet\Configuration\Modules"
    119             }
    120             ([TestModuleLocation]::Custom)
    121             {
    122                 return Join-Path $env:TEMP "E2EPesterCustomModules"
    123             }
    124             default
    125             {
    126                 throw $testModuleLocation
    127             }
    128         }
    129     }

    131     function CleanupPsModulePath()
    132     {
    133         $wingetPath = GetExpectedModulePath DefaultLocation
    134         $customPath = GetExpectedModulePath Custom
    135         $modulePath = $env:PsModulePath
    136         $newModulePath = ($modulePath.Split(';') | Where-Object { $_ -ne $wingetPath } | Where-Object { $_ -ne $customPath }) -join ';'
    137         $env:PsModulePath = $newModulePath
    138     }

    140     function EnsureModuleState([string]$moduleName, [bool]$present, [string]$repository = $null, [TestModuleLocation]$testModuleLocation = [TestModuleLocation]::CurrentUser)
    141     {
    142         CleanupPsModulePath
    143         $wingetPath = GetExpectedModulePath DefaultLocation
    144         $customPath = GetExpectedModulePath Custom
    145         $env:PsModulePath += ";$wingetPath;$customPath"
    146 
    147         $availableModules = Get-Module $moduleName -ListAvailable
    148         $isPresent = $null -ne $availableModules
    149 
    150         if ($isPresent)
    151         {
    152             foreach ($module in $availableModules)
    153             {
    154                 try
    155                 {
    156                     $item = Get-Item $module.Path -ErrorAction Stop
    157                     while ($item.Name -ne $moduleName)
    158                     {
    159                         $item = Get-Item $item.PSParentPath -ErrorAction Stop
    160                     }
    161 
    162                     if (-not $present)
    163                     {
    164                         Get-ChildItem $item.FullName -Recurse | Remove-Item -Force -Recurse -ErrorAction Stop
    165                         Remove-Item $item -ErrorAction Stop
    166                     }
    167                     else
    168                     {
    169                         # Must be in the right location
    170                         $expected = GetExpectedModulePath $testModuleLocation
    171                         if ($expected -ne $item.Parent.FullName)
    172                         {
    173                             Get-ChildItem $item.FullName -Recurse | Remove-Item -Force -Recurse -ErrorAction Stop
    174                             Remove-Item $item -ErrorAction Stop
    175                             $isPresent = $false
    176                         }
    177                     }
    178                 }
    179                 catch [System.Management.Automation.ItemNotFoundException]
    180                 {
    181                     Write-Host "Item not found, ignoring..." $_.Exception.Message
    182                 }
    183             }
    184         }
    185 
    186         if ((-not $isPresent) -and $present)
    187         {
    188             $params = @{
    189                 Name = $moduleName
    190                 Force = $true
    191             }
    192 
    193             if (-not [string]::IsNullOrEmpty($repository))
    194             {
    195                 $params.Add('Repository', $repository)
    196             }
    197 
    198             if (($testModuleLocation -eq [TestModuleLocation]::CurrentUser) -or
    199                 ($testModuleLocation -eq [TestModuleLocation]::AllUsers))
    200             {
    201                 if ($testModuleLocation -eq [TestModuleLocation]::AllUsers)
    202                 {
    203                     $params.Add('Scope', 'AllUsers')
    204                 }
    205 
    206                 Install-Module @params
    207             }
    208             else
    209             {
    210                 $path = $customPath
    211                 if (($testModuleLocation -eq [TestModuleLocation]::WinGetModulePath) -or
    212                     ($testModuleLocation -eq [TestModuleLocation]::DefaultLocation))
    213                 {
    214                     $path = $wingetPath
    215                 }
    216                 $params.Add('Path', $path)
    217 
    218                 Save-Module @params
    219             }
    220         }
    221 
    222         CleanupPsModulePath
    223     }
