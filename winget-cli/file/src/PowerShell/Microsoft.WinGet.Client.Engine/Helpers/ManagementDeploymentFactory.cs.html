    179         [System.Diagnostics.CodeAnalysis.SuppressMessage("Interoperability", "CA1416:Validate platform compatibility", Justification = "COM only usage.")]
    180         private static T Create<T>(Type? type, in Guid iid)
    181             where T : new()
    182         {
    183             if (type == null)
    184             {
    185                 throw new ArgumentNullException(iid.ToString());
    186             }
    187 
    188             if (Utilities.UsesInProcWinget)
    189             {
    190                 // This doesn't work on Windows PowerShell
    191                 // If we want to support it, we need something that loads the
    192                 // Microsoft.Management.Deployment.dll for .NET framework as CsWinRT
    193                 // does for .NET Core
    194                 return new T();
    195             }
    196 
    197             object? instance = null;
    198 
    199             if (Utilities.ExecutingAsAdministrator)
    200             {
    201                 int hr = WinRTHelpers.ManualActivation(type.GUID, iid, 0, out instance);
    202 
    203                 if (hr < 0)
    204                 {
    205                     if (hr == ErrorCode.FileNotFound || hr == ErrorCode.PackageNotRegisteredForUser)
    206                     {
    207                         throw new WinGetIntegrityException(IntegrityCategory.AppInstallerNotInstalled);
    208                     }
    209                     else
    210                     {
    211                         throw new COMException($"Failed to create instance: {hr}", hr);
    212                     }
    213                 }
    214             }
    215             else
    216             {
    217                 instance = Activator.CreateInstance(type);
    218             }
    219 
    220             if (instance == null)
    221             {
    222                 throw new ArgumentNullException();
    223             }
    224 
    225 #if NET
    226             IntPtr pointer = Marshal.GetIUnknownForObject(instance);
    227             return MarshalInterface<T>.FromAbi(pointer);
    228 #else
    229             return (T)instance;
    230 #endif
    231         }
