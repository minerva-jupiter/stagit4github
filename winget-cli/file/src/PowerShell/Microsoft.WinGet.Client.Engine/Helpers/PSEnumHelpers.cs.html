     20         /// Checks if the provided enum string value matches the 'Default' value for PS Enums.
     21         /// </summary>
     22         /// <param name="value">Enum string value.</param>
     23         /// <returns>Boolean value.</returns>
     24         public static bool IsDefaultEnum(string value)
     25         {
     26             return string.Equals(value, "Default", StringComparison.OrdinalIgnoreCase);
     27         }
