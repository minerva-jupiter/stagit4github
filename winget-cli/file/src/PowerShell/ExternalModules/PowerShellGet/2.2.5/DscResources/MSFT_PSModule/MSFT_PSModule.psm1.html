    339     [CmdletBinding()]

    343         [ValidateSet('Present', 'Absent')]

    348         [System.String]
    349         $Name,
    350 
    351         [Parameter()]
    352         [System.String]

    384 
    385     # Validate the repository argument
    386     if ($PSBoundParameters.ContainsKey('Repository')) {
    387         Test-ParameterValue -Value $Repository -Type 'PackageSource' -ProviderName 'PowerShellGet' -Verbose
    388     }
    389 
    390     if ($Ensure -ieq 'Present') {
    391         # Version check
    392         $extractedArguments = New-SplatParameterHashTable -FunctionBoundParameters $PSBoundParameters `
    393             -ArgumentNames ('MinimumVersion', 'MaximumVersion', 'RequiredVersion')
    394 
    395         $null = Test-VersionParameter @extractedArguments
    396 
    397         try {
    398             $extractedArguments = New-SplatParameterHashTable -FunctionBoundParameters $PSBoundParameters `
    399                 -ArgumentNames ('Name', 'Repository', 'MinimumVersion', 'MaximumVersion', 'RequiredVersion')
    400 
    401             Write-Verbose -Message ($localizedData.StartFindModule -f $Name)
    402 
    403             $modules = Find-Module @extractedArguments -ErrorVariable ev
    404         }
    405         catch {
    406             $errorMessage = $script:localizedData.ModuleNotFoundInRepository -f $Name
    407             New-InvalidOperationException -Message $errorMessage -ErrorRecord $_
    408         }
    409 
    410         $trusted = $null
    411         $moduleFound = $null
    412 
    413         foreach ($m in $modules) {
    414             # Check for the installation policy.
    415             $trusted = Get-InstallationPolicy -RepositoryName $m.Repository -ErrorAction SilentlyContinue -WarningAction SilentlyContinue
    416 
    417             # Stop the loop if found a trusted repository.
    418             if ($trusted) {
    419                 $moduleFound = $m
    420                 break;
    421             }
    422         }
    423 
    424         try {
    425             # The repository is trusted, so we install it.
    426             if ($trusted) {
    427                 Write-Verbose -Message ($localizedData.StartInstallModule -f $Name, $moduleFound.Version.toString(), $moduleFound.Repository)
    428 
    429                 # Extract the installation options.
    430                 $extractedSwitches = New-SplatParameterHashTable -FunctionBoundParameters $PSBoundParameters -ArgumentNames ('Force', 'AllowClobber', 'SkipPublisherCheck')
    431 
    432                 $moduleFound | Install-Module @extractedSwitches 2>&1 | out-string | Write-Verbose
    433             }
    434             # The repository is untrusted but user's installation policy is trusted, so we install it with a warning.
    435             elseif ($InstallationPolicy -ieq 'Trusted') {
    436                 Write-Warning -Message ($localizedData.InstallationPolicyWarning -f $Name, $modules[0].Repository, $InstallationPolicy)
    437 
    438                 # Extract installation options (Force implied by InstallationPolicy).
    439                 $extractedSwitches = New-SplatParameterHashTable -FunctionBoundParameters $PSBoundParameters -ArgumentNames ('AllowClobber', 'SkipPublisherCheck')
    440 
    441                 # If all the repositories are untrusted, we choose the first one.
    442                 $modules[0] | Install-Module @extractedSwitches -Force 2>&1 | out-string | Write-Verbose
    443             }
    444             # Both user and repository is untrusted
    445             else {
    446                 $errorMessage = $script:localizedData.InstallationPolicyFailed -f $InstallationPolicy, 'Untrusted'
    447                 New-InvalidOperationException -Message $errorMessage
    448             }
    449 
    450             Write-Verbose -Message ($localizedData.InstalledSuccess -f $Name)
    451         }
    452         catch {
    453             $errorMessage = $script:localizedData.FailToInstall -f $Name
    454             New-InvalidOperationException -Message $errorMessage -ErrorRecord $_
    455         }
    456     }
    457     # Ensure=Absent
    458     else {
    459 
    460         $extractedArguments = New-SplatParameterHashTable -FunctionBoundParameters $PSBoundParameters `
    461             -ArgumentNames ('Name', 'Repository', 'MinimumVersion', 'MaximumVersion', 'RequiredVersion')
    462 
    463         # Get the module with the right version and repository properties.
    464         $modules = Get-RightModule @extractedArguments
    465 
    466         if (-not $modules) {
    467             $errorMessage = $script:localizedData.ModuleWithRightPropertyNotFound -f $Name
    468             New-InvalidOperationException -Message $errorMessage
    469         }
    470 
    471         foreach ($module in $modules) {
    472             # Get the path where the module is installed.
    473             $path = $module.ModuleBase
    474 
    475             Write-Verbose -Message ($localizedData.StartUnInstallModule -f $Name)

    477             try {

    487                 $errorMessage = $script:localizedData.FailToUninstall -f $module.Name

    513 function Get-RightModule {
    514     <#
    515         These suppressions are added because this repository have other Visual Studio Code workspace
    516         settings than those in DscResource.Tests DSC test framework.
    517         Only those suppression that contradict this repository guideline is added here.
    518     #>
    519     [Diagnostics.CodeAnalysis.SuppressMessageAttribute('DscResource.AnalyzerRules\Measure-ForEachStatement', '')]
    520     [Diagnostics.CodeAnalysis.SuppressMessageAttribute('DscResource.AnalyzerRules\Measure-FunctionBlockBraces', '')]
    521     [Diagnostics.CodeAnalysis.SuppressMessageAttribute('DscResource.AnalyzerRules\Measure-IfStatement', '')]
    522     [CmdletBinding()]
    523     param
    524     (
    525         [Parameter(Mandatory = $true)]
    526         [ValidateNotNullOrEmpty()]
    527         [System.String]
    528         $Name,
    529 
    530         [Parameter()]
    531         [System.String]
    532         $RequiredVersion,
    533 
    534         [Parameter()]
    535         [System.String]
    536         $MinimumVersion,
    537 
    538         [Parameter()]
    539         [System.String]
    540         $MaximumVersion,
    541 
    542         [Parameter()]
    543         [System.String]
    544         $Repository
    545     )
    546 
    547     Write-Verbose -Message ($localizedData.StartGetModule -f $($Name))
    548 
    549     $modules = Microsoft.PowerShell.Core\Get-Module -Name $Name -ListAvailable -ErrorAction SilentlyContinue -WarningAction SilentlyContinue
    550 
    551     if (-not $modules) {
    552         return $null
    553     }
    554 
    555     <#
    556         As Get-Module does not take RequiredVersion, MinimumVersion, MaximumVersion, or Repository,
    557         below we need to check whether the modules are containing the right version and repository
    558         location.
    559     #>
    560 
    561     $extractedArguments = New-SplatParameterHashTable -FunctionBoundParameters $PSBoundParameters `
    562         -ArgumentNames ('MaximumVersion', 'MinimumVersion', 'RequiredVersion')
    563     $returnVal = @()
    564 
    565     foreach ($m in $modules) {
    566         $versionMatch = $false
    567         $installedVersion = $m.Version
    568 
    569         # Case 1 - a user provides none of RequiredVersion, MinimumVersion, MaximumVersion
    570         if ($extractedArguments.Count -eq 0) {
    571             $versionMatch = $true
    572         }
    573 
    574         # Case 2 - a user provides RequiredVersion
    575         elseif ($extractedArguments.ContainsKey('RequiredVersion')) {
    576             # Check if it matches with the installed version
    577             $versionMatch = ($installedVersion -eq [System.Version] $RequiredVersion)
    578         }
    579         else {
    580 
    581             # Case 3 - a user provides MinimumVersion
    582             if ($extractedArguments.ContainsKey('MinimumVersion')) {
    583                 $versionMatch = ($installedVersion -ge [System.Version] $extractedArguments['MinimumVersion'])
    584             }
    585 
    586             # Case 4 - a user provides MaximumVersion
    587             if ($extractedArguments.ContainsKey('MaximumVersion')) {
    588                 $isLessThanMax = ($installedVersion -le [System.Version] $extractedArguments['MaximumVersion'])
    589 
    590                 if ($extractedArguments.ContainsKey('MinimumVersion')) {
    591                     $versionMatch = $versionMatch -and $isLessThanMax
    592                 }
    593                 else {
    594                     $versionMatch = $isLessThanMax
    595                 }
    596             }
    597 
    598             # Case 5 - Both MinimumVersion and MaximumVersion are provided. It's covered by the above.
    599             # Do not return $false yet to allow the foreach to continue
    600             if (-not $versionMatch) {
    601                 Write-Verbose -Message ($localizedData.VersionMismatch -f $Name, $installedVersion)
    602                 $versionMatch = $false
    603             }
    604         }
    605 
    606         # Case 6 - Version matches but need to check if the module is from the right repository.
    607         if ($versionMatch) {
    608             # A user does not provide Repository, we are good
    609             if (-not $PSBoundParameters.ContainsKey('Repository')) {
    610                 Write-Verbose -Message ($localizedData.ModuleFound -f "$Name $installedVersion")
    611                 $returnVal += $m
    612             }
    613             else {
    614                 # Check if the Repository matches
    615                 $sourceName = Get-ModuleRepositoryName -Module $m
    616 
    617                 if ($Repository -ieq $sourceName) {
    618                     Write-Verbose -Message ($localizedData.ModuleFound -f "$Name $installedVersion")
    619                     $returnVal += $m
    620                 }
    621                 else {
    622                     Write-Verbose -Message ($localizedData.RepositoryMismatch -f $($Name), $($sourceName))
    623                 }
    624             }
    625         }
    626     } # foreach
    627 
    628     return $returnVal
    629 }
