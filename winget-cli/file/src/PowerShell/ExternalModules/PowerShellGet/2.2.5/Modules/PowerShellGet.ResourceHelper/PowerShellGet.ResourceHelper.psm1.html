     26 <#
     27     .SYNOPSIS
     28         This is a helper function that extract the parameters from a given table.
     29 
     30     .PARAMETER FunctionBoundParameters
     31         Specifies the hash table containing a set of parameters to be extracted.
     32 
     33     .PARAMETER ArgumentNames
     34         Specifies a list of arguments you want to extract.

     36 function New-SplatParameterHashTable {
     37     [CmdletBinding()]
     38     [OutputType([System.Collections.Hashtable])]
     39     param
     40     (
     41         [Parameter(Mandatory = $true)]
     42         [System.Collections.Hashtable]
     43         $FunctionBoundParameters,
     44 
     45         [Parameter(Mandatory = $true)]
     46         [System.String[]]
     47         $ArgumentNames
     48     )
     49 
     50     Write-Verbose -Message ($script:localizedData.CallingFunction -f $($MyInvocation.MyCommand))
     51 
     52     $returnValue = @{}
     53 
     54     foreach ($arg in $ArgumentNames) {
     55         if ($FunctionBoundParameters.ContainsKey($arg)) {
     56             # Found an argument we are looking for, so we add it to return collection.
     57             $returnValue.Add($arg, $FunctionBoundParameters[$arg])
     58         }
     59     }
     60 
     61     return $returnValue
     62 }

     65     .SYNOPSIS
     66         This is a helper function that validate that a value is correct and used correctly.
     67 
     68     .PARAMETER Value
     69         Specifies the value to be validated.
     70 
     71     .PARAMETER Type
     72         Specifies the type of argument.
     73 
     74     .PARAMETER Type
     75         Specifies the name of the provider.
     76 
     77     .OUTPUTS
     78         None. Throws an error if the test fails.
     79 #>
     80 function Test-ParameterValue {
     81     [CmdletBinding()]
     82     param
     83     (
     84         [Parameter(Mandatory = $true)]
     85         [ValidateNotNullOrEmpty()]
     86         [System.String]
     87         $Value,
     88 
     89         [Parameter(Mandatory = $true)]
     90         [ValidateNotNullOrEmpty()]
     91         [System.String]
     92         $Type,
     93 
     94         [Parameter(Mandatory = $true)]
     95         [ValidateNotNullOrEmpty()]
     96         [System.String]
     97         $ProviderName
     98     )
     99 
    100     Write-Verbose -Message ($script:localizedData.CallingFunction -f $($MyInvocation.MyCommand))
    101 
    102     switch ($Type) {
    103         'SourceUri' {
    104             # Checks whether given URI represents specific scheme
    105             # Most common schemes: file, http, https, ftp
    106             $scheme = @('http', 'https', 'file', 'ftp')
    107 
    108             $newUri = $Value -as [System.URI]
    109             $returnValue = ($newUri -and $newUri.AbsoluteURI -and ($scheme -icontains $newUri.Scheme))
    110 
    111             if ($returnValue -eq $false) {
    112                 $errorMessage = $script:localizedData.InValidUri -f $Value
    113                 New-InvalidArgumentException -ArgumentName $Type -Message $errorMessage
    114             }
    115         }
    116 
    117         'DestinationPath' {
    118             $returnValue = Test-Path -Path $Value
    119 
    120             if ($returnValue -eq $false) {
    121                 $errorMessage = $script:localizedData.PathDoesNotExist -f $Value
    122                 New-InvalidArgumentException -ArgumentName $Type -Message $errorMessage
    123             }
    124         }
    125 
    126         'PackageSource' {
    127             # Value can be either the package source Name or source Uri.
    128 
    129             # Check if the source is a Uri.
    130             $uri = $Value -as [System.URI]
    131 
    132             if ($uri -and $uri.AbsoluteURI) {
    133                 # Check if it's a valid Uri.
    134                 Test-ParameterValue -Value $Value -Type 'SourceUri' -ProviderName $ProviderName
    135             }
    136             else {
    137                 # Check if it's a registered package source name.
    138                 $source = PackageManagement\Get-PackageSource -Name $Value -ProviderName $ProviderName -ErrorVariable ev
    139 
    140                 if ((-not $source) -or $ev) {
    141                     # We do not need to throw error here as Get-PackageSource does already.
    142                     Write-Verbose -Message ($script:localizedData.SourceNotFound -f $source)
    143                 }
    144             }
    145         }
    146 
    147         default {
    148             $errorMessage = $script:localizedData.UnexpectedArgument -f $Type
    149             New-InvalidArgumentException -ArgumentName $Type -Message $errorMessage
    150         }
    151     }
    152 }

    154 <#
    155     .SYNOPSIS
    156         This is a helper function that does the version validation.
    157 
    158     .PARAMETER RequiredVersion
    159         Provides the required version.
    160 
    161     .PARAMETER MaximumVersion
    162         Provides the maximum version.
    163 
    164     .PARAMETER MinimumVersion
    165         Provides the minimum version.
    166 #>
    167 function Test-VersionParameter {
    168     [CmdletBinding()]
    169     param
    170     (
    171         [Parameter()]
    172         [System.String]
    173         $RequiredVersion,
    174 
    175         [Parameter()]
    176         [System.String]
    177         $MinimumVersion,
    178 
    179         [Parameter()]
    180         [System.String]
    181         $MaximumVersion
    182     )
    183 
    184     Write-Verbose -Message ($localizedData.CallingFunction -f $($MyInvocation.MyCommand))
    185 
    186     $isValid = $false
    187 
    188     # Case 1: No further check required if a user provides either none or one of these: minimumVersion, maximumVersion, and requiredVersion.
    189     if ($PSBoundParameters.Count -le 1) {
    190         return $true
    191     }
    192 
    193     # Case 2: #If no RequiredVersion is provided.
    194     if (-not $PSBoundParameters.ContainsKey('RequiredVersion')) {
    195         # If no RequiredVersion, both MinimumVersion and MaximumVersion are provided. Otherwise fall into the Case #1.
    196         $isValid = $PSBoundParameters['MinimumVersion'] -le $PSBoundParameters['MaximumVersion']
    197     }
    198 
    199     # Case 3: RequiredVersion is provided.
    200     #        In this case  MinimumVersion and/or MaximumVersion also are provided. Otherwise fall in to Case #1.
    201     #        This is an invalid case. When RequiredVersion is provided, others are not allowed. so $isValid is false, which is already set in the init.
    202 
    203     if ($isValid -eq $false) {
    204         $errorMessage = $script:localizedData.VersionError
    205         New-InvalidArgumentException `
    206             -ArgumentName 'RequiredVersion, MinimumVersion or MaximumVersion' `
    207             -Message $errorMessage
    208     }
    209 }

    236 <#
    237     .SYNOPSIS
    238         This method is used to compare current and desired values for any DSC resource.
    239 
    240     .PARAMETER CurrentValues
    241         This is hash table of the current values that are applied to the resource.
    242 
    243     .PARAMETER DesiredValues
    244         This is a PSBoundParametersDictionary of the desired values for the resource.
    245 
    246     .PARAMETER ValuesToCheck
    247         This is a list of which properties in the desired values list should be checked.
    248         If this is empty then all values in DesiredValues are checked.
    249 #>
    250 function Test-DscParameterState {
    251     [CmdletBinding()]
    252     param
    253     (
    254         [Parameter(Mandatory = $true)]
    255         [System.Collections.Hashtable]
    256         $CurrentValues,
    257 
    258         [Parameter(Mandatory = $true)]
    259         [System.Object]
    260         $DesiredValues,
    261 
    262         [Parameter()]
    263         [System.Array]
    264         $ValuesToCheck
    265     )
    266 
    267     $returnValue = $true
    268 
    269     if (($DesiredValues.GetType().Name -ne 'HashTable') `
    270             -and ($DesiredValues.GetType().Name -ne 'CimInstance') `
    271             -and ($DesiredValues.GetType().Name -ne 'PSBoundParametersDictionary')) {
    272         $errorMessage = $script:localizedData.PropertyTypeInvalidForDesiredValues -f $($DesiredValues.GetType().Name)
    273         New-InvalidArgumentException -ArgumentName 'DesiredValues' -Message $errorMessage
    274     }
    275 
    276     if (($DesiredValues.GetType().Name -eq 'CimInstance') -and ($null -eq $ValuesToCheck)) {
    277         $errorMessage = $script:localizedData.PropertyTypeInvalidForValuesToCheck
    278         New-InvalidArgumentException -ArgumentName 'ValuesToCheck' -Message $errorMessage
    279     }
    280 
    281     if (($null -eq $ValuesToCheck) -or ($ValuesToCheck.Count -lt 1)) {
    282         $keyList = $DesiredValues.Keys
    283     }
    284     else {
    285         $keyList = $ValuesToCheck
    286     }
    287 
    288     $keyList | ForEach-Object -Process {
    289         if (($_ -ne 'Verbose')) {
    290             if (($CurrentValues.ContainsKey($_) -eq $false) `
    291                     -or ($CurrentValues.$_ -ne $DesiredValues.$_) `
    292                     -or (($DesiredValues.GetType().Name -ne 'CimInstance' -and $DesiredValues.ContainsKey($_) -eq $true) -and ($null -ne $DesiredValues.$_ -and $DesiredValues.$_.GetType().IsArray))) {
    293                 if ($DesiredValues.GetType().Name -eq 'HashTable' -or `
    294                         $DesiredValues.GetType().Name -eq 'PSBoundParametersDictionary') {
    295                     $checkDesiredValue = $DesiredValues.ContainsKey($_)
    296                 }
    297                 else {
    298                     # If DesiredValue is a CimInstance.
    299                     $checkDesiredValue = $false
    300                     if (([System.Boolean]($DesiredValues.PSObject.Properties.Name -contains $_)) -eq $true) {
    301                         if ($null -ne $DesiredValues.$_) {
    302                             $checkDesiredValue = $true
    303                         }
    304                     }
    305                 }
    306 
    307                 if ($checkDesiredValue) {
    308                     $desiredType = $DesiredValues.$_.GetType()
    309                     $fieldName = $_
    310                     if ($desiredType.IsArray -eq $true) {
    311                         if (($CurrentValues.ContainsKey($fieldName) -eq $false) `
    312                                 -or ($null -eq $CurrentValues.$fieldName)) {
    313                             Write-Verbose -Message ($script:localizedData.PropertyValidationError -f $fieldName) -Verbose
    314 
    315                             $returnValue = $false
    316                         }
    317                         else {
    318                             $arrayCompare = Compare-Object -ReferenceObject $CurrentValues.$fieldName `
    319                                 -DifferenceObject $DesiredValues.$fieldName
    320                             if ($null -ne $arrayCompare) {
    321                                 Write-Verbose -Message ($script:localizedData.PropertiesDoesNotMatch -f $fieldName) -Verbose
    322 
    323                                 $arrayCompare | ForEach-Object -Process {
    324                                     Write-Verbose -Message ($script:localizedData.PropertyThatDoesNotMatch -f $_.InputObject, $_.SideIndicator) -Verbose
    325                                 }
    326 
    327                                 $returnValue = $false
    328                             }
    329                         }
    330                     }
    331                     else {
    332                         switch ($desiredType.Name) {
    333                             'String' {
    334                                 if (-not [System.String]::IsNullOrEmpty($CurrentValues.$fieldName) -or `
    335                                         -not [System.String]::IsNullOrEmpty($DesiredValues.$fieldName)) {
    336                                     Write-Verbose -Message ($script:localizedData.ValueOfTypeDoesNotMatch `
    337                                             -f $desiredType.Name, $fieldName, $($CurrentValues.$fieldName), $($DesiredValues.$fieldName)) -Verbose
    338 
    339                                     $returnValue = $false
    340                                 }
    341                             }
    342 
    343                             'Int32' {
    344                                 if (-not ($DesiredValues.$fieldName -eq 0) -or `
    345                                         -not ($null -eq $CurrentValues.$fieldName)) {
    346                                     Write-Verbose -Message ($script:localizedData.ValueOfTypeDoesNotMatch `
    347                                             -f $desiredType.Name, $fieldName, $($CurrentValues.$fieldName), $($DesiredValues.$fieldName)) -Verbose
    348 
    349                                     $returnValue = $false
    350                                 }
    351                             }
    352 
    353                             { $_ -eq 'Int16' -or $_ -eq 'UInt16'} {
    354                                 if (-not ($DesiredValues.$fieldName -eq 0) -or `
    355                                         -not ($null -eq $CurrentValues.$fieldName)) {
    356                                     Write-Verbose -Message ($script:localizedData.ValueOfTypeDoesNotMatch `
    357                                             -f $desiredType.Name, $fieldName, $($CurrentValues.$fieldName), $($DesiredValues.$fieldName)) -Verbose
    358 
    359                                     $returnValue = $false
    360                                 }
    361                             }
    362 
    363                             default {
    364                                 Write-Warning -Message ($script:localizedData.UnableToCompareProperty `
    365                                         -f $fieldName, $desiredType.Name)
    366 
    367                                 $returnValue = $false
    368                             }
    369                         }
    370                     }
    371                 }
    372             }
    373         }
    374     }
    375 
    376     return $returnValue
    377 }
