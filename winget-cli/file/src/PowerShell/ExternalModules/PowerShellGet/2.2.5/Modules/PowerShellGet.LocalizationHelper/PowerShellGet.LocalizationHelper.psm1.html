    186 <#
    187     .SYNOPSIS
    188         Retrieves the localized string data based on the machine's culture.
    189         Falls back to en-US strings if the machine's culture is not supported.
    190 
    191     .PARAMETER ResourceName
    192         The name of the resource as it appears before '.strings.psd1' of the localized string file.
    193         For example:
    194             For WindowsOptionalFeature: MSFT_WindowsOptionalFeature
    195             For Service: MSFT_ServiceResource
    196             For Registry: MSFT_RegistryResource
    197             For Helper: SqlServerDscHelper
    198 
    199     .PARAMETER ScriptRoot
    200         Optional. The root path where to expect to find the culture folder. This is only needed
    201         for localization in helper modules. This should not normally be used for resources.
    202 #>
    203 function Get-LocalizedData {
    204     [CmdletBinding()]
    205     param
    206     (
    207         [Parameter(Mandatory = $true)]
    208         [ValidateNotNullOrEmpty()]
    209         [System.String]
    210         $ResourceName,
    211 
    212         [Parameter()]
    213         [ValidateNotNullOrEmpty()]
    214         [System.String]
    215         $ScriptRoot
    216     )
    217 
    218     if ( -not $ScriptRoot ) {
    219         $resourceDirectory = Join-Path -Path $PSScriptRoot -ChildPath $ResourceName
    220         $localizedStringFileLocation = Join-Path -Path $resourceDirectory -ChildPath $PSUICulture
    221     }
    222     else {
    223         $localizedStringFileLocation = Join-Path -Path $ScriptRoot -ChildPath $PSUICulture
    224     }
    225 
    226     if (-not (Test-Path -Path $localizedStringFileLocation)) {
    227         # Fallback to en-US
    228         if ( -not $ScriptRoot ) {
    229             $localizedStringFileLocation = Join-Path -Path $resourceDirectory -ChildPath 'en-US'
    230         }
    231         else {
    232             $localizedStringFileLocation = Join-Path -Path $ScriptRoot -ChildPath 'en-US'
    233         }
    234     }
    235 
    236     Import-LocalizedData `
    237         -BindingVariable 'localizedData' `
    238         -FileName "$ResourceName.strings.psd1" `
    239         -BaseDirectory $localizedStringFileLocation
    240 
    241     return $localizedData
    242 }
