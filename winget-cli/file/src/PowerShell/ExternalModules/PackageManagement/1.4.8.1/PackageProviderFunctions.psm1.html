     18 <#
     19 	Overrides the default Write-Debug so that the output gets routed back thru the
     20 	$request.Debug() function
     21 #>
     22 function Write-Debug {
     23 	param(
     24 	[Parameter(Mandatory=$true)][string] $message,
     25 	[parameter(ValueFromRemainingArguments=$true)]
     26 	[object[]]
     27 	 $args= @()
     28 	)
     29 
     30 	if( -not $request  ) {
     31 		if( -not $args  ) {
     32 			Microsoft.PowerShell.Utility\write-verbose $message
     33 			return
     34 		}
     35 
     36 		$msg = [system.string]::format($message, $args)
     37 		Microsoft.PowerShell.Utility\write-verbose $msg
     38 		return
     39 	}
     40 
     41 	if( -not $args  ) {
     42 		$null = $request.Debug($message);
     43 		return
     44 	}
     45 	$null = $request.Debug($message,$args);
     46 }

     64 function Write-Progress {
     65     param(
     66         [CmdletBinding()]
     67 
     68         [Parameter(Position=0)]
     69         [string]
     70         $Activity,
     71 
     72         # This parameter is not supported by request object
     73         [Parameter(Position=1)]
     74         [ValidateNotNullOrEmpty()]
     75         [string]
     76         $Status,
     77 
     78         [Parameter(Position=2)]
     79         [ValidateRange(0,[int]::MaxValue)]
     80         [int]
     81         $Id,
     82 
     83         [Parameter()]
     84         [int]
     85         $PercentComplete=-1,
     86 
     87         # This parameter is not supported by request object
     88         [Parameter()]
     89         [int]
     90         $SecondsRemaining=-1,
     91 
     92         # This parameter is not supported by request object
     93         [Parameter()]
     94         [string]
     95         $CurrentOperation,        
     96 
     97         [Parameter()]
     98         [ValidateRange(-1,[int]::MaxValue)]
     99         [int]
    100         $ParentID=-1,
    101 
    102         [Parameter()]
    103         [switch]
    104         $Completed,
    105 
    106         # This parameter is not supported by request object
    107         [Parameter()]
    108         [int]
    109         $SourceID,
    110 
    111 	    [object[]]
    112         $args= @()
    113     )
    114 
    115     $params = @{}
    116 
    117     if ($PSBoundParameters.ContainsKey("Activity")) {
    118         $params.Add("Activity", $PSBoundParameters["Activity"])
    119     }
    120 
    121     if ($PSBoundParameters.ContainsKey("Status")) {
    122         $params.Add("Status", $PSBoundParameters["Status"])
    123     }
    124 
    125     if ($PSBoundParameters.ContainsKey("PercentComplete")) {
    126         $params.Add("PercentComplete", $PSBoundParameters["PercentComplete"])
    127     }
    128 
    129     if ($PSBoundParameters.ContainsKey("Id")) {
    130         $params.Add("Id", $PSBoundParameters["Id"])
    131     }
    132 
    133     if ($PSBoundParameters.ContainsKey("ParentID")) {
    134         $params.Add("ParentID", $PSBoundParameters["ParentID"])
    135     }
    136 
    137     if ($PSBoundParameters.ContainsKey("Completed")) {
    138         $params.Add("Completed", $PSBoundParameters["Completed"])
    139     }
    140 
    141 	if( -not $request  ) {    
    142 		if( -not $args  ) {
    143 			Microsoft.PowerShell.Utility\Write-Progress @params
    144 			return
    145 		}
    146 
    147 		$params["Activity"] = [system.string]::format($Activity, $args)
    148 		Microsoft.PowerShell.Utility\Write-Progress @params
    149 		return
    150 	}
    151 
    152 	if( -not $args  ) {
    153         $request.Progress($Activity, $Status, $Id, $PercentComplete, $SecondsRemaining, $CurrentOperation, $ParentID, $Completed)
    154 	}
    155 
    156 }
