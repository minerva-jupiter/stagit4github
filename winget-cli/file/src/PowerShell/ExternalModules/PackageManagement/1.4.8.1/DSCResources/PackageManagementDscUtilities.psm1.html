     17  Function ExtractArguments
     18 {
     19     <#
     20     .SYNOPSIS
     21 
     22     This is a helper function that extract the parameters from a given table. 
     23 
     24     .PARAMETER FunctionBoundParameters
     25     Specifies the hashtable containing a set of parameters to be extracted
     26 
     27     .PARAMETER ArgumentNames
     28     Specifies A list of arguments you want to extract
     29     #>
     30 
     31     Param
     32     (
     33         [parameter(Mandatory = $true)]
     34         [System.Collections.Hashtable]
     35         $FunctionBoundParameters,
     36 
     37         #A list of arguments you want to extract
     38         [parameter(Mandatory = $true)]
     39         [System.String[]]$ArgumentNames
     40     )
     41 
     42     Write-Verbose -Message ($LocalizedData.CallingFunction -f $($MyInvocation.mycommand))
     43 
     44     $returnValue=@{}
     45 
     46     foreach ($arg in $ArgumentNames)
     47     {
     48         if($FunctionBoundParameters.ContainsKey($arg))
     49         {
     50             #Found an argument we are looking for, so we add it to return collection
     51             $returnValue.Add($arg,$FunctionBoundParameters[$arg])
     52         }
     53     }
     54 
     55     return $returnValue
     56  }

     58 function ThrowError
     59 {
     60     <#
     61     .SYNOPSIS
     62 
     63     This is a helper function that throws an error. 
     64 
     65     .PARAMETER ExceptionName
     66     Specifies the type of errors, e.g. System.ArgumentException
     67 
     68     .PARAMETER ExceptionMessage
     69     Specifies the exception message
     70 
     71     .PARAMETER ErrorId
     72     Specifies an identifier of the error
     73 
     74     .PARAMETER ErrorCategory
     75     Specifies the error category, e.g., InvalidArgument defined in System.Management.Automation. 
     76 
     77     #>
     78 
     79     param
     80     (        
     81         [parameter(Mandatory = $true)]
     82         [ValidateNotNullOrEmpty()]
     83         [System.String]        
     84         $ExceptionName,
     85 
     86         [parameter(Mandatory = $true)]
     87         [ValidateNotNullOrEmpty()]
     88         [System.String]
     89         $ExceptionMessage,      
     90         
     91         [parameter(Mandatory = $true)]
     92         [ValidateNotNullOrEmpty()]
     93         [System.String]
     94         $ErrorId,
     95 
     96         [parameter(Mandatory = $true)]
     97         [ValidateNotNull()]
     98         [System.Management.Automation.ErrorCategory]
     99         $ErrorCategory
    100     )
    101     
    102     Write-Verbose -Message ($LocalizedData.CallingFunction -f $($MyInvocation.mycommand))
    103         
    104     $exception   = New-Object -TypeName $ExceptionName -ArgumentList $ExceptionMessage;
    105     $errorRecord = New-Object -TypeName System.Management.Automation.ErrorRecord -ArgumentList ($exception, $ErrorId, $ErrorCategory, $null)    
    106     throw $errorRecord
    107 }

    109 Function ValidateArgument
    110 {
    111     <#
    112     .SYNOPSIS
    113 
    114     This is a helper function that validates the arguments. 
    115 
    116     .PARAMETER Argument
    117     Specifies the argument to be validated.
    118 
    119     .PARAMETER Type
    120     Specifies the type of argument.
    121     #>
    122 
    123     [CmdletBinding()]
    124     param
    125     (
    126         [parameter(Mandatory = $true)]
    127         [ValidateNotNullOrEmpty()]
    128         [string]$Argument,
    129 
    130         [parameter(Mandatory = $true)]
    131         [ValidateNotNullOrEmpty()]
    132         [String]$Type,
    133 
    134         [parameter(Mandatory = $true)]
    135         [ValidateNotNullOrEmpty()]
    136         [String]$ProviderName
    137     )
    138 
    139     Write-Verbose -Message ($LocalizedData.CallingFunction -f $($MyInvocation.mycommand))
    140 
    141     switch ($Type)
    142     {
    143 
    144         "SourceUri"
    145         {
    146             # Checks whether given URI represents specific scheme
    147             # Most common schemes: file, http, https, ftp        
    148             $scheme =@('http', 'https', 'file', 'ftp')
    149  
    150             $newUri = $Argument -as [System.URI]  
    151             $returnValue = ($newUri -and $newUri.AbsoluteURI -and ($scheme -icontains $newuri.Scheme)) 
    152 
    153             if ($returnValue -eq $false)
    154             {                
    155                 ThrowError  -ExceptionName "System.ArgumentException" `
    156                             -ExceptionMessage ($LocalizedData.InValidUri -f $Argument)`
    157                             -ErrorId "InValidUri" `
    158                             -ErrorCategory InvalidArgument
    159             }
    160             
    161             #Check whether it's a valid uri. Wait for the response within 2mins.
    162             <#$result = Invoke-WebRequest $newUri -TimeoutSec 120 -UseBasicParsing -ErrorAction SilentlyContinue
    163 
    164             if ($null -eq (([xml]$result.Content).service ))
    165             {
    166                 ThrowError  -ExceptionName "System.ArgumentException" `
    167                             -ExceptionMessage ($LocalizedData.InValidUri -f $Argument)`
    168                             -ErrorId "InValidUri" `
    169                             -ErrorCategory InvalidArgument
    170             }#>
    171                                          
    172         }
    173         "DestinationPath"
    174         {
    175             $returnValue = Test-Path -Path $Argument
    176             if ($returnValue -eq $false)
    177             {
    178                 ThrowError  -ExceptionName "System.ArgumentException" `
    179                             -ExceptionMessage ($LocalizedData.PathDoesNotExist -f $Argument)`
    180                             -ErrorId "PathDoesNotExist" `
    181                             -ErrorCategory InvalidArgument
    182             }
    183         }
    184         "PackageSource"
    185         {      
    186             #Argument can be either the package source Name or source Uri.  
    187             
    188             #Check if the source is a uri 
    189             $uri = $Argument -as [System.URI]  
    190 
    191             if($uri -and $uri.AbsoluteURI) 
    192             {
    193                 # Check if it's a valid Uri
    194                 ValidateArgument -Argument $Argument -Type "SourceUri" -ProviderName $ProviderName
    195             }
    196             else
    197             {
    198                 #Check if it's a registered package source name                                                             
    199                 $source = PackageManagement\Get-PackageSource -Name $Argument -ProviderName $ProviderName -verbose -ErrorVariable ev
    200                 if ((-not $source) -or $ev) 
    201                 {
    202                     #We do not need to throw error here as Get-PackageSource does already
    203                     Write-Verbose -Message ($LocalizedData.SourceNotFound -f $source)                
    204                 }
    205             }
    206         }
    207         default
    208         {
    209             ThrowError  -ExceptionName "System.ArgumentException" `
    210                         -ExceptionMessage ($LocalizedData.UnexpectedArgument -f $Type)`
    211                         -ErrorId "UnexpectedArgument" `
    212                         -ErrorCategory InvalidArgument
    213         }
    214      }           
    215 }

    217 Function ValidateVersionArgument
    218 {
    219     <#
    220     .SYNOPSIS
    221 
    222     This is a helper function that does the version validation. 
    223 
    224     .PARAMETER RequiredVersion
    225     Provides the required version.
    226 
    227     .PARAMETER MaximumVersion
    228     Provides the maximum version.
    229 
    230     .PARAMETER MinimumVersion
    231     Provides the minimum version.
    232     #>
    233 
    234     [CmdletBinding()]
    235     param
    236     (
    237         [string]$RequiredVersion,
    238         [string]$MinimumVersion,
    239         [string]$MaximumVersion
    240 
    241     )
    242          
    243     Write-Verbose -Message ($LocalizedData.CallingFunction -f $($MyInvocation.mycommand))
    244 
    245     $isValid = $false
    246          
    247     #Case 1: No further check required if a user provides either none or one of these: minimumVersion, maximumVersion, and requiredVersion
    248     if ($PSBoundParameters.Count -le 1)
    249     {
    250         return $true
    251     }
    252 
    253     #Case 2: #If no RequiredVersion is provided 
    254     if (-not $PSBoundParameters.ContainsKey('RequiredVersion'))
    255     {
    256         #If no RequiredVersion, both MinimumVersion and MaximumVersion are provided. Otherwise fall into the Case #1
    257         $isValid = $PSBoundParameters['MinimumVersion'] -le $PSBoundParameters['MaximumVersion']
    258     }
    259     
    260     #Case 3: RequiredVersion is provided. 
    261     #        In this case  MinimumVersion and/or MaximumVersion also are provided. Otherwise fall in to Case #1.
    262     #        This is an invalid case. When RequiredVersion is provided, others are not allowed. so $isValid is false, which is already set in the init
    263 
    264     if ($isValid -eq $false)
    265     {        
    266         ThrowError  -ExceptionName "System.ArgumentException" `
    267                     -ExceptionMessage ($LocalizedData.VersionError)`
    268                     -ErrorId "VersionError" `
    269                     -ErrorCategory InvalidArgument
    270     }
    271 }

    299     }
