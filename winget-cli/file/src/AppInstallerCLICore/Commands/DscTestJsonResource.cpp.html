     19         struct TestJsonFunctionData
     20         {
     21             TestJsonFunctionData()
     22             {
     23                 InitializeFileData();
     24             }

     26             TestJsonFunctionData(const std::optional<Json::Value>& json) : Input(json), Output(Input.CopyForOutput())

     28                 InitializeFileData();
     29             }

     36             static std::filesystem::path GetFilePath()
     37             {
     38                 std::filesystem::path result = Runtime::GetPathTo(Runtime::PathName::LocalState);
     39                 result /= "test-json-file.json";
     40                 return result;
     41             }
     42 
     43             // Fills the Output object with the current state
     44             void Get()
     45             {
     46                 const std::string& propertyName = Input.Property().value();
     47                 const Json::Value* propertyValue = RootValue.find(propertyName.data(), propertyName.data() + propertyName.length());
     48 
     49                 if (propertyValue)
     50                 {
     51                     Output.Exist(true);
     52 
     53                     Output.Value(*propertyValue);
     54                 }
     55                 else
     56                 {
     57                     Output.Exist(false);
     58                 }
     59             }

     61         private:
     62             void InitializeFileData()
     63             {
     64                 FilePath = GetFilePath();
     65                 RootValue = GetJsonFromFile();
     66             }

     68             Json::Value GetJsonFromFile() const
     69             {
     70                 Json::Value result;
     71                 Json::CharReaderBuilder builder;
     72                 Json::String errors;
     73 
     74                 std::ifstream stream{ FilePath, std::ios::binary };
     75 
     76                 if (stream)
     77                 {
     78                     if (!Json::parseFromStream(builder, stream, &result, &errors))
     79                     {
     80                         AICLI_LOG(CLI, Warning, << "Failed to read test JSON file: " << errors);
     81                         result = Json::Value{};
     82                     }
     83                 }
     84                 else
     85                 {
     86                     AICLI_LOG(CLI, Warning, << "Couldn't open test JSON file: " << FilePath);
     87                 }
     88 
     89                 return result;
     90             }

    134     void DscTestJsonResource::ResourceFunctionGet(Execution::Context& context) const

    146     void DscTestJsonResource::ResourceFunctionSet(Execution::Context& context) const
    147     {
    148         if (auto json = GetJsonFromInput(context))
    149         {
    150             TestJsonFunctionData data{ json };
    151 
    152             data.Get();
    153 
    154             if (data.RootValue.isNull())
    155             {
    156                 data.RootValue = Json::Value{ Json::objectValue };
    157             }
    158 
    159             if (data.Input.ShouldExist())
    160             {
    161                 data.RootValue[data.Input.Property().value()] = data.Input.Value().value_or(Json::Value{ Json::nullValue });
    162                 data.Output.Exist(true);
    163                 data.Output.Value(data.RootValue[data.Input.Property().value()]);
    164             }
    165             else if (data.Output.Exist().value())
    166             {
    167                 data.RootValue.removeMember(data.Input.Property().value());
    168                 data.Output.Exist(false);
    169             }
    170 
    171             std::ofstream stream{ data.FilePath, std::ios::binary };
    172 
    173             Json::StreamWriterBuilder writerBuilder;
    174             writerBuilder.settings_["indentation"] = "  ";
    175 
    176             stream << Json::writeString(writerBuilder, data.RootValue);
    177 
    178             WriteJsonOutputLine(context, data.Output.ToJson());
    179         }
    180     }

    182     void DscTestJsonResource::ResourceFunctionExport(Execution::Context& context) const
    183     {
    184         TestJsonFunctionData data;
    185 
    186         if (data.RootValue.isObject())
    187         {
    188             for (const auto& member : data.RootValue.getMemberNames())
    189             {
    190                 const Json::Value* memberValue = data.RootValue.find(member.data(), member.data() + member.length());
    191 
    192                 if (memberValue)
    193                 {
    194                     TestJsonObject output;
    195                     output.Property(member);
    196                     output.Value(*memberValue);
    197 
    198                     WriteJsonOutputLine(context, output.ToJson());
    199                 }
    200             }
    201         }
    202     }
