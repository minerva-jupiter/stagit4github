     62             const Json::Value& GetResolvedInput()
     63             {
     64                 THROW_HR_IF(E_UNEXPECTED, !Input.Settings().has_value());
     65                 if (!_resolvedInputUserSettings)
     66                 {
     67                     if(Input.Action().has_value() && Utility::CaseInsensitiveEquals(Input.Action().value(), ACTION_FULL))

     69                         Output.Action(ACTION_FULL);

    103             Json::Value MergeUserSettingsFiles(const Json::Value& overlay)
    104             {
    105                 Json::Value mergedUserSettingsFile = GetUserSettings();
    106                 MergeUserSettingsFiles(mergedUserSettingsFile, overlay);
    107                 return mergedUserSettingsFile;

    110             // Merges the overlay settings into the target settings.
    111             void MergeUserSettingsFiles(Json::Value& target, const Json::Value& overlay)
    112             {
    113                 // If either is not an object, we can't merge.
    114                 if (!overlay.isObject() || !target.isObject())
    115                 {
    116                     return;
    117                 }
    118 
    119                 // Iterate through the overlay settings and merge them into the target.
    120                 for (const auto& overlayKey : overlay.getMemberNames())
    121                 {
    122                     const Json::Value& overlayValue = overlay[overlayKey];
    123                     if (target.isMember(overlayKey))
    124                     {
    125                         Json::Value& targetValue = target[overlayKey];
    126                         if (targetValue.isObject() && overlayValue.isObject())
    127                         {
    128                             // Recursively merge the objects.
    129                             MergeUserSettingsFiles(targetValue, overlayValue);
    130                         }
    131                         else
    132                         {
    133                             // Replace the value in the target.
    134                             // Note: Arrays are not merged, they are replaced.
    135                             target[overlayKey] = overlayValue;
    136                         }
    137                     }
    138                     else
    139                     {
    140                         // Add the overlay key to the target.
    141                         target[overlayKey] = overlayValue;
    142                     }
    143                 }
    144             }

    146             const Json::Value& GetUserSettings()
    147             {
    148                 if (!_userSettings)
    149                 {
    150                     _userSettings = Json::objectValue;
    151                     std::ifstream file(_userSettingsPath, std::ios::binary);
    152                     if (file)
    153                     {
    154                         Json::CharReaderBuilder builder;
    155                         std::string errs;
    156                         Json::Value jsonRoot;
    157                         if (Json::parseFromStream(builder, file, &jsonRoot, &errs))
    158                         {
    159                             _userSettings = jsonRoot;
    160                         }
    161                         else
    162                         {
    163                             AICLI_LOG(Config, Warning, << "Failed to parse user settings file: " << _userSettingsPath << ", error: " << errs);
    164                         }
    165                     }
    166                     else
    167                     {
    168                         AICLI_LOG(Config, Warning, << "Failed to open user settings file: " << _userSettingsPath);
    169                     }
    170                 }
    171 
    172                 return *_userSettings;
    173             }

    204     void DscUserSettingsFileResource::ResourceFunctionSet(Execution::Context& context) const
    205     {
    206         if (auto json = GetJsonFromInput(context))
    207         {
    208             UserSettingsFileFunctionData data{ json };
    209 
    210             data.Get();
    211 
    212             // Capture the diff before updating the output
    213             auto diff = data.DiffJson();
    214 
    215             if (!data.Test())
    216             {
    217                 data.Output.Settings(data.GetResolvedInput());
    218                 if (!data.WriteOutput())
    219                 {
    220                     AICLI_TERMINATE_CONTEXT(HRESULT_FROM_WIN32(ERROR_OPEN_FAILED));
    221                     return;
    222                 }
    223             }
    224 
    225             WriteJsonOutputLine(context, data.Output.ToJson());
    226             WriteJsonOutputLine(context, diff);
    227         }
    228     }
