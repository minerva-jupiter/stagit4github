     18     // The kind of resource that this command is implementing.
     19     enum class DscResourceKind
     20     {
     21         // Standard resource
     22         Resource,
     23         // Group resource
     24         Group,
     25         // Adapter resource
     26         Adapter,
     27         // Import resource
     28         Import,
     29     };
     30 
     31     // The functions that a DSC resource can provide.
     32     enum class DscFunctions
     33     {
     34         None        = 0x000,
     35         // Gets the current state; should always be implemented.
     36         Get         = 0x001,
     37         // Sets the state; should always be implemented.
     38         Set         = 0x002,
     39         // Produces the output of Set without modifying state.
     40         WhatIf      = 0x004,
     41         // Determines if the current state matches the given state.
     42         Test        = 0x008,
     43         // Deletes the given state.
     44         Delete      = 0x010,
     45         // Gets all instances of the resource.
     46         Export      = 0x020,
     47         // Required for a Group resource, ignored for all others.
     48         Validate    = 0x040,
     49         // Required for an Import resource.
     50         Resolve     = 0x080,
     51         // Required for an Adapter resource.
     52         Adapter     = 0x100,
     53         // Gets the schema for the resource's properties.
     54         Schema      = 0x200,
     55     };
     56 
     57     DEFINE_ENUM_FLAG_OPERATORS(DscFunctions);
     58 
     59     // Behavior changes for DSC functions.
     60     enum class DscFunctionModifiers
     61     {
     62         None                = 0x00,
     63         // The resource implements a check during Set (and WhatIf) to determine if already in the correct state.
     64         // If not provided, DSC will ensure that the state is tested beforehand.
     65         ImplementsPretest   = 0x01,
     66         // The resource will act on the `_exist` property during Set (and WhatIf).
     67         // If not provided, the resource should implement Delete.
     68         HandlesExist        = 0x02,
     69         // Functions that may return state information (set, what-if, test) return only the state.
     70         ReturnsState        = 0x04,
     71         // Functions that may return state information (set, what-if, test) return the state and property difference.
     72         ReturnsStateAndDiff = 0x08,
     73     };
     74 
     75     DEFINE_ENUM_FLAG_OPERATORS(DscFunctionModifiers);
