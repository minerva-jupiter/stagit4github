     55     void RepairCommand::Complete(Execution::Context& context, Execution::Args::Type valueType) const
     56     {
     57         if (valueType == Execution::Args::Type::Manifest ||
     58             valueType == Execution::Args::Type::Log)
     59         {
     60             // Intentionally output nothing to allow pass through to filesystem.
     61             return;
     62         }
     63 
     64         context <<
     65             Workflow::OpenSource() <<
     66             Workflow::OpenCompositeSource(Repository::PredefinedSource::Installed);
     67 
     68         switch (valueType)
     69         {
     70         case Execution::Args::Type::Id:
     71         case Execution::Args::Type::Name:
     72         case Execution::Args::Type::Moniker:
     73         case Execution::Args::Type::TargetVersion:
     74         case Execution::Args::Type::Channel:
     75         case Execution::Args::Type::Source:
     76             context <<
     77                 Workflow::CompleteWithSingleSemanticsForValueUsingExistingSource(valueType);
     78             break;
     79         }
     80     }

     93     void RepairCommand::ExecuteInternal(Execution::Context& context) const
     94     {
     95         context.SetFlags(Execution::ContextFlag::InstallerExecutionUseRepair);
     96 
     97         context <<
     98             Workflow::InitializeInstallerDownloadAuthenticatorsMap <<
     99             Workflow::ReportExecutionStage(ExecutionStage::Discovery) <<
    100             Workflow::OpenSource() <<
    101             Workflow::OpenCompositeSource(DetermineInstalledSource(context));
    102 
    103         if (context.Args.Contains(Args::Type::Manifest))
    104         {
    105             context <<
    106                 Workflow::GetManifestFromArg <<
    107                 Workflow::ReportManifestIdentity <<
    108                 Workflow::SearchSourceUsingManifest <<
    109                 Workflow::EnsureOneMatchFromSearchResult(OperationType::Repair);
    110         }
    111         else
    112         {
    113             context <<
    114                 Workflow::SearchSourceForSingle <<
    115                 Workflow::HandleSearchResultFailures <<
    116                 Workflow::EnsureOneMatchFromSearchResult(OperationType::Repair) <<
    117                 Workflow::ReportPackageIdentity;
    118         }
    119 
    120         context <<
    121             Workflow::GetInstalledPackageVersion <<
    122             Workflow::SelectApplicableInstallerIfNecessary <<
    123             Workflow::RepairSinglePackage;
    124     }
