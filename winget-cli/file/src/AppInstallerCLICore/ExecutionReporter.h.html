     39     // Reporter should be the central place to show workflow status to user.
     40     struct Reporter : public IProgressSink
     41     {
     42         // The channel that the reporter is targeting.
     43         // Based on commands/arguments, only one of these channels can be chosen.
     44         enum class Channel
     45         {
     46             Output,
     47             Completion,
     48             Json,
     49             Disabled,
     50         };
     51 
     52         // The level for the Output channel.
     53         enum class Level : uint32_t
     54         {
     55             None = 0x0,
     56             Verbose = 0x1,
     57             Info = 0x2,
     58             Warning = 0x4,
     59             Error = 0x8,
     60             All = Verbose | Info | Warning | Error,
     61         };
