     49     // A task in the workflow.
     50     struct WorkflowTask
     51     {
     52         using Func = void (*)(Execution::Context&);
     53 
     54         WorkflowTask(Func f) : m_isFunc(true), m_func(f) {}
     55         WorkflowTask(std::string_view name, bool executeAlways = false) : m_name(name), m_executeAlways(executeAlways) {}
     56 
     57         virtual ~WorkflowTask() = default;
     58 
     59         WorkflowTask(const WorkflowTask&) = default;
     60         WorkflowTask& operator=(const WorkflowTask&) = default;
     61 
     62         WorkflowTask(WorkflowTask&&) = default;
     63         WorkflowTask& operator=(WorkflowTask&&) = default;
     64 
     65         bool operator==(const WorkflowTask& other) const;
     66 
     67         virtual void operator()(Execution::Context& context) const;
     68 
     69         const std::string& GetName() const { return m_name; }
     70         bool IsFunction() const { return m_isFunc; }
     71         Func Function() const { return m_func; }
     72         bool ExecuteAlways() const { return m_executeAlways; }
     73         void Log() const;
     74 
     75     private:
     76         bool m_isFunc = false;
     77         Func m_func = nullptr;
     78         std::string m_name;
     79         bool m_executeAlways = false;
     80     };
