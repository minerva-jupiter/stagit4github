    327     std::vector<ArgumentCommon> ArgumentCommon::GetFromExecArgs(const Execution::Args& execArgs)
    328     {
    329         auto argTypes = execArgs.GetTypes();
    330         std::vector<ArgumentCommon> result;
    331         std::transform(argTypes.begin(), argTypes.end(), std::back_inserter(result), ArgumentCommon::ForType);
    332         return result;
    333     }

    513     void Argument::ValidateExclusiveArguments(const Execution::Args& args)
    514     {
    515         auto argProperties = ArgumentCommon::GetFromExecArgs(args);
    516 
    517         using ExclusiveSet_t = std::underlying_type_t<ArgTypeExclusiveSet>;
    518         for (ExclusiveSet_t i = 1 + static_cast<ExclusiveSet_t>(ArgTypeExclusiveSet::None); i < static_cast<ExclusiveSet_t>(ArgTypeExclusiveSet::Max); i <<= 1)
    519         {
    520             std::vector<ArgumentCommon> argsFromSet;
    521             std::copy_if(
    522                 argProperties.begin(),
    523                 argProperties.end(),
    524                 std::back_inserter(argsFromSet),
    525                 [=](const ArgumentCommon& arg) { return static_cast<ExclusiveSet_t>(arg.ExclusiveSet) & i; });
    526 
    527             if (argsFromSet.size() > 1)
    528             {
    529                 // Create a string showing the exclusive args.
    530                 std::string argsString;
    531                 for (const auto& arg : argsFromSet)
    532                 {
    533                     if (!argsString.empty())
    534                     {
    535                         argsString += '|';
    536 
    537                     }
    538 
    539                     argsString += arg.Name;
    540                 }
    541 
    542                 throw CommandException(Resource::String::MultipleExclusiveArgumentsProvided(Utility::LocIndString{ argsString }));
    543             }
    544         }
    545     }

    557     ArgTypeCategory Argument::GetCategoriesPresent(const Execution::Args& args)
    558     {
    559         auto argProperties = ArgumentCommon::GetFromExecArgs(args);
    560 
    561         ArgTypeCategory result = ArgTypeCategory::None;

    563         {

    565         }

    570     ArgTypeCategory Argument::GetCategoriesAndValidateCommonArguments(const Execution::Args& args, bool requirePackageSelectionArg)
    571     {
    572         const auto categories = GetCategoriesPresent(args);
    573 
    574         // Commands like install require some argument to select a package
    575         if (requirePackageSelectionArg)
    576         {
    577             if (WI_AreAllFlagsClear(categories, ArgTypeCategory::Manifest | ArgTypeCategory::PackageQuery | ArgTypeCategory::SinglePackageQuery))
    578             {
    579                 throw CommandException(Resource::String::NoPackageSelectionArgumentProvided);
    580             }
    581         }
    582 
    583         // If a manifest is specified, we cannot also have arguments for searching
    584         if (WI_IsFlagSet(categories, ArgTypeCategory::Manifest) &&
    585             WI_IsAnyFlagSet(categories, ArgTypeCategory::PackageQuery | ArgTypeCategory::QuerySource))
    586         {
    587             throw CommandException(Resource::String::BothManifestAndSearchQueryProvided);
    588         }
    589 
    590         // If we have multiple packages, we cannot have arguments that only make sense for a single package
    591         if (WI_IsFlagSet(categories, ArgTypeCategory::MultiplePackages) &&
    592             WI_IsAnyFlagSet(categories, ArgTypeCategory::SinglePackageQuery | ArgTypeCategory::SingleInstallerBehavior))
    593         {
    594             throw CommandException(Resource::String::ArgumentForSinglePackageProvidedWithMultipleQueries);
    595         }
    596 
    597         return categories;
    598     }
