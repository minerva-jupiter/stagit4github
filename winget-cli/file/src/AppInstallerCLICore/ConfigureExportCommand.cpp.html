     55     void ConfigureExportCommand::ValidateArgumentsInternal(Execution::Args& execArgs) const
     56     {
     57         Configuration::ValidateCommonArguments(execArgs);
     58 
     59         if (!execArgs.Contains(Execution::Args::Type::ConfigurationExportModule, Execution::Args::Type::ConfigurationExportResource) &&
     60             !execArgs.Contains(Execution::Args::Type::ConfigurationExportPackageId) &&
     61             !execArgs.Contains(Execution::Args::Type::ConfigurationExportAll))
     62         {
     63             throw CommandException(Resource::String::ConfigureExportArgumentRequiredError);
     64         }
     65 
     66         if (execArgs.Contains(Execution::Args::Type::ConfigurationExportAll) &&
     67             (execArgs.Contains(Execution::Args::Type::ConfigurationExportPackageId) ||
     68              execArgs.Contains(Execution::Args::Type::ConfigurationExportModule) ||
     69              execArgs.Contains(Execution::Args::Type::ConfigurationExportResource)))
     70         {
     71             throw CommandException(Resource::String::ConfigureExportArgumentConflictWithAllError);
     72         }
     73     }
