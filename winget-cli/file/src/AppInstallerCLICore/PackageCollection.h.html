     14     // Container for data to identify multiple packages to be installed from multiple sources.
     15     struct PackageCollection
     16     {
     17         // Description of a package.
     18         // Does not represent the actual package, just enough to find and install it.
     19         struct Package
     20         {
     21             Package() = default;
     22             Package(Utility::LocIndString&& id) :
     23                 Id(std::move(id)) {}
     24             Package(Utility::LocIndString&& id, Utility::Version&& version, Utility::Channel&& channel) :
     25                 Id(std::move(id)), VersionAndChannel(std::move(version), std::move(channel)) {}
     26             Package(Utility::LocIndString&& id, Utility::VersionAndChannel&& versionAndChannel) :
     27                 Id(std::move(id)), VersionAndChannel(std::move(versionAndChannel)) {}
     28 
     29             Utility::LocIndString Id;
     30             Utility::VersionAndChannel VersionAndChannel;
     31             Manifest::ScopeEnum Scope = Manifest::ScopeEnum::Unknown;
     32             std::filesystem::path InstalledLocation;
     33         };
     34 
     35         // A source along with a set of packages available from it.
     36         struct Source
     37         {
     38             Source() = default;
     39             Source(const Repository::SourceDetails& sourceDetails) : Details(sourceDetails) {}
     40             Source(Repository::SourceDetails&& sourceDetails) : Details(std::move(sourceDetails)) {}
     41 
     42             Repository::SourceDetails Details;
     43             std::vector<Package> Packages;
     44         };
     45 
     46         // Version of the WinGet client that produced this collection.
     47         std::string ClientVersion;
     48 
     49         // Requests from each individual source.
     50         std::vector<Source> Sources;
     51     };
