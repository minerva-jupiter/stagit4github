     74         /// <inheritdoc/>
     75         public CreateManifestResult CreateManifest(string manifestPath, string mergedManifestPath, WinGetCreateManifestOption option)
     76         {
     77             try
     78             {
     79                 bool succeeded = false;
     80                 IntPtr manifestHandle = IntPtr.Zero;
     81                 string failureOrWarningMessage = null;
     82                 WinGetCreateManifest(
     83                     manifestPath,
     84                     out succeeded,
     85                     out manifestHandle,
     86                     out failureOrWarningMessage,
     87                     mergedManifestPath,
     88                     option);
     89 
     90                 // WinGetUtil uses exceptions for passing warning messages. When a warning is detected,
     91                 // WinGetUtil will throw so a manifest handle will not be returned, in this case, we try
     92                 // to create the manifest handle without any validation.
     93                 if (succeeded && manifestHandle == IntPtr.Zero)
     94                 {
     95                     WinGetCreateManifest(
     96                         manifestPath,
     97                         out succeeded,
     98                         out manifestHandle,
     99                         out string failureNotExpected,
    100                         null,
    101                         WinGetCreateManifestOption.NoValidation);
    102                 }
    103 
    104                 return new CreateManifestResult(succeeded, failureOrWarningMessage, succeeded ? new WinGetManifest(manifestHandle) : null);
    105             }
    106             catch (Exception e)
    107             {
    108                 throw new WinGetManifestException(e);
    109             }
    110         }
