    401         /// <summary>
    402         /// Returns a List of strings containing the URIs contained within this manifest.
    403         /// </summary>
    404         /// <returns>List of strings.</returns>
    405         public List<string> GetURIs()
    406         {
    407             List<string> uris = new List<string>();
    408             if (!string.IsNullOrEmpty(this.PublisherUrl))
    409             {
    410                 uris.Add(this.PublisherUrl);
    411             }
    412 
    413             if (!string.IsNullOrEmpty(this.PublisherSupportUrl))
    414             {
    415                 uris.Add(this.PublisherSupportUrl);
    416             }
    417 
    418             if (!string.IsNullOrEmpty(this.PrivacyUrl))
    419             {
    420                 uris.Add(this.PrivacyUrl);
    421             }
    422 
    423             if (!string.IsNullOrEmpty(this.PackageUrl))
    424             {
    425                 uris.Add(this.PackageUrl);
    426             }
    427 
    428             if (!string.IsNullOrEmpty(this.LicenseUrl))
    429             {
    430                 uris.Add(this.LicenseUrl);
    431             }
    432 
    433             if (!string.IsNullOrEmpty(this.CopyrightUrl))
    434             {
    435                 uris.Add(this.CopyrightUrl);
    436             }
    437 
    438             if (!string.IsNullOrEmpty(this.ReleaseNotesUrl))
    439             {
    440                 uris.Add(this.ReleaseNotesUrl);
    441             }
    442 
    443             if (this.Agreements != null)
    444             {
    445                 foreach (var agreement in this.Agreements)
    446                 {
    447                     if (!string.IsNullOrEmpty(agreement.AgreementUrl))
    448                     {
    449                         uris.Add(agreement.AgreementUrl);
    450                     }
    451                 }
    452             }
    453 
    454             foreach (ManifestInstaller installer in this.Installers)
    455             {
    456                 uris.AddRange(installer.GetURIs());
    457             }
    458 
    459             if (this.Localization != null)
    460             {
    461                 foreach (ManifestLocalization localization in this.Localization)
    462                 {
    463                     uris.AddRange(localization.GetURIs());
    464                 }
    465             }
    466 
    467             if (this.Documentations != null)
    468             {
    469                 foreach (var docs in this.Documentations)
    470                 {
    471                     if (!string.IsNullOrEmpty(docs.DocumentUrl))
    472                     {
    473                         uris.Add(docs.DocumentUrl);
    474                     }
    475                 }
    476             }
    477 
    478             if (this.Icons != null)
    479             {
    480                 foreach (var icon in this.Icons)
    481                 {
    482                     if (!string.IsNullOrEmpty(icon.IconUrl))
    483                     {
    484                         uris.Add(icon.IconUrl);
    485                     }
    486                 }
    487             }
    488 
    489             return uris;
    490         }

    492         /// <summary>
    493         /// Checks if two manifests are equivalent.
    494         /// </summary>
    495         /// <param name="other">other object.</param>
    496         /// <returns>boolean indicating if the objects are equals.</returns>
    497         public bool IsManifestEquivalent(Manifest other)
    498         {
    499             // If parameter is null, return false.
    500             if (ReferenceEquals(other, null))
    501             {
    502                 return false;
    503             }
    504 
    505             // Optimization for a common success case.
    506             if (ReferenceEquals(this, other))
    507             {
    508                 return true;
    509             }
    510 
    511             // If run-time types are not exactly the same, return false.
    512             if (this.GetType() != other.GetType())
    513             {
    514                 return false;
    515             }
    516 
    517             // Equality of Manifest consist on only these properties.
    518             return (this.Id == other.Id) &&
    519                    (this.Version == other.Version) &&
    520                    this.CompareInstallers(other.Installers);
    521         }

    523         private bool CompareInstallers(List<ManifestInstaller> installers)
    524         {
    525             ISet<ManifestInstaller> first =
    526                 new HashSet<ManifestInstaller>(
    527                     this.Installers != null ?
    528                     this.Installers :
    529                     new List<ManifestInstaller>());
    530 
    531             return first.SetEquals(installers != null
    532                 ? installers : new List<ManifestInstaller>());
    533         }
