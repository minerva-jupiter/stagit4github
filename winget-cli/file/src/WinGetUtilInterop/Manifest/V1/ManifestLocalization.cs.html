    130         public List<string> GetURIs()
    131         {
    132             List<string> uris = new List<string>();
    133             if (!string.IsNullOrEmpty(this.PublisherUrl))
    134             {
    135                 uris.Add(this.PublisherUrl);
    136             }
    137 
    138             if (!string.IsNullOrEmpty(this.PublisherSupportUrl))
    139             {
    140                 uris.Add(this.PublisherSupportUrl);
    141             }
    142 
    143             if (!string.IsNullOrEmpty(this.PrivacyUrl))
    144             {
    145                 uris.Add(this.PrivacyUrl);
    146             }
    147 
    148             if (!string.IsNullOrEmpty(this.PackageUrl))
    149             {
    150                 uris.Add(this.PackageUrl);
    151             }
    152 
    153             if (!string.IsNullOrEmpty(this.LicenseUrl))
    154             {
    155                 uris.Add(this.LicenseUrl);
    156             }
    157 
    158             if (!string.IsNullOrEmpty(this.CopyrightUrl))
    159             {
    160                 uris.Add(this.CopyrightUrl);
    161             }
    162 
    163             if (!string.IsNullOrEmpty(this.ReleaseNotesUrl))
    164             {
    165                 uris.Add(this.ReleaseNotesUrl);
    166             }
    167 
    168             if (this.Agreements != null)
    169             {
    170                 foreach (var agreement in this.Agreements)
    171                 {
    172                     if (!string.IsNullOrEmpty(agreement.AgreementUrl))
    173                     {
    174                         uris.Add(agreement.AgreementUrl);
    175                     }
    176                 }
    177             }
    178 
    179             if (this.Documentations != null)
    180             {
    181                 foreach (var docs in this.Documentations)
    182                 {
    183                     if (!string.IsNullOrEmpty(docs.DocumentUrl))
    184                     {
    185                         uris.Add(docs.DocumentUrl);
    186                     }
    187                 }
    188             }
    189 
    190             if (this.Icons != null)
    191             {
    192                 foreach (var icon in this.Icons)
    193                 {
    194                     if (!string.IsNullOrEmpty(icon.IconUrl))
    195                     {
    196                         uris.Add(icon.IconUrl);
    197                     }
    198                 }
    199             }
    200 
    201             return uris;
    202         }
