    114         /// <summary>

    119         /// <summary>
    120         /// Arp version validation failure.
    121         /// </summary>
    122         ArpVersionValidationFailure = 0x2,
    123 
    124         /// <summary>
    125         /// Installer validation failure.
    126         /// </summary>
    127         InstallerValidationFailure = 0x4,

    130 
    131         /// <summary>
    132         /// Single Manifest package has dependencies.
    133         /// </summary>
    134         SingleManifestPackageHasDependencies = 0x10000,
    135 
    136         /// <summary>

    138         /// </summary>
    139         MultiManifestPackageHasDependencies = 0x20000,

    151         /// <summary>

    153         /// </summary>
    154         FoundDependencyLoop = 0x100000,

    158         /// <summary>
    159         /// Manifest validation internal error.
