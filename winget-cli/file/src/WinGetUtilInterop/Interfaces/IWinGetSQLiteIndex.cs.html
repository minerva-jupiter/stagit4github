      7 namespace Microsoft.WinGetUtil.Interfaces

      9     using System;

     11     /// <summary>
     12     /// The properties that can be set with IWinGetSQLiteIndex::SetProperty.
     13     /// The values must match those in WinGetUtil.h.
     14     /// </summary>
     15     public enum SQLiteIndexProperty
     16     {
     17         /// <summary>
     18         /// The base time to use for update tracking. The value is in the Unix epoch.
     19         /// Set to an empty string to use the current time.
     20         /// Set to 0 to force all files to be output.
     21         /// </summary>
     22         PackageUpdateTrackingBaseTime = 0,
     23 
     24         /// <summary>
     25         /// The full path to a base directory where intermediate files will be output.
     26         /// The path does not need to exist, and may not be created if no files need to be written.
     27         /// </summary>
     28         IntermediateFileOutputPath = 1,
     29     }

     33     /// </summary>

     35     {
     36         /// <summary>
     37         /// Migrates the index to the given version.
     38         /// </summary>

     42 
     43         /// <summary>
     44         /// Sets the given property to the given value.
     45         /// </summary>
