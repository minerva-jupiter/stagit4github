    156         /// <summary>
    157         /// Get CLSID based on the provided context for the specified type
    158         /// </summary>
    159         /// <typeparam name="T">Projected class type</typeparam>
    160         /// <param name="context">Context</param>
    161         /// <returns>CLSID for the provided context and type, or throw an exception if not found.</returns>
    162         public static Guid GetClsid<T>(ClsidContext context)
    163         {
    164             ValidateType(typeof(T));
    165             return Classes[typeof(T)].GetClsid(context);
    166         }

    168         /// <summary>
