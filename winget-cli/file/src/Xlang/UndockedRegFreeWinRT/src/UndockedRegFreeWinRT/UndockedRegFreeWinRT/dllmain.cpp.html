    106 /* 
    107 In the context callback call to the MTA apartment, there is a bug that prevents COM 
    108 from automatically initializing MTA remoting. It only allows NTA to be intialized 
    109 outside of the NTA and blocks all others. The workaround for this is to spin up another 
    110 thread that is not been CoInitialize. COM treats this thread as a implicit MTA and 
    111 when we call CoGetObjectContext on it we implicitily initialized the MTA. 
    112 */
    113 HRESULT EnsureMTAInitialized()
    114 {
    115     TP_CALLBACK_ENVIRON callBackEnviron;
    116     InitializeThreadpoolEnvironment(&callBackEnviron);
    117     PTP_POOL pool = CreateThreadpool(nullptr);
    118     if (pool == nullptr)
    119     {
    120         return HRESULT_FROM_WIN32(GetLastError());
    121     }
    122     SetThreadpoolThreadMaximum(pool, 1);
    123     if (!SetThreadpoolThreadMinimum(pool, 1)) 
    124     {
    125         return HRESULT_FROM_WIN32(GetLastError());
    126     }
    127     PTP_CLEANUP_GROUP cleanupgroup = CreateThreadpoolCleanupGroup();
    128     if (cleanupgroup == nullptr)
    129     {
    130         return HRESULT_FROM_WIN32(GetLastError());
    131     }
    132     SetThreadpoolCallbackPool(&callBackEnviron, pool);
    133     SetThreadpoolCallbackCleanupGroup(&callBackEnviron,
    134         cleanupgroup,
    135         nullptr);
    136     PTP_WORK ensureMTAInitializedWork = CreateThreadpoolWork(
    137         &EnsureMTAInitializedCallBack,
    138         nullptr,
    139         &callBackEnviron);
    140     if (ensureMTAInitializedWork == nullptr)
    141     {
    142         return HRESULT_FROM_WIN32(GetLastError());
    143     }
    144     SubmitThreadpoolWork(ensureMTAInitializedWork);
    145     CloseThreadpoolCleanupGroupMembers(cleanupgroup,
    146         false,
    147         nullptr);
    148     return S_OK;
    149 }

    151 HRESULT GetActivationLocation(HSTRING activatableClassId, ActivationLocation &activationLocation)
    152 {
    153     APTTYPE aptType;
    154     APTTYPEQUALIFIER aptQualifier;
    155     RETURN_IF_FAILED(CoGetApartmentType(&aptType, &aptQualifier));
    156 
    157     ABI::Windows::Foundation::ThreadingType threading_model;
    158     RETURN_IF_FAILED(WinRTGetThreadingModel(activatableClassId, &threading_model)); //REGDB_E_CLASSNOTREG
    159     switch (threading_model)
    160     {
    161     case ABI::Windows::Foundation::ThreadingType_BOTH:
    162         activationLocation = ActivationLocation::CurrentApartment;
    163         break;
    164     case ABI::Windows::Foundation::ThreadingType_STA:
    165         if (aptType == APTTYPE_MTA)
    166         {
    167             return RO_E_UNSUPPORTED_FROM_MTA;
    168         }
    169         else
    170         {
    171             activationLocation = ActivationLocation::CurrentApartment;
    172         }
    173         break;
    174     case ABI::Windows::Foundation::ThreadingType_MTA:
    175         if (aptType == APTTYPE_MTA)
    176         {
    177             activationLocation = ActivationLocation::CurrentApartment;
    178         }
    179         else
    180         {
    181             activationLocation = ActivationLocation::CrossApartmentMTA;
    182         }
    183         break;
    184     }
    185     return S_OK;
    186 }

    189 HRESULT WINAPI RoActivateInstanceDetour(HSTRING activatableClassId, IInspectable** instance)
    190 {
    191     ActivationLocation location;
    192     HRESULT hr = GetActivationLocation(activatableClassId, location);
    193     if (hr == REGDB_E_CLASSNOTREG)
    194     {
    195         return TrueRoActivateInstance(activatableClassId, instance);
    196     }
    197     RETURN_IF_FAILED(hr);
    198 
    199     // Activate in current apartment
    200     if (location == ActivationLocation::CurrentApartment)
    201     {
    202         Microsoft::WRL::ComPtr<IActivationFactory> pFactory;
    203         RETURN_IF_FAILED(WinRTGetActivationFactory(activatableClassId, __uuidof(IActivationFactory), (void**)&pFactory));
    204         return pFactory->ActivateInstance(instance);
    205     }
    206 
    207     // Cross apartment MTA activation
    208     struct CrossApartmentMTAActData {
    209         HSTRING activatableClassId;
    210         IStream *stream;
    211     };
    212 
    213     CrossApartmentMTAActData cbdata{ activatableClassId };
    214     CO_MTA_USAGE_COOKIE mtaUsageCookie;
    215     RETURN_IF_FAILED(CoIncrementMTAUsage(&mtaUsageCookie));
    216     RETURN_IF_FAILED(EnsureMTAInitialized());
    217     Microsoft::WRL::ComPtr<IContextCallback> defaultContext;
    218     ComCallData data;
    219     data.pUserDefined = &cbdata;
    220     RETURN_IF_FAILED(CoGetDefaultContext(APTTYPE_MTA, IID_PPV_ARGS(&defaultContext)));
    221     RETURN_IF_FAILED(defaultContext->ContextCallback(
    222         [](_In_ ComCallData* pComCallData) -> HRESULT
    223         {
    224             CrossApartmentMTAActData* data = reinterpret_cast<CrossApartmentMTAActData*>(pComCallData->pUserDefined);
    225             Microsoft::WRL::ComPtr<IInspectable> instance;
    226             Microsoft::WRL::ComPtr<IActivationFactory> pFactory;
    227             RETURN_IF_FAILED(WinRTGetActivationFactory(data->activatableClassId, __uuidof(IActivationFactory), (void**)&pFactory));
    228             RETURN_IF_FAILED(pFactory->ActivateInstance(&instance));
    229             RETURN_IF_FAILED(CoMarshalInterThreadInterfaceInStream(IID_IInspectable, instance.Get(), &data->stream));
    230             return S_OK;
    231         },
    232         &data, IID_ICallbackWithNoReentrancyToApplicationSTA, 5, nullptr)); // 5 is meaningless.
    233     RETURN_IF_FAILED(CoGetInterfaceAndReleaseStream(cbdata.stream, IID_IInspectable, (LPVOID*)instance));
    234     return S_OK;
    235 }

    297     const HSTRING name,
    298     const HSTRING windowsMetaDataDir,
    299     const DWORD packageGraphDirsCount,
    300     const HSTRING* packageGraphDirs,
    301     DWORD* metaDataFilePathsCount,

    320             packageGraphDirsCount, packageGraphDirs,
    321             metaDataFilePathsCount, metaDataFilePaths,
    322             subNamespacesCount, subNamespaces);
    323     }

    327 void InstallHooks()
    328 {
    329     if (DetourIsHelperProcess()) {
    330         return;
    331     }
    332 
    333     DetourRestoreAfterWith();
    334

    356 HRESULT ExtRoLoadCatalog()
    357 {
    358     WCHAR filePath[MAX_PATH];
    359     if (!GetModuleFileNameW(nullptr, filePath, _countof(filePath)))
    360     {
    361         return HRESULT_FROM_WIN32(GetLastError());
    362     }
    363     std::wstring manifestPath(filePath);
    364     HANDLE hActCtx = INVALID_HANDLE_VALUE;
    365     auto exit = wil::scope_exit([&]
    366     {
    367         if (hActCtx != INVALID_HANDLE_VALUE)
    368         {
    369             ReleaseActCtx(hActCtx);
    370         }
    371     });
    372     wil::unique_hmodule exeModule;
    373     RETURN_IF_WIN32_BOOL_FALSE(GetModuleHandleExW(0, nullptr, &exeModule));
    374     ACTCTXW acw = { sizeof(acw) };
    375     acw.lpSource = manifestPath.c_str();
    376     acw.hModule = exeModule.get();
    377     acw.lpResourceName = MAKEINTRESOURCEW(1);
    378     acw.dwFlags = ACTCTX_FLAG_HMODULE_VALID | ACTCTX_FLAG_RESOURCE_NAME_VALID;
    379 
    380     hActCtx = CreateActCtxW(&acw);
    381     RETURN_LAST_ERROR_IF(!hActCtx);
    382     if (hActCtx == INVALID_HANDLE_VALUE)
    383     {
    384         SetLastError(ERROR_OUTOFMEMORY);
    385         return HRESULT_FROM_WIN32(GetLastError());
    386     }
    387 
    388     PACTIVATION_CONTEXT_DETAILED_INFORMATION actCtxInfo = nullptr;
    389     SIZE_T bufferSize = 0;
    390     (void)QueryActCtxW(0,
    391         hActCtx,
    392         nullptr,
    393         ActivationContextDetailedInformation,
    394         nullptr,
    395         0,
    396         &bufferSize);
    397     RETURN_HR_IF(HRESULT_FROM_WIN32(GetLastError()), bufferSize == 0);
    398     auto actCtxInfoBuffer = std::make_unique<BYTE[]>(bufferSize);
    399     actCtxInfo = reinterpret_cast<PACTIVATION_CONTEXT_DETAILED_INFORMATION>(actCtxInfoBuffer.get());
    400     if (!actCtxInfo)
    401     {
    402         SetLastError(ERROR_OUTOFMEMORY);
    403         return HRESULT_FROM_WIN32(GetLastError());
    404     }
    405 
    406     RETURN_IF_WIN32_BOOL_FALSE(QueryActCtxW(0,
    407         hActCtx,
    408         nullptr,
    409         ActivationContextDetailedInformation,
    410         actCtxInfo,
    411         bufferSize,
    412         nullptr));
    413 
    414     for (DWORD index = 1; index <= actCtxInfo->ulAssemblyCount; index++)
    415     {
    416         bufferSize = 0;
    417         PACTIVATION_CONTEXT_ASSEMBLY_DETAILED_INFORMATION asmInfo = nullptr;
    418         (void)QueryActCtxW(0,
    419             hActCtx,
    420             &index,
    421             AssemblyDetailedInformationInActivationContext,
    422             nullptr,
    423             0,
    424             &bufferSize);
    425         RETURN_HR_IF(HRESULT_FROM_WIN32(GetLastError()), bufferSize == 0);
    426         auto asmInfobuffer = std::make_unique<BYTE[]>(bufferSize);
    427         asmInfo = reinterpret_cast<PACTIVATION_CONTEXT_ASSEMBLY_DETAILED_INFORMATION>(asmInfobuffer.get());
    428         if (!asmInfo)
    429         {
    430             SetLastError(ERROR_OUTOFMEMORY);
    431             return HRESULT_FROM_WIN32(GetLastError());
    432         }
    433 
    434         RETURN_IF_WIN32_BOOL_FALSE(QueryActCtxW(0,
    435             hActCtx,
    436             &index,
    437             AssemblyDetailedInformationInActivationContext,
    438             asmInfo,
    439             bufferSize,
    440             nullptr));
    441         RETURN_IF_FAILED(LoadManifestFromPath(asmInfo->lpAssemblyManifestPath));
    442     }
    443     return S_OK;
    444 }

    446 BOOL WINAPI DllMain(HINSTANCE hmodule, DWORD reason, LPVOID /*lpvReserved*/)
    447 {
    448     if (reason == DLL_PROCESS_ATTACH)
    449     {
    450         DisableThreadLibraryCalls(hmodule);
    451         InstallHooks();
    452         try
    453         {
    454             ExtRoLoadCatalog();
    455         }
    456         catch (...)
    457         {
    458             LOG_CAUGHT_EXCEPTION();
    459             return false;
    460         }
    461     }
    462     if (reason == DLL_PROCESS_DETACH)
    463     {
    464         RemoveHooks();
    465     }
    466     return true;
    467 }
