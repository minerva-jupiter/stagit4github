     36 struct component
     37 {
     38     wstring module_name;

     44     ~component()
     45     {
     46         if (handle)

     49         }

     52     HRESULT LoadModule()
     53     {
     54         if (handle == nullptr)
     55         {
     56             handle = LoadLibraryExW(module_name.c_str(), nullptr, LOAD_WITH_ALTERED_SEARCH_PATH);
     57             if (handle == nullptr)
     58             {
     59                 return HRESULT_FROM_WIN32(GetLastError());
     60             }
     61             this->get_activation_factory = (activation_factory_type)GetProcAddress(handle, "DllGetActivationFactory");
     62             if (this->get_activation_factory == nullptr)
     63             {
     64                 return HRESULT_FROM_WIN32(GetLastError());
     65             }
     66         }
     67         return (handle != nullptr && this->get_activation_factory != nullptr) ? S_OK : E_FAIL;
     68     }
     69 
     70     HRESULT GetActivationFactory(HSTRING className, REFIID  iid, void** factory)
     71     {
     72         RETURN_IF_FAILED(LoadModule());
     73 
     74         IActivationFactory* ifactory = nullptr;
     75         HRESULT hr = this->get_activation_factory(className, &ifactory);
     76         // optimize for IActivationFactory?
     77         if (SUCCEEDED(hr))
     78         {
     79             hr = ifactory->QueryInterface(iid, factory);
     80             ifactory->Release();
     81         }
     82         return hr;
     83     }

     94 HRESULT LoadManifestFromPath(std::wstring path)
     95 {
     96     if (path.size() < 4)
     97     {
     98         return COR_E_ARGUMENT;
     99     }
    100     std::wstring ext(path.substr(path.size() - 4, path.size()));
    101     wstr_tolower(ext);
    102     if (ext.compare(L".exe") == 0 || ext.compare(L".dll") == 0)
    103     {
    104         return LoadFromEmbeddedManifest(path.c_str());
    105     }
    106     else
    107     {
    108         return LoadFromSxSManifest(path.c_str());
    109     }

    117 HRESULT LoadFromEmbeddedManifest(PCWSTR path)
    118 {
    119     wil::unique_hmodule handle(LoadLibraryExW(path, nullptr, LOAD_LIBRARY_AS_DATAFILE_EXCLUSIVE));
    120     RETURN_HR_IF(HRESULT_FROM_WIN32(ERROR_FILE_NOT_FOUND), !handle);
    121 
    122     // Try both just to be on the safe side
    123     HRSRC hrsc = FindResourceW(handle.get(), MAKEINTRESOURCEW(1), RT_MANIFEST);
    124     if (!hrsc)
    125     {
    126         hrsc = FindResourceW(handle.get(), MAKEINTRESOURCEW(2), RT_MANIFEST);
    127         RETURN_HR_IF(HRESULT_FROM_WIN32(ERROR_FILE_NOT_FOUND), !hrsc);
    128     }
    129     HGLOBAL embeddedManifest = LoadResource(handle.get(), hrsc);
    130     RETURN_HR_IF(HRESULT_FROM_WIN32(ERROR_FILE_NOT_FOUND), !embeddedManifest);
    131 
    132     DWORD length = SizeofResource(handle.get(), hrsc);
    133     void* data = LockResource(embeddedManifest);
    134     RETURN_HR_IF(HRESULT_FROM_WIN32(ERROR_FILE_NOT_FOUND), !data);
    135 
    136     return WinRTLoadComponentFromString(std::string_view((char*)data, length));
    137 }

    172 HRESULT ParseRootManifestFromXmlReaderInput(IUnknown* input)
    173 {
    174     XmlNodeType nodeType;
    175     PCWSTR localName = nullptr;
    176     auto locale = _create_locale(LC_ALL, "C");
    177     ComPtr<IXmlReader> xmlReader;
    178     RETURN_IF_FAILED(CreateXmlReader(__uuidof(IXmlReader), (void**)&xmlReader, nullptr));
    179     RETURN_IF_FAILED(xmlReader->SetInput(input));
    180     while (S_OK == xmlReader->Read(&nodeType))
    181     {
    182         if (nodeType == XmlNodeType_Element)
    183         {
    184             RETURN_IF_FAILED((xmlReader->GetLocalName(&localName, nullptr)));
    185 
    186             if (_wcsicmp_l(localName, L"file", locale) == 0)
    187             {
    188                 RETURN_IF_FAILED(ParseFileTag(xmlReader));
    189             }
    190         }
    191     }
    192 
    193     return S_OK;
    194 }

    196 HRESULT ParseFileTag(ComPtr<IXmlReader> xmlReader)
    197 {
    198     HRESULT hr = S_OK;
    199     XmlNodeType nodeType;
    200     PCWSTR localName = nullptr;
    201     wstring fileName;
    202 
    203     {
    204         PCWSTR fileNameStr = nullptr;
    205         hr = xmlReader->MoveToAttributeByName(L"name", nullptr);
    206         RETURN_HR_IF(HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR), hr != S_OK);
    207         RETURN_IF_FAILED(xmlReader->GetValue(&fileNameStr, nullptr));
    208         RETURN_HR_IF(HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR), fileNameStr == nullptr || !fileNameStr[0]);
    209         
    210         fileName = fileNameStr;
    211     }
    212     
    213     auto locale = _create_locale(LC_ALL, "C");
    214     while (S_OK == xmlReader->Read(&nodeType))
    215     {
    216         if (nodeType == XmlNodeType_Element)
    217         {
    218             RETURN_IF_FAILED(xmlReader->GetLocalName(&localName, nullptr));
    219             if (localName != nullptr && _wcsicmp_l(localName, L"activatableClass", locale) == 0)
    220             {
    221                 RETURN_IF_FAILED(ParseActivatableClassTag(xmlReader, fileName.c_str()));
    222             }
    223         }
    224         else if (nodeType == XmlNodeType_EndElement)
    225         {
    226             RETURN_IF_FAILED(xmlReader->GetLocalName(&localName, nullptr));
    227             RETURN_HR_IF(S_OK, localName != nullptr && _wcsicmp_l(localName, L"file", locale) == 0);
    228         }
    229     }
    230     return S_OK;
    231 }

    233 HRESULT ParseActivatableClassTag(ComPtr<IXmlReader> xmlReader, PCWSTR fileName)
    234 {
    235     PCWSTR localName = nullptr;
    236     auto locale = _create_locale(LC_ALL, "C");
    237     auto this_component = make_shared<component>();
    238     this_component->module_name = fileName;
    239     HRESULT hr = xmlReader->MoveToFirstAttribute();
    240     // Using this pattern intead of calling multiple MoveToAttributeByName improves performance
    241     const WCHAR* activatableClass = nullptr;
    242     const WCHAR* threadingModel = nullptr;
    243     const WCHAR* xmlns = nullptr;
    244     if (S_FALSE == hr)
    245     {
    246         return HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR);
    247     }
    248     else
    249     {
    250         while (TRUE)
    251         {
    252             const WCHAR* pwszLocalName;
    253             const WCHAR* pwszValue;
    254             if (FAILED_LOG(xmlReader->GetLocalName(&pwszLocalName, NULL)))
    255             {
    256                 return HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR);
    257             }
    258             if (FAILED_LOG(xmlReader->GetValue(&pwszValue, NULL)))
    259             {
    260                 return HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR);
    261             }
    262             if (pwszLocalName != nullptr)
    263             {
    264                 if (_wcsicmp_l(L"threadingModel", pwszLocalName, locale) == 0)
    265                 {
    266                     threadingModel = pwszValue;
    267                 }
    268                 else if (_wcsicmp_l(L"name", pwszLocalName, locale) == 0)
    269                 {
    270                     activatableClass = pwszValue;
    271                 }
    272                 else if (_wcsicmp_l(L"xmlns", pwszLocalName, locale) == 0)
    273                 {
    274                     xmlns = pwszValue;
    275                 }
    276             }
    277             if (xmlReader->MoveToNextAttribute() != S_OK)
    278             {
    279                 break;
    280             }
    281         }
    282     }
    283     if (threadingModel == nullptr)
    284     {
    285         return HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR);
    286     }
    287     if (_wcsicmp_l(L"sta", threadingModel, locale) == 0)
    288     {
    289         this_component->threading_model = ABI::Windows::Foundation::ThreadingType::ThreadingType_STA;
    290     }
    291     else if (_wcsicmp_l(L"mta", threadingModel, locale) == 0)
    292     {
    293         this_component->threading_model = ABI::Windows::Foundation::ThreadingType::ThreadingType_MTA;
    294     }
    295     else if (_wcsicmp_l(L"both", threadingModel, locale) == 0)
    296     {
    297         this_component->threading_model = ABI::Windows::Foundation::ThreadingType::ThreadingType_BOTH;
    298     }
    299     else
    300     {
    301         return HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR);
    302     }
    303 
    304     if (activatableClass == nullptr || !activatableClass[0])
    305     {
    306         return HRESULT_FROM_WIN32(ERROR_SXS_MANIFEST_PARSE_ERROR);
    307     }
    308     this_component->xmlns = xmlns; // Should we care if this value is blank or missing?
    309     // Check for duplicate activatable classes
    310     auto component_iter = g_types.find(activatableClass);
    311     if (component_iter != g_types.end()) 
    312     {
    313         return HRESULT_FROM_WIN32(ERROR_SXS_DUPLICATE_ACTIVATABLE_CLASS);
    314     }
    315     g_types[activatableClass] = this_component;
    316     return S_OK;
    317 }

    319 HRESULT WinRTGetThreadingModel(HSTRING activatableClassId, ABI::Windows::Foundation::ThreadingType* threading_model)
    320 {
    321     auto raw_class_name = WindowsGetStringRawBuffer(activatableClassId, nullptr);
    322     auto component_iter = g_types.find(raw_class_name);
    323     if (component_iter != g_types.end())
    324     {
    325         *threading_model = component_iter->second->threading_model;

    327     }
    328     return REGDB_E_CLASSNOTREG;

    345 HRESULT WinRTGetMetadataFile(
    346     const HSTRING        name,
    347     IMetaDataDispenserEx* metaDataDispenser,
    348     HSTRING* metaDataFilePath,
    349     IMetaDataImport2** metaDataImport,
    350     mdTypeDef* typeDefToken)
    351 {
    352     wchar_t folderPrefix[9];
    353     PCWSTR pszFullName = WindowsGetStringRawBuffer(name, nullptr);
    354     HRESULT hr = StringCchCopyW(folderPrefix, _countof(folderPrefix), pszFullName);
    355     if (hr != S_OK && hr != STRSAFE_E_INSUFFICIENT_BUFFER)
    356     {
    357         return hr;
    358     }
    359     if (CompareStringOrdinal(folderPrefix, -1, L"Windows.", -1, false) == CSTR_EQUAL)
    360     {
    361         return RO_E_METADATA_NAME_NOT_FOUND;
    362     }
    363 
    364     if (metaDataFilePath != nullptr)
    365     {
    366         *metaDataFilePath = nullptr;
    367     }
    368     if (metaDataImport != nullptr)
    369     {
    370         *metaDataImport = nullptr;
    371     }
    372     if (typeDefToken != nullptr)
    373     {
    374         *typeDefToken = mdTypeDefNil;
    375     }
    376 
    377     if (((metaDataImport == nullptr) && (typeDefToken != nullptr)) ||
    378         ((metaDataImport != nullptr) && (typeDefToken == nullptr)))
    379     {
    380         return E_INVALIDARG;
    381     }
    382 
    383     ComPtr<IMetaDataDispenserEx> spMetaDataDispenser;
    384     // The API uses the caller's passed-in metadata dispenser. If null, it
    385     // will create an instance of the metadata reader to dispense metadata files.
    386     if (metaDataDispenser == nullptr)
    387     {
    388         RETURN_IF_FAILED(MetaDataGetDispenser(CLSID_CorMetaDataDispenser,
    389             IID_IMetaDataDispenser,
    390             &spMetaDataDispenser));
    391         {
    392             variant_t version{ L"WindowsRuntime 1.4" };
    393             RETURN_IF_FAILED(spMetaDataDispenser->SetOption(MetaDataRuntimeVersion, &version.GetVARIANT()));
    394         }
    395     }
    396     return UndockedRegFreeWinRT::ResolveThirdPartyType(
    397         spMetaDataDispenser.Get(),
    398         pszFullName,
    399         metaDataFilePath,
    400         metaDataImport,
    401         typeDefToken);
