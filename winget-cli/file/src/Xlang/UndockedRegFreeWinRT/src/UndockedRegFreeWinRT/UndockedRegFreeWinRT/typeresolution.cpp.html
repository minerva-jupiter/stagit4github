     25             SetLastError(hr);

    106     HRESULT FindTypeInMetaDataFile(
    107         _In_ IMetaDataDispenserEx* pMetaDataDispenser,
    108         _In_ PCWSTR pszFullName,
    109         _In_ PCWSTR pszCandidateFilePath,
    110         _In_ TYPE_RESOLUTION_OPTIONS resolutionOptions,
    111         _COM_Outptr_opt_result_maybenull_ IMetaDataImport2** ppMetaDataImport,
    112         _Out_opt_ mdTypeDef* pmdTypeDef)
    113     {
    114         HRESULT hr = S_OK;
    115         Microsoft::WRL::ComPtr<IMetaDataImport2> spMetaDataImport;
    116         MetaDataImportersLRUCache* pMetaDataImporterCache = MetaDataImportersLRUCache::GetMetaDataImportersLRUCacheInstance();
    117         if (pMetaDataImporterCache != nullptr)
    118         {
    119             hr = pMetaDataImporterCache->GetMetaDataImporter(
    120                 pMetaDataDispenser,
    121                 pszCandidateFilePath,
    122                 &spMetaDataImport);
    123         }
    124         else
    125         {
    126             hr = E_OUTOFMEMORY;
    127         }
    128 
    129         if (SUCCEEDED(hr))
    130         {
    131             const size_t cFullName = wcslen(pszFullName);
    132             wchar_t pszRetrievedName[g_uiMaxTypeName];
    133             HCORENUM hEnum = nullptr;
    134             mdTypeDef rgTypeDefs[32];
    135             ULONG cTypeDefs;
    136             DWORD dwTypeDefProps;
    137             hr = RO_E_METADATA_NAME_NOT_FOUND;
    138 
    139             if (TRO_RESOLVE_TYPE & resolutionOptions)
    140             {
    141                 hr = spMetaDataImport->FindTypeDefByName(pszFullName, mdTokenNil, &rgTypeDefs[0]);
    142                 if (SUCCEEDED(hr))
    143                 {
    144                     //  Check to confirm that the type we just found is a
    145                     //  winrt type.  If it is, we're good, otherwise we
    146                     //  want to fail with RO_E_INVALID_METADATA_FILE.
    147                     hr = spMetaDataImport->GetTypeDefProps(rgTypeDefs[0], nullptr, 0, nullptr, &dwTypeDefProps, nullptr);
    148                     if (SUCCEEDED(hr))
    149                     {
    150                         //  If we found the type but it's not a winrt type,
    151                         //  it's an error.
    152                         //
    153                         //  If the type is public, than the metadata file
    154                         //  is corrupt (all public types in a winrt file
    155                         //  must be tdWindowsRuntime).  If the type is
    156                         //  private, then we just want to report that the
    157                         //  type wasn't found.
    158                         if (!IsTdWindowsRuntime(dwTypeDefProps))
    159                         {
    160                             if (IsTdPublic(dwTypeDefProps))
    161                             {
    162                                 hr = RO_E_INVALID_METADATA_FILE;
    163                             }
    164                             else
    165                             {
    166                                 hr = RO_E_METADATA_NAME_NOT_FOUND;
    167                             }
    168                         }
    169                     }
    170                     else
    171                     {
    172                         hr = RO_E_INVALID_METADATA_FILE;
    173                     }
    174                     if (SUCCEEDED(hr))
    175                     {
    176                         if (pmdTypeDef != nullptr)
    177                         {
    178                             *pmdTypeDef = rgTypeDefs[0];
    179                         }
    180                         if (ppMetaDataImport != nullptr)
    181                         {
    182                             *ppMetaDataImport = spMetaDataImport.Detach();
    183                         }
    184                     }
    185                 }
    186                 else if (hr == CLDB_E_RECORD_NOTFOUND)
    187                 {
    188                     hr = RO_E_METADATA_NAME_NOT_FOUND;
    189                 }
    190             }
    191 
    192             if ((hr == RO_E_METADATA_NAME_NOT_FOUND) &&
    193                 (TRO_RESOLVE_NAMESPACE & resolutionOptions))
    194             {
    195                 // Check whether the name is a namespace rather than a type.
    196                 do
    197                 {
    198                     hr = spMetaDataImport->EnumTypeDefs(
    199                         &hEnum,
    200                         rgTypeDefs,
    201                         ARRAYSIZE(rgTypeDefs),
    202                         &cTypeDefs);
    203 
    204                     if (hr == S_OK)
    205                     {
    206                         for (UINT32 iTokenIndex = 0; iTokenIndex < cTypeDefs; ++iTokenIndex)
    207                         {
    208                             hr = spMetaDataImport->GetTypeDefProps(
    209                                 rgTypeDefs[iTokenIndex],
    210                                 pszRetrievedName,
    211                                 ARRAYSIZE(pszRetrievedName),
    212                                 nullptr,
    213                                 &dwTypeDefProps,
    214                                 nullptr);
    215 
    216                             if (FAILED(hr))
    217                             {
    218                                 break;
    219                             }
    220 
    221                             hr = RO_E_METADATA_NAME_NOT_FOUND;
    222 
    223                             // Only consider windows runtime types when
    224                             // trying to determine if the name is a
    225                             // namespace.
    226                             if (IsTdWindowsRuntime(dwTypeDefProps) && (wcslen(pszRetrievedName) > cFullName))
    227                             {
    228                                 if ((wcsncmp(pszRetrievedName, pszFullName, cFullName) == 0) &&
    229                                     (pszRetrievedName[cFullName] == L'.'))
    230                                 {
    231                                     hr = RO_E_METADATA_NAME_IS_NAMESPACE;
    232                                     break;
    233                                 }
    234                             }
    235                         }
    236                     }
    237                 } while (hr == RO_E_METADATA_NAME_NOT_FOUND);
    238 
    239                 // There were no more tokens to enumerate, but the type was still not found.
    240                 if (hr == S_FALSE)
    241                 {
    242                     hr = RO_E_METADATA_NAME_NOT_FOUND;
    243                 }
    244 
    245                 if (hEnum != nullptr)
    246                 {
    247                     spMetaDataImport->CloseEnum(hEnum);
    248                     hEnum = nullptr;
    249                 }
    250             }
    251         }
    252         return hr;
    253     }

    255     HRESULT FindTypeInDirectory(
    256         _In_ IMetaDataDispenserEx* pMetaDataDispenser,
    257         _In_ PCWSTR pszFullName,
    258         _In_ PCWSTR pszDirectoryPath,
    259         _Out_opt_ HSTRING* phstrMetaDataFilePath,
    260         _COM_Outptr_opt_result_maybenull_ IMetaDataImport2** ppMetaDataImport,
    261         _Out_opt_ mdTypeDef* pmdTypeDef)
    262     {
    263         HRESULT hr;
    264 
    265         wchar_t szCandidateFilePath[MAX_PATH + 1] = { 0 };
    266         wchar_t szCandidateFileName[MAX_PATH + 1] = { 0 };
    267         PWSTR pszLastDot;
    268 
    269         hr = StringCchCopy(szCandidateFileName, ARRAYSIZE(szCandidateFileName), pszFullName);
    270 
    271         if (SUCCEEDED(hr))
    272         {
    273             // To resolve type SomeNamespace.B.C, first check if SomeNamespace.B.C is a type or
    274             // a namespace in the metadata files in the directory in this order:
    275             // 1. SomeNamespace.B.C.WinMD
    276             // 2. SomeNamespace.B.WinMD
    277             // 3. SomeNamespace.WinMD
    278             do
    279             {
    280                 pszLastDot = nullptr;
    281 
    282                 hr = StringCchPrintfEx(
    283                     szCandidateFilePath,
    284                     ARRAYSIZE(szCandidateFilePath),
    285                     nullptr,
    286                     nullptr,
    287                     0,
    288                     METADATA_FILE_PATH_FORMAT,
    289                     pszDirectoryPath,
    290                     szCandidateFileName);
    291 
    292                 if (SUCCEEDED(hr))
    293                 {
    294                     hr = FindTypeInMetaDataFile(
    295                         pMetaDataDispenser,
    296                         pszFullName,
    297                         szCandidateFilePath,
    298                         TRO_RESOLVE_TYPE_AND_NAMESPACE,
    299                         ppMetaDataImport,
    300                         pmdTypeDef);
    301 
    302                     if (SUCCEEDED(hr))
    303                     {
    304                         if (phstrMetaDataFilePath != nullptr)
    305                         {
    306                             hr = WindowsCreateString(
    307                                 szCandidateFilePath,
    308                                 static_cast<UINT32>(wcslen(szCandidateFilePath)),
    309                                 phstrMetaDataFilePath);
    310                         }
    311                         break;
    312                     }
    313                 }
    314 
    315                 hr = RO_E_METADATA_NAME_NOT_FOUND;
    316                 pszLastDot = wcsrchr(szCandidateFileName, '.');
    317 
    318                 if (pszLastDot != nullptr)
    319                 {
    320                     *pszLastDot = '\0';
    321                 }
    322             } while (pszLastDot != nullptr);
    323 
    324             // If name was not found when searching in the "upward direction", then
    325             // the name might be a namespace name in a down-level file.
    326             if (hr == RO_E_METADATA_NAME_NOT_FOUND)
    327             {
    328                 wchar_t szFilePathSearchTemplate[MAX_PATH + 1] = { 0 };
    329 
    330                 hr = StringCchPrintfEx(
    331                     szFilePathSearchTemplate,
    332                     ARRAYSIZE(szFilePathSearchTemplate),
    333                     nullptr,
    334                     nullptr,
    335                     0,
    336                     METADATA_FILE_SEARCH_FORMAT,
    337                     pszDirectoryPath,
    338                     pszFullName);
    339 
    340                 if (SUCCEEDED(hr))
    341                 {
    342                     WIN32_FIND_DATA fd;
    343                     HANDLE hFindFile;
    344 
    345                     // Search in all files in the directory whose name begin with the input string.
    346                     hFindFile = FindFirstFile(szFilePathSearchTemplate, &fd);
    347 
    348                     if (hFindFile != INVALID_HANDLE_VALUE)
    349                     {
    350                         PWSTR pszFilePathPart;
    351                         size_t cchRemaining;
    352 
    353                         do
    354                         {
    355                             if (fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY)
    356                             {
    357                                 continue;
    358                             }
    359 
    360                             pszFilePathPart = szCandidateFilePath;
    361                             cchRemaining = ARRAYSIZE(szCandidateFilePath);
    362                             hr = StringCchCopyEx(
    363                                 pszFilePathPart,
    364                                 cchRemaining,
    365                                 pszDirectoryPath,
    366                                 &pszFilePathPart,
    367                                 &cchRemaining,
    368                                 0);
    369 
    370                             if (SUCCEEDED(hr))
    371                             {
    372                                 hr = StringCchCopyEx(
    373                                     pszFilePathPart,
    374                                     cchRemaining,
    375                                     fd.cFileName,
    376                                     &pszFilePathPart,
    377                                     &cchRemaining,
    378                                     0);
    379                             }
    380 
    381                             if (SUCCEEDED(hr))
    382                             {
    383                                 hr = FindTypeInMetaDataFile(
    384                                     pMetaDataDispenser,
    385                                     pszFullName,
    386                                     szCandidateFilePath,
    387                                     TRO_RESOLVE_NAMESPACE,
    388                                     ppMetaDataImport,
    389                                     pmdTypeDef);
    390 
    391                                 if (hr == S_OK)
    392                                 {
    393                                     hr = E_UNEXPECTED;
    394                                     break;
    395                                 }
    396 
    397                                 if (hr == RO_E_METADATA_NAME_IS_NAMESPACE)
    398                                 {
    399                                     break;
    400                                 }
    401                             }
    402                         } while (FindNextFile(hFindFile, &fd));
    403 
    404                         FindClose(hFindFile);
    405                     }
    406                     else
    407                     {
    408                         hr = RO_E_METADATA_NAME_NOT_FOUND;
    409                     }
    410                 }
    411             }
    412         }
    413 
    414         if (hr == STRSAFE_E_INSUFFICIENT_BUFFER)
    415         {
    416             hr = RO_E_METADATA_NAME_NOT_FOUND;
    417         }
    418 
    419         return hr;
    420     }

    423         _In_ IMetaDataDispenserEx* pMetaDataDispenser,
    424         _In_ PCWSTR pszFullName,
    425         _In_ PCWSTR pszDirectoryPath,

    432         size_t cchPackagePathRemaining = ARRAYSIZE(pszPackagePath);
    433 
    434         HRESULT hr = StringCchCopyEx(
    435             pszPackagePath,
    436             ARRAYSIZE(pszPackagePath),

    441 
    442         if (SUCCEEDED(hr))

    446             {

    448                     pszPackagePathWritePtr,
    449                     cchPackagePathRemaining,
    450                     L"\\",
    451                     &pszPackagePathWritePtr,
    452                     &cchPackagePathRemaining,
    453                     0);
    454             }
    455         }
    456 
    457         if (SUCCEEDED(hr))
    458         {
    459             hr = FindTypeInDirectory(
    460                 pMetaDataDispenser,
    461                 pszFullName,
    462                 pszPackagePath,
    463                 phstrMetaDataFilePath,
    464                 ppMetaDataImport,
    465                 pmdTypeDef);

    471     HRESULT ResolveThirdPartyType(
    472         _In_ IMetaDataDispenserEx* pMetaDataDispenser,
    473         _In_ PCWSTR pszFullName,
    474         _Out_opt_ HSTRING* phstrMetaDataFilePath,
    475         _COM_Outptr_opt_result_maybenull_ IMetaDataImport2** ppMetaDataImport,
    476         _Out_opt_ mdTypeDef* pmdTypeDef)
    477     {
    478         HRESULT hr = S_OK;
    479         UINT32 dwPackagesCount = 0;
    480         UINT32 dwBufferLength = 0;
    481 
    482         const UINT32 filter = PACKAGE_FILTER_HEAD | PACKAGE_FILTER_DIRECT | PACKAGE_FILTER_IS_IN_RELATED_SET;
    483         hr = HRESULT_FROM_WIN32(GetCurrentPackageInfo(filter, &dwBufferLength, nullptr, &dwPackagesCount));
    484         // Only find the type next to the exe if the it is an unpackaged app. Packaged apps can have their exe on 
    485         // their package graph, which will allow type resolution against adjacent WinMDs.
    486         if (hr == HRESULT_FROM_WIN32(APPMODEL_ERROR_NO_PACKAGE))
    487         {
    488             PCWSTR exeDir = nullptr;  // Never freed; owned by process global.
    489             RETURN_IF_FAILED(GetProcessExeDir(&exeDir));
    490 
    491             hr = FindTypeInDirectoryWithNormalization(
    492                 pMetaDataDispenser,
    493                 pszFullName,
    494                 exeDir,
    495                 phstrMetaDataFilePath,
    496                 ppMetaDataImport,
    497                 pmdTypeDef);
    498 
    499             if (hr == RO_E_METADATA_NAME_NOT_FOUND)
    500             {
    501                 // For compatibility purposes, if we fail to find the type in the unpackaged location, we should return
    502                 // HRESULT_FROM_WIN32(APPMODEL_ERROR_NO_PACKAGE) instead of a "not found" error. This preserves the
    503                 // behavior that existed before unpackaged type resolution was implemented.
    504                 hr = HRESULT_FROM_WIN32(APPMODEL_ERROR_NO_PACKAGE);
    505             }
    506         }
    507         else
    508         {
    509             hr = RO_E_METADATA_NAME_NOT_FOUND;
    510         }
    511 
    512         // If not successful looking next to an unpackaged exe, try looking next to our DLL.
    513         if (FAILED(hr))
    514         {
    515             PCWSTR dllDir = nullptr;  // Never freed; owned by process global.
    516             RETURN_IF_FAILED(GetProcessDllDir(&dllDir));
    517 
    518             HRESULT dllHR = FindTypeInDirectoryWithNormalization(
    519                 pMetaDataDispenser,
    520                 pszFullName,
    521                 dllDir,
    522                 phstrMetaDataFilePath,
    523                 ppMetaDataImport,
    524                 pmdTypeDef);
    525 
    526             if (SUCCEEDED(dllHR))
    527             {
    528                 hr = dllHR;
    529             }
    530         }
    531 
    532         return hr;
    533     }

    656         if (SUCCEEDED(hr))
    657         {
    658             // Make room for new element.
    659             for (int i = g_dwMetaDataImportersLRUCacheSize - 2; i >= 0; i--)
    660             {
    661                 _arFilePaths[i + 1] = _arFilePaths[i];
    662             }
    663 
    664             _arFilePaths[0] = AllocateAndCopyString(pszFilePath);
    665 
    666             if (_arFilePaths[0] == nullptr)
    667             {
    668                 hr = E_OUTOFMEMORY;
    669             }
    670         }
    671 
    672         return hr;
    673     }

    675     bool MetaDataImportersLRUCache::IsFilePathCached(PCWSTR pszFilePath)
    676     {
    677         int filePathIndex = GetFilePathIndex(pszFilePath);
    678 
    679         if (filePathIndex != -1)
    680         {
    681             MoveElementToFrontOfList(filePathIndex);
    682             return true;
    683         }
    684         else
    685         {
    686             return false;
    687         }
    688     }
    689 
    690     int MetaDataImportersLRUCache::GetFilePathIndex(PCWSTR pszFilePath)
    691     {
    692         int filePathIndex = -1;
    693 
    694         for (int i = 0; (i < g_dwMetaDataImportersLRUCacheSize) && (_arFilePaths[i] != nullptr); i++)
    695         {
    696             if (wcscmp(pszFilePath, _arFilePaths[i]) == 0)
    697             {
    698                 filePathIndex = i;
    699                 break;
    700             }
    701         }
    702 
    703         return filePathIndex;
    704     }
    705 
    706     void MetaDataImportersLRUCache::MoveElementToFrontOfList(int elementIndex)
    707     {
    708         PWSTR pszFoundFilePath = _arFilePaths[elementIndex];
    709 
    710         for (int i = elementIndex - 1; i >= 0; i--)
    711         {
    712             _arFilePaths[i + 1] = _arFilePaths[i];
    713         }
    714 
    715         _arFilePaths[0] = pszFoundFilePath;
    716     }

    718     HRESULT MetaDataImportersLRUCache::RemoveLeastRecentlyUsedItemIfListIsFull()
    719     {
    720         HRESULT hr = S_OK;
    721         PWSTR pszLastFilePathInList = _arFilePaths[g_dwMetaDataImportersLRUCacheSize - 1];
    722 
    723         if (pszLastFilePathInList != nullptr)
    724         {
    725             IMetaDataImport2* value = _metadataImportersMap[pszLastFilePathInList];
    726             if (value != nullptr)
    727             {
    728                 value->Release();
    729                 value = nullptr;
    730             }
    731             if (!_metadataImportersMap.erase(pszLastFilePathInList))
    732             {
    733                 hr = E_UNEXPECTED;
    734             }
    735 
    736             delete[] pszLastFilePathInList;
    737             _arFilePaths[g_dwMetaDataImportersLRUCacheSize - 1] = nullptr;
    738         }
    739 
    740         return hr;
    741     }
