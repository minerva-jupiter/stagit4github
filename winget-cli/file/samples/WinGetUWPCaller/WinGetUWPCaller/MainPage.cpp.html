    147         std::wstring MakeCompactByteString(uint64_t bytes)
    148         {
    149             static constexpr std::array<std::wstring_view, 4> s_sizeStrings = { L"B"sv, L"KB"sv, L"MB"sv, L"GB"sv };
    150             static constexpr size_t s_sizeIncrement = 1000;
    151 
    152             size_t sizeIndex = 0;
    153             while (sizeIndex < s_sizeStrings.size() && bytes > s_sizeIncrement)
    154             {
    155                 sizeIndex += 1;
    156                 bytes /= s_sizeIncrement;
    157             }
    158 
    159             return std::to_wstring(bytes).append(L" ").append(s_sizeStrings[sizeIndex]);

    292     IAsyncAction MainPage::FindPackageAsync()
    293     {
    294         hstring queryInput = queryTextBox().Text();
    295         auto selectedItems = catalogsListBox().SelectedItems();
    296         int32_t searchType = searchField().SelectedIndex();
    297         PackageCatalog catalog = m_catalog;
    298 
    299         BackgroundActionData actionData{ { searchButton() }, operationStatusText() };
    300 
    301         co_await winrt::resume_background();
    302 
    303         actionData.Status(EnsurePackageManager());
    304 
    305         if (!catalog)
    306         {
    307             actionData.RunAndCatchStatus(L"Connect catalog(s)", [&]() {
    308                 PackageCatalogReference catalogReference{ nullptr };
    309 
    310                 if (selectedItems.Size() == 0)
    311                 {
    312                     // If no items are selected, we use all available catalogs.
    313                     CreateCompositePackageCatalogOptions createCompositePackageCatalogOptions;
    314                     createCompositePackageCatalogOptions.CompositeSearchBehavior(CompositeSearchBehavior::RemotePackagesFromRemoteCatalogs);
    315 
    316                     for (const auto& item : m_packageManager.GetPackageCatalogs())
    317                     {
    318                         createCompositePackageCatalogOptions.Catalogs().Append(item);
    319                     }
    320 
    321                     catalogReference = m_packageManager.CreateCompositePackageCatalog(createCompositePackageCatalogOptions);
    322                 }
    323                 else if (selectedItems.Size() == 1)
    324                 {
    325                     // If one items is selected, we can directly use this catalog.
    326                     catalogReference = selectedItems.GetAt(0).as<PackageCatalogReference>();
    327                 }
    328                 else
    329                 {
    330                     // If multiple items are selected, we create a composite catalog using those catalogs.
    331                     CreateCompositePackageCatalogOptions createCompositePackageCatalogOptions;
    332                     createCompositePackageCatalogOptions.CompositeSearchBehavior(CompositeSearchBehavior::RemotePackagesFromRemoteCatalogs);
    333 
    334                     for (const auto& item : selectedItems)
    335                     {
    336                         createCompositePackageCatalogOptions.Catalogs().Append(item.as<PackageCatalogReference>());
    337                     }
    338 
    339                     catalogReference = m_packageManager.CreateCompositePackageCatalog(createCompositePackageCatalogOptions);
    340                 }
    341 
    342                 ConnectResult connectResult{ catalogReference.Connect() };
    343 
    344                 switch (connectResult.Status())
    345                 {
    346                 case ConnectResultStatus::Ok: break;
    347                 case ConnectResultStatus::CatalogError: throw std::exception{ "Catalog connection error." };
    348                 case ConnectResultStatus::SourceAgreementsNotAccepted: throw std::exception{ "Required catalog agreements not accepted." };
    349                 }
    350 
    351                 catalog = connectResult.PackageCatalog();
    352             });
    353         }
    354 
    355         CatalogPackage package{ nullptr };
    356 
    357         actionData.RunAndCatchStatus(L"Find package", [&]() {
    358             FindPackagesOptions findPackagesOptions;
    359             PackageMatchFilter filter;
    360 
    361             switch (searchType)
    362             {
    363             case 0: // Generic query
    364                 filter.Field(PackageMatchField::CatalogDefault);
    365                 filter.Option(PackageFieldMatchOption::ContainsCaseInsensitive);
    366                 break;
    367             case 1: // Identifier (case-insensitive)
    368                 filter.Field(PackageMatchField::Id);
    369                 filter.Option(PackageFieldMatchOption::EqualsCaseInsensitive);
    370                 break;
    371             case 2: // Name (substring)
    372                 filter.Field(PackageMatchField::Name);
    373                 filter.Option(PackageFieldMatchOption::ContainsCaseInsensitive);
    374                 break;
    375             }
    376 
    377             filter.Value(queryInput);
    378             findPackagesOptions.Selectors().Append(filter);
    379             FindPackagesResult findPackagesResult = catalog.FindPackages(findPackagesOptions);
    380 
    381             winrt::IVectorView<MatchResult> matches = findPackagesResult.Matches();
    382             if (matches.Size() == 0)
    383             {
    384                 throw std::exception{ "No package found matching input" };
    385             }
    386             else if (matches.Size() > 1)
    387             {
    388                 throw std::exception{ "Multiple packages found matching input; refine query." };
    389             }
    390             else
    391             {
    392                 package = matches.GetAt(0).CatalogPackage();
    393             }
    394         });
    395 
    396         if (package)
    397         {
    398             // Display the package name using the user's default localization information.
    399             std::wostringstream stream;
    400             stream << L"Found package: " <<
    401                 static_cast<std::wstring_view>(package.DefaultInstallVersion().GetCatalogPackageMetadata().PackageName()) << L" [" <<
    402                 static_cast<std::wstring_view>(package.Id()) << L"]";
    403             actionData.Status(std::move(stream).str());
    404         }
    405 
    406         co_await winrt::resume_foreground(actionData.Dispatcher());
    407 
    408         m_catalog = catalog;
    409         m_package = package;
    410 
    411         bool operationButtonsEnabled = static_cast<bool>(m_package);
    412         installButton().IsEnabled(operationButtonsEnabled);
    413         upgradeButton().IsEnabled(operationButtonsEnabled);
    414         downloadButton().IsEnabled(operationButtonsEnabled);
    415 
    416         actionData.Finalize();
    417     }

    419     IAsyncAction MainPage::InstallOrUpgradeAsync(bool upgrade)
    420     {
    421         PackageManager packageManager = m_packageManager;
    422         CatalogPackage package = m_package;
    423         auto progressBar = operationProgressBar();
    424         auto statusText = operationStatusText();
    425 
    426         BackgroundActionData actionData{ { installButton(), upgradeButton(), downloadButton() }, { cancelButton() }, statusText };
    427 
    428         co_await winrt::resume_background();
    429 
    430         Windows::Foundation::IAsyncOperationWithProgress<Deployment::InstallResult, Deployment::InstallProgress> packageOperation;
    431 
    432         actionData.RunAndCatchStatus(L"Begin install", [&]() {
    433             InstallOptions installOptions;
    434 
    435             // Passing PackageInstallScope::User causes the install to fail if there's no installer that supports that.
    436             installOptions.PackageInstallScope(PackageInstallScope::Any);
    437             installOptions.PackageInstallMode(PackageInstallMode::Silent);
    438 
    439             if (upgrade)
    440             {
    441                 packageOperation = packageManager.UpgradePackageAsync(package, installOptions);
    442             }
    443             else
    444             {
    445                 packageOperation = packageManager.InstallPackageAsync(package, installOptions);
    446             }
    447         });
    448 
    449         actionData.Dispatcher().RunAsync(Windows::UI::Core::CoreDispatcherPriority::Normal,
    450             [packageOperation, this]() { m_packageOperation = packageOperation; });
    451 
    452         actionData.RunAndCatchStatus(L"Set progress handler", [&]() {
    453             packageOperation.Progress([&](
    454                 IAsyncOperationWithProgress<InstallResult, InstallProgress> const& /* sender */,
    455                 InstallProgress const& progress)
    456                 {
    457                     actionData.Dispatcher().RunAsync(Windows::UI::Core::CoreDispatcherPriority::Normal,
    458                         [progressBar, statusText, progress]() {
    459                             progressBar.Value(progress.DownloadProgress * 100);
    460 
    461                             switch (progress.State)
    462                             {
    463                             case PackageInstallProgressState::Queued:
    464                                 statusText.Text(L"Queued");
    465                                 break;
    466                             case PackageInstallProgressState::Downloading:
    467                             {
    468                                 std::wstring downloadText{ L"Downloaded " };
    469                                 downloadText += MakeCompactByteString(progress.BytesDownloaded) + L" of " + MakeCompactByteString(progress.BytesRequired);
    470                                 statusText.Text(downloadText);
    471                             }
    472                             break;
    473                             case PackageInstallProgressState::Installing:
    474                                 statusText.Text(L"Installer running");
    475                                 progressBar.IsIndeterminate(true);
    476                                 break;
    477                             case PackageInstallProgressState::PostInstall:
    478                                 statusText.Text(L"Post install bookkeeping");
    479                                 break;
    480                             case PackageInstallProgressState::Finished:
    481                                 statusText.Text(L"Done");
    482                                 progressBar.IsIndeterminate(false);
    483                                 break;
    484                             default:
    485                                 statusText.Text(L"");
    486                             }
    487                         });
    488                 });
    489             });
    490 
    491         InstallResult installResult{ nullptr };
    492 
    493         actionData.RunAndCatchStatus(L"Install", [&]() {
    494             installResult = packageOperation.get();
    495         });
    496 
    497         if (packageOperation && packageOperation.Status() == AsyncStatus::Canceled)
    498         {
    499             actionData.Status(L"Cancelled");
    500         }
    501         else if (installResult)
    502         {
    503             // Error handling for the installResult is done by first examining the Status.
    504             // Any status value other than Ok will have additional error detail in the
    505             // ExtendedErrorCode property. This HRESULT value will typically (but not always)
    506             // have the Windows Package Manager facility value (0xA15). The symbolic names and
    507             // meanings of these error codes can be found at:
    508             // https://github.com/microsoft/winget-cli/blob/master/doc/windows/package-manager/winget/returnCodes.md
    509             // or by using the winget CLI:
    510             // > winget error 0x8A150049
    511             // > winget error -- -2146762487
    512             switch (installResult.Status())
    513             {
    514             case InstallResultStatus::Ok:
    515                 actionData.Status(installResult.RebootRequired() ? L"Reboot required" : L"Done");
    516                 break;
    517             case InstallResultStatus::BlockedByPolicy:
    518                 // See installResult.ExtendedErrorCode for more detail.
    519                 // This is typically caused by system configuration applied by policy.
    520                 actionData.Status(L"Blocked by policy");
    521                 break;
    522             case InstallResultStatus::CatalogError:
    523                 // See installResult.ExtendedErrorCode for more detail.
    524                 // This is typically an issue with an external service.
    525                 actionData.Status(L"Catalog error");
    526                 break;
    527             case InstallResultStatus::InternalError:
    528                 // See installResult.ExtendedErrorCode for more detail.
    529                 // This is typically an issue with the Windows Package Manager code.
    530                 actionData.Status(L"Internal error");
    531                 break;
    532             case InstallResultStatus::InvalidOptions:
    533                 // See installResult.ExtendedErrorCode for more detail.
    534                 // This is caused by invalid input combinations.
    535                 actionData.Status(L"Invalid options");
    536                 break;
    537             case InstallResultStatus::DownloadError:
    538                 // See installResult.ExtendedErrorCode for more detail.
    539                 // This is typically a transient network error.
    540                 actionData.Status(L"Download error");
    541                 break;
    542             case InstallResultStatus::InstallError:
    543                 // See installResult.ExtendedErrorCode and installResult.InstallerErrorCode for more detail.
    544                 // This is caused by an error in the installer or an issue with the system state.
    545                 // InstallerErrorCode is the value returned by the installer technology in use for the install
    546                 // attempt and may or may not be an HRESULT.
    547                 actionData.Status(L"Installation error");
    548                 break;
    549             case InstallResultStatus::ManifestError:
    550                 // See installResult.ExtendedErrorCode for more detail.
    551                 // This is an issue with the catalog providing the package.
    552                 actionData.Status(L"Manifest error");
    553                 break;
    554             case InstallResultStatus::NoApplicableInstallers:
    555                 // No applicable installers were available due the combination of the current system,
    556                 // user settings, and parameters provided to the install request.
    557                 actionData.Status(L"No applicable installers");
    558                 break;
    559             case InstallResultStatus::NoApplicableUpgrade:
    560                 // No upgrade was available due the combination of available versions, the current system,
    561                 // user settings, and parameters provided to the upgrade request.
    562                 actionData.Status(L"No applicable upgrade");
    563                 break;
    564             case InstallResultStatus::PackageAgreementsNotAccepted:
    565                 // The user has not accepted the agreements required by the package.
    566                 actionData.Status(L"Package agreements not accepted");
    567                 break;
    568             }
    569         }
    570 
    571         // Switch back to ui thread context.
    572         co_await winrt::resume_foreground(actionData.Dispatcher());
    573 
    574         progressBar.IsIndeterminate(false);
    575 
    576         actionData.Finalize();
    577     }

    762     IAsyncAction MainPage::UninstallAsync()
    763     {
    764         PackageManager packageManager = m_packageManager;
    765 
    766         auto progressBar = uninstallProgressBar();
    767         auto statusText = uninstallStatusText();
    768 
    769         IInspectable selectedValue = installedListBox().SelectedValue();
    770         CatalogPackage package{ nullptr };
    771         if (selectedValue)
    772         {
    773             package = selectedValue.as<CatalogPackage>();
    774         }
    775         else
    776         {
    777             statusText.Text(L"Select a package to uninstall");
    778             co_return;
    779         }
    780 
    781         BackgroundActionData actionData{ { uninstallButton() }, statusText };
    782 
    783         co_await winrt::resume_background();
    784 
    785         Windows::Foundation::IAsyncOperationWithProgress<Deployment::UninstallResult, Deployment::UninstallProgress> packageOperation;
    786 
    787         actionData.RunAndCatchStatus(L"Begin uninstall", [&]() {
    788             UninstallOptions uninstallOptions;
    789 
    790             uninstallOptions.PackageUninstallScope(PackageUninstallScope::Any);
    791             uninstallOptions.PackageUninstallMode(PackageUninstallMode::Silent);
    792 
    793             packageOperation = packageManager.UninstallPackageAsync(package, uninstallOptions);
    794         });
    795 
    796         actionData.RunAndCatchStatus(L"Set progress handler", [&]() {
    797             packageOperation.Progress([&](
    798                 IAsyncOperationWithProgress<UninstallResult, UninstallProgress> const& /* sender */,
    799                 UninstallProgress const& progress)
    800                 {
    801                     actionData.Dispatcher().RunAsync(Windows::UI::Core::CoreDispatcherPriority::Normal,
    802                         [progressBar, statusText, progress]() {
    803                             progressBar.Value(progress.UninstallationProgress * 100);
    804 
    805                             switch (progress.State)
    806                             {
    807                             case PackageUninstallProgressState::Queued:
    808                                 statusText.Text(L"Queued");
    809                                 break;
    810                             case PackageUninstallProgressState::Uninstalling:
    811                                 statusText.Text(L"Uninstaller running");
    812                                 progressBar.IsIndeterminate(true);
    813                                 break;
    814                             case PackageUninstallProgressState::PostUninstall:
    815                                 statusText.Text(L"Post uninstall bookkeeping");
    816                                 break;
    817                             case PackageUninstallProgressState::Finished:
    818                                 statusText.Text(L"Done");
    819                                 progressBar.IsIndeterminate(false);
    820                                 break;
    821                             default:
    822                                 statusText.Text(L"");
    823                             }
    824                         });
    825                 });
    826             });
    827 
    828         UninstallResult uninstallResult{ nullptr };
    829 
    830         actionData.RunAndCatchStatus(L"Uninstall", [&]() {
    831             uninstallResult = packageOperation.get();
    832         });
    833 
    834         if (packageOperation && packageOperation.Status() == AsyncStatus::Canceled)
    835         {
    836             actionData.Status(L"Cancelled");
    837         }
    838         else if (uninstallResult)
    839         {
    840             // Error handling for the installResult is done by first examining the Status.
    841             // Any status value other than Ok will have additional error detail in the
    842             // ExtendedErrorCode property. This HRESULT value will typically (but not always)
    843             // have the Windows Package Manager facility value (0xA15). The symbolic names and
    844             // meanings of these error codes can be found at:
    845             // https://github.com/microsoft/winget-cli/blob/master/doc/windows/package-manager/winget/returnCodes.md
    846             // or by using the winget CLI:
    847             // > winget error 0x8A150049
    848             // > winget error -- -2146762487
    849             switch (uninstallResult.Status())
    850             {
    851             case UninstallResultStatus::Ok:
    852                 actionData.Status(uninstallResult.RebootRequired() ? L"Reboot required" : L"Done");
    853                 break;
    854             case UninstallResultStatus::BlockedByPolicy:
    855                 // See installResult.ExtendedErrorCode for more detail.
    856                 // This is typically caused by system configuration applied by policy.
    857                 actionData.Status(L"Blocked by policy");
    858                 break;
    859             case UninstallResultStatus::CatalogError:
    860                 // See installResult.ExtendedErrorCode for more detail.
    861                 // This is typically an issue with an external service.
    862                 actionData.Status(L"Catalog error");
    863                 break;
    864             case UninstallResultStatus::InternalError:
    865                 // See installResult.ExtendedErrorCode for more detail.
    866                 // This is typically an issue with the Windows Package Manager code.
    867                 actionData.Status(L"Internal error");
    868                 break;
    869             case UninstallResultStatus::InvalidOptions:
    870                 // See installResult.ExtendedErrorCode for more detail.
    871                 // This is caused by invalid input combinations.
    872                 actionData.Status(L"Invalid options");
    873                 break;
    874             case UninstallResultStatus::UninstallError:
    875                 // See installResult.ExtendedErrorCode and installResult.UninstallerErrorCode for more detail.
    876                 // This is caused by an error in the uninstaller or an issue with the system state.
    877                 // UninstallerErrorCode is the value returned by the uninstaller technology in use for the uninstall
    878                 // attempt and may or may not be an HRESULT.
    879                 actionData.Status(L"Uninstallation error");
    880                 break;
    881             case UninstallResultStatus::ManifestError:
    882                 // See installResult.ExtendedErrorCode for more detail.
    883                 // This is an issue with the catalog providing the package.
    884                 actionData.Status(L"Manifest error");
    885                 break;
    886             }
    887         }
    888 
    889         // Switch back to ui thread context.
    890         co_await winrt::resume_foreground(actionData.Dispatcher());
    891 
    892         progressBar.IsIndeterminate(false);
    893 
    894         actionData.Finalize();
    895     }

    898     {
    899         BackgroundActionData actionData{ { activeRefreshButton() }, activeStatusText() };
    900 
    901         co_await winrt::resume_background();
    902 
    903         actionData.Status(EnsurePackageManager());
    904 
    905         PackageCatalog catalog{ nullptr };
    906 
    907         actionData.RunAndCatchStatus(L"Connect installed catalog", [&]() {
    908             PackageCatalogReference catalogReference = m_packageManager.GetLocalPackageCatalog(LocalPackageCatalog::InstallingPackages);
    909             ConnectResult connectResult{ catalogReference.Connect() };
    910 
    911             switch (connectResult.Status())
    912             {
    913             case ConnectResultStatus::Ok: break;
    914             case ConnectResultStatus::CatalogError: throw std::exception{ "Catalog connection error." };
    915             }
    916 
    917             catalog = connectResult.PackageCatalog();
    918         });
    919 
    920         winrt::IVectorView<MatchResult> matches;
    921 
    922         actionData.RunAndCatchStatus(L"Find package", [&]() {
    923             FindPackagesOptions findPackagesOptions;
    924             FindPackagesResult findPackagesResult = catalog.FindPackages(findPackagesOptions);
    925 
    926             matches = findPackagesResult.Matches();
    927         });
    928 
    929         co_await winrt::resume_foreground(actionData.Dispatcher());
    930 
    931         m_activePackageViews.Clear();
    932 
    933         if (matches)
    934         {
    935             for (auto const& match : matches)
    936             {
    937                 WinGetUWPCaller::ActivePackageView activeView;
    938                 activeView.Package(match.CatalogPackage());
    939                 auto installOperation = m_packageManager.GetInstallProgress(activeView.Package(), nullptr);
    940                 if (installOperation)
    941                 {
    942                     activeView.Dispatcher(actionData.Dispatcher());
    943                     activeView.AsyncOperation(installOperation);
    944                     m_activePackageViews.Append(activeView);
    945                 }
    946             }
    947         }
    948 
    949         actionData.Finalize();
    950     }
