     64 template <typename T>
     65 std::ostream& operator<<(std::ostream& out, const JSONPair<T>& pair)
     66 {
     67     out << '"' << pair.Name << "\": ";
     68     if (JSONControl<T>::quote)
     69     {
     70         out << '"';
     71     }
     72     if (JSONControl<T>::output)
     73     {
     74         out << pair.Value;
     75     }
     76     if (JSONControl<T>::quote)
     77     {
     78         out << '"';
     79     }
     80     if (pair.Comma)
     81     {
     82         out << ',';
     83     }
     84     return out << std::endl;
     85 }

     87 std::string ConvertToUTF8(std::wstring_view input)
     88 {
     89     if (input.empty())
     90     {
     91         return {};
     92     }
     93 
     94     int utf8ByteCount = WideCharToMultiByte(CP_UTF8, 0, input.data(), wil::safe_cast<int>(input.length()), nullptr, 0, nullptr, nullptr);
     95     THROW_LAST_ERROR_IF(utf8ByteCount == 0);
     96 
     97     // Since the string view should not contain the null char, the result won't either.
     98     // This allows us to use the resulting size value directly in the string constructor.
     99     std::string result(wil::safe_cast<size_t>(utf8ByteCount), '\0');
    100 
    101     int utf8BytesWritten = WideCharToMultiByte(CP_UTF8, 0, input.data(), wil::safe_cast<int>(input.length()), &result[0], wil::safe_cast<int>(result.size()), nullptr, nullptr);
    102     FAIL_FAST_HR_IF(E_UNEXPECTED, utf8ByteCount != utf8BytesWritten);
    103 
    104     return result;
    105 }

    107 std::wstring ConvertToUTF16(std::string_view input, UINT codePage = CP_UTF8)
    108 {
    109     if (input.empty())
    110     {
    111         return {};
    112     }
    113 
    114     int utf16CharCount = MultiByteToWideChar(codePage, 0, input.data(), wil::safe_cast<int>(input.length()), nullptr, 0);
    115     THROW_LAST_ERROR_IF(utf16CharCount == 0);
    116 
    117     // Since the string view should not contain the null char, the result won't either.
    118     // This allows us to use the resulting size value directly in the string constructor.
    119     std::wstring result(wil::safe_cast<size_t>(utf16CharCount), L'\0');
    120 
    121     int utf16CharsWritten = MultiByteToWideChar(codePage, 0, input.data(), wil::safe_cast<int>(input.length()), &result[0], wil::safe_cast<int>(result.size()));
    122     FAIL_FAST_HR_IF(E_UNEXPECTED, utf16CharCount != utf16CharsWritten);
    123 
    124     return result;
    125 }

    153     int ParseArgs(int argc, char** argv)
    154     {
    155         // Supports the following arguments:
    156         //  -id : [Required] The PackageIdentifier to install and check for correlation
    157         //  -src : [Required] The source name for the package to install
    158         //  -out : [Required] The file to write results to
    159         //  -dev : [Optional] Use the dev CLSIDs
    160         //  -cor : [Optional] Only correlate the package
    161 
    162         for (int i = 1; i < argc; ++i)
    163         {
    164             if ("-id"sv == argv[i] && i + 1 < argc)
    165             {
    166                 packageIdentifier = argv[++i];
    167             }
    168             else if ("-src"sv == argv[i] && i + 1 < argc)
    169             {
    170                 sourceName = argv[++i];
    171             }
    172             else if ("-out"sv == argv[i] && i + 1 < argc)
    173             {
    174                 outputPath = argv[++i];
    175             }
    176             else if ("-meta"sv == argv[i] && i + 1 < argc)
    177             {
    178                 metadataCollection = true;
    179                 wingetUtilPath = argv[++i];
    180             }
    181             else if ("-sys32"sv == argv[i] && i + 1 < argc)
    182             {
    183                 sys32Path = argv[++i];
    184             }
    185             else if ("-dev"sv == argv[i])
    186             {
    187                 useDevCLSIDs = true;
    188             }
    189             else if ("-cor"sv == argv[i])
    190             {
    191                 onlyCorrelate = true;
    192             }
    193         }
    194 
    195         // Check inputs
    196         if (outputPath.empty())
    197         {
    198             std::cout << "No output file path specified, use -out" << std::endl;
    199             return 2;
    200         }
    201 
    202         if (!outputPath.has_stem())
    203         {
    204             std::cout << "Output path is not a file" << std::endl;
    205             return 3;
    206         }
    207 
    208         std::filesystem::create_directories(outputPath.parent_path());
    209 
    210         outputStream.open(outputPath);
    211 
    212         if (!outputStream)
    213         {
    214             std::cout << "Output file could not be created" << std::endl;
    215             return 4;
    216         }
    217 
    218         return 0;
    219     }

    392     void Install()
    393     {
    394         try
    395         {
    396             action = "Create package manager";
    397             auto packageManager = CreatePackageManager();
    398 
    399             action = "Get source reference";
    400             auto sourceRef = packageManager.GetPackageCatalogByName(ConvertToUTF16(sourceName));
    401 
    402             action = "Connecting to catalog";
    403             auto connectResult = sourceRef.Connect();
    404 
    405             if (connectResult.Status() != ConnectResultStatus::Ok)
    406             {
    407                 hr = E_FAIL;
    408                 error = "Error connecting to catalog";
    409                 return;
    410             }
    411 
    412             auto catalog = connectResult.PackageCatalog();
    413 
    414             action = "Create find options";
    415             auto findOptions = CreateFindPackagesOptions();
    416 
    417             action = "Add package id filter";
    418             auto filter = CreatePackageMatchFilter();
    419             filter.Field(PackageMatchField::Id);
    420             filter.Option(PackageFieldMatchOption::Equals);
    421             filter.Value(ConvertToUTF16(packageIdentifier));
    422             findOptions.Filters().Append(filter);
    423 
    424             action = "Find package";
    425             auto findResult = catalog.FindPackages(findOptions);
    426 
    427             if (findResult.Status() != FindPackagesResultStatus::Ok)
    428             {
    429                 hr = E_FAIL;
    430                 std::ostringstream stream;
    431                 stream << "Error " << static_cast<int>(findResult.Status()) << " finding package";
    432                 error = std::move(stream).str();
    433                 return;
    434             }
    435 
    436             action = "Get match";
    437             auto matches = findResult.Matches();
    438 
    439             if (matches.Size() == 0)
    440             {
    441                 hr = E_NOT_SET;
    442                 error = "Package not found";
    443                 return;
    444             }
    445 
    446             auto package = matches.GetAt(0).CatalogPackage();
    447 
    448             action = "Inspect package";
    449             auto installVersion = package.DefaultInstallVersion();
    450             packageName = ConvertToUTF8(installVersion.DisplayName());
    451             packagePublisher = ConvertToUTF8(installVersion.Publisher());
    452 
    453             if (metadataCollection)
    454             {
    455                 BeginMetadataCollection();
    456                 if (FAILED(hr))
    457                 {
    458                     return;
    459                 }
    460             }
    461 
    462             if (!onlyCorrelate)
    463             {
    464                 action = "Create install options";
    465                 auto installOptions = CreateInstallOptions();
    466 
    467                 installOptions.PackageInstallScope(PackageInstallScope::Any);
    468                 installOptions.PackageInstallMode(PackageInstallMode::Silent);
    469 
    470                 std::cout << "Beginning to install " << packageIdentifier << " (" << packageName << ") from " << sourceName << "..." << std::endl;
    471                 action = "Install package";
    472                 auto installOperation = packageManager.InstallPackageAsync(package, installOptions);
    473 
    474                 if (installOperation.wait_for(std::chrono::minutes(10)) != AsyncStatus::Completed)
    475                 {
    476                     hr = E_FAIL;
    477                     error = "Install operation timed out";
    478                     return;
    479                 }
    480 
    481                 auto installResult = installOperation.GetResults();
    482                 if (installResult.Status() != InstallResultStatus::Ok)
    483                 {
    484                     hr = installResult.ExtendedErrorCode();
    485                     error = "Error installing package";
    486                     return;
    487                 }
    488             }
    489 
    490             if (metadataCollection)
    491             {
    492                 CompleteMetadataCollection();
    493                 if (FAILED(hr))
    494                 {
    495                     return;
    496                 }
    497             }
    498         }
    499         catch (const winrt::hresult_error& hre)
    500         {
    501             hr = hre.code();
    502             error = ConvertToUTF8(hre.message());
    503         }
    504     }

    629 
    630             action = "Get matches";
    631             auto matches = findResult.Matches();
    632 
    633             action = "Get source info";
    634             auto sourceInfo = sourceRef.Info();
    635             auto sourceIdentifier = sourceInfo.Id();
    636             auto sourceType = sourceInfo.Type();
    637 
    638             for (const auto& match : matches)
    639             {
    640                 auto package = match.CatalogPackage();
    641 
    642                 if (ConvertToUTF8(package.Id()) != packageIdentifier)
    643                 {
    644                     continue;
    645                 }
    646 
    647                 auto installed = package.InstalledVersion();
    648 
    649                 if (installed)
    650                 {
    651                     auto installedCatalogInfo = installed.PackageCatalog().Info();
    652 
    653                     if (installedCatalogInfo.Id() == sourceIdentifier && installedCatalogInfo.Type() == sourceType)
    654                     {
    655                         correlateArchive = true;
    656                         archiveName = ConvertToUTF8(installed.DisplayName());
    657                         archivePublisher = ConvertToUTF8(installed.Publisher());
    658                         break;
    659                     }
    660                 }
    661             }
    662         }
    663         catch (const winrt::hresult_error& hre)
    664         {
    665             hr = hre.code();
    666             error = ConvertToUTF8(hre.message());
    667         }
    668     }
    669 
    670     void ReportResult()
    671     {
    672         if (outputStream)
    673         {
    674             outputStream << "{" << std::endl;
    675             outputStream << JSONPair{ "PackageIdentifier", packageIdentifier };
    676             outputStream << JSONPair{ "Source", sourceName };
    677             outputStream << JSONPair{ "UseDev", useDevCLSIDs };
    678             outputStream << JSONPair{ "Error", error };
    679             outputStream << JSONPair{ "Phase", phase };
    680             outputStream << JSONPair{ "Action", action };
    681             outputStream << JSONPair{ "PackageName", packageName };
    682             outputStream << JSONPair{ "PackagePublisher", packagePublisher };
    683             outputStream << JSONPair{ "CorrelatePackageKnown", correlatePackageKnown };
    684             outputStream << JSONPair{ "PackageKnownName", packageKnownName };
    685             outputStream << JSONPair{ "PackageKnownPublisher", packageKnownPublisher };
    686             outputStream << JSONPair{ "CorrelateArchive", correlateArchive };
    687             outputStream << JSONPair{ "ArchiveName", archiveName };
    688             outputStream << JSONPair{ "ArchivePublisher", archivePublisher };
    689             // Keep at the end to prevent a dangling comma
    690             outputStream << JSONPair{ "HRESULT", hr, false } << "}" << std::endl;
    691         }
    692     }
    693 
    694     void main(int argc, char** argv)
    695     {
    696         hr = ParseArgs(argc, argv);
    697         if (hr != 0)
    698         {
    699             return;
    700         }
    701 
    702         ValidateArgs();
    703         if (FAILED(hr))
    704         {
    705             return;
    706         }
    707 
    708         if (metadataCollection)
    709         {
    710             LoadWingetUtil();
    711             if (FAILED(hr))
    712             {
    713                 return;
    714             }
    715         }
    716 
    717         auto co_uninitialize = wil::CoInitializeEx();
    718 
    719         // Execute the install step
    720         phase = "Install";
    721         std::cout << "Connecting to PackageManager..." << std::endl;
    722         Install();
    723         if (FAILED(hr))
    724         {
    725             return;
    726         }
    727 
    728         // Check for the installed package being correlated when the remote package is known,
    729         // as when trying to determine information about a single known package.
    730         phase = "Correlate when package known";
    731         std::cout << "Correlating package when known..." << std::endl;
    732         CorrelatePackageKnown();
    733 
    734         // Check for the installed package being correlated when archiving local package information.
    735         phase = "Correlate when archiving";
    736         std::cout << "Correlating package when archiving..." << std::endl;
    737         CorrelateArchive();
    738 
    739         std::cout << "Done" << std::endl;
    740         phase = "Completed";
    741         action.clear();
    742     }
    743 };
