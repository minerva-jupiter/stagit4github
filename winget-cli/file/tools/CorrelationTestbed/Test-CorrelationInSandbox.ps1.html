    109 function Close-WindowsSandbox {
    110     $sandbox = Get-Process 'WindowsSandboxClient' -ErrorAction SilentlyContinue
    111     if ($sandbox)
    112     {
    113       Write-Host '--> Closing Windows Sandbox'
    114 
    115       $sandboxServer = Get-Process 'WindowsSandbox' -ErrorAction SilentlyContinue
    116 
    117       $sandbox | Stop-Process
    118       $sandbox | Wait-Process -Timeout 120
    119 
    120       # Also wait for the server to close
    121       if ($sandboxServer)
    122       {
    123         try
    124         {
    125             $sandboxServer | Wait-Process -Timeout 120
    126         }
    127         catch [System.TimeoutException]
    128         {
    129             # Force stop the server if it does not automatically stop
    130             $sandboxServer | Stop-Process
    131             $sandboxServer | Wait-Process -Timeout 120
    132         }
    133 
    134       }
    135 
    136       Write-Host
    137     }
    138     Remove-Variable sandbox
    139 }
