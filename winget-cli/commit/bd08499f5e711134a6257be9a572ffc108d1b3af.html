+        /// </summary>
+        public bool Completed
+        {
+            set
+            {
+                this.completed = value;
+                if (value)
+                {
+                    this.resetEvent.Set();
+                }
+            }
+        }

+        /// This should be called on the main thread to wait for the asynchronous operation to complete.
+        /// </summary>
+        public void Wait()
+        {
+            while (!this.completed)
+            {
+                lock (this.records)
+                {
+                    this.Flush();
+                }

+                this.resetEvent.WaitOne();
+            }

+            this.Flush();
+        }

+                }

+            }
