+        {

+            // TODO: go to list filtered to packages with update available
+        }
+        else if (context.Args.Contains(Execution::Args::Type::All))
+        {
+            // --all switch updates all packages found
+            context <<
+                SearchSourceForMany <<
+                EnsureMatchesFromSearchResult <<
+                UpdateAllApplicable;
+        }

+            context <<
+                SearchSourceForSingle <<
