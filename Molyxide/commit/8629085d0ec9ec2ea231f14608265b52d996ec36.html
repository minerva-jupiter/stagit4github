+/// ZDF SVF (Topology Preserving Transform)
+struct ZdfSvf {
+    s1: f32,
+    s2: f32,
+}
+
+impl ZdfSvf {
+    fn new() -> Self {
+        Self { s1: 0.0, s2: 0.0 }
+    }
+    fn process(&mut self, input: f32, mode: ResonatorMode, freq: f32, emph: f32, sr: f32) -> f32 {
+        let q = (0.5 + emph * 9.5).max(0.01);
+        let g = (std::f32::consts::PI * freq / sr).tan();
+        let k = 1.0 / q;
+        let a1 = 1.0 / (1.0 + g * (g + k));
+        let a2 = g * a1;
+        let a3 = g * a2;
+        let m_v3 = input - self.s2;
+        let v1 = a1 * self.s1 + a2 * m_v3;
+        let v2 = self.s2 + a2 * self.s1 + a3 * m_v3;
+        self.s1 = 2.0 * v1 - self.s1;
+        self.s2 = 2.0 * v2 - self.s2;
+        match mode {
+            ResonatorMode::LowPass => v2,
+            ResonatorMode::BandPass => v1,
+            ResonatorMode::HighPass => input - k * v1 - v2,
+        }
+    }
+}

+/// ZDF Ladder Filter (24dB/oct)
+/// Moog スタイルの 4-pole ローパスフィルター
+struct LadderFilter {
+    s: [f32; 4],
+}
+
+impl LadderFilter {
+    fn new() -> Self {
+        Self { s: [0.0; 4] }
+    }
+    fn process(&mut self, input: f32, cutoff: f32, resonance: f32, sr: f32) -> f32 {
+        let g = (std::f32::consts::PI * cutoff / sr).tan();
+        let k = 4.0 * resonance.clamp(0.0, 0.99); // 自励振を避けるため微調整
+
+        // 1-pole 係数
+        let a = 1.0 / (1.0 + g);
+
+        // フィードバック補正
+        let g_pow4 = g * g * g * g;
+        let sigma = g_pow4 * a * a * a * a;
+        let gamma = 1.0 / (1.0 + k * sigma);
+
+        // 内部状態からのフィードバック
+        let input_fb = (input
+            - k * (self.s[0] * a
+                + self.s[1] * a * a
+                + self.s[2] * a * a * a
+                + self.s[3] * a * a * a * a))
+            * gamma;
+
+        // 4段の 1-pole フィルターの直列処理
+        let mut vin = input_fb;
+        for i in 0..4 {
+            let v = (vin - self.s[i]) * g * a;
+            let out = v + self.s[i];
+            self.s[i] = out + v;
+            vin = out;
+        }
+
+        vin // 4段目の出力
+    }
+}
