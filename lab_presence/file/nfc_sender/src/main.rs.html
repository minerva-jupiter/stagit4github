     20     fn init(&mut self) -> Result<(), Box<dyn std::error::Error>> {
     21         self.core_url =
     22             std::env::var("CORE_URL").unwrap_or_else(|_| "https://localhost:8443".to_string());
     23         self.member_directory = serde_json::from_str(&std::env::var("MEMBER_DIRECTORY")?)?;
     24         let allowed_devices: serde_json::Value =
     25             serde_json::from_str(&std::env::var("ALLOWED_DEVICES")?)?;
     26         let secret: &str = allowed_devices["nfc"]
     27             .as_str()
     28             .ok_or("ALLOWED_DEVICES.nfc is not a string or missing")?;
     29         self.login(secret.to_string())?;
     30         Ok(())
     31     }

     33     fn login(&mut self, secret: String) -> Result<(), Box<dyn std::error::Error>> {
     34         let login_request = shared::LoginRequest {
     35             device_id: "nfc".to_string(),
     36             device_secret: secret,
     37         };
     38         let response: shared::LoginResponse =
     39             ureq::post(&format!("{}/api/v1/login", self.core_url))
     40                 .send_json(login_request)?
     41                 .body_mut()
     42                 .read_json::<shared::LoginResponse>()?;
     43         self.token = response.access_token;
     44         Ok(())
     45     }
     46     fn update_presence(&self, user_id: &UserId, presence: &str) -> Result<(), ureq::Error> {
     47         let request_body = shared::PresenceUpdateRequest {
     48             presence: presence.to_string(),
     49             status_msg: "NFC toggle request sent".to_string(),
     50         };
     51         let _ = ureq::put(&format!(
     52             "{}/api/v1/presence/{}/status",
     53             self.core_url, user_id.id
     54         ))
     55         .header("Authorization", &format!("Bearer {}", self.token))
     56         .send_json(request_body)?;
     57         Ok(())
     58     }
     59     fn get_presence(&self, user_id: &UserId) -> Result<String, ureq::Error> {
     60         let now_presence_res = ureq::get(&format!(
     61             "{}/api/v1/presence/{}/status",
     62             self.core_url, user_id.id
     63         ))
     64         .header("Authorization", &format!("Bearer {}", self.token))
     65         .call()?
     66         .body_mut()
     67         .read_json::<shared::PresenceResponse>()?;

     69     }
     70     fn toggled_presence(presence: &str) -> &'static str {

     76     }

     78 fn main() -> Result<(), Box<dyn std::error::Error>> {
     79     let _ = dotenvy::dotenv();
     80     let ctx = Context::establish(Scope::User)?;
     81 
     82     let mut nfc_sender: NfcSender = NfcSender::new();
     83     nfc_sender.init()?;
     84 
     85     println!("NFC sender has started...");
     86 
     87     loop {
     88         let mut readers_buf = [0u8; 2048];
     89         let mut readers = match ctx.list_readers(&mut readers_buf) {
     90             Ok(r) => r,
     91             Err(_) => {
     92                 sleep(Duration::from_secs(1));
     93                 continue;
     94             }
     95         };
     96 
     97         let reader_name = match readers.next() {
     98             Some(name) => name,
     99             None => {
    100                 sleep(Duration::from_secs(1));
    101                 continue;
    102             }
    103         };
    104 
    105         if let Ok(card) = ctx.connect(reader_name, ShareMode::Shared, Protocols::ANY) {
    106             let felica_poll_cmd = [
    107                 0xFF, 0x00, 0x00, 0x00, 0x08, 0xD4, 0x40, 0x01, 0x00, 0xFF, 0xFF, 0x01, 0x00,
    108             ];
    109             let mut rapdu_buf = [0u8; 258];
    110 
    111             if let Ok(rapdu) = card.transmit(&felica_poll_cmd, &mut rapdu_buf) {
    112                 if rapdu.len() >= 18 {
    113                     let idm_bytes = &rapdu[10..18];
    114                     let idm_str = idm_bytes
    115                         .iter()
    116                         .map(|b| format!("{:02X}", b))
    117                         .collect::<String>();
    118 
    119                     if let Ok(user_id) = nfc_sender.member_directory.inquire(&idm_str) {
    120                         if let Ok(current_presence) = nfc_sender.get_presence(&user_id) {
    121                             let toggled = NfcSender::toggled_presence(&current_presence);
    122                             let _ = nfc_sender.update_presence(&user_id, toggled);
    123                         }
    124                     }
    125 
    126                     sleep(Duration::from_secs(2));
    127                 }
    128             }
    129         }
    130 
    131         sleep(Duration::from_millis(500));
    132     }
    133 }
