      9 export default function Home() {

     20       if (user) {
     21         setIsSignedIn(true);

     23 
     24         try {
     25           await OneSignal.init({
     26             appId: process.env.NEXT_PUBLIC_ONESIGNAL_APP_ID!,
     27           });
     28

     32           console.error("OneSignal operation failed:", error);
     33         }

     36         console.log("User is signed out");
     37       }
     38     });
     39

     41   }
     42   function Signin() {
     43     const handleSignIn = async () => {
     44       try {
     45         await signInWithPopup(auth, authProvider);
     46         console.log("Sign-in popup successful");
     47       } catch (error) {
     48         console.error("Error signing in:", error);
     49       }
     50     };
     51     const handleSignOut = async () => {
     52       try {
     53         await signOut(auth);

     56         console.error("Error signing out:", error);
     57       }
     58     };
     59     if (isSignedIn == false) {
     60       return <button onClick={handleSignIn}>Sign in with Google</button>;
     61     } else {
     62       return <button onClick={handleSignOut}>Sign out</button>;
     63     }
     64   }

     67       <h1 className="text-3xl font-bold text-center text-gray-800">
     68         Café Timer
     69       </h1>
     70       {Signin()}

     72 
     73       <button onClick={() => handleTap("normal")}>Taped !!!</button>
     74 
     75       <button onClick={() => handleTap("ticket1")}>Ticket 1</button>
