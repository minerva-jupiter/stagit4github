     35 fn score_text(bytes: &[u8]) -> f64 {
     36     let mut score = 0.0;
     37     for &b in bytes {
     38         match b.to_ascii_lowercase() {
     39             b'e' => score += 12.02,
     40             b't' => score += 9.10,
     41             b'a' => score += 8.12,
     42             b'o' => score += 7.68,
     43             b'i' => score += 7.31,
     44             b'n' => score += 6.95,
     45             b's' => score += 6.28,
     46             b'r' => score += 6.02,
     47             b'h' => score += 5.92,
     48             b'd' => score += 4.32,
     49             b'l' => score += 3.98,
     50             b'u' => score += 2.88,
     51             b' ' => score += 15.0,
     52             0..=31 | 127 => score -= 50.0,
     53             32..=126 => score += 1.0,
     54             _ => score -= 10.0,
     55         }
     56     }

     60 fn crack_single_byte_xor(crypted: &[u8]) -> (u8, Vec<u8>, f64) {
     61     let mut best_score = f64::MIN;
     62     let mut best_key = 0;
     63     let mut best_msg = Vec::new();
     64
