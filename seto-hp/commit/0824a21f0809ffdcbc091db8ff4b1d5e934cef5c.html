+          setProgress((xhr.loaded / xhr.total) * 100)

+        // called when loading has errors
+        (error) => {
+          console.log("An error happened")

+        }
+      )
+    }
+  }, [gltf])
+
+  return (
+    <>
+      {gltf ? (
+        <primitive object={gltf.scene} />
+      ) : (
+        <Html center>{progress} % loaded</Html>

+    </>
