+import type { Temporal } from '@js-temporal/polyfill';
+import type { Map } from 'immutable';
+
+// 学年
+export type Grade = 1 | 2 | 3;
+// クラス
+export type Class = 1 | 2 | 3 | 4 | 5 | 6 | 7 | 8;
+
+type Weekday = '月' | '火' | '水' | '木' | '金';
+type SchoolPeriod = 1 | 2 | 3 | 4 | 5 | 6 | 7;
+
+// 月1 から 金7 (木7 以外)
+export type AbstractLessonName = Exclude<`${Weekday}${SchoolPeriod}`, '木7'>;
+// 授業名
+export type ActualLessonName = string;
+
+export type Lesson = AbstractLessonName | ActualLessonName;
+
+// 1日の授業
+type DayLessons = Record<SchoolPeriod, string | null>;
+
+// クラスの標準時間割
+export type Standard = Readonly<Record<AbstractLessonName, string>>;
+
+// 曜日変更込みの時間割表 (月1, 月2,… など) または、授業名代入済みの時間割表
+export type Calendar = Map<Temporal.PlainDate, DayLessons>;
+
+// クラスの臨時変更
+export type Unusual = Map<Temporal.PlainDate, Partial<DayLessons>>;

+const csvToArray = async (path: PathLike): Promise<string[][]> => {
+  const csv = await readFile(path);

+    .toString()
+    .trim()
+    .split('\n')
+    .map((s) => s.split(','));

+const arrayToCsv = async (path: PathLike, arr: (string | null)[][]): Promise<void> => {
+  const csvString = arr.map((v) => (v ?? '').join(',')).join('\n');

+};
