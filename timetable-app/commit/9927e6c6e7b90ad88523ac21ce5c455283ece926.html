+import { useLocalStorage } from '@mantine/hooks';
+
+import formatDate from 'common/utils/formatDate';
+
+import type { GradeClass, SchoolTime } from 'common/types';
+
+const getStorageKey = (gradeClass: string, date: Date, period: SchoolTime) => {
+	const dateString = formatDate(date);
+	return `specialChange_data_${gradeClass}_${dateString}_${period}`;
+};
+
+type Params = {
+	gradeClass: GradeClass | null;
+	date: Date;
+	period: SchoolTime;
+};
+
+type Return = [string | null, (value: string) => void];
+
+const useSpecialChange = ({ gradeClass, date, period }: Params): Return => {
+	const [specialChangeSubject, setSpecialChangeSubject] = useLocalStorage<
+		string | undefined
+	>({
+		key: getStorageKey(gradeClass ?? '', date, period),
+		getInitialValueInEffect: true,
+	});
+	return [specialChangeSubject ?? null, setSpecialChangeSubject];
+};
+
+export default useSpecialChange;
