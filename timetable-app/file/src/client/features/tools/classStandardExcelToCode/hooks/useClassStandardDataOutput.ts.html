     16 const useDayPeriodDataOutput = (): Result => {
     17 	const [value, setValue] = useState<string>('');
     18 
     19 	const transformIntoObject = (data: InputSchema) => {
     20 		const output = structuredClone(initialData);
     21 
     22 		// 横方向曜日時限、縦方向クラスの2次元配列に転置する
     23 		const transposedData = data[0]?.map((_, i) => data.map((row) => row[i]));
     24 
     25 		// 各クラスについて実行
     26 		transposedData?.map((row, i) => {
     27 			const gradeClass = GRADE_CLASSES[i];
     28 
     29 			// 各曜日時限について実行
     30 			row.map((subjects, j) => {
     31 				const dayPeriod = DAY_PERIOD_JA[j];
     32 				if (gradeClass && dayPeriod && subjects) {
     33 					output[gradeClass][dayPeriod] = subjects
     34 						.split('/')
     35 						.map((subject) => ({
     36 							shortName: subject,
     37 						}));
     38 				}
     39 			});
     40 		});
     41 
     42 		format(JSON.stringify(output), {
     43 			useTabs: true,
     44 			parser: 'json',
     45 			plugins: [prettierPluginBabel, prettierPluginEstree],
     46 		})
     47 			.then((v) => {
     48 				setValue(v);
     49 			})
     50 			.catch((err) => {
     51 				console.log(err);
     52 			});
     53 	};
     54 
     55 	return {
