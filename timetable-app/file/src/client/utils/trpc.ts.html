      7 function getBaseUrl() {
      8 	if (typeof window !== 'undefined')
      9 		// browser should use relative path
     10 		return '';
     11 	if (process.env['VERCEL_URL'])
     12 		// reference for vercel.com
     13 		return `https://${process.env['VERCEL_URL']}`;

     15 		// reference for render.com
     16 		return `http://${process.env['RENDER_INTERNAL_HOSTNAME']}:${process.env['PORT']}`;
     17 	// assume localhost
     18 	return `http://localhost:${process.env['PORT'] ?? 3000}`;
     19 }
