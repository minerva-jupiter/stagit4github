    177         static double InputNormalizerToDouble(string input,double oldValue)
    178         {
    179             if(input == null || input == ""){
    180                 return oldValue;
    181             }
    182             bool NormalizationCan = double.TryParse(input, out double value);
    183             if (NormalizationCan == false || value < 0 || value > 100)
    184             {
    185                 string messageBoxText = "Please enter up to 100";
    186                 string caption = "unexpected input";
    187                 MessageBoxButton button = MessageBoxButton.OK;
    188                 MessageBoxImage icon = MessageBoxImage.Warning;
    189 
    190                 System.Windows.MessageBox.Show(messageBoxText, caption, button, icon);
    191                 return oldValue;
    192             }
    193             else
    194             {
    195                 return value;
    196             }
    197         }
