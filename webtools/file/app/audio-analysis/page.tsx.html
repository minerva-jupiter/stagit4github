     39     const maxPeriodSamples = sampleRate / 50;

     41     for (
     42       let offset = Math.floor(minPeriodSamples);

     46       let correlation = 0;

     48         correlation += timeDomainBuffer[i] * timeDomainBuffer[i + offset];
     49       }
     50       if (correlation > maxCorrelation) {
     51         maxCorrelation = correlation;
     52         bestOffset = offset;
     53       }
     54     }
     55 
     56     if (bestOffset > 0) {

     59

     67           ctx.lineWidth = 3;
     68           ctx.beginPath();
     69 
     70           let startSampleIndex = 0;
     71           for (let i = 1; i < timeDomainBuffer.length / 2; i++) {
     72             if (timeDomainBuffer[i - 1] < 0 && timeDomainBuffer[i] >= 0) {
     73               startSampleIndex = i;
     74               break;
     75             }
     76           }
     77 
     78           const xStep = canvas.width / bestOffset;
     79           for (let i = 0; i < bestOffset; i++) {
     80             const sampleValue = timeDomainBuffer[startSampleIndex + i] || 0;
     81             const x = i * xStep;
     82             const y = (1 - sampleValue) * (canvas.height / 2);
     83             i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
     84           }
     85           ctx.stroke();
     86         }
     87       }
     88     }
