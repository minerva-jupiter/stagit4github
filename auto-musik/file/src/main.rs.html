      9 fn main() -> Result<(), Box<dyn std::error::Error>> {
     10     let target_port_name = "Microsoft GS Wavetable Synth";
     11     let mut transmitter = MidiTransmitter::new(target_port_name)?;
     12 
     13     let mut generator = MusicGenerator::new();

     15     // 1分(60,000ms) / 120 = 500ms が四分音符の長さ
     16     const QUARTER_NOTE_DURATION_MS: u64 = 500;
     17 
     18     let start_time = std::time::Instant::now();
     19     let mut current_time_ms: u64 = 0;
     20 
     21     loop {
     22         let next_bar: Bar = generator.generate_next_bar();
     23         println!("Generated Chord is: {:?}", next_bar.chord); // どのコードが選ばれたか表示
     24 
     25         let bar_duration_ms = 4 * QUARTER_NOTE_DURATION_MS;
     26 
     27         for (event_time_offset, event) in next_bar.events.iter() {
     28             let target_absolute_time_ms = current_time_ms + event_time_offset;
     29 
     30             let elapsed_time = start_time.elapsed().as_millis() as u64;
     31 
     32             if target_absolute_time_ms > elapsed_time {
     33                 let wait_time_ms = target_absolute_time_ms - elapsed_time;
     34                 std::thread::sleep(std::time::Duration::from_millis(wait_time_ms));
     35             }
     36 
     37             if let Some(message) = note_event_to_midi_message(*event)
     38                 && let Err(e) = transmitter.send_message(&message)
     39             {
     40                 eprintln!("Midi sending error: {}", e);
     41             }
     42         }
     43 
     44         current_time_ms += bar_duration_ms;
     45 
     46         let elapsed_time = start_time.elapsed().as_millis() as u64;
     47         if current_time_ms > elapsed_time {
     48             let wait_time_ms = current_time_ms - elapsed_time;
     49             std::thread::sleep(std::time::Duration::from_millis(wait_time_ms));
     50         }
     51     }
     52 }

     54 fn note_event_to_midi_message(event: NoteEvent) -> Option<[u8; 3]> {
     55     match event {
     56         NoteEvent::NoteOn { note, velocity } => Some([0x90, note, velocity]),
     57         NoteEvent::NoteOff { note } => Some([0x80, note, 0]),
     58     }
     59 }
